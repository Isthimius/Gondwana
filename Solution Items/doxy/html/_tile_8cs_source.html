150 public int CompareTo(Tile? tile)
151 {
152 if (tile is null)
153 return -1;
154
155 float thisLoc = GetTileLocForCompare(this);
156 float tileLoc = GetTileLocForCompare(tile);
157
158 // Handle fixed position vs non-fixed first
159 if (IsPositionFixed && !tile.IsPositionFixed)
160 return -1;
161
162 if (!IsPositionFixed && tile.IsPositionFixed)
163 return 1;
164
165 // Use tuple comparison for the rest (Y, Z, X)
166 return (thisLoc, zOrder, SceneLayerCoordinates.X)
167 .CompareTo((tileLoc, tile.zOrder, tile.SceneLayerCoordinates.X));
168 }
169
174 private static float GetTileLocForCompare(Tile tile)
175 {
176 return tile.IsPositionFixed
177 ? tile.DrawLocationWorld.Top + tile.OverhangPixels.Top
178 : tile.DrawLocationWorld.Bottom - tile.OverhangPixels.Bottom - 1;
179 }
