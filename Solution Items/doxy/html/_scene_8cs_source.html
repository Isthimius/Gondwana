182 public RectangleF GetWorldBoundsPx()
183 {
184 if (_sceneLayers.Count == 0)
185 return RectangleF.Empty;
186
187 RectangleF result = RectangleF.Empty;
188 bool hasBounds = false;
189
190 foreach (var layer in _sceneLayers)
191 {
192 var lb = layer.GetLayerBoundsPx();
193 if (lb.IsEmpty)
194 continue;
195
196 if (!hasBounds)
197 {
198 result = lb;
199 hasBounds = true;
200 }
201 else
202 {
203 result = RectangleF.Union(result, lb);
204 }
205 }
206
207 return result;
