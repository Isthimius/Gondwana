223 private static SKRect ComputeDestRect(RectangleF bounds, int srcW, int srcH, StretchMode mode)
224 {
225 var b = bounds;
226 var dst = new SKRect(b.Left, b.Top, b.Right, b.Bottom);
227 if (mode == StretchMode.Fill || srcW <= 0 || srcH <= 0) return dst;
228
229 float bw = b.Width, bh = b.Height;
230 float arSrc = (float)srcW / srcH;
231 float arDst = bw / bh;
232
233 if (mode == StretchMode.None)
234 return new SKRect(b.Left, b.Top, b.Left + srcW, b.Top + srcH);
235
236 if (mode == StretchMode.Uniform)
237 {
238 if (arSrc > arDst)
239 {
240 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
241 return new SKRect(b.Left, y, b.Left + w, y + h);
242 }
243 else
244 {
245 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
246 return new SKRect(x, b.Top, x + w, b.Top + h);
247 }
248 }
249
250 // UniformToFill
251 if (arSrc < arDst)
252 {
253 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
254 return new SKRect(b.Left, y, b.Left + w, y + h);
255 }
256 else
257 {
258 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
259 return new SKRect(x, b.Top, x + w, b.Top + h);
260 }

265 private void OnFrameReady(object? sender, VideoFrameReadyEventArgs e)
266 {
267 // Copy the RGBA buffer into a reusable SKBitmap.
268 // NOTE: do not hold the lock while touching Skia; just protect shared refs/copies.
269 SKBitmap? local;
270 lock (_frameLock)
271 {
272 if (_frame is null || _srcW != e.Width || _srcH != e.Height)
273 {
274 _frame?.Dispose();
275 _frame = new SKBitmap(new SKImageInfo(e.Width, e.Height, SKColorType.Rgba8888));
276 _srcW = e.Width;
277 _srcH = e.Height;
278 _srcStride = e.Stride;
279 }
280 local = _frame;
281 }
282
283 if (local is null) return;
284
285 CopyFrameToBitmap(e.Pixels, e.Stride, local, _srcW, _srcH);
286
287 // Mark dirty so the direct-draw manager repaints our area.
288 ForceRefresh();
289 }

291 private static void CopyFrameToBitmap(IntPtr srcPixels, int srcStride, SKBitmap dst, int width, int height)
292 {
293#if !BROWSER && !NO_UNSAFE
294 // Fast path: single memcpy using pointers
295 unsafe
296 {
297 using var pix = dst.PeekPixels();
298 var dstPtr = (byte*)pix.GetPixels().ToPointer();
299 Buffer.MemoryCopy(
300 source: (void*)srcPixels,
301 destination: dstPtr,
302 destinationSizeInBytes: pix.RowBytes * height,
303 sourceBytesToCopy: (long)srcStride * height);
304 }
305#else
306 // Safe path: row-by-row Marshal.Copy (no unsafe)
307 using var pix = dst.PeekPixels();
308 IntPtr dstBase = pix.GetPixels();
309 int dstRB = (int)pix.RowBytes;
310 int srcRB = srcStride;
311 int rowBytes = Math.Min(dstRB, srcRB);
312
313 var rowBuf = System.Buffers.ArrayPool<byte>.Shared.Rent(rowBytes);
314 try
315 {
316 for (int y = 0; y < height; y++)
317 {
318 var srcRow = IntPtr.Add(srcPixels, y * srcRB);
319 var dstRow = IntPtr.Add(dstBase, y * dstRB);
320 Marshal.Copy(srcRow, rowBuf, 0, rowBytes);
321 Marshal.Copy(rowBuf, 0, dstRow, rowBytes);
322 }
323 }
324 finally
325 {
326 System.Buffers.ArrayPool<byte>.Shared.Return(rowBuf, clearArray: false);
327 }
328#endif
329 }
