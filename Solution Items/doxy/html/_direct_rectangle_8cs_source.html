325 public override void Update(long tick)
326 {
327 if (tick <= _lastTick)
328 return;
329
330 // Compute dt from ticks (seconds)
331 float dt = 0f;
332
333 // no previous tick assume first frame, so skip dt-based updates
334 if (_pulseLastTick > 0)
335 {
336 dt = HighResTimer.GetDuration(_pulseLastTick, tick);
337 _timeSec += dt;
338 }
339
340 _pulseLastTick = tick;
341
342 bool changed = false;
343
344 if (_pulseFillEnabled)
345 {
346 float t = PulseT(_timeSec, _pulseFillPeriodSec, _pulseFillWave);
347 var c = LerpColor(_pulseFillFrom, _pulseFillTo, t);
348 if (_fillPaint.Color != c) { _fillPaint.Color = c; changed = true; }
349 }
350
351 if (_pulseBorderEnabled)
352 {
353 float t = PulseT(_timeSec, _pulseBorderPeriodSec, _pulseBorderWave);
354 var c = LerpColor(_pulseBorderFrom, _pulseBorderTo, t);
355
356 if (_strokePaint.Color != c)
357 {
358 _strokePaint.Color = c; // paint-only override
359 ForceRefresh();
360 }
361 }
362
363 if (changed)
364 {
365 _needsRebuildPaints = false; // we already set paints directly
366 ForceRefresh(); // request redraw
367 }
368
369 base.Update(tick);
370 }

372 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
373 {
374 var canvas = backbuffer.Canvas;
375
376 if (_needsRebuildPaints)
377 RebuildPaints();
378
379 var fillRect = destRectScreen.ToSKRect();
380 var strokeRect = fillRect;
381
382 bool willDrawStroke = !_isFilled || _borderColor.HasValue || _strokePaint.StrokeWidth > 0.01f;
383 float half = _strokePaint.StrokeWidth * 0.5f;
384
385 // 1) APPLY STROKE ALIGNMENT (use HALF the width; path is centered)
386 if (willDrawStroke && _strokeAlign != StrokeAlign.Center)
387 {
388 if (_strokeAlign == StrokeAlign.Inside)
389 strokeRect.Inflate(-half, -half);
390 else if (_strokeAlign == StrokeAlign.Outside)
391 strokeRect.Inflate(half, half);
392 }
393
394 // 1.5) corner radius for the stroke path, adjusted to keep the inner/outer arcs aligned to the fill
395 float strokeCornerRadius;
396 switch (_strokeAlign)
397 {
398 case StrokeAlign.Outside:
399 strokeCornerRadius = MathF.Max(0f, _cornerRadius - half);
400 break;
401 case StrokeAlign.Inside:
402 strokeCornerRadius = _cornerRadius + half;
403 break;
404 case StrokeAlign.Center:
405 default:
406 strokeCornerRadius = _cornerRadius;
407 break;
408 }
409
410 // 2) Dash for stroke only
411 _strokePaint.PathEffect = _dashPattern is { Length: > 0 }
412 ? SKPathEffect.CreateDash(_dashPattern, 0)
413 : null;
414
415 // 3) Draw fill (unmodified rect)
416 if (_isFilled)
417 {
418 if (_cornerRadius > 0)
419 canvas.DrawRoundRect(fillRect, strokeCornerRadius, strokeCornerRadius, _fillPaint);
420 else
421 canvas.DrawRect(fillRect, _fillPaint);
422 }
423
424 // 4) Draw stroke on its aligned rect (pure/opaque so it stays white)
425 if (willDrawStroke)
426 {
427 var prevBlend = _strokePaint.BlendMode;
428 var prevColor = _strokePaint.Color;
429 var prevAA = _strokePaint.IsAntialias;
430
431 _strokePaint.IsAntialias = false;
432 _strokePaint.BlendMode = SKBlendMode.Src;
433
434 // If pulsing, keep color from Update(),
435 // else use base border color (or previous).
436 var strokeColor = _pulseBorderEnabled ? _strokePaint.Color
437 : (_borderColor ?? prevColor);
438
439 _strokePaint.Color = strokeColor.WithAlpha(255);
440
441 if (_cornerRadius > 0)
442 canvas.DrawRoundRect(strokeRect, _cornerRadius, _cornerRadius, _strokePaint);
443 else
444 canvas.DrawRect(strokeRect, _strokePaint);
445
446 _strokePaint.IsAntialias = prevAA;
447 _strokePaint.BlendMode = prevBlend;
448 _strokePaint.Color = prevColor;
449 _strokePaint.StrokeJoin = SKStrokeJoin.Round;
450 _strokePaint.StrokeCap = SKStrokeCap.Round;
451 }
452 }

457 private void RebuildPaints()
458 {
459 // Ensure styles/AA set
460 _fillPaint.IsAntialias = true;
461 _fillPaint.Style = SKPaintStyle.Fill;
462
463 _strokePaint.IsAntialias = true;
464 _strokePaint.Style = SKPaintStyle.Stroke;
465
466 // Only rebuild stroke color if not pulsing border
467 if (!_pulseBorderEnabled)
468 {
469 // If border color set, use it for stroke; else match fill/base color
470 if (_borderColor.HasValue)
471 {
472 var sc = _strokePaint.Color; // preserve alpha if you want; otherwise:
473 _strokePaint.Color = _borderColor.Value;
474 }
475 else
476 {
477 _strokePaint.Color = _fillPaint.Color; // same as fill if no explicit border
478 }
479
480 _needsRebuildPaints = false;
481 }

491 [MethodImpl(MethodImplOptions.AggressiveInlining)]
492 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
493 {
494 float phase = (timeSec / periodSec) % 1f;
495 if (wave == PulseWave.Sine)
496 {
497 // (sin(2πx)+1)/2 in [0..1]
498 return 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f));
499 }
500 else // Triangle: ramp up 0..1 then down 1..0
501 {
502 return phase < 0.5f ? (phase * 2f) : (1f - ((phase - 0.5f) * 2f));
503 }
504 }
505
506 [MethodImpl(MethodImplOptions.AggressiveInlining)]
507 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
508 {
509 t01 = Math.Clamp(t01, 0f, 1f);
510
511 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
512 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
513 byte bch = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
514 byte aA = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
515
516 return new SKColor(r, g, bch, aA);
517 }
