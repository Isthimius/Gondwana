141 internal Frame AdvanceFrame()
142 {
143 switch (SequenceCycleType)
144 {
145 case CycleType.PingPong:
146 currentFrameIdx += curFrameIncrement;
147 if ((currentFrameIdx <= 0) || (currentFrameIdx >= frameList.Count - 1))
148 curFrameIncrement *= -1;
149
150 if (currentFrameIdx < 0)
151 currentFrameIdx = 0;
152
153 if (currentFrameIdx > frameList.Count - 1)
154 currentFrameIdx = frameList.Count - 1;
155
156 cycleFinished = false;
157 break;
158
159 case CycleType.Repeating:
160 if (++currentFrameIdx >= frameList.Count)
161 currentFrameIdx = 0;
162
163 cycleFinished = false;
164 break;
165
166 case CycleType.Simple:
167 if (++currentFrameIdx > frameList.Count - 1)
168 {
169 currentFrameIdx = frameList.Count - 1;
170 cycleFinished = true;
171 }
172 else
173 cycleFinished = false;
174 break;
175
176 default:
177 throw new InvalidOperationException("Invalid CycleType: " + SequenceCycleType.ToString());
178 }
179
180 return frameList[currentFrameIdx];
181 }
