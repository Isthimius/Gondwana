Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

64 private void OnTouchBegan(object? sender, TouchEventArgs e)
65 {
66 _activeContacts.Add(e.Touch.Id);
67
68 if (_activeContacts.Count > 1)
69 {
70 _isMultiTouchSequence = true;
71 _activeSwipes.Clear();
72 return;
73 }
74
75 if (!_isMultiTouchSequence)
76 _activeSwipes[e.Touch.Id] = new SwipeState(e.Touch.Position, e.Tick);
77 }
78
79 private void OnTouchEnded(object? sender, TouchEventArgs e)
80 {
81 _activeContacts.Remove(e.Touch.Id);
82
83 if (_isMultiTouchSequence)
84 {
85 _activeSwipes.Remove(e.Touch.Id);
86 if (_activeContacts.Count == 0)
87 _isMultiTouchSequence = false;
88 return;
89 }
90
91 if (!_activeSwipes.TryGetValue(e.Touch.Id, out var state))
92 return;
93
94 _activeSwipes.Remove(e.Touch.Id);
95
96 // A system-cancelled contact must never produce a swipe.
97 if (e.Touch.Phase == TouchPhase.Cancelled)
98 return;
99
100 var endPos = e.Touch.Position;
101 var dx = endPos.X - state.StartPosition.X;
102 var dy = endPos.Y - state.StartPosition.Y;
103
104 var distance = Math.Sqrt(dx * dx + dy * dy);
105 var elapsed = HighResTimer.GetDuration(state.StartTick, e.Tick);
106
107 if (elapsed <= 0)
108 return;
109
110 var speed = distance / elapsed;
111
112 if (!WouldRecognize(distance, elapsed))
113 return;
114
115 var direction = Math.Abs(dx) >= Math.Abs(dy)
116 ? (dx >= 0 ? SwipeDirection.Right : SwipeDirection.Left)
117 : (dy >= 0 ? SwipeDirection.Down : SwipeDirection.Up);
118
119 Swiped?.Invoke(this, new SwipedEventArgs(direction, state.StartPosition, endPos, speed));
120 }
