Gondwana Game Engine v2.3.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

59 private void OnTouchEnded(object? sender, TouchEventArgs e)
60 {
61 if (!_activeSwipes.TryGetValue(e.Touch.Id, out var state))
62 return;
63
64 _activeSwipes.Remove(e.Touch.Id);
65
66 // A system-cancelled contact must never produce a swipe.
67 if (e.Touch.Phase == TouchPhase.Cancelled)
68 return;
69
70 var endPos = e.Touch.Position;
71 var dx = endPos.X - state.StartPosition.X;
72 var dy = endPos.Y - state.StartPosition.Y;
73
74 var distance = Math.Sqrt(dx * dx + dy * dy);
75 var elapsed = (DateTime.UtcNow - state.StartTime).TotalSeconds;
76
77 if (elapsed <= 0)
78 return;
79
80 var speed = distance / elapsed;
81
82 if (speed < MinimumSwipeSpeedPixelsPerSecond)
83 return;
84
85 var direction = Math.Abs(dx) >= Math.Abs(dy)
86 ? (dx >= 0 ? SwipeDirection.Right : SwipeDirection.Left)
87 : (dy >= 0 ? SwipeDirection.Down : SwipeDirection.Up);
88
89 Swiped?.Invoke(this, new SwipedEventArgs(direction, state.StartPosition, endPos, speed));
90 }
