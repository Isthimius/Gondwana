Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

154 protected sealed override void ProcessPointerMove(WidgetPointerEventArgs args)
155 {
156 base.ProcessPointerMove(args);
157
158 if (!_isPointerDown || args.PointerId != _dragPointerId)
159 return;
160
161 Vector2 totalScreenDeltaPx = GetTotalScreenDelta(args.ScreenPositionPx);
162
163 if (!IsDragging)
164 {
165 float thresholdSquared = DragThresholdPx * DragThresholdPx;
166
167 if (totalScreenDeltaPx.LengthSquared() < thresholdSquared)
168 return;
169
170 IsDragging = true;
171 _suppressNextPointerClick = true;
172
173 DispatchDragStarted(CreateDragEventArgs(args));
174
175 if (!IsDragging)
176 return;
177 }
178
179 ApplyDragPosition(
180 totalScreenDeltaPx,
181 _dragView ?? args.View);
182
183 DispatchDragged(CreateDragEventArgs(args));
184 }
185
187 protected sealed override void ProcessPointerUp(WidgetPointerEventArgs args)
188 {
189 base.ProcessPointerUp(args);
190
191 if (!_isPointerDown || !IsMatchingRelease(args))
192 return;
193
194 WidgetDragEventArgs? dragArgs = null;
195
196 if (IsDragging)
197 {
198 Vector2 totalScreenDeltaPx = GetTotalScreenDelta(args.ScreenPositionPx);
199 ApplyDragPosition(
200 totalScreenDeltaPx,
201 _dragView ?? args.View);
202
203 dragArgs = CreateDragEventArgs(args);
204 }
205
206 bool completedDrag = IsDragging;
207 ResetPointerState();
208
209 if (completedDrag && dragArgs is not null)
210 {
211 _suppressNextPointerClick = true;
212 DispatchDragEnded(dragArgs);
213 }
214 }
