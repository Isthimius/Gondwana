Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

40 internal void Register(Tilesheet sheet, bool disposeReplaced = true)
41 {
42 if (sheet is null)
43 {
44 Engine.Logger.LogError("TilesheetRegistry: Attempt to register a null Tilesheet.");
45 return;
46 }
47
48 Tilesheet? replaced = null;
49
50 lock (_gate)
51 {
52 if (_sheets.TryGetValue(sheet.Name, out var existing) && !ReferenceEquals(existing, sheet))
53 {
54 _sheets[sheet.Name] = sheet;
55 replaced = existing;
56 }
57 else
58 {
59 _sheets[sheet.Name] = sheet;
60 }
61
62 // Avoid duplicate subscriptions if the same instance is registered again.
63 sheet.Disposed -= OnTilesheetDisposed;
64 sheet.Disposed += OnTilesheetDisposed;
65 }
66
67 if (replaced is not null)
68 {
69 replaced.Disposed -= OnTilesheetDisposed;
70
71 if (disposeReplaced)
72 replaced.Dispose();
73 }
74 }

309 internal void OnTilesheetRenamed(string oldName, string newName, Tilesheet sheet, bool disposeReplaced = true)
310 {
311 if (oldName is null)
312 throw new ArgumentNullException(nameof(oldName));
313
314 if (newName is null)
315 throw new ArgumentNullException(nameof(newName));
316
317 if (sheet is null)
318 throw new ArgumentNullException(nameof(sheet));
319
320 Tilesheet? replaced = null;
321
322 lock (_gate)
323 {
324 // If old mapping points to this sheet, remove it.
325 if (_sheets.TryGetValue(oldName, out var existingAtOld) && ReferenceEquals(existingAtOld, sheet))
326 _sheets.Remove(oldName);
327
328 // If something else is already at newName, warn and replace.
329 if (_sheets.TryGetValue(newName, out var existingAtNew) && !ReferenceEquals(existingAtNew, sheet))
330 {
331 Engine.Logger.LogWarning(
332 "TilesheetRegistry: Renaming '{Old}'→'{New}' replaced a different registered Tilesheet instance.",
333 oldName, newName);
334 replaced = existingAtNew;
335 }
336
337 _sheets[newName] = sheet;
338 }
339
340 if (replaced is not null)
341 {
342 replaced.Disposed -= OnTilesheetDisposed;
343
344 if (disposeReplaced)
345 replaced.Dispose();
346 }
347 }
