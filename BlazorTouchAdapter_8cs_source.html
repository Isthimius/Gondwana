Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

57 private void OnTouchStart(BrowserTouchEventArgs e)

74 var point = new GondwanaTouchPoint((int)t.Identifier, GetPosition(t), TouchPhase.Moved);
75 _activeTouches[t.Identifier] = point;
76 }

80 private void OnTouchEnd(BrowserTouchEventArgs e)
81 {
82 if (_isDisposed) return;
83 foreach (var t in e.ChangedTouches)
84 {
85 if (!_activeTouches.ContainsKey(t.Identifier)) continue;
86 var point = new GondwanaTouchPoint((int)t.Identifier, GetPosition(t), TouchPhase.Ended);
87 _activeTouches.Remove(t.Identifier);
88 _pendingEnds.Enqueue(point);

93 private void OnTouchCancel(BrowserTouchEventArgs e)
94 {
95 if (_isDisposed) return;
96 foreach (var t in e.ChangedTouches)
97 {
98 if (!_activeTouches.TryGetValue(t.Identifier, out var existing)) continue;
99 var point = new GondwanaTouchPoint(existing.Id, existing.Position, TouchPhase.Cancelled);
100 _activeTouches.Remove(t.Identifier);
101 _pendingEnds.Enqueue(point);
102 }
103 RebuildSnapshot();
104 }

107 public IReadOnlyList<GondwanaTouchPoint> ConsumeEndedTouches()
108 {
109 if (_pendingEnds.IsEmpty)
110 return Array.Empty<GondwanaTouchPoint>();
111
112 var snapshot = new List<GondwanaTouchPoint>(_pendingEnds.Count);
113 while (_pendingEnds.TryDequeue(out var point))
114 snapshot.Add(point);
115 return snapshot;
116 }
