Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

392 private static SKRect ComputeDestRect(RectangleF bounds, int srcW, int srcH, StretchMode mode)
393 {
394 var b = bounds;
395 var dst = new SKRect(b.Left, b.Top, b.Right, b.Bottom);
396 if (mode == StretchMode.Fill || srcW <= 0 || srcH <= 0) return dst;
397
398 float bw = b.Width, bh = b.Height;
399 float arSrc = (float)srcW / srcH;
400 float arDst = bw / bh;
401
402 if (mode == StretchMode.None)
403 return new SKRect(b.Left, b.Top, b.Left + srcW, b.Top + srcH);
404
405 if (mode == StretchMode.Uniform)
406 {
407 if (arSrc > arDst)
408 {
409 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
410 return new SKRect(b.Left, y, b.Left + w, y + h);
411 }
412 else
413 {
414 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
415 return new SKRect(x, b.Top, x + w, b.Top + h);
416 }
417 }
418
419 // UniformToFill
420 if (arSrc < arDst)
421 {
422 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
423 return new SKRect(b.Left, y, b.Left + w, y + h);
424 }
425 else
426 {
427 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
428 return new SKRect(x, b.Top, x + w, b.Top + h);
429 }
430 }

439 private void OnFrameReady(object? sender, VideoFrameReadyEventArgs e)
440 {
441 // Copy the RGBA buffer into a reusable SKBitmap.
442 // NOTE: do not hold the lock while touching Skia; just protect shared refs/copies.
443 SKBitmap? local;
444 lock (_frameLock)
445 {
446 if (_frame is null || _srcW != e.Width || _srcH != e.Height)
447 {
448 _frame?.Dispose();
449 _frame = new SKBitmap(new SKImageInfo(e.Width, e.Height, SKColorType.Rgba8888));
450 _srcW = e.Width;
451 _srcH = e.Height;
452 _srcStride = e.Stride;
453 }
454 local = _frame;
455 }
456
457 if (local is null) return;
458
459 CopyFrameToBitmap(e.Pixels, e.Stride, local, _srcW, _srcH);
460
461 // Mark dirty so the direct-draw manager repaints our area.
462 ForceRefresh();
463 }

474 private static void CopyFrameToBitmap(IntPtr srcPixels, int srcStride, SKBitmap dst, int width, int height)
475 {
476#if !BROWSER && !NO_UNSAFE
477 // Fast path: single memcpy using pointers
478 unsafe
479 {
480 using var pix = dst.PeekPixels();
481 var dstPtr = (byte*)pix.GetPixels().ToPointer();
482 Buffer.MemoryCopy(
483 source: (void*)srcPixels,
484 destination: dstPtr,
485 destinationSizeInBytes: pix.RowBytes * height,
486 sourceBytesToCopy: (long)srcStride * height);
487 }
488#else
489 // Safe path: row-by-row Marshal.Copy (no unsafe)
490 using var pix = dst.PeekPixels();
491 IntPtr dstBase = pix.GetPixels();
492 int dstRB = (int)pix.RowBytes;
493 int srcRB = srcStride;
494 int rowBytes = Math.Min(dstRB, srcRB);
495
496 var rowBuf = System.Buffers.ArrayPool<byte>.Shared.Rent(rowBytes);
497 try
498 {
499 for (int y = 0; y < height; y++)
500 {
501 var srcRow = IntPtr.Add(srcPixels, y * srcRB);
502 var dstRow = IntPtr.Add(dstBase, y * dstRB);
503 Marshal.Copy(srcRow, rowBuf, 0, rowBytes);
504 Marshal.Copy(rowBuf, 0, dstRow, rowBytes);
505 }
506 }
507 finally
508 {
509 System.Buffers.ArrayPool<byte>.Shared.Return(rowBuf, clearArray: false);
510 }
511#endif
512 }
