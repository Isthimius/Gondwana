Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

151 internal void UpdateZoom(float dtSeconds)
152 {
153 if (_zoomTarget is null)
154 return;
155
156 float target = _zoomTarget.Value;
157 float dt = Math.Max(0f, dtSeconds);
158
159 if (_zoomDurationSeconds is { } durationSeconds)
160 {
161 _zoomElapsedSeconds = Math.Min(_zoomElapsedSeconds + dt, durationSeconds);
162
163 float progress = durationSeconds > 0f ? _zoomElapsedSeconds / durationSeconds : 1f;
164
165 // Preserve the existing exponential ease-out character, but normalize
166 // it so progress reaches exactly 1 at the requested duration.
167 const float decay = 4.605170186f; // -ln(0.01)
168
169 float normalization = 1f - MathF.Exp(-decay);
170 float easedProgress = (1f - MathF.Exp(-decay * progress)) / normalization;
171
172 Zoom = _zoomStart + (target - _zoomStart) * easedProgress;
173
174 if (_zoomElapsedSeconds >= durationSeconds)
175 {
176 // This no longer causes a visible jump: easedProgress is already 1.
177 SnapZoom(target);
178 }
179
180 return;
181 }
182
183 if (_zoomLerpPerSecond <= 0f)
184 return;
185
186 float t = 1f - MathF.Exp(-_zoomLerpPerSecond * dt);
187
188 float newZoom = _zoom + (target - _zoom) * t;
189
190 Zoom = newZoom;
191
192 if (Math.Abs(newZoom - target) < 1e-4f)
193 SnapZoom(target);
194 }
