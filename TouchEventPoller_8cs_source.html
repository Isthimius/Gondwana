Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

178 internal void PollForEvents(long tick)
179 {
180 if (_isDisposed || Adapter is null) return;
181
182 var began = Adapter.ConsumeBeganTouches();
183 var ended = Adapter.ConsumeEndedTouches();
184
185 // A paused/stopped poller deliberately establishes a new logical contact boundary.
186 // Drain transient queues, forget prior contacts, and begin fresh on resume.
187 if (Configuration is null || Configuration.IsPaused)
188 {
189 ClearContactState();
190 return;
191 }
192
193 // Lifecycle transitions are lossless and are never throttled.
194 foreach (var beganPoint in began)
195 {
196 if (_activeTouches.ContainsKey(beganPoint.Id))
197 continue;
198
199 var normalized = new TouchPoint(
200 beganPoint.Id,
201 beganPoint.Position,
202 TouchPhase.Began);
203
204 _activeTouches[normalized.Id] = normalized;
205 TouchBegan?.Invoke(this, new TouchEventArgs(normalized, tick));
206 }
207
208 foreach (var endedPoint in ended)
209 {
210 // A beginning and ending may both have occurred between engine polls. The queued
211 // beginning above ensures consumers still receive a complete lifecycle.
212 if (_activeTouches.Remove(endedPoint.Id))
213 TouchEnded?.Invoke(this, new TouchEventArgs(endedPoint, tick));
214 }
215
216 var currentActive = Adapter.ActiveTouches;
217
218 // Discover active contacts for adapters that use the default empty beginning queue.
219 foreach (var point in currentActive)
220 {
221 if (!_activeTouches.ContainsKey(point.Id))
222 {
223 var normalized = new TouchPoint(point.Id, point.Position, TouchPhase.Began);
224 _activeTouches[normalized.Id] = normalized;
225 TouchBegan?.Invoke(this, new TouchEventArgs(normalized, tick));
226 }
227 }
228
229 bool movementEmitted = false;
230 if (Configuration.ReadyForNextEvent(tick))
231 {
232 foreach (var point in currentActive)
233 {
234 if (!_activeTouches.TryGetValue(point.Id, out var known) ||
235 known.Position == point.Position)
236 continue;
237
238 var movedPoint = new TouchPoint(point.Id, point.Position, TouchPhase.Moved);
239 _activeTouches[point.Id] = movedPoint;
240 TouchMoved?.Invoke(this, new TouchEventArgs(movedPoint, tick));
241 movementEmitted = true;
242 }
243 }
244
245 _activeTouchesSnapshot = _activeTouches.Values.ToArray();
246
247 if (movementEmitted)
248 Configuration._lastEventTick = tick;
249 }
