Gondwana Game Engine v2.4.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

159 internal void PollForEvents(long tick)
160 {
161 if (Adapter is null) return;
162
163 bool eventEmitted = false;
164
165 // Step 1: Always drain ended/cancelled contacts — prompt delivery is required for
166 // gesture recognizers (e.g. tap duration) regardless of throttle state.
167 var ended = Adapter.ConsumeEndedTouches();
168 foreach (var endedPoint in ended)
169 {
170 if (_activeTouches.Remove(endedPoint.Id))
171 {
172 TouchEnded?.Invoke(this, new TouchEventArgs(endedPoint));
173 eventEmitted = true;
174 }
175 }
176
177 // Step 2: Only emit Began/Moved events when not paused and throttle interval has elapsed.
178 if (Configuration is not null && !Configuration.IsPaused && Configuration.ReadyForNextEvent(tick))
179 {
180 var currentActive = Adapter.ActiveTouches;
181 foreach (var point in currentActive)
182 {
183 if (!_activeTouches.TryGetValue(point.Id, out var known))
184 {
185 // New contact
186 _activeTouches[point.Id] = point;
187 TouchBegan?.Invoke(this, new TouchEventArgs(point));
188 eventEmitted = true;
189 }
190 else if (known.Position != point.Position)
191 {
192 // Existing contact moved. The poller owns phase semantics for transition events,
193 // so TouchPhase.Moved is always correct here regardless of the adapter's stored phase.
194 var movedPoint = new TouchPoint(point.Id, point.Position, TouchPhase.Moved);
195 _activeTouches[point.Id] = movedPoint;
196 TouchMoved?.Invoke(this, new TouchEventArgs(movedPoint));
197 eventEmitted = true;
198 }
199 }
200 }
201
202 // Step 3: Always update snapshot so ActiveTouches reflects current reality.
203 _activeTouchesSnapshot = _activeTouches.Values.ToArray();
204
205 // Step 4: Advance the throttle timestamp only when an event was actually emitted,
206 // so an idle poller never resets the throttle window and defers real touches.
207 if (eventEmitted && Configuration is not null)
208 Configuration._lastEventTick = tick;
209 }
