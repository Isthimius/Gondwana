Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

802 private void Cycle()
803 {
804 EngineDispatcher.Drain();
805
806 long tick = HighResTimer.GetCurrentTick();
807 var deltaMs = HighResTimer.GetDuration(_lastCycleTick, tick);
808 _lastCycleTick = tick;
809
810 EnginePluginRegistry.InvokePreCycle(this, deltaMs);
811
812 DoBackgroundTasks(tick);
813
814 // if TargetFPS <= 0, render to screen unbounded
815 // otherwise, check if throttle time has passed since last tick...
816 if ((Configuration.TargetFPS <= 0)
817 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS)
818 {
819 EnginePluginRegistry.InvokePreFrameRender(this, deltaMs);
820
821 DoForegroundTasks(tick);
822
823 EnginePluginRegistry.InvokePostFrameRender(this, deltaMs);
824
825 // save time of this last tick; increment CPS counter
826 _lastForegroundTick = tick;
827 _netCyclesThisMeasure++;
828 }
829
830 // increment CPS counter
831 _grossCyclesThisMeasure++;
832
833 // if 0 or negative, sampling is turned off
834 if (Configuration.SamplingTimeForCPS > 0)
835 CalculateCPS(tick);
836
837 EnginePluginRegistry.InvokePostCycle(this, deltaMs);
838 }

840 private void DoBackgroundTasks(long tick)
841 {
842 // find total real seconds passed since last background loop
843 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
844
845 BeforeBackgroundTasksExecute?.Invoke();
846
847 // raise pre-cycle timer events
848 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
849
850 // check for keyboard events
851 KeyboardEventPoller.Instance?.PollForEvents(tick);
852
853 // check for mouse events
854 MouseEventPoller.Instance?.PollForEvents(tick);
855
856 // check for touch events
857 TouchEventPoller.Instance?.PollForEvents(tick);
858
859 // check for gamepad events
860 GamepadEventPoller.Instance?.PollForEvents(tick);
861
862 // cycle Animator frames
863 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
864 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
865
866 // advance Sprite Movement paths
867 SpriteManager.Instance.MoveSprites(tick);
868
869 // resolve collisions after movement
870 foreach (var scene in Scenes.Scene.GetAllScenes())
871 {
872 foreach (var layer in scene.SceneLayers)
873 layer.CollisionResolver.Resolve();
874 }
875
876 // update cameras so any movement can mark RefreshNeeded = All.
877 foreach (var surface in RenderSurfaceHostRegistry.All)
878 surface.ViewManager.UpdateCameras(deltaSeconds);
879
880 AfterBackgroundTasksExecute?.Invoke();
881
882 _lastBackgroundTick = tick;
883 }
884
885 private void DoForegroundTasks(long tick)
886 {
887 // raise event
888 BeforeFrameRender?.Invoke();
889
890 // update the DirectDrawing instances' states
891 DirectDrawingManager.Instance.UpdateAll(tick);
892
893 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
894 foreach (var surface in RenderSurfaceHostRegistry.All)
895 if (!surface.Backbuffer.IsGlThreadRendered)
896 surface.RenderToBackbuffer(tick);
897
898 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
899 foreach (var surface in RenderSurfaceHostRegistry.All)
900 if (!surface.Backbuffer.IsGlThreadRendered)
901 surface.PresentBackbufferToAdapter();
902
903 // update state of gamepad(s)
904 Input.GamepadManager?.Update();
905
906 // raise event
907 // GL thread rendering is done as part of this invocation
908 AfterFrameRender?.Invoke();
909
910 // raise post-cycle timer events
911 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
912 }

914 private void CalculateCPS(long tick)
915 {
916 // Has the sampling interval elapsed?
917 long elapsedTicks = tick - _lastCPSSamplingTick;
918 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
919 return;
920
921 // SNAPSHOT the counters BEFORE resetting or posting
922 long grossCycles = _grossCyclesThisMeasure;
923 long netCycles = _netCyclesThisMeasure;
924
925 // Compute using the snapshot
926 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
927 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
928 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
929
930 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
931 long totalGpuFrames = 0;
932 int gpuSurfaceCount = 0;
933 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
934 {
935 if (surface.Backbuffer is GpuBackbuffer gpuBb)
936 {
937 totalGpuFrames += gpuBb.ConsumeFrameCount();
938 gpuSurfaceCount++;
939 }
940 }
941 double? gpuFps = gpuSurfaceCount > 0
942 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
943 : null;
944
945 // Build immutable args NOW (so lambda doesn't read changing fields later)
946 var args = new CyclesPerSecondCalculatedEventArgs(
947 grossCycles,
948 netCycles,
949 grossCps,
950 netCps,
951 elapsedSec,
952 gpuFps
953 );
954
955 // Post the snapshot
956 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
957
958 _grossCPS = grossCps;
959 _netFPS = netCps;
960
961 // Reset for next window
962 _lastCPSSamplingTick = tick;
963 _grossCyclesThisMeasure = 0;
964 _netCyclesThisMeasure = 0;
965 }

987 private void Dispose(bool disposing)
988 {
989 if (!IsDisposed)
990 {
991 if (disposing)
992 {
993 IsDisposing = true;
994
995 // stop the loop first so handlers don't race the cycle thread
996 try
997 {
998 Stop();
999 }
1000 catch (Exception ex)
1001 {
1002 Logger.LogError(ex, "Unhandled exception calling Stop()");
1003 }
1004
1005 // wait for the background loop to actually exit
1006 try
1007 {
1008 _cycleTask?.Wait();
1009 }
1010 catch (Exception ex)
1011 {
1012 Logger.LogError(ex, "Error waiting for engine loop to exit.");
1013 }
1014
1015 // raise Disposing on UI thread if possible; otherwise inline
1016 if (UiDispatcher is not null)
1017 UiDispatcher.Post(() => SafeInvoke(Disposing));
1018 else
1019 SafeInvoke(Disposing);
1020
1021 // managed cleanup...
1022 Input.KeyboardEventPoller?.StopMonitoringAllKeys();
1023 MouseEventPoller.Reset();
1024 TouchEventPoller.Reset();
1025
1026 if (Input.GamepadManager is not null)
1027 foreach (var gamepadAdapter in Input.GamepadManager.ConnectedAdapters)
1028 Input.GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
1029
1030 try
1031 {
1032 _configurationFile?.Dispose();
1033 }
1034 catch (Exception ex)
1035 {
1036 Logger.LogError(ex, "Error saving the engine configuration during disposal.");
1037 }
1038 finally
1039 {
1040 _configurationFile = null;
1041 }
1042
1043 if (EngineLogger.Mode == EngineLoggingMode.Asynchronous)
1044 EngineLogger.StopAsyncLogging(flush: Configuration.FlushAsyncLogsOnShutdown);
1045
1046 Timer.ClearAll();
1047 State.Clear();
1048 }
1049
1050 // unmanaged cleanup...
1051 IsDisposed = true;
1052
1053 if (disposing)
1054 {
1055 // now signal we're fully torn down
1056 if (UiDispatcher is not null)
1057 UiDispatcher.Post(() => SafeInvoke(Disposed));
1058 else
1059 SafeInvoke(Disposed);
1060 }
1061 }
1062 }
