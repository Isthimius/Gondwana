Gondwana Game Engine v2.2.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

668 private void Cycle()
669 {
670 EngineDispatcher.Drain();
671
672 long tick = HighResTimer.GetCurrentTick();
673
674 DoBackgroundTasks(tick);
675
676 // if TargetFPS <= 0, render to screen unbounded
677 // otherwise, check if throttle time has passed since last tick...
678 if ((Configuration.TargetFPS <= 0)
679 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS)
680 {
681 DoForegroundTasks(tick);
682
683 // save time of this last tick; increment CPS counter
684 _lastForegroundTick = tick;
685 _netCyclesThisMeasure++;
686 }
687
688 // increment CPS counter
689 _grossCyclesThisMeasure++;
690
691 // if 0 or negative, sampling is turned off
692 if (Configuration.SamplingTimeForCPS > 0)
693 CalculateCPS(tick);
694 }
695
696 private void DoBackgroundTasks(long tick)
697 {
698 // find total real seconds passed since last background loop
699 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
700
701 BeforeBackgroundTasksExecute?.Invoke();
702
703 // raise pre-cycle timer events
704 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
705
706 // check for keyboard events
707 KeyboardEventPoller.Instance?.PollForEvents(tick);
708
709 // check for mouse events
710 MouseEventPoller.Instance?.PollForEvents(tick);
711
712 // check for gamepad events
713 GamepadEventPoller.Instance?.PollForEvents(tick);
714
715 // cycle Animator frames
716 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
717 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
718
719 // advance Sprite Movement paths
720 SpriteManager.Instance.MoveSprites(tick);
721
722 // resolve collisions after movement
723 foreach (var scene in Scenes.Scene.GetAllScenes())
724 {
725 foreach (var layer in scene.SceneLayers)
726 layer.CollisionResolver.Resolve();
727 }
728
729 // update cameras so any movement can mark RefreshNeeded = All.
730 foreach (var surface in RenderSurfaceHostRegistry.All)
731 surface.ViewManager.UpdateCameras(deltaSeconds);
732
733 AfterBackgroundTasksExecute?.Invoke();
734
735 _lastBackgroundTick = tick;
736 }
737
738 private void DoForegroundTasks(long tick)
739 {
740 // raise event
741 BeforeFrameRender?.Invoke();
742
743 // update the DirectDrawing instances' states
744 DirectDrawingManager.Instance.UpdateAll(tick);
745
746 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
747 foreach (var surface in RenderSurfaceHostRegistry.All)
748 if (!surface.Backbuffer.IsGlThreadRendered)
749 surface.RenderToBackbuffer(tick);
750
751 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
752 foreach (var surface in RenderSurfaceHostRegistry.All)
753 if (!surface.Backbuffer.IsGlThreadRendered)
754 surface.PresentBackbufferToAdapter();
755
756 // update state of gamepad(s)
757 Input.GamepadManager?.Update();
758
759 // raise event
760 AfterFrameRender?.Invoke();
761
762 // raise post-cycle timer events
763 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
764 }

766 private void CalculateCPS(long tick)
767 {
768 // Has the sampling interval elapsed?
769 long elapsedTicks = tick - _lastCPSSamplingTick;
770 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
771 return;
772
773 // SNAPSHOT the counters BEFORE resetting or posting
774 long grossCycles = _grossCyclesThisMeasure;
775 long netCycles = _netCyclesThisMeasure;
776
777 // Compute using the snapshot
778 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
779 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
780 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
781
782 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
783 long totalGpuFrames = 0;
784 int gpuSurfaceCount = 0;
785 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
786 {
787 if (surface.Backbuffer is GpuBackbuffer gpuBb)
788 {
789 totalGpuFrames += gpuBb.ConsumeFrameCount();
790 gpuSurfaceCount++;
791 }
792 }
793 double? gpuFps = gpuSurfaceCount > 0
794 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
795 : null;
796
797 // Build immutable args NOW (so lambda doesn't read changing fields later)
798 var args = new CyclesPerSecondCalculatedEventArgs(
799 grossCycles,
800 netCycles,
801 grossCps,
802 netCps,
803 elapsedSec,
804 gpuFps
805 );
806
807 // Post the snapshot
808 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
809
810 _grossCPS = grossCps;
811 _netFPS = netCps;
812
813 // Reset for next window
814 _lastCPSSamplingTick = tick;
815 _grossCyclesThisMeasure = 0;
816 _netCyclesThisMeasure = 0;
817 }

823 private void Dispose(bool disposing)
824 {
825 if (!IsDisposed)
826 {
827 if (disposing)
828 {
829 IsDisposing = true;
830
831 // stop the loop first so handlers don't race the cycle thread
832 try
833 {
834 Stop();
835 }
836 catch (Exception ex)
837 {
838 Logger.LogError(ex, "Unhandled exception calling Stop()");
839 }
840
841 // wait for the background loop to actually exit
842 try
843 {
844 _cycleTask?.Wait();
845 }
846 catch (Exception ex)
847 {
848 Logger.LogError(ex, "Error waiting for engine loop to exit.");
849 }
850
851 // raise Disposing on UI thread if possible; otherwise inline
852 if (UiDispatcher is not null)
853 UiDispatcher.Post(() => SafeInvoke(Disposing));
854 else
855 SafeInvoke(Disposing);
856
857 // managed cleanup...
858 Input.KeyboardEventPoller?.StopMonitoringAllKeys();
859 Input.MouseEventPoller?.StopMonitoringMouse();
860
861 if (Input.GamepadManager is not null)
862 foreach (var gamepadAdapter in Input.GamepadManager.ConnectedAdapters)
863 Input.GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
864
865 if (Configuration.LoggingMode == EngineLoggingMode.Asynchronous && Configuration.FlushAsyncLogsOnShutdown)
866 EngineLogger.StopAsyncLogging(flush: true);
867
868 Timer.ClearAll();
869 State.Clear();
870 }
871
872 // unmanaged cleanup...
873 IsDisposed = true;
874
875 if (disposing)
876 {
877 // now signal we're fully torn down
878 if (UiDispatcher is not null)
879 UiDispatcher.Post(() => SafeInvoke(Disposed));
880 else
881 SafeInvoke(Disposed);
882 }
883 }
884 }

886 private static void SafeInvoke(Action? evnt)

888 try { evnt?.Invoke(); }
889 catch (Exception ex)
890 {
891 // Keep disposal robust; log and continue
892 Logger.LogError(ex, "Unhandled exception in disposal event handler.");
893 }
894 }
