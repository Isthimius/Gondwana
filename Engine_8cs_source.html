Gondwana Game Engine v2.4.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

791 private void Cycle()
792 {
793 EngineDispatcher.Drain();
794
795 long tick = HighResTimer.GetCurrentTick();
796 var deltaMs = HighResTimer.GetDuration(_lastCycleTick, tick);
797 _lastCycleTick = tick;
798
799 EnginePluginRegistry.InvokePreCycle(this, deltaMs);
800
801 DoBackgroundTasks(tick);
802
803 // if TargetFPS <= 0, render to screen unbounded
804 // otherwise, check if throttle time has passed since last tick...
805 if ((Configuration.TargetFPS <= 0)
806 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS)
807 {
808 EnginePluginRegistry.InvokePreFrameRender(this, deltaMs);
809
810 DoForegroundTasks(tick);
811
812 EnginePluginRegistry.InvokePostFrameRender(this, deltaMs);
813
814 // save time of this last tick; increment CPS counter
815 _lastForegroundTick = tick;
816 _netCyclesThisMeasure++;
817 }
818
819 // increment CPS counter
820 _grossCyclesThisMeasure++;
821
822 // if 0 or negative, sampling is turned off
823 if (Configuration.SamplingTimeForCPS > 0)
824 CalculateCPS(tick);
825
826 EnginePluginRegistry.InvokePostCycle(this, deltaMs);
827 }

829 private void DoBackgroundTasks(long tick)
830 {
831 // find total real seconds passed since last background loop
832 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
833
834 BeforeBackgroundTasksExecute?.Invoke();
835
836 // raise pre-cycle timer events
837 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
838
839 // check for keyboard events
840 KeyboardEventPoller.Instance?.PollForEvents(tick);
841
842 // check for mouse events
843 MouseEventPoller.Instance?.PollForEvents(tick);
844
845 // check for touch events
846 TouchEventPoller.Instance?.PollForEvents(tick);
847
848 // check for gamepad events
849 GamepadEventPoller.Instance?.PollForEvents(tick);
850
851 // cycle Animator frames
852 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
853 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
854
855 // advance Sprite Movement paths
856 SpriteManager.Instance.MoveSprites(tick);
857
858 // resolve collisions after movement
859 foreach (var scene in Scenes.Scene.GetAllScenes())
860 {
861 foreach (var layer in scene.SceneLayers)
862 layer.CollisionResolver.Resolve();
863 }
864
865 // update cameras so any movement can mark RefreshNeeded = All.
866 foreach (var surface in RenderSurfaceHostRegistry.All)
867 surface.ViewManager.UpdateCameras(deltaSeconds);
868
869 AfterBackgroundTasksExecute?.Invoke();
870
871 _lastBackgroundTick = tick;
872 }
873
874 private void DoForegroundTasks(long tick)
875 {
876 // raise event
877 BeforeFrameRender?.Invoke();
878
879 // update the DirectDrawing instances' states
880 DirectDrawingManager.Instance.UpdateAll(tick);
881
882 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
883 foreach (var surface in RenderSurfaceHostRegistry.All)
884 if (!surface.Backbuffer.IsGlThreadRendered)
885 surface.RenderToBackbuffer(tick);
886
887 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
888 foreach (var surface in RenderSurfaceHostRegistry.All)
889 if (!surface.Backbuffer.IsGlThreadRendered)
890 surface.PresentBackbufferToAdapter();
891
892 // update state of gamepad(s)
893 Input.GamepadManager?.Update();
894
895 // raise event
896 AfterFrameRender?.Invoke();
897
898 // raise post-cycle timer events
899 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
900 }

902 private void CalculateCPS(long tick)
903 {
904 // Has the sampling interval elapsed?
905 long elapsedTicks = tick - _lastCPSSamplingTick;
906 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
907 return;
908
909 // SNAPSHOT the counters BEFORE resetting or posting
910 long grossCycles = _grossCyclesThisMeasure;
911 long netCycles = _netCyclesThisMeasure;
912
913 // Compute using the snapshot
914 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
915 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
916 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
917
918 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
919 long totalGpuFrames = 0;
920 int gpuSurfaceCount = 0;
921 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
922 {
923 if (surface.Backbuffer is GpuBackbuffer gpuBb)
924 {
925 totalGpuFrames += gpuBb.ConsumeFrameCount();
926 gpuSurfaceCount++;
927 }
928 }
929 double? gpuFps = gpuSurfaceCount > 0
930 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
931 : null;
932
933 // Build immutable args NOW (so lambda doesn't read changing fields later)
934 var args = new CyclesPerSecondCalculatedEventArgs(
935 grossCycles,
936 netCycles,
937 grossCps,
938 netCps,
939 elapsedSec,
940 gpuFps
941 );
942
943 // Post the snapshot
944 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
945
946 _grossCPS = grossCps;
947 _netFPS = netCps;
948
949 // Reset for next window
950 _lastCPSSamplingTick = tick;
951 _grossCyclesThisMeasure = 0;
952 _netCyclesThisMeasure = 0;
953 }

959 private void Dispose(bool disposing)
960 {
961 if (!IsDisposed)
962 {
963 if (disposing)
964 {
965 IsDisposing = true;
966
967 // stop the loop first so handlers don't race the cycle thread
968 try
969 {
970 Stop();
971 }
972 catch (Exception ex)
973 {
974 Logger.LogError(ex, "Unhandled exception calling Stop()");
975 }
976
977 // wait for the background loop to actually exit
978 try
979 {
980 _cycleTask?.Wait();
981 }
982 catch (Exception ex)
983 {
984 Logger.LogError(ex, "Error waiting for engine loop to exit.");
985 }
986
987 // raise Disposing on UI thread if possible; otherwise inline
988 if (UiDispatcher is not null)
989 UiDispatcher.Post(() => SafeInvoke(Disposing));
990 else
991 SafeInvoke(Disposing);
992
993 // managed cleanup...
994 Input.KeyboardEventPoller?.StopMonitoringAllKeys();
995 Input.MouseEventPoller?.StopMonitoringMouse();
996 Input.TouchEventPoller?.StopMonitoringTouch();
997 (Input.TouchEventPoller?.Adapter as IDisposable)?.Dispose();
998
999 if (Input.GamepadManager is not null)
1000 foreach (var gamepadAdapter in Input.GamepadManager.ConnectedAdapters)
1001 Input.GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
1002
1003 if (Configuration.LoggingMode == EngineLoggingMode.Asynchronous && Configuration.FlushAsyncLogsOnShutdown)
1004 EngineLogger.StopAsyncLogging(flush: true);
1005
1006 Timer.ClearAll();
1007 State.Clear();
1008 }
1009
1010 // unmanaged cleanup...
1011 IsDisposed = true;
1012
1013 if (disposing)
1014 {
1015 // now signal we're fully torn down
1016 if (UiDispatcher is not null)
1017 UiDispatcher.Post(() => SafeInvoke(Disposed));
1018 else
1019 SafeInvoke(Disposed);
1020 }
1021 }
1022 }

1031 }
