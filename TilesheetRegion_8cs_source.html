Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

299 internal void BuildTileCache()
300 {
301 ThrowIfDisposed();
302 ClearTileCache();
303
304 if (Tilesheet == null)
305 return;
306
307 if (Tilesheet.SkBitmap == null || Tilesheet.SkBitmap.IsEmpty)
308 return;
309
310 if (_tileSize.Width <= 0 || _tileSize.Height <= 0)
311 return;
312
313 if (_area.Width <= 0 || _area.Height <= 0)
314 return;
315
316 if (_tilePadding.Left < 0 || _tilePadding.Top < 0 || _tilePadding.Right < 0 || _tilePadding.Bottom < 0)
317 throw new InvalidOperationException("Tilesheet region tile padding cannot be negative.");
318
319 if (_regionMargin.Left < 0 || _regionMargin.Top < 0 || _regionMargin.Right < 0 || _regionMargin.Bottom < 0)
320 throw new InvalidOperationException("Tilesheet region margin cannot be negative.");
321
322 if (TileWidthIncludingPadding <= 0 || TileHeightIncludingPadding <= 0)
323 return;
324
325 int xTiles = (_area.Width - _regionMargin.Left - _regionMargin.Right) / TileWidthIncludingPadding;
326 int yTiles = (_area.Height - _regionMargin.Top - _regionMargin.Bottom) / TileHeightIncludingPadding;
327
328 if (xTiles <= 0 || yTiles <= 0)
329 return;
330
331 _tileCache = new TilesheetRegionSlice?[xTiles, yTiles];
332
333 var regionArea = Area;
334 var bitmapBounds = Tilesheet.SkBitmap.Info.Rect;
335
336 for (int y = 0; y < yTiles; y++)
337 {
338 for (int x = 0; x < xTiles; x++)
339 {
340 var srcRect = GetTileBounds(x, y);
341
342 // Prevent this region from bleeding into another differently-sized row/section.
343 if (!regionArea.Contains(srcRect))
344 continue;
345
346 // Prevent invalid reads outside the source image.
347 if (!bitmapBounds.Contains(srcRect.ToSKRectI()))
348 continue;
349
350 var slice = CreateSlice(srcRect);
351
352 if (slice.HasValue)
353 _tileCache[x, y] = slice.Value;
354 }
355 }
356 }
357
361 internal void ClearTileCache()
362 {
363 if (_tileCache == null)
364 return;
365
366 for (int y = 0; y < _tileCache.GetLength(1); y++)
367 {
368 for (int x = 0; x < _tileCache.GetLength(0); x++)
369 {
370 _tileCache[x, y]?.Bitmap.Dispose();
371 _tileCache[x, y]?.Image.Dispose();
372 _tileCache[x, y] = null;
373 }
374 }
375
376 _tileCache = null;
377 }

383 private Rectangle GetTileBounds(int xTile, int yTile)
384 {
385 int x = _area.X + _regionMargin.Left + (xTile * TileWidthIncludingPadding);
386 int y = _area.Y + _regionMargin.Top + (yTile * TileHeightIncludingPadding);
387
388 return new Rectangle(x + _tilePadding.Left, y + _tilePadding.Top, _tileSize.Width, _tileSize.Height);
389 }

391 private TilesheetRegionSlice? CreateSlice(Rectangle srcRect)
392 {
393 var srcInfo = Tilesheet.SkBitmap.Info;
394
395 var sliceInfo = new SKImageInfo(
396 srcRect.Width,
397 srcRect.Height,
398 srcInfo.ColorType,
399 srcInfo.AlphaType);
400
401 var bmp = new SKBitmap(sliceInfo);
402 bmp.Erase(SKColors.Transparent);
403
404 if (!Tilesheet.SkBitmap.ExtractSubset(bmp, srcRect.ToSKRectI()))
405 {
406 bmp.Dispose();
407 return null;
408 }
409
410 var img = SKImage.FromBitmap(bmp);
411
412 return new TilesheetRegionSlice(bmp, img);
413 }
