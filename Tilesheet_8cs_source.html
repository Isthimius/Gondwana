Gondwana Game Engine v2.5.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

307 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
308 {
309 if (SkBitmap == null || SkBitmap.IsEmpty)
310 throw new ArgumentException("Invalid bitmap.");
311
312 var targetColor = maskColor ?? SKColors.White;
313
314 MaskColor = targetColor;
315 MaskTolerance = tolerance;
316 Premultiplied = true;
317
318 ClearTileCache();
319
320 // ensure the bitmap actually supports alpha
321 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
322 {
323 var info = new SKImageInfo(
324 SkBitmap.Width,
325 SkBitmap.Height,
326 SkBitmap.Info.ColorType,
327 SKAlphaType.Premul
328 );
329
330 var withAlpha = new SKBitmap(info);
331 using (var canvas = new SKCanvas(withAlpha))
332 {
333 canvas.DrawBitmap(SkBitmap, 0, 0);
334 }
335
336 SkBitmap.Dispose();
337 SkBitmap = withAlpha;
338 }
339
340 SkBitmapOriginal?.Dispose();
341 SkBitmapOriginal = SkBitmap.Copy();
342
343 SkBitmap.ApplyAlphaMask(targetColor, tolerance);
344 SkBitmap = SkBitmap.PremultiplyAlpha();
345
346 BuildTileCache();
347 }

562 private void Dispose(bool disposing)
563 {
564 if (_disposed) return;
565 _disposed = true;
566
567 if (disposing)
568 {
569 // clean up tile cache
570 foreach (var region in Regions)
571 {
572 region.Dispose();
573 }
574
575 Regions.Clear();
576
577 // dispose the main bitmaps
578 SkBitmap?.Dispose();
579 SkBitmapOriginal?.Dispose();
580
581 try
582 {
583 Disposed?.Invoke(this);
584 }
585 catch (Exception ex)
586 {
587 Engine.Logger.LogError(ex, "Error during Tilesheet Disposed event handling.");
588 }
589
590 // break delegate references
591 Disposed = null;
592 }
