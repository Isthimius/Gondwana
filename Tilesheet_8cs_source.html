Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

75 {
76 ArgumentNullException.ThrowIfNull(baseSheet);

78 Name = name;
79 SkBitmap = SKBitmap.Decode(file)

82 ImageFilePath = file;
83 ValueBag = new(baseSheet.ValueBag);

88 region.Name,

90 region.TileSize,
91 region.TilePadding,
92 region.RegionMargin,
93 region.Overhang,
94 region.CollisionAdjust);

98 for (int x = 0; x < region.Columns; x++)
99 {

101 if (frameCollisionAdjust != region.CollisionAdjust)
102 {
103 copiedRegion.SetFrameCollisionAdjust(
104 x,
105 y,
106 frameCollisionAdjust);
107 }
108 }

191 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
192 {
193 if (SkBitmap == null || SkBitmap.IsEmpty)
194 throw new ArgumentException("Invalid bitmap.");
195
196 var targetColor = maskColor ?? SKColors.White;
197
198 MaskColor = targetColor;
199 MaskTolerance = tolerance;
200 Premultiplied = true;
201
202 ClearTileCache();
203
204 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)

207 SkBitmap.Width,
208 SkBitmap.Height,
209 SkBitmap.Info.ColorType,
210 SKAlphaType.Premul);
211
212 var withAlpha = new SKBitmap(info);
213 using (var canvas = new SKCanvas(withAlpha))

216 SkBitmap.Dispose();
