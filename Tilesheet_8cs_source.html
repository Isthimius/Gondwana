Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

207 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
208 {
209 if (SkBitmap == null || SkBitmap.IsEmpty)
210 throw new ArgumentException("Invalid bitmap.");
211
212 var targetColor = maskColor ?? SKColors.White;
213
214 MaskColor = targetColor;
215 MaskTolerance = tolerance;
216 Premultiplied = true;
217
218 ClearTileCache();
219
220 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
221 {
222 var info = new SKImageInfo(
223 SkBitmap.Width,
224 SkBitmap.Height,
225 SkBitmap.Info.ColorType,
226 SKAlphaType.Premul);
227
228 var withAlpha = new SKBitmap(info);
229 using (var canvas = new SKCanvas(withAlpha))
230 canvas.DrawBitmap(SkBitmap, 0, 0);
231
232 SkBitmap.Dispose();
233 SkBitmap = withAlpha;
234 }
235
236 SkBitmapOriginal?.Dispose();
237 SkBitmapOriginal = SkBitmap.Copy();
238
239 SkBitmap.ApplyAlphaMask(targetColor, tolerance);
240 SkBitmap = SkBitmap.PremultiplyAlpha();
241
242 BuildTileCache();
243 }
