Gondwana Game Engine v2.1.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

81 {

87
88 AssetIdentifier = new AssetsFileIdentifier(resFile, AssetTypes.Image, entryName);

92 $"Tilesheet asset '{entryName}' could not be loaded from AssetsFile '{resFile.FilePath}'. " +
93 "The asset entry does not exist or returned a null data stream."
94 );
95
96 SkBitmap = SKBitmap.Decode(assetStream)
97 ?? throw new ArgumentException(
98 $"Failed to decode tilesheet bitmap from asset '{entryName}' in AssetsFile '{resFile.FilePath}'. " +

100 );
101
102 _name = entryName;
103
104 // Register AFTER successful decode so the registry never contains a half-constructed tilesheet
105 TilesheetRegistry.Instance.Register(this);

313 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
314 {
315 if (SkBitmap == null || SkBitmap.IsEmpty)
316 throw new ArgumentException("Invalid bitmap.");
317
318 MaskColor = maskColor;
319 MaskTolerance = tolerance;
320 Premultiplied = true;
321
322 var targetColor = maskColor ?? SKColors.White;
323
324 // ensure the bitmap actually supports alpha
325 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
326 {
327 var info = new SKImageInfo(
328 SkBitmap.Width,
329 SkBitmap.Height,
330 SkBitmap.Info.ColorType,
331 SKAlphaType.Premul
332 );
333
334 var withAlpha = new SKBitmap(info);
335 using (var canvas = new SKCanvas(withAlpha))
336 {
337 canvas.DrawBitmap(SkBitmap, 0, 0);
338 }
339
340 SkBitmap.Dispose();
341 SkBitmap = withAlpha;
342 }
343
344 SkBitmapOriginal = SkBitmap.Copy();
345
346 SkiaHelper.ApplyAlphaMask(SkBitmap, targetColor, tolerance);
347 SkBitmap = SkiaHelper.PremultiplyAlpha(SkBitmap);
348
349 BuildTileCache();
350 }

391 private void BuildTileCache()
392 {
393 ClearCache();
394
395 if (TileSize.Width <= 0 || TileSize.Height <= 0)
396 return;
397
398 int xTiles = (SkBitmap.Width - InitialOffsetX + XPixelsBetweenTiles) / (_tileSize.Width + XPixelsBetweenTiles);
399 int yTiles = (SkBitmap.Height - InitialOffsetY + YPixelsBetweenTiles) / (_tileSize.Height + YPixelsBetweenTiles);
400
401 _tileCache = new TilesheetSlice?[xTiles, yTiles];
402
403 for (int y = 0; y < yTiles; y++)
404 {
405 for (int x = 0; x < xTiles; x++)
406 {
407 var srcRect = GetTileBounds(x, y);
408 if (!SkBitmap.Info.Rect.Contains(srcRect.ToSKRectI()))
409 continue;
410
411 var srcInfo = SkBitmap.Info;
412
413 // IMPORTANT: preserve alpha + color type from the masked tilesheet bitmap
414 var sliceInfo = new SKImageInfo(
415 _tileSize.Width,
416 _tileSize.Height,
417 srcInfo.ColorType,
418 srcInfo.AlphaType
419 );
420
421 var bmp = new SKBitmap(sliceInfo);
422
423 // Optional but nice: ensure any untouched pixels are transparent
424 bmp.Erase(SKColors.Transparent);
425
426 if (SkBitmap.ExtractSubset(bmp, srcRect.ToSKRectI()))
427 {
428 var img = SKImage.FromBitmap(bmp);
429 _tileCache[x, y] = new TilesheetSlice(bmp, img);
430 }
431 else
432 {
433 bmp.Dispose();
434 }
435 }
436 }
437 }

589 private void Dispose(bool disposing)
590 {
591 if (_disposed) return;
592 _disposed = true;
593
594 if (disposing)
595 {
596 // unregister from registry
597 TilesheetRegistry.Instance.Remove(_name, this, dispose: false);
598
599 // clean up tile cache
600 if (_tileCache != null)
601 {
602 for (int x = 0; x < _tileCache.GetLength(0); x++)
603 {
604 for (int y = 0; y < _tileCache.GetLength(1); y++)
605 {
606 _tileCache[x, y]?.Bitmap?.Dispose();
607 _tileCache[x, y]?.Image?.Dispose();
608 }
609 }
610 _tileCache = null;
611 }
612
613 // dispose the main bitmaps
614 SkBitmap?.Dispose();
615 SkBitmapOriginal?.Dispose();
616
617 try
618 {
619 Disposed?.Invoke(this, new TilesheetDisposedEventArgs(this));
620 }
621 catch (Exception ex)
622 {
623 Engine.Logger.LogError(ex, "Error during Tilesheet Disposed event handling.");
624 }
625
626 // break delegate references
627 Disposed = null;
628 }
629
630 // no unmanaged resources to free
631 }
