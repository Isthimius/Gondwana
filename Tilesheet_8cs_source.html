Gondwana Game Engine v2.4.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

68 public Tilesheet(AssetsFile resFile, string entryName)
69 {
70 if (resFile is null)
71 throw new ArgumentNullException(nameof(resFile));
72
73 if (string.IsNullOrWhiteSpace(entryName))
74 throw new ArgumentException("Asset entry name must be a non-empty string.", nameof(entryName));
75
76 AssetIdentifier = new AssetsFileIdentifier(resFile, AssetTypes.Image, entryName);
77
78 using var assetStream = AssetIdentifier.Data

81 "The asset entry does not exist or returned a null data stream."
82 );
83
84 SkBitmap = SKBitmap.Decode(assetStream)
85 ?? throw new ArgumentException(
86 $"Failed to decode tilesheet bitmap from asset '{entryName}' in AssetsFile '{resFile.FilePath}'. " +
87 "The asset data is corrupt or not a supported image format."
88 );
89
90 _name = entryName;
91
92 // Register AFTER successful decode so the registry never contains a half-constructed tilesheet
93 TilesheetRegistry.Instance.Register(this);

333 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
334 {
335 if (SkBitmap == null || SkBitmap.IsEmpty)
336 throw new ArgumentException("Invalid bitmap.");
337
338 var targetColor = maskColor ?? SKColors.White;
339
340 MaskColor = targetColor;
341 MaskTolerance = tolerance;
342 Premultiplied = true;
343
344 ClearTileCache();
345
346 // ensure the bitmap actually supports alpha
347 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
348 {
349 var info = new SKImageInfo(
350 SkBitmap.Width,
351 SkBitmap.Height,
352 SkBitmap.Info.ColorType,
353 SKAlphaType.Premul
354 );
355
356 var withAlpha = new SKBitmap(info);
357 using (var canvas = new SKCanvas(withAlpha))
358 {
359 canvas.DrawBitmap(SkBitmap, 0, 0);
360 }
361
362 SkBitmap.Dispose();
363 SkBitmap = withAlpha;
364 }
365
366 SkBitmapOriginal?.Dispose();
367 SkBitmapOriginal = SkBitmap.Copy();
368
369 SkBitmap.ApplyAlphaMask(targetColor, tolerance);
370 SkBitmap = SkBitmap.PremultiplyAlpha();
371
372 BuildTileCache();
373 }

580 private void Dispose(bool disposing)
581 {
582 if (_disposed) return;
583 _disposed = true;
584
585 if (disposing)
586 {
587 // unregister from registry
588 TilesheetRegistry.Instance.Remove(_name, this, dispose: false);
589
590 // clean up tile cache
591 foreach (var region in Regions)
592 {
593 region.Dispose();
594 }
595
596 Regions.Clear();
597
598 // dispose the main bitmaps
599 SkBitmap?.Dispose();
600 SkBitmapOriginal?.Dispose();
601
602 try
603 {
604 Disposed?.Invoke();
605 }
606 catch (Exception ex)
607 {
608 Engine.Logger.LogError(ex, "Error during Tilesheet Disposed event handling.");
609 }
610
611 // break delegate references
612 Disposed = null;
613 }

615 // no unmanaged resources to free
