Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

120 public MenuBarWidget AddMenu(string text,
121 Action<MenuDropDownWidget>? configure = null,
122 int? headerWidth = null)
123 {
124 ArgumentNullException.ThrowIfNull(text);
125
126 int resolvedHeaderWidth = headerWidth ?? EstimateHeaderWidth(text);
127
128 if (resolvedHeaderWidth <= 0)
129 {
130 throw new ArgumentOutOfRangeException(
131 nameof(headerWidth),
132 resolvedHeaderWidth,
133 "Menu header width must be positive.");
134 }
135
136 var headerBounds = new Rectangle(
137 _nextHeaderX,
138 _bounds.Y,
139 resolvedHeaderWidth,
140 _bounds.Height);
141
142 int menuIndex = _menus.Count;
143
144 var header = new MenuHeaderWidget(
145 RenderSurfaceHost,
146 _view,
147 headerBounds,
148 text,
149 _theme,
150 $"{Nickname}.header.{menuIndex}");
151
152 var dropDown = new MenuDropDownWidget(
153 RenderSurfaceHost,
154 _view,
155 new Point(headerBounds.X, _bounds.Bottom),
156 _theme,
157 $"{Nickname}.dropdown.{menuIndex}");
158
159 configure?.Invoke(dropDown);
160 PositionDropDown(headerBounds, dropDown);
161
162 var menu = new MenuBarMenu(header, dropDown);
163 _menus.Add(menu);
164
165 header.Invoked += OnHeaderInvoked;
166 header.Hovered += OnHeaderHovered;
167 dropDown.ItemInvoked += OnDropDownItemInvoked;
168
169 Add(
170 header,
171 new Vector2(
172 headerBounds.X - _bounds.X,
173 headerBounds.Y - _bounds.Y));
174
175 Add(
176 dropDown,
177 new Vector2(
178 dropDown.GetPosition().X - _bounds.X,
179 dropDown.GetPosition().Y - _bounds.Y));
180
181 // Dropdown drawings are visible immediately after construction; menus start closed.
182 dropDown.CloseAnimated(MenuDropDownAnimation.None, 0f, immediate: true);
183
184 _nextHeaderX += resolvedHeaderWidth;
185 ApplyMenuZOrder(_menuZOrder);
186
187 return this;
188 }

191 public void OpenMenuAt(int index)
192 {
193 if (index < 0 || index >= _menus.Count)
194 throw new ArgumentOutOfRangeException(nameof(index));
195
196 if (_openMenuIndex == index)
197 return;
198
199 CloseMenuInternal(immediate: true);
200
201 _openMenuIndex = index;
202 MenuBarMenu menu = _menus[index];
203
204 menu.Header.SetOpen(true);
205
206 foreach (MenuBarMenu candidate in _menus)
207 candidate.Header.SetMenuBarActive(true);
208
209 _dismissLayer.SetDismissEnabled(true);
210
211 // Input routing is registration ordered rather than drawing-Z ordered.
212 _dismissLayer.Activate();
213
214 foreach (MenuBarMenu candidate in _menus)
215 candidate.Header.Activate();
216
217 menu.DropDown.OpenAnimated(
218 DropDownAnimation,
219 Math.Max(0f, DropDownAnimationDurationSec));
220
221 menu.DropDown.Activate();
222 menu.DropDown.SelectFirstEnabled();
223
224 MenuOpened?.Invoke(menu);
225 }

275 protected override void OnKeyboardInput(WidgetKeyboardEventArgs args)
276 {
277 base.OnKeyboardInput(args);
278
279 if (args.Handled ||
280 args.KeyAction != KeyAction.Pressed ||
281 _openMenuIndex < 0)
282 {
283 return;
284 }
285
286 MenuDropDownWidget dropDown = _menus[_openMenuIndex].DropDown;
287
288 switch (args.Key)
289 {
290 case 27: // Escape
291 args.Handled = true;
292 CloseMenu();
293 break;
294
295 case 37: // Left Arrow
296 args.Handled = true;
297 OpenMenuAt((_openMenuIndex - 1 + _menus.Count) % _menus.Count);
298 break;
299
300 case 39: // Right Arrow
301 args.Handled = true;
302 OpenMenuAt((_openMenuIndex + 1) % _menus.Count);
303 break;
304
305 case 38: // Up Arrow
306 args.Handled = true;
307 dropDown.SelectPreviousEnabled();
308 break;
309
310 case 40: // Down Arrow
311 args.Handled = true;
312 dropDown.SelectNextEnabled();
313 break;
314
315 case 13: // Enter
316 args.Handled = true;
317 dropDown.InvokeSelectedItem();
318 break;
319 }
