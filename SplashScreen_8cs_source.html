Gondwana Game Engine v2.4.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

163 // Subscribes to terminal events before starting the fade so no completion/disposal event is
164 // missed, then returns a Task that resolves when the transition finishes or is canceled by
165 // disposal.
166 private static Task FadeAndWaitAsync(DirectDrawingBase drawing, float targetOpacity, float durationSec)
167 {
168 var tcs = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
169
170 void Cleanup()
171 {
172 drawing.FadeToCompleted -= OnComplete;
173 drawing.Disposing -= OnDisposing;
174 }
175
176 void OnComplete(object? sender, DirectDrawingBase _)
177 {
178 Cleanup();
179 tcs.TrySetResult();
180 }
181
182 void OnDisposing(object? sender, IDirectDrawable _)
183 {
184 Cleanup();
185 tcs.TrySetCanceled();
186 }
187
188 drawing.FadeToCompleted += OnComplete;
189 drawing.Disposing += OnDisposing;
190
191 try
192 {
193 drawing.FadeTo(targetOpacity, durationSec);
194 }
195 catch
196 {
197 Cleanup();
198 throw;
199 }
200
201 return tcs.Task;
202 }
