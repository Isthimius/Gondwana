Gondwana Game Engine v2.5.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

82 public static SplashScreen? TryCreate(Stream imageStream,
83 RenderSurfaceHostBase host,
84 View view,
85 float fadeInSec = 0.45f,
86 float holdSec = 3f,
87 float fadeOutSec = 0.45f)
88 {
89 if (host.ViewManager.Views.Count == 0)
90 return null;
91
92 using var bitmap = SKBitmap.Decode(imageStream);
93 var sourceImage = bitmap == null ? null : SKImage.FromBitmap(bitmap);
94
95 if (sourceImage == null)
96 return null;
97
98 var vp = view.Viewport.TargetRectPx;
99 var screenBounds = new Rectangle(0, 0, vp.Width, vp.Height);
100
101 var image = new DirectImage(sourceImage, host, view, screenBounds, SplashImageNickname)
102 .SetScaleMode(DirectImage.ScaleMode.Fit);
103
104 // DirectImage does not own SKImage lifetime; dispose the decoded image when the drawable is disposed.
105 image.Disposing += (_, _) => sourceImage.Dispose();
106
107 image.ZOrder = int.MaxValue;
108 image.Opacity = 0f;
109
110 var splashScreen = new SplashScreen(image, fadeInSec, holdSec, fadeOutSec);
111 return splashScreen;
112 }
