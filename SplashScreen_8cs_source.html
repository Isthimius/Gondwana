Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

298 private async Task RunHoldPhaseAsync(
299 Action? onHoldingSync,
300 Func<Task>? onHoldingAsync,
301 Action? onSplashCompleted)
302 {
303 Logger.LogDebug("Starting splash screen hold phase");
304
305 try
306 {
307 var holdDurationTask = WaitForHoldDurationAsync();
308
309 var delegateTask = ExecuteHoldDelegatesAsync(onHoldingSync, onHoldingAsync);
310
311 await Task.WhenAll(
312 holdDurationTask,
313 delegateTask);
314 }
315 catch (OperationCanceledException)
316 when (_disposed)
317 {
318 return;
319 }
320 catch (Exception ex)
321 {
322 Logger.LogError(ex, "Unexpected error during SplashScreen hold phase");
323 }
324 finally
325 {
326 _holdTimer?.Dispose();
327 _holdTimer = null;
328
329 _holdDurationCompletionSource = null;
330 }
331
332 if (_disposed)
333 return;
334
335 Engine.Instance.EngineDispatcher.Post(() =>
336 {
337 if (_disposed)
338 return;
339
340 if (CurrentState != State.Holding)
341 return;
342
343 StartFadeOut(onSplashCompleted);
344 });
345 }

347 private Task WaitForHoldDurationAsync()
348 {
349 var completionSource = new TaskCompletionSource(
350 TaskCreationOptions.RunContinuationsAsynchronously);
351
352 _holdDurationCompletionSource =
353 completionSource;
354
355 var timer = Timer.Add(
356 TimerType.PostCycle,
357 TimerCycles.Once,
358 HoldSec);
359
360 _holdTimer = timer;
361
362 void OnHoldTimerElapsed()
363 {
364 timer.Tick -= OnHoldTimerElapsed;
365
366 completionSource.TrySetResult();
367 }
368
369 timer.Tick += OnHoldTimerElapsed;
370
371 return completionSource.Task;
372 }

374 private static async Task ExecuteHoldDelegatesAsync(
375 Action? onHoldingSync,
376 Func<Task>? onHoldingAsync)
377 {
378 if (onHoldingSync is null &&
379 onHoldingAsync is null)
380 {
381 return;
382 }
383
384 var dispatcher =
385 Engine.Instance.EngineDispatcher;
386
387 try
388 {
389 await dispatcher.PostAsync(async () =>
390 {
391 Task asyncTask =
392 Task.CompletedTask;
393
394 Task syncTask =
395 Task.CompletedTask;
396
397 if (onHoldingAsync is not null)
398 {
399 Logger.LogDebug("Calling onHoldingAsync in SplashScreen");
400
401 try
402 {
403 asyncTask = onHoldingAsync() ?? Task.CompletedTask;
404 }
405 catch (Exception ex)
406 {
407 asyncTask = Task.FromException(ex);
408 }
409 }
410
411 if (onHoldingSync is not null)
412 {
413 Logger.LogDebug("Calling onHoldingSync in SplashScreen");
414
415 try
416 {
417 onHoldingSync();
418 }
419 catch (Exception ex)
420 {
421 syncTask = Task.FromException(ex);
422 }
423 }
424
425 await Task.WhenAll(asyncTask, syncTask);
426 });
427 }
428 catch (Exception ex)
429 {
430 Logger.LogError(ex, "Error executing SplashScreen hold delegates");
431 }
