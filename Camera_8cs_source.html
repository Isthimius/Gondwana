Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

233 public void PanToOverDuration(PointF worldTopLeftPx, float durationSeconds)
234 {
235 if (durationSeconds <= 0f)
236 {
237 // Degenerate case — behave like SnapTo().
238 _panTargetUpperLeftPx = null;
239 SnapTo(worldTopLeftPx);
240 return;
241 }
242
243 // In an exponential smoothing system:
244 // pos(t) = target + (pos0 - target) * exp(-k * t)
245 //
246 // To reach ~99% of target position in T seconds:
247 // exp(-k*T) = 0.01 → k = -ln(0.01) / T
248 //
249 // This produces a visually nice, predictable time-to-arrive.
250 float k = -(float)Math.Log(0.01f) / durationSeconds;
251
252 // Cancel following and start a timed pan using your existing machinery.
253 _followWorldPx = null;
254 _hardFollow = false;
255
256 _panLerpPerSecond = k;
257 _panTargetUpperLeftPx = ClampToWorldBounds(worldTopLeftPx);
258 }

477 internal void Update(float dtSeconds)
478 {
479 // 1) Explicit pan-to-UL (PanTo) takes priority over center-follow.
480 if (_panTargetUpperLeftPx is { } panTarget)
481 {
482 var clamped = ClampToWorldBounds(panTarget);
483
484 if (_hardFollow || _panLerpPerSecond <= 0f)
485 {
486 PositionPx = clamped;
487 _panTargetUpperLeftPx = null;
488 }
489 else
490 {
491 float t = 1f - (float)Math.Exp(-_panLerpPerSecond * Math.Max(0f, dtSeconds));
492 var newPos = new PointF(
493 PositionPx.X + (clamped.X - PositionPx.X) * t,
494 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
495
496 PositionPx = newPos;
497
498 // Close enough → snap and finish.
499 if (Math.Abs(newPos.X - clamped.X) < 0.5f &&
500 Math.Abs(newPos.Y - clamped.Y) < 0.5f)
501 {
502 PositionPx = clamped;
503 _panTargetUpperLeftPx = null;
504 }
505 }
506
507 // If we're actively panning, don't also run center-follow this frame.
508 if (_panTargetUpperLeftPx is not null)
509 return;
510 }
511
512 // 2) Center-follow / dead-zone logic.
513 if (_followWorldPx is null)
514 return;
515
516 var desiredUL = DesiredUpperLeftToContainTarget(_followWorldPx());
517 if (_hardFollow || FollowLerpPerSecond <= 0f)
518 {
519 PositionPx = ClampToWorldBounds(desiredUL);
520 }
521 else
522 {
523 float t = 1f - (float)Math.Exp(-FollowLerpPerSecond * Math.Max(0f, dtSeconds));
524 var clamped = ClampToWorldBounds(desiredUL);
525 PositionPx = new PointF(
526 PositionPx.X + (clamped.X - PositionPx.X) * t,
527 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
528 }
529 }

533 private PointF DesiredUpperLeftToContainTarget(PointF targetWorldPx)
534 {
535 var vis = GetVisibleWorldSizePx();
536 if (DeadZonePx == Rectangle.Empty)
537 return new PointF(targetWorldPx.X - vis.Width * 0.5f,
538 targetWorldPx.Y - vis.Height * 0.5f);
539
540 var viewWorld = new RectangleF(PositionPx.X, PositionPx.Y, vis.Width, vis.Height);
541 var dzWorld = new RectangleF(viewWorld.X + DeadZonePx.X,
542 viewWorld.Y + DeadZonePx.Y,
543 DeadZonePx.Width, DeadZonePx.Height);
544
545 if (dzWorld.Contains(targetWorldPx))
546 return PositionPx;
547
548 float newX = PositionPx.X;
549 float newY = PositionPx.Y;
550
551 if (targetWorldPx.X < dzWorld.Left)
552 newX -= dzWorld.Left - targetWorldPx.X;
553
554 if (targetWorldPx.X > dzWorld.Right)
555 newX += targetWorldPx.X - dzWorld.Right;
556
557 if (targetWorldPx.Y < dzWorld.Top)
558 newY -= dzWorld.Top - targetWorldPx.Y;
559
560 if (targetWorldPx.Y > dzWorld.Bottom)
561 newY += targetWorldPx.Y - dzWorld.Bottom;
562
563 return new PointF(newX, newY);
564 }

566 private PointF ClampToWorldBounds(PointF ul)
567 {
568 if (WorldBoundsPx == RectangleF.Empty)
569 return ul;
570
571 var vis = GetVisibleWorldSizePx();
572 float minX = WorldBoundsPx.Left;
573 float minY = WorldBoundsPx.Top;
574 float maxX = WorldBoundsPx.Right - vis.Width;
575 float maxY = WorldBoundsPx.Bottom - vis.Height;
576
577 if (maxX < minX)
578 maxX = minX;
579
580 if (maxY < minY)
581 maxY = minY;
582
583 return new PointF(
584 Math.Clamp(ul.X, minX, maxX),
585 Math.Clamp(ul.Y, minY, maxY));
586 }
