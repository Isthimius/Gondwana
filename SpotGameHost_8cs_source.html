Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

308 private void KeyboardEventPoller_KeyDown(KeyDownEventArgs args)
309 {
310 if (args.KeyAction != KeyAction.Pressed)
311 return;
312
313 var key = WinFormsKeyboardAdapter.GetKeyFromString(args.KeyConfig.Key);
314 switch (key)
315 {
316 case Keys.S:
317 SetScoreVisible(!_showScores);
318 break;
319 default:
320 break;
321 }
322 }

324 private void MouseEventPoller_MouseEvent(Gondwana.Input.Mouse.MouseEventArgs args)
325 {
326 if (!_handleHumanInput)
327 return;
328
329 if (Scene is null || Scene.SceneLayers.Count == 0)
330 return;
331
332 if (RenderSurface.Host.ViewManager.Views.Count == 0)
333 return;
334
335 var view = RenderSurface.Host.ViewManager.Views[0];
336 var layer = Scene.SceneLayers[0];
337
338 var screenPos = args.CurrentPosition;
339
340 if (args.LeftButtonJustPressed)
341 {
342 var selectedCoord = view.ScreenPxToGrid(layer, screenPos);
343
344 if (selectedCoord.X >= 0 && selectedCoord.X < layer.GridColumnCount &&
345 selectedCoord.Y >= 0 && selectedCoord.Y < layer.GridRowCount)
346 {
347 var cell = SpotGame.SpotGameField.GetCell((int)selectedCoord.X, (int)selectedCoord.Y);
348
349 if (SpotGame.AttemptSelectCell(cell, out var playerMovement))
350 {
351 if (playerMovement != null)
352 SpotGame.ExecuteMove(playerMovement.Value);
353 }
354 }
355 }
356 }

415 SKColor[] colors =
416 {
417 SKColors.Red,
418 SKColors.Blue,
419 SKColors.Yellow,
420 SKColors.Green,

424 return new ParticleEmitter
425 {

427 JitterY = height * 0.5f,
428
429 EmitRate = 0.65f,
430 LifeRange = (1000f, 2000f),
431
432 VelocityRangeX = (-100f, -50f),
433 VelocityRangeY = (-1f, 1f),
434
435 SizeRange = (40f, 80f),
436
437 GravityY = 0f,
438 BlendMode = SKBlendMode.SrcOver,

639 private void CreateGameOverText(List<Player> winningPlayers)
640 {
641 bool multipleWinners;
642 string message;
643
644 Color primaryTextColor = winningPlayers[0].ColorItem.TextColor.ToColor();
645 Color? secondaryTextColor = null;
646 Color primaryFillColor = winningPlayers[0].ColorItem.Color.ToColor();
647 Color? secondaryFillColor = null;
648
649 if (winningPlayers.Count == 1)
650 {
651 multipleWinners = false;
652 message = winningPlayers[0].Name + " wins!";
653 }
654 else
655 {
656 multipleWinners = true;
657
658 var names = winningPlayers.Select(p => p.Name).ToList();
659 var formatted = names.Count == 2
660 ? string.Join(" and ", names)
661 : string.Join(", ", names.Take(names.Count - 1)) + $", and {names.Last()}";
662
663 message = $"{formatted} tie!";
664
665 secondaryTextColor = winningPlayers[1].ColorItem.TextColor.ToColor();
666 secondaryFillColor = winningPlayers[1].ColorItem.Color.ToColor();
667 }

741 private void OnGameStarted(SpotGame game)
742 {
743 Engine.Logger.LogDebug("Game started with players: {0}", string.Join(", ", game.Players.Select(p => p.Name)));
744
745 if (MusicEnabled)
746 {
