Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

482 public void Draw(BackbufferBase backbuffer, RectangleF destRectScreen)
483 {
484 if (!Visible)
485 return;
486
487 var canvas = backbuffer.Canvas;
488
489 // Compute reveal clip rect (screen pixel space) from bounds
490 bool useClip = _revealT < 0.999f;
491
492 if (useClip)
493 {
494 var r = new SKRect(destRectScreen.Left, destRectScreen.Top, destRectScreen.Right, destRectScreen.Bottom);
495
496 SKRect clipRect = _revealDir switch
497 {
498 RevealDirection.LeftToRight => new SKRect(r.Left, r.Top, r.Left + r.Width * _revealT, r.Bottom),
499 RevealDirection.RightToLeft => new SKRect(r.Right - r.Width * _revealT, r.Top, r.Right, r.Bottom),
500 RevealDirection.TopToBottom => new SKRect(r.Left, r.Top, r.Right, r.Top + r.Height * _revealT),
501 RevealDirection.BottomToTop => new SKRect(r.Left, r.Bottom - r.Height * _revealT, r.Right, r.Bottom),
502 _ => r
503 };
504
505 // Early-out if reveal window is empty
506 if (clipRect.Width <= 0f || clipRect.Height <= 0f)
507 return;
508
509 // Outer save owns the clip lifetime
510 canvas.Save();
511
512 // Capture current matrix
513 var m = canvas.TotalMatrix;
514
515 canvas.ResetMatrix();
516 canvas.ClipRect(clipRect, SKClipOperation.Intersect, antialias: false);
517
518 // Put the prior matrix back so Draw() sees the same transform state
519 canvas.SetMatrix(m);
520 }
521
522 if (_opacity >= 0.999f)
523 {
524 OnDraw(backbuffer, destRectScreen);
525 }
526 else
527 {
528 using var layerPaint = new SKPaint
529 {
530 Color = new SKColor(255, 255, 255, (byte)(_opacity * 255))
531 };
532
533 canvas.SaveLayer(destRectScreen.ToSKRect(), layerPaint);
534 OnDraw(backbuffer, destRectScreen);
535 canvas.Restore(); // end SaveLayer
536 }
537
538 if (useClip)
539 canvas.Restore(); // end outer clip Save
540 }

786 protected internal void ForceRefresh()
787 {
788 switch (Mode)
789 {
790 case DirectDrawingMode.SceneLayer:
791 // bounds is WORLD-space
792 // there will only be one SceneLayer per DirectDrawing in this mode
793 SceneLayer!.RefreshQueue.AddWorldRect(_worldBounds);
794 break;
795
796 case DirectDrawingMode.View:
797 // bounds is SCREEN-space
798 // need to cycle through all SceneLayers for the View to which this DirectDrawing belongs
799 foreach (var sceneLayer in RenderSurfaceHost.Scene.SceneLayers)
800 {
801 sceneLayer.RefreshQueue.AddViewScreenRect(View!, sceneLayer, _screenBounds);
802 }
803 break;
804
805 default:
806 break;
807 }

831 public virtual void Update(long tick)
832 {
833 if (tick <= _lastTick)
834 return;
835
836 // Advance fade tween
837 if (_isFading)
838 {
839 float dt = HighResTimer.GetDuration(_lastTick, tick);
840
841 _fadeElapsedSec += dt;
842 float timeElapsed = Math.Clamp(_fadeElapsedSec / _fadeDurationSec, 0f, 1f);
843 // linear; swap in easing if you like
844 _opacity = _fadeFrom + (_fadeTo - _fadeFrom) * timeElapsed;
845
846 if (HideWhenFullyTransparent)
847 Visible = _opacity > 0f; // hides when hit zero
848
849 ForceRefresh();
850
851 if (timeElapsed >= 1f)
852 {
853 _isFading = false;
854 FadeToCompleted?.Invoke(this, this);
855 }
856 }
857
858 // Advance reveal tween
859 if (_revealAnimating)
860 {
861 float dt = HighResTimer.GetDuration(_lastTick, tick);
862 _revealElapsedSec = Math.Min(_revealElapsedSec + dt, _revealDurationSec);
863
864 float u = _revealElapsedSec / _revealDurationSec;
865 _revealT = (_revealEasing is null ? u : _revealEasing(u));
866 _revealT = _revealStart + (_revealTarget - _revealStart) * _revealT;
867
868 ForceRefresh();
869
870 if (_revealElapsedSec >= _revealDurationSec)
871 _revealAnimating = false;
872 }
873
874 _lastTick = tick;
875 }
