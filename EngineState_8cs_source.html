Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

369 private static void ApplySnapshot(
370 EngineStateSnapshot snapshot,
371 bool clearExisting,
372 bool overwriteExisting,
373 EngineStateParts parts,
374 string? baseDirectory)
375 {
376 parts = NormalizeParts(parts);
377
378 // clear only what we're about to load.
379 if (clearExisting)
380 ClearSelected(parts);
381
382 if (parts.HasFlag(EngineStateParts.AssetsFiles))
383 LoadAssetsFiles(snapshot.AssetsFiles ?? Enumerable.Empty<AssetsFile>(), overwriteExisting);
384
385 if (parts.HasFlag(EngineStateParts.Audio))
386 MergeAudio(snapshot.AssetsFiles, snapshot.SoundResources, overwriteExisting);
387
388 if (parts.HasFlag(EngineStateParts.Tilesheets))
389 MergeTilesheets(snapshot.Tilesheets, overwriteExisting, baseDirectory);
390
391 if (parts.HasFlag(EngineStateParts.Cycles))
392 MergeCycles(snapshot.Cycles, overwriteExisting);
393
394 if (parts.HasFlag(EngineStateParts.Scenes))
395 MergeScenes(snapshot.Scenes, overwriteExisting);
396
397 if (parts.HasFlag(EngineStateParts.Sprites))
398 MergeSprites(snapshot.Sprites, overwriteExisting);

425 private static void LoadAssetsFiles(IEnumerable<AssetsFile> resourceFiles, bool overwriteExisting)
426 {
427 // Replace raw deserialized resource files with proper loaded instances
428 if (resourceFiles.Any())
429 {
430 foreach (var raw in resourceFiles)
431 {
432 try
433 {
434 var loaded = AssetsFile.LoadOrCreate(raw.FilePath, raw.Password, raw.UseEncryption);
435
436 if (overwriteExisting)
437 {
438 foreach (var entry in loaded.GetAllEntries().Where(e => e.AssetType == AssetTypes.Svg))
439 SvgResourceManager.Instance.Unload(entry.AssetName);
440 }
441
442 SvgResourceManager.Instance.LoadFromEngineAssetsFile(loaded);
443 }
444 catch (Exception ex)
445 {
446 Engine.Logger.LogError(ex, "Failed to load resource file '{FilePath}'", raw.FilePath);
447 throw;
448 }
449 }
450 }

453 private static void MergeAudio(
454 List<AssetsFile>? assetsFiles,
455 Dictionary<string, AudioResource>? soundSpecs,
456 bool overwriteExisting)
457 {
458 // 1) Load from asset packs
459 if (assetsFiles is not null)
460 {
461 foreach (var af in assetsFiles)
462 {
463 if (overwriteExisting)
464 {
465 foreach (var entry in af.GetAllEntries())
466 {
467 if (entry.AssetType == AssetTypes.Audio)
468 AudioResourceManager.Instance.Unload(entry.AssetName);
469 }
470 }
471
472 AudioResourceManager.Instance.LoadFromEngineAssetsFile(af);
473 }
474 }
475
476 // 2) Apply loose-file specs / overrides
477 if (soundSpecs is null)
478 return;
479
480 foreach (var (key, spec) in soundSpecs)
481 {
482 if (AudioResourceManager.Instance.Contains(key))
483 {
484 if (!overwriteExisting)
485 {
486 var existing = AudioResourceManager.Instance.Get(key);
487 if (existing is not null)
488 {
489 existing.Volume = spec.Volume;
490 existing.Pan = spec.Pan;
491 existing.IsLooping = spec.IsLooping;
492 }
493 continue;
494 }
495
496 AudioResourceManager.Instance.Unload(key);
497 }
498
499 // Ensure the audio spec is (re)created/registered in the manager.
500 spec.ReloadIntoManager();
501 }

516 private static Dictionary<string, TilesheetStateEntry> CaptureTilesheetEntries(
517 string? baseDirectory,
518 string engineStatePath,
519 bool separateGtsFiles)
520 {
521 var result = new Dictionary<string, TilesheetStateEntry>(StringComparer.Ordinal);
522
523 foreach (var (key, tilesheet) in TilesheetRegistry.Instance.GetAll())
524 {
525 if (separateGtsFiles)
526 {
527 var gtsDirectory = GetTilesheetStateDirectory(engineStatePath);
528 Directory.CreateDirectory(gtsDirectory);
529
530 var gtsFileName = $"{SanitizeFileName(key)}.gts";
531 var gtsFullPath = Path.Combine(gtsDirectory, gtsFileName);
532
533 // Important:
534 // Save the GTS using the GTS serializer, not EngineState.JsonSerializerSettings.
535 // This avoids $id/$values noise in the .gts file.
536 TilesheetDefinitionSerializer.Save(
537 gtsFullPath,
538 tilesheet,
539 makePathsRelative: true);
540
541 var gtsPathForState = MakeRelativePath(
542 gtsFullPath,
543 baseDirectory);
544
545 result[key] = new TilesheetStateEntry
546 {
547 GtsPath = gtsPathForState
548 };
549 }
550 else
551 {
552 result[key] = new TilesheetStateEntry
553 {
554 Definition = TilesheetDefinitionSerializer.FromTilesheet(
555 tilesheet,
556 baseDirectory,
557 makePathsRelative: !string.IsNullOrWhiteSpace(baseDirectory))
558 };
559 }
560 }
561
562 return result;
563 }

583 private static string SanitizeFileName(string value)
584 {
585 if (string.IsNullOrWhiteSpace(value))
586 return Guid.NewGuid().ToString("N");
587
588 var invalidChars = Path.GetInvalidFileNameChars();
589 var sanitized = new string(
590 value.Select(ch => invalidChars.Contains(ch) ? '_' : ch).ToArray());
591
592 return string.IsNullOrWhiteSpace(sanitized)
593 ? Guid.NewGuid().ToString("N")
594 : sanitized;
595 }

597 private static void MergeTilesheets(
598 Dictionary<string, TilesheetStateEntry>? tilesheets,
599 bool overwriteExisting,
600 string? baseDirectory)
601 {
602 if (tilesheets is null || tilesheets.Count == 0)
603 return;
604
605 var registry = TilesheetRegistry.Instance.GetAll();
606
607 foreach (var (key, entry) in tilesheets)
608 {
609 if (entry is null)
610 continue;
611
612 var existingKey = key;
613
614 if (!overwriteExisting && registry.ContainsKey(existingKey))
615 continue;
616
617 Tilesheet rebuilt;
618
619 if (!string.IsNullOrWhiteSpace(entry.GtsPath))
620 {
621 var gtsPath = ResolvePath(entry.GtsPath, baseDirectory);
622
623 rebuilt = TilesheetFactory.FromDefinitionFile(gtsPath);
624 }
625 else if (entry.Definition is not null)
626 {
627 if (string.IsNullOrWhiteSpace(entry.Definition.Name))
628 entry.Definition.Name = key;
629
630 rebuilt = TilesheetFactory.FromDefinition(
631 entry.Definition,
632 baseDirectory);
633 }
634 else
635 {
636 throw new InvalidDataException(
637 $"Tilesheet state entry '{key}' does not contain a GTS path or inline definition.");
638 }
639
640 TilesheetRegistry.Instance.Register(
641 rebuilt,
642 disposeReplaced: overwriteExisting);
643 }
644 }

671 private static void MergeScenes(List<Scene>? scenes, bool overwriteExisting)
672 {
673 if (scenes is null || scenes.Count == 0)
674 return;
675
676 // Index existing scenes by ID (case-sensitive; change if you prefer)
677 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
678 for (int i = 0; i < Scene._allScenes.Count; i++)
679 {
680 var id = Scene._allScenes[i].ID;
681 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
682 existingIndexById.Add(id, i);
683 }
684
685 // Avoid duplicating the same incoming ID twice (keeps last one)
686 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
687
688 foreach (var incoming in scenes)
689 {
690 if (incoming is null)
691 continue;
692
693 // Ensure ID exists (important if something created scenes without IDs)
694 if (string.IsNullOrWhiteSpace(incoming.ID))
695 incoming.ID = Guid.NewGuid().ToString();
696
697 // If the incoming list contains the same ID multiple times, last one wins.
698 if (!seenIncoming.Add(incoming.ID))
699 {
700 // Replace the previously added/replaced incoming with this one:
701 // easiest way: treat it as overwriteExisting=true for that ID
702 overwriteExisting = true;
703 }
704
705 if (existingIndexById.TryGetValue(incoming.ID, out int existingIndex))
706 {
707 if (!overwriteExisting)
708 continue;
709
710 Scene._allScenes[existingIndex] = incoming;
711 }
712 else
713 {
714 existingIndexById[incoming.ID] = Scene._allScenes.Count;
715 Scene._allScenes.Add(incoming);
716 }
717 }
718 }

720 private static void MergeSprites(List<Sprite>? sprites, bool overwriteExisting)
721 {
722 if (sprites is null || sprites.Count == 0)
723 return;
724
725 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
726 for (int i = 0; i < SpriteManager.Instance._spriteList.Count; i++)
727 {
728 var id = SpriteManager.Instance._spriteList[i].Nickname;
729 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
730 existingIndexById.Add(id, i);
731 }
732
733 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
734
735 foreach (var incoming in sprites)
736 {
737 if (incoming is null)
738 continue;
739
740 if (string.IsNullOrWhiteSpace(incoming.Nickname))
741 incoming.Nickname = Guid.NewGuid().ToString();
742
743 if (!seenIncoming.Add(incoming.Nickname))
744 {
745 // Same-ID appears again in the incoming list: last one wins.
746 overwriteExisting = true;
747 }
748
749 if (existingIndexById.TryGetValue(incoming.Nickname, out int existingIndex))
750 {
751 if (!overwriteExisting)
752 continue;
753
754 SpriteManager.Instance._spriteList[existingIndex] = incoming;
755 }
756 else
757 {
758 existingIndexById[incoming.Nickname] = SpriteManager.Instance._spriteList.Count;
759 SpriteManager.Instance.AddSprite(incoming);
760 }
761 }
