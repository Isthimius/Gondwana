Gondwana Game Engine v2.1.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

320 private static void ApplySnapshot(
321 EngineStateSnapshot snapshot,
322 bool clearExisting,
323 bool overwriteExisting,
324 EngineStateParts parts)
325 {
326 parts = NormalizeParts(parts);
327
328 // clear only what we're about to load.
329 if (clearExisting)
330 ClearSelected(parts);
331
332 if (parts.HasFlag(EngineStateParts.AssetsFiles))
333 LoadAssetsFiles(snapshot.AssetsFiles ?? Enumerable.Empty<AssetsFile>());
334
335 if (parts.HasFlag(EngineStateParts.Audio))
336 MergeAudio(snapshot.AssetsFiles, snapshot.SoundResources, overwriteExisting);
337
338 if (parts.HasFlag(EngineStateParts.Tilesheets))
339 MergeTilesheets(snapshot.Tilesheets, overwriteExisting);
340
341 if (parts.HasFlag(EngineStateParts.Cycles))
342 MergeCycles(snapshot.Cycles, overwriteExisting);
343
344 if (parts.HasFlag(EngineStateParts.Scenes))
345 MergeScenes(snapshot.Scenes, overwriteExisting);
346
347 if (parts.HasFlag(EngineStateParts.Sprites))
348 MergeSprites(snapshot.Sprites, overwriteExisting);
349
350 if (parts.HasFlag(EngineStateParts.ValueBag))
351 Engine.Instance.State.ValueBag.MergeFrom(snapshot.ValueBag, overwriteExisting);
352 }

398 private static Tilesheet? RebuildTilesheetFromSaved(string key, Tilesheet saved)
399 {
400 Tilesheet rebuilt;
401
402 // 1) Rehydrate bitmap from AssetsFile entry (preferred) or file path (fallback)
403 if (saved.AssetIdentifier is not null && saved.AssetIdentifier.IsValid)
404 {
405 var id = saved.AssetIdentifier;
406 rebuilt = new Tilesheet(id.AssetsFile, id.AssetName);
407 }
408 else if (!string.IsNullOrWhiteSpace(saved.ImageFilePath) && File.Exists(saved.ImageFilePath))
409 {
410 rebuilt = new Tilesheet(saved.Name, saved.ImageFilePath);
411 }
412 else
413 {
414 Engine.Logger.LogWarning(
415 "EngineState: Skipping tilesheet '{Key}' because it has no valid AssetIdentifier and no ImageFilePath.",
416 key);
417 return null;
418 }
419
420 // 2) Restore metadata (these trigger cache rebuild as needed)
421 rebuilt.Name = saved.Name;
422 rebuilt.TileSize = saved.TileSize;
423 rebuilt.InitialOffsetX = saved.InitialOffsetX;
424 rebuilt.InitialOffsetY = saved.InitialOffsetY;
425 rebuilt.XPixelsBetweenTiles = saved.XPixelsBetweenTiles;
426 rebuilt.YPixelsBetweenTiles = saved.YPixelsBetweenTiles;
427 rebuilt.OverhangPixels = saved.OverhangPixels;
428
429 // 3) Restore extensible tilesheet metadata
430 rebuilt.ValueBag = saved.ValueBag.Clone();
431
432 // 4) Reapply bitmap transforms recorded in the save.
433 //
434 // IMPORTANT: SkBitmap is not serialized, so these operations must be replayed here.
435 // ApplyMask() also premultiplies alpha internally in the implementation.
436 if (saved.MaskColor is not null)
437 {
438 rebuilt.ApplyMask(saved.MaskColor, saved.MaskTolerance);
439 }
440 else if (saved.Premultiplied)
441 {
442 // ApplyMask also premultiplies alpha internally,
443 // so only call if Premultiplied and no MaskColor
444 rebuilt.ApplyPremultiplyAlpha();
445 }
446
447 return rebuilt;
448 }

450 private static void MergeAudio(
451 List<AssetsFile>? assetsFiles,
452 Dictionary<string, AudioResource>? soundSpecs,
453 bool overwriteExisting)
454 {
455 // 1) Load from asset packs
456 if (assetsFiles is not null)
457 {
458 foreach (var af in assetsFiles)
459 {
460 if (overwriteExisting)
461 {
462 foreach (var entry in af.GetAllEntries())
463 {
464 if (entry.AssetType == AssetTypes.Audio)
465 AudioResourceManager.Instance.Unload(entry.AssetName);
466 }
467 }
468
469 AudioResourceManager.Instance.LoadFromEngineAssetsFile(af);
470 }
471 }
472
473 // 2) Apply loose-file specs / overrides
474 if (soundSpecs is null)
475 return;
476
477 foreach (var (key, spec) in soundSpecs)
478 {
479 if (AudioResourceManager.Instance.Contains(key))
480 {
481 if (!overwriteExisting)
482 {
483 var existing = AudioResourceManager.Instance.Get(key);
484 if (existing is not null)
485 {
486 existing.Volume = spec.Volume;
487 existing.Pan = spec.Pan;
488 existing.IsLooping = spec.IsLooping;
489 }
490 continue;
491 }
492
493 AudioResourceManager.Instance.Unload(key);
494 }
495
496 // Ensure the audio spec is (re)created/registered in the manager.
497 spec.ReloadIntoManager();
498 }
499 }

533 private static void MergeScenes(List<Scene>? scenes, bool overwriteExisting)
534 {
535 if (scenes is null || scenes.Count == 0)
536 return;
537
538 // Index existing scenes by ID (case-sensitive; change if you prefer)
539 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
540 for (int i = 0; i < Scene._allScenes.Count; i++)
541 {
542 var id = Scene._allScenes[i].ID;
543 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
544 existingIndexById.Add(id, i);
545 }
546
547 // Avoid duplicating the same incoming ID twice (keeps last one)
548 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
549
550 foreach (var incoming in scenes)
551 {
552 if (incoming is null)
553 continue;
554
555 // Ensure ID exists (important if something created scenes without IDs)
556 if (string.IsNullOrWhiteSpace(incoming.ID))
557 incoming.ID = Guid.NewGuid().ToString();
558
559 // If the incoming list contains the same ID multiple times, last one wins.
560 if (!seenIncoming.Add(incoming.ID))
561 {
562 // Replace the previously added/replaced incoming with this one:
563 // easiest way: treat it as overwriteExisting=true for that ID
564 overwriteExisting = true;
565 }
566
567 if (existingIndexById.TryGetValue(incoming.ID, out int existingIndex))
568 {
569 if (!overwriteExisting)
570 continue;
571
572 Scene._allScenes[existingIndex] = incoming;
573 }
574 else
575 {
576 existingIndexById[incoming.ID] = Scene._allScenes.Count;
577 Scene._allScenes.Add(incoming);
578 }
579 }
580 }

582 private static void MergeSprites(List<Sprite>? sprites, bool overwriteExisting)
583 {
584 if (sprites is null || sprites.Count == 0)
585 return;
586
587 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
588 for (int i = 0; i < SpriteManager.Instance._spriteList.Count; i++)
589 {
590 var id = SpriteManager.Instance._spriteList[i].Nickname;
591 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
592 existingIndexById.Add(id, i);
593 }
594
595 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
596
597 foreach (var incoming in sprites)
598 {
599 if (incoming is null)
600 continue;
601
602 if (string.IsNullOrWhiteSpace(incoming.Nickname))
603 incoming.Nickname = Guid.NewGuid().ToString();
604
605 if (!seenIncoming.Add(incoming.Nickname))
606 {
607 // Same-ID appears again in the incoming list: last one wins.
608 overwriteExisting = true;
609 }
610
611 if (existingIndexById.TryGetValue(incoming.Nickname, out int existingIndex))
612 {
613 if (!overwriteExisting)
614 continue;
615
616 SpriteManager.Instance._spriteList[existingIndex] = incoming;
617 }
618 else
619 {
620 existingIndexById[incoming.Nickname] = SpriteManager.Instance._spriteList.Count;
621 SpriteManager.Instance._spriteList.Add(incoming);
622 }
623 }
