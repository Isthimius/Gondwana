Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

754 public virtual RectangleF GetLayerBoundsPx()
755 {
756 if (GridColumnCount == 0 || GridRowCount == 0)
757 return RectangleF.Empty;
758
759 int maxCol = GridColumnCount - 1;
760 int maxRow = GridRowCount - 1;
761
762 // Corner tiles
763 var corners = new[]
764 {
765 this[0, 0],
766 this[maxCol, 0],
767 this[0, maxRow],
768 this[maxCol, maxRow]
769 };
770
771 RectangleF result = RectangleF.Empty;
772 bool hasBounds = false;
773
774 foreach (var tile in corners)
775 {
776 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
777 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
778
779 if (!hasBounds)
780 {
781 result = rf;
782 hasBounds = true;
783 }
784 else
785 {
786 result = RectangleF.Union(result, rf);
787 }
788 }
789
790 return result;
791 }

844 internal virtual List<IDrawable> GetDrawablesInWorldRect(Rectangle worldRect, bool includeOverhang = true)
845 {
846 // Make selection rect covering so we never miss the edge tile.
847 // Drawing is still clipped later, so over-selecting is safe.
848 var queryRect = worldRect;
849 queryRect.Inflate(TileWidth, TileHeight); // <- KEY (tile-sized)
850 queryRect.Inflate(1, 1); // optional boundary insurance
851
852 // Gather into a list so we can sort it.
853 var list = new List<IDrawable>(64);
854
855 // 1) Grid tiles
856 var sceneLayerTiles = CoordinateSystem.GetSceneLayerTilesInPixelRange(
857 this,
858 worldRect,
859 includeOverhang: includeOverhang);
860
861 if (sceneLayerTiles != null)
862 {
863 for (int i = 0; i < sceneLayerTiles.Count; i++)
864 {
865 var tile = sceneLayerTiles[i];
866
867 if (tile is null)
868 continue;
869
870 if (!tile.Visible)
871 continue;
872
873 // Defensive overlap check (same idea as sprites)
874 if (!tile.DrawLocationWorld.IntersectsWith(queryRect))
875 continue;
876
877 list.Add(tile);
878 }
879 }
880
881 // 2) Sprites
882 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(queryRect, this, fullEnclosures: false);
883
884 for (int i = 0; i < sprites.Count; i++)
885 {
886 var sprite = sprites[i];
887
888 if (sprite is null)
889 continue;
890
891 // Defensive overlap check (cheap)
892 if (!sprite.VisualBoundsWorld.IntersectsWith(queryRect))
893 continue;
894
895 list.Add(sprite);
896 }
897
898 // 3) DirectDrawing instances
899 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
900
901 for (int i = 0; i < drawings.Count; i++)
902 {
903 var drawing = drawings[i];
904
905 if (!drawing.Visible)
906 continue;
907
908 // Must be SceneLayer-mode by definition if it's "for layer", but be defensive:
909 if (drawing.Mode != DirectDrawingMode.SceneLayer)
910 continue;
911
912 // Only include if it intersects this dirty rect
913 if (!drawing.WorldBounds.IntersectsWith(worldRect))
914 continue;
915
916 list.Add(drawing);
917 }
918
919 // 4) Sort using Tile.CompareTo
920 list.Sort(CompareDrawables); // ← this calls Tile.CompareTo internally
921 return list;
922 }

924 private static int CompareDrawables(IDrawable a, IDrawable b)
925 {
926 int z = a.ZOrder.CompareTo(b.ZOrder);
927 if (z != 0)
928 return z;
929
930 // Preserve legacy ordering for tiles/sprites when Z ties
931 if (a is Tile ta && b is Tile tb)
932 return ta.CompareTo(tb);
933
934 // Stable tie-breaker (avoid flicker)
935 return a.Id.CompareTo(b.Id);
936 }
