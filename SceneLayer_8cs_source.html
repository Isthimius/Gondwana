Gondwana Game Engine v2.2.4
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

690 public virtual RectangleF GetLayerBoundsPx()
691 {
692 if (GridColumnCount == 0 || GridRowCount == 0)
693 return RectangleF.Empty;
694
695 int maxCol = GridColumnCount - 1;
696 int maxRow = GridRowCount - 1;
697
698 // Corner tiles
699 var corners = new[]
700 {
701 this[0, 0],
702 this[maxCol, 0],
703 this[0, maxRow],
704 this[maxCol, maxRow]
705 };
706
707 RectangleF result = RectangleF.Empty;
708 bool hasBounds = false;
709
710 foreach (var tile in corners)
711 {
712 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
713 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
714
715 if (!hasBounds)
716 {
717 result = rf;
718 hasBounds = true;
719 }
720 else
721 {
722 result = RectangleF.Union(result, rf);
723 }
724 }
725
726 return result;
727 }

778 internal virtual List<IDrawable> GetDrawablesInWorldRect(Rectangle worldRect, bool includeOverhang = true)
779 {
780 // Make selection rect covering so we never miss the edge tile.
781 // Drawing is still clipped later, so over-selecting is safe.
782 var queryRect = worldRect;
783 queryRect.Inflate(TileWidth, TileHeight); // <- KEY (tile-sized)
784 queryRect.Inflate(1, 1); // optional boundary insurance
785
786 // Gather into a list so we can sort it.
787 var list = new List<IDrawable>(64);
788
789 // 1) Grid tiles
790 var sceneLayerTiles = CoordinateSystem.GetSceneLayerTilesInPixelRange(
791 this,
792 worldRect,
793 includeOverhang: includeOverhang);
794
795 if (sceneLayerTiles != null)
796 {
797 for (int i = 0; i < sceneLayerTiles.Count; i++)
798 {
799 var tile = sceneLayerTiles[i];
800
801 if (tile is null)
802 continue;
803
804 if (!tile.Visible)
805 continue;
806
807 // Defensive overlap check (same idea as sprites)
808 if (!tile.DrawLocationWorld.IntersectsWith(queryRect))
809 continue;
810
811 list.Add(tile);
812 }
813 }
814
815 // 2) Sprites
816 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(queryRect, this, fullEnclosures: false);
817
818 for (int i = 0; i < sprites.Count; i++)
819 {
820 var sprite = sprites[i];
821
822 if (sprite is null)
823 continue;
824
825 // Defensive overlap check (cheap)
826 if (!sprite.DrawLocationWorld.IntersectsWith(queryRect))
827 continue;
828
829 list.Add(sprite);
830 }
831
832 // 3) DirectDrawing instances
833 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
834
835 for (int i = 0; i < drawings.Count; i++)
836 {
837 var drawing = drawings[i];
838
839 if (!drawing.Visible)
840 continue;
841
842 // Must be SceneLayer-mode by definition if it's "for layer", but be defensive:
843 if (drawing.Mode != DirectDrawingMode.SceneLayer)
844 continue;
845
846 // Only include if it intersects this dirty rect
847 if (!drawing.WorldBounds.IntersectsWith(worldRect))
848 continue;
849
850 list.Add(drawing);
851 }
852
853 // 4) Sort using Tile.CompareTo
854 list.Sort(CompareDrawables); // ← this calls Tile.CompareTo internally
855 return list;
856 }
857
858 private static int CompareDrawables(IDrawable a, IDrawable b)
859 {
860 int z = a.ZOrder.CompareTo(b.ZOrder);
861 if (z != 0)
862 return z;
863
864 // Preserve legacy ordering for tiles/sprites when Z ties
865 if (a is Tile ta && b is Tile tb)
866 return ta.CompareTo(tb);
867
868 // Stable tie-breaker (avoid flicker)
869 return a.Id.CompareTo(b.Id);
870 }
