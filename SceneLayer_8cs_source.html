Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

703 public virtual RectangleF GetLayerBoundsPx()
704 {
705 if (GridColumnCount == 0 || GridRowCount == 0)
706 return RectangleF.Empty;
707
708 int maxCol = GridColumnCount - 1;
709 int maxRow = GridRowCount - 1;
710
711 // Corner tiles
712 var corners = new[]
713 {
714 this[0, 0],
715 this[maxCol, 0],
716 this[0, maxRow],
717 this[maxCol, maxRow]
718 };
719
720 RectangleF result = RectangleF.Empty;
721 bool hasBounds = false;
722
723 foreach (var tile in corners)
724 {
725 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
726 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
727
728 if (!hasBounds)
729 {
730 result = rf;
731 hasBounds = true;
732 }
733 else
734 {
735 result = RectangleF.Union(result, rf);
736 }
737 }
738
739 return result;
740 }

791 internal virtual List<IDrawable> GetDrawablesInWorldRect(Rectangle worldRect, bool includeOverhang = true)
792 {
793 // Make selection rect covering so we never miss the edge tile.
794 // Drawing is still clipped later, so over-selecting is safe.
795 var queryRect = worldRect;
796 queryRect.Inflate(TileWidth, TileHeight); // <- KEY (tile-sized)
797 queryRect.Inflate(1, 1); // optional boundary insurance
798
799 // Gather into a list so we can sort it.
800 var list = new List<IDrawable>(64);
801
802 // 1) Grid tiles
803 var sceneLayerTiles = CoordinateSystem.GetSceneLayerTilesInPixelRange(
804 this,
805 worldRect,
806 includeOverhang: includeOverhang);
807
808 if (sceneLayerTiles != null)
809 {
810 for (int i = 0; i < sceneLayerTiles.Count; i++)
811 {
812 var tile = sceneLayerTiles[i];
813
814 if (tile is null)
815 continue;
816
817 if (!tile.Visible)
818 continue;
819
820 // Defensive overlap check (same idea as sprites)
821 if (!tile.DrawLocationWorld.IntersectsWith(queryRect))
822 continue;
823
824 list.Add(tile);
825 }
826 }
827
828 // 2) Sprites
829 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(queryRect, this, fullEnclosures: false);
830
831 for (int i = 0; i < sprites.Count; i++)
832 {
833 var sprite = sprites[i];
834
835 if (sprite is null)
836 continue;
837
838 // Defensive overlap check (cheap)
839 if (!sprite.DrawLocationWorld.IntersectsWith(queryRect))
840 continue;
841
842 list.Add(sprite);
843 }
844
845 // 3) DirectDrawing instances
846 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
847
848 for (int i = 0; i < drawings.Count; i++)
849 {
850 var drawing = drawings[i];
851
852 if (!drawing.Visible)
853 continue;
854
855 // Must be SceneLayer-mode by definition if it's "for layer", but be defensive:
856 if (drawing.Mode != DirectDrawingMode.SceneLayer)
857 continue;
858
859 // Only include if it intersects this dirty rect
860 if (!drawing.WorldBounds.IntersectsWith(worldRect))
861 continue;
862
863 list.Add(drawing);
864 }
865
866 // 4) Sort using Tile.CompareTo
867 list.Sort(CompareDrawables); // ← this calls Tile.CompareTo internally
868 return list;
869 }
870
871 private static int CompareDrawables(IDrawable a, IDrawable b)
872 {
873 int z = a.ZOrder.CompareTo(b.ZOrder);
874 if (z != 0)
875 return z;
876
877 // Preserve legacy ordering for tiles/sprites when Z ties
878 if (a is Tile ta && b is Tile tb)
879 return ta.CompareTo(tb);
880
881 // Stable tie-breaker (avoid flicker)
882 return a.Id.CompareTo(b.Id);
883 }
