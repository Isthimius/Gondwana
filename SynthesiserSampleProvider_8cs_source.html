Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

123 public int Read(float[] buffer, int offset, int count)
124 {
125 int framesRequested = count / 2;
126 int framesRendered = 0;
127
128 while (framesRendered < framesRequested)
129 {
130 int remaining = framesRequested - framesRendered;
131 int renderCount = Math.Min(remaining, _left.Length);
132
133 var leftSpan = _left.AsSpan(0, renderCount);
134 var rightSpan = _right.AsSpan(0, renderCount);
135
136 // IMPORTANT: render via the sequencer, not directly via the synth.
137 _sequencer.Render(leftSpan, rightSpan);
138
139 bool silent = true;
140
141 for (int i = 0; i < renderCount; i++)
142 {
143 float left = leftSpan[i];
144 float right = rightSpan[i];
145
146 buffer[offset++] = left;
147 buffer[offset++] = right;
148
149 if (Math.Abs(left) > Tolerance || Math.Abs(right) > Tolerance)
150 silent = false;
151 }
152
153 framesRendered += renderCount;
154
155 if (silent)
156 {
157 if (_loop)
158 {
159 _sequencer.Play(_midiFile, loop: false); // manual restart
160 }
161 else
162 {
163 break;
164 }
165 }
166 }
167
168 return framesRendered * 2;
169 }
