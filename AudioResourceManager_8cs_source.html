Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

88 public List<AudioResource> LoadFromEngineAssetsFile(AssetsFile resourceFile, float defaultVolume = 1.0f, float defaultPan = 0.0f)
89 {
90 List<AudioResource> loadedSounds = new();
91
92 foreach (var entry in resourceFile.GetAllEntries())
93 {
94 if (entry.AssetType != AssetTypes.Audio)
95 continue;
96
97 if (_soundResources.ContainsKey(entry.AssetName))
98 {
99 Engine.Logger.LogDebug("AudioResource '{Key}' already loaded. Skipping.", entry.AssetName);
100 continue;
101 }
102
103 var stream = resourceFile.Get(entry.AssetType, entry.AssetName);
104 if (stream == null)
105 {
106 Engine.Logger.LogWarning("Failed to retrieve stream for audio resource: {Key}", entry.AssetName);
107 continue;
108 }
109
110 try
111 {
112 using var ms = new MemoryStream();
113 stream.CopyTo(ms);
114 var bytes = ms.ToArray();
115 Engine.Logger.LogInformation("Loaded sound: {Key}", entry.AssetName);
116 loadedSounds.Add(LoadFromBytes(entry.AssetName, bytes, entry.AssetName, defaultVolume, defaultPan));
117 }
118 catch (Exception ex)
119 {
120 Engine.Logger.LogError(ex, "Error loading sound from asset file for key: {Key}", entry.AssetName);
121 throw;
122 }
123 }
124
125 return loadedSounds;
126 }

138 public AudioResource? Clone(string key, string? newKey = null, float? volume = null, float? pan = null)
139 {
140 if (!_soundResources.TryGetValue(key, out var original))
141 {
142 Engine.Logger.LogWarning("Attempted to clone non-existent AudioResource with key: {Key}", key);
143 return null;
144 }
145
146 newKey ??= $"{key}_clone_{Guid.NewGuid()}";
147
148 if (_soundResources.ContainsKey(newKey))
149 {
150 Engine.Logger.LogWarning("AudioResource with key '{Key}' already exists. Cannot clone.", newKey);
151 return null;
152 }
153
154 if (original.soundResource.OriginalBytes == null)
155 {
156 Engine.Logger.LogWarning("Cannot clone AudioResource '{Key}' – missing original bytes.", key);
157 return null;
158 }
159
160 if (string.IsNullOrEmpty(original.soundResource.SourceExtension))
161 {
162 Engine.Logger.LogWarning("Cannot clone AudioResource '{Key}' – missing original extension.", key);
163 return null;
164 }
165
166 return LoadFromStream(
167 newKey,
168 new MemoryStream(original.soundResource.OriginalBytes),
169 original.soundResource.SourceExtension,
170 volume ?? original.soundResource.Volume,
171 pan ?? original.soundResource.Pan
172 );

175 private AudioResource LoadFromBytes(string key, byte[] bytes, string fileHint, float volume, float pan)
176 {
177 string ext = Path.GetExtension(fileHint);
178
179 if (string.IsNullOrWhiteSpace(ext))
180 {
181 throw new InvalidOperationException(
182 $"Audio asset '{key}' has no file extension. " +
183 "Ensure audio AssetsFile entries retain their extension."
184 );
185 }
186
187 var (readerFactory, requiresFile) = PlatformAudioFactory.GetReaderFactory(ext);
188
189 Stream streamForReader;
190 string? tempFilePath = null;
191
192 if (requiresFile)
193 {
194 // TODO: how does this play with the WinForms implementation of PlatformAudioFactory?
195 tempFilePath = SaveStreamToTempFile(new MemoryStream(bytes), ext);
196 streamForReader = File.OpenRead(tempFilePath);
197 }
198 else
199 {
200 streamForReader = new MemoryStream(bytes);
201 }
202
203 var reader = readerFactory(streamForReader);
204
205 var sound = new AudioResource(
206 key,
207 reader,
208 volume,
209 pan,
210 fileHint,
211 bytes,
212 tempFilePath
213 );
214
215 _soundResources[key] = (sound, requiresFile ? tempFilePath : null);
216 RegisterLoadedSound(key, sound);
217 return sound;
218 }
