Gondwana Game Engine v2.4.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

69 private void OnTouchBegan(object? sender, TouchEventArgs e)
70 {
71 _activeTaps[e.Touch.Id] = new TapState(e.Touch.Position);
72 }
73
74 private void OnTouchMoved(object? sender, TouchEventArgs e)
75 {
76 if (!_activeTaps.TryGetValue(e.Touch.Id, out var state))
77 return;
78
79 if (state.Cancelled)
80 return;
81
82 var dx = e.Touch.Position.X - state.StartPosition.X;
83 var dy = e.Touch.Position.Y - state.StartPosition.Y;
84 var distance = Math.Sqrt(dx * dx + dy * dy);
85
86 if (distance > MaxTapMovementPixels)
87 _activeTaps[e.Touch.Id] = state with { Cancelled = true };
88 }
89
90 private void OnTouchEnded(object? sender, TouchEventArgs e)
91 {
92 if (!_activeTaps.TryGetValue(e.Touch.Id, out var state))
93 return;
94
95 _activeTaps.Remove(e.Touch.Id);
96
97 // A system-cancelled contact (e.g. incoming call) must never fire Tapped.
98 if (state.Cancelled || e.Touch.Phase == TouchPhase.Cancelled)
99 return;
100
101 var elapsed = (DateTime.UtcNow - state.StartTime).TotalSeconds;
102 if (elapsed <= MaxTapDurationSeconds)
103 Tapped?.Invoke(this, new TappedEventArgs(e.Touch.Id, state.StartPosition));
104 }
105
110 public void Dispose()
111 {
112 if (_isDisposed) return;
113 _isDisposed = true;
114
115 _touchInput.TouchBegan -= OnTouchBegan;
116 _touchInput.TouchMoved -= OnTouchMoved;
117 _touchInput.TouchEnded -= OnTouchEnded;
118 }
