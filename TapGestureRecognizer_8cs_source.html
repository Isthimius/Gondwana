Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

74 private void OnTouchBegan(object? sender, TouchEventArgs e)
75 {
76 _activeContacts.Add(e.Touch.Id);
77
78 if (_activeContacts.Count > 1)
79 {
80 _isMultiTouchSequence = true;
81 _activeTaps.Clear();
82 return;
83 }
84
85 if (!_isMultiTouchSequence)
86 _activeTaps[e.Touch.Id] = new TapState(e.Touch.Position, e.Tick);
87 }
88
89 private void OnTouchMoved(object? sender, TouchEventArgs e)
90 {
91 if (!_activeTaps.TryGetValue(e.Touch.Id, out var state))
92 return;
93
94 if (state.Cancelled)
95 return;
96
97 var dx = e.Touch.Position.X - state.StartPosition.X;
98 var dy = e.Touch.Position.Y - state.StartPosition.Y;
99 var distance = Math.Sqrt(dx * dx + dy * dy);
100
101 if (distance > MaxTapMovementPixels)
102 _activeTaps[e.Touch.Id] = state with { Cancelled = true };
103 }
104
105 private void OnTouchEnded(object? sender, TouchEventArgs e)
106 {
107 _activeContacts.Remove(e.Touch.Id);
108
109 if (_isMultiTouchSequence)
110 {
111 _activeTaps.Remove(e.Touch.Id);
112 if (_activeContacts.Count == 0)
113 _isMultiTouchSequence = false;
114 return;
115 }
116
117 if (!_activeTaps.TryGetValue(e.Touch.Id, out var state))
118 return;
119
120 _activeTaps.Remove(e.Touch.Id);
121
122 // A system-cancelled contact (e.g. incoming call) must never fire Tapped.
123 if (state.Cancelled || e.Touch.Phase == TouchPhase.Cancelled)
124 return;
125
126 var dx = e.Touch.Position.X - state.StartPosition.X;
127 var dy = e.Touch.Position.Y - state.StartPosition.Y;
128 var distance = Math.Sqrt(dx * dx + dy * dy);
129 var elapsed = HighResTimer.GetDuration(state.StartTick, e.Tick);
130
131 if (distance <= MaxTapMovementPixels &&
132 elapsed <= MaxTapDurationSeconds &&
133 !(CompetingSwipeRecognizer?.WouldRecognize(distance, elapsed) ?? false))
134 {
135 Tapped?.Invoke(this, new TappedEventArgs(e.Touch.Id, state.StartPosition));
136 }
137 }
