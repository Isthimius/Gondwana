Gondwana Game Engine v2.2.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

84 private void BlitAndInvalidate()
85 {
86 if (_disposed) return;
87
88 var img = _currentImage;
89 if (img == null) return;
90
91 var w = img.Width;
92 var h = img.Height;
93 if (w <= 0 || h <= 0) return;
94
95 // Recreate the writable bitmap only when dimensions change.
96 if (_bitmap == null || _bitmap.PixelSize.Width != w || _bitmap.PixelSize.Height != h)
97 {
98 _bitmap?.Dispose();
99 _bitmap = new WriteableBitmap(
100 new PixelSize(w, h),
101 new Vector(96, 96),
102 PixelFormat.Bgra8888,
103 AlphaFormat.Premul);
104 }
105
106 // Read pixels from SKImage (BGRA8888 premul) straight into the writable bitmap.
107 using (var fb = _bitmap.Lock())
108 {
109 var info = new SKImageInfo(w, h, SKColorType.Bgra8888, SKAlphaType.Premul);
110 img.ReadPixels(info, fb.Address, fb.RowBytes);
111 }
112
113 _control.SetBitmap(_bitmap);
114 _control.InvalidateVisual();
115
116 while (_toDispose.TryDequeue(out var stale))
117 stale.Dispose();
118 }
