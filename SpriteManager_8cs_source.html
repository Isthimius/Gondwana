Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

286 internal void MoveSprites(long tick)
287 {
288 if (tick <= _lastTick)
289 return;
290
291 float duration = HighResTimer.GetDuration(_lastTick, tick);
292
293 lock (_spriteListLock)
294 {
295 for (int i = 0; i < _spriteList.Count; i++)
296 {
297 var sprite = _spriteList[i];
298
299 if (sprite._pendingDispose)
300 continue;
301
302 sprite.Movement.AdvanceMovement(duration);
303 sprite.AdvanceResize(duration);
304 sprite.AdvanceJiggle(duration);
305 }
306
307 SweepDisposedSprites();
308 }
309
310 _lastTick = tick;
311 }
312
313 private void SweepDisposedSprites()
314 {
315 int i = 0;
316 while (i < _spriteList.Count)
317 {
318 if (!_spriteList[i]._pendingDispose)
319 {
320 i++;
321 continue;
322 }
323
324 _spriteList[i].DisposeImmediate();
325
326 int last = _spriteList.Count - 1;
327 _spriteList[i] = _spriteList[last];
328 _spriteList.RemoveAt(last);
329 }
330 }
