Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

279 internal void MoveSprites(long tick)
280 {
281 if (tick <= _lastTick)
282 return;
283
284 float duration = HighResTimer.GetDuration(_lastTick, tick);
285
286 lock (_spriteListLock)
287 {
288 for (int i = 0; i < _spriteList.Count; i++)
289 {
290 var sprite = _spriteList[i];
291
292 if (sprite._pendingDispose)
293 continue;
294
295 sprite.Movement.AdvanceMovement(duration);
296 sprite.AdvanceResize(duration);
297 sprite.AdvanceJiggle(duration);
298 }
299
300 SweepDisposedSprites();
301 }
302
303 _lastTick = tick;
304 }
305
306 private void SweepDisposedSprites()
307 {
308 int i = 0;
309 while (i < _spriteList.Count)
310 {
311 if (!_spriteList[i]._pendingDispose)
312 {
313 i++;
314 continue;
315 }
316
317 _spriteList[i].DisposeImmediate();
318
319 int last = _spriteList.Count - 1;
320 _spriteList[i] = _spriteList[last];
321 _spriteList.RemoveAt(last);
322 }
323 }
