Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

69 private void Step(float dt)
70 {
71 // integrate kinematics in the mover's own space
72 _state.Velocity += _state.Acceleration * dt;
73 ClampVelocity();
74
75 // exponential damping (frame-rate independent)
76 if (_state.LinearDamping > 0f)
77 _state.Velocity *= MathF.Exp(-_state.LinearDamping * dt);
78
79 // kill tiny velocities when no accel
80 const float StopEpsilon = 0.01f;
81 if (_state.Acceleration == Vector2.Zero &&
82 _state.Velocity.LengthSquared() < StopEpsilon * StopEpsilon)
83 _state.Velocity = Vector2.Zero;
84
85 // move in the mover's space
86 var pos = _mover.GetPosition();
87 var newPos = pos + _state.Velocity * dt;
88
89 // optional wrapping if mover lives in Grid space
90 if (_mover.PositionSpace == MovementSpace.Grid && (WrapX || WrapY))
91 {
92 if (_coords is null || _sceneLayer is null)
93 throw new InvalidOperationException("Grid wrapping requires coordinates/layer.");
94
95 var wrapped = _coords.FindEquivalentSceneLayerCoordinates(
96 new PointF(newPos.X, newPos.Y),
97 _sceneLayer.GridColumnCount - 1, _sceneLayer.GridRowCount - 1);
98
99 float newX = newPos.X;
100 float newY = newPos.Y;
101
102 if (WrapX)
103 newX = wrapped.X;
104
105 if (WrapY)
106 newY = wrapped.Y;
107
108 newPos = new Vector2(newX, newY);
109 }
110
111 _mover.SetPosition(newPos);
112 }
