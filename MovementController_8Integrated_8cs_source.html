Gondwana Game Engine v2.2.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

65 private void Step(float dt)
66 {
67 // integrate kinematics in the mover's own space
68 _state.Velocity += _state.Acceleration * dt;
69 ClampVelocity();
70
71 // exponential damping (frame-rate independent)
72 if (_state.LinearDamping > 0f)
73 _state.Velocity *= MathF.Exp(-_state.LinearDamping * dt);
74
75 // kill tiny velocities when no accel
76 const float StopEpsilon = 0.01f;
77 if (_state.Acceleration == Vector2.Zero &&
78 _state.Velocity.LengthSquared() < StopEpsilon * StopEpsilon)
79 _state.Velocity = Vector2.Zero;
80
81 // move in the mover's space
82 var pos = _mover.GetPosition();
83 var newPos = pos + _state.Velocity * dt;
84
85 // optional wrapping if mover lives in Grid space
86 if (_mover.PositionSpace == MovementSpace.Grid && (WrapX || WrapY))
87 {
88 if (_coords is null || _sceneLayer is null)
89 throw new InvalidOperationException("Grid wrapping requires coordinates/layer.");
90
91 var wrapped = _coords.FindEquivalentSceneLayerCoordinates(
92 new PointF(newPos.X, newPos.Y),
93 _sceneLayer.GridColumnCount - 1, _sceneLayer.GridRowCount - 1);
94
95 float newX = newPos.X;
96 float newY = newPos.Y;
97
98 if (WrapX)
99 newX = wrapped.X;
100
101 if (WrapY)
102 newY = wrapped.Y;
103
104 newPos = new Vector2(newX, newY);
105 }
106
107 _mover.SetPosition(newPos);
108 }
