Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

59 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(SceneLayer sceneLayer, Rectangle worldPixelRange, bool includeOverhang)
60 {
61 var retVal = new List<SceneLayerTile>();
62
63 // 1) Find coarse grid bounds via inverse transform (unchanged)
64 PointF ptUL = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Left, worldPixelRange.Top));
65 PointF ptBR = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Right - 1, worldPixelRange.Bottom - 1));
66
67 int minY = (int)Math.Floor(ptUL.Y) - 1;
68 int maxY = (int)Math.Ceiling(ptBR.Y) + 1;
69 int minX = (int)Math.Floor(ptUL.X) - 1;
70 int maxX = (int)Math.Ceiling(ptBR.X) + 1;
71
72 // 2) Scan candidate grid cells and include if their overhang-aware rect intersects
73 for (int y = minY; y <= maxY; y++)
74 {
75 for (int x = minX; x <= maxX; x++)
76 {
77 var gPt = sceneLayer[x, y];
78 if (gPt == null) continue;
79
80 // Overhang-aware pixel rect
81 var rect = GetPixelRangeForTile(gPt, includeOverhang);
82 if (rect.IntersectsWith(worldPixelRange))
83 retVal.Add(gPt);
84 }
85 }
86
87 return retVal;
88 }
