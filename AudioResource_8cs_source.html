Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

83 private ISampleProvider BuildAudioGraph(WaveStream source, float volume, float pan)
84 {
85 _pan = Math.Clamp(pan, -1f, 1f);
86 ISampleProvider baseProvider = source.ToSampleProvider();
87
88 int ch = baseProvider.WaveFormat.Channels;
89 if (ch < 1)
90 {
91 Engine.Logger.LogWarning(
92 "AudioResource {Key} has invalid channel count: {ChannelCount}", Key, ch);
93
94 // just pass through, no pan stage
95 volumeProvider = new VolumeSampleProvider(baseProvider)
96 {
97 Volume = Math.Clamp(volume, 0f, 1f)
98 };
99
100 return volumeProvider;
101 }
102
103 switch (ch)
104 {
105 case 1:
106 // MONO -> use PanningSampleProvider (expects mono, outputs stereo)
107 monoPanProvider = new PanningSampleProvider(baseProvider)
108 {
109 Pan = Math.Clamp(pan, -1f, 1f)
110 };
111 stereoPanProvider = null;
112 baseProvider = monoPanProvider; // now stereo
113 break;
114
115 case 2:
116 // STEREO -> use StereoSampleProvider for balance/pan
117 stereoPanProvider = new StereoPanSampleProvider(baseProvider);
118 ApplyStereoPan(stereoPanProvider, pan); // set L/R gains from pan
119 monoPanProvider = null;
120 baseProvider = stereoPanProvider; // stays stereo
121 break;
122
123 default:
124 // >2 CH -> pick first two channels, then treat as stereo
125 var mux = new MultiplexingSampleProvider(new[] { baseProvider }, 2);
126 mux.ConnectInputToOutput(0, 0); // channel 0 -> output L
127 mux.ConnectInputToOutput(1, 1); // channel 1 -> output R
128 baseProvider = mux;
129
130 stereoPanProvider = new StereoPanSampleProvider(baseProvider);
131 ApplyStereoPan(stereoPanProvider, pan);
132 monoPanProvider = null;
133 baseProvider = stereoPanProvider;
134 break;
135 }
136
137 // final stage: master volume
138 volumeProvider = new VolumeSampleProvider(baseProvider)
139 {
140 Volume = Math.Clamp(volume, 0f, 1f)
141 };
142
143 return volumeProvider;
144 }

356 internal void ReloadIntoManager(bool forceReload = false)
357 {
358 if (string.IsNullOrWhiteSpace(Key))
359 throw new InvalidOperationException("AudioResource has no Key and cannot be reloaded.");
360
361 var mgr = AudioResourceManager.Instance;
362
363 // If already loaded, just apply settings and bail (idempotent).
364 if (!forceReload && mgr.TryGet(Key, out var existing) && existing is not null)
365 {
366 existing.Volume = Volume;
367 existing.Pan = Pan;
368 existing.IsLooping = IsLooping;
369 return;
370 }
371
372 if (forceReload && mgr.Contains(Key))
373 mgr.Unload(Key); // safe: manager owns the live instance :contentReference[oaicite:2]{index=2}
374
375 // Load from persisted source
376 if (AssetIdentifier is not null && AssetIdentifier.IsValid)
377 {
378 using var s = AssetIdentifier.Data;
379 if (s is null)
380 throw new InvalidOperationException($"Missing asset data for {Key}.");
381
382 mgr.LoadFromStream(Key, s, SourceExtension ?? ".wav", Volume, Pan);
383 }
384 else if (!string.IsNullOrWhiteSpace(SourceFilePath))
385 {
386 mgr.LoadFromFile(Key, SourceFilePath, Volume, Pan);
387 }
388 else
389 {
390 throw new InvalidOperationException($"AudioResource '{Key}' has no persisted source.");
391 }
392
393 // Apply looping after load (LoadFromStream/File sets volume/pan during graph creation)
394 if (mgr.TryGet(Key, out var loaded) && loaded is not null)
395 loaded.IsLooping = IsLooping;
396 }

414 private void HandlePlaybackStopped()
415 {
416 try
417 {
418 if (_stopRequested)
419 {
420 _stopRequested = false;
421 return;
422 }
423
424 bool reachedEnd = waveStream.Position >= waveStream.Length;
425
426 if (IsLooping && reachedEnd)
427 {
428 Play(true);
429 }
430 else
431 {
432 if (PlaybackCompletedAsync is not null)
433 {
434 _ = Task.Run(async () =>
435 {
436 try
437 {
438 await PlaybackCompletedAsync();
439 }
440 catch (Exception ex)
441 {
442 Engine.Logger.LogError(ex, "PlaybackCompletedAsync threw an exception for audio resource: {Key}", Key);
443 }
444 });
445 }
446
447 PlaybackCompleted?.Invoke(this, EventArgs.Empty);
448 }
449 }
450 catch (Exception ex)
451 {
452 Engine.Logger.LogError(ex, "Error during playback completion handling for audio resource: {Key}", Key);
453 }

456 private static void ApplyStereoPan(StereoPanSampleProvider s, float pan)
457 {
458 pan = Math.Clamp(pan, -1f, 1f);
459 // equal-power: map [-1..1] to [0..pi/2]
460 float angle = (pan + 1f) * 0.5f * (float)(Math.PI / 2);
461 s.LeftVolume = MathF.Cos(angle);
462 s.RightVolume = MathF.Sin(angle);
463 }
