Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

233 public virtual DirectComposite Add(IDirectCompositeChild child,
234 bool keepCurrentOffset = true,
235 Vector2? explicitLocalOffsetPx = null)
236 {
237 ObjectDisposedException.ThrowIf(_disposed, this);
238 ArgumentNullException.ThrowIfNull(child);
239
240 if (_children.Contains(child))
241 return this;
242
243 ValidateChildIdentity(child);
244 ValidateParentRelationship(child);
245 ValidateAndAssignTarget(child);
246
247 Vector2 offset = keepCurrentOffset ? child.GetPosition() - new Vector2(_anchor.X, _anchor.Y) : explicitLocalOffsetPx ?? Vector2.Zero;
248
249 RegisterParent(child);
250
251 try
252 {
253 _children.Add(child);
254 _localOffsetPx[child] = offset;
255 child.Disposing += OnChildDisposing;
256
257 child.SetPosition(new Vector2(_anchor.X, _anchor.Y) + offset);
258
259 OnChildAdded(child);
260 }
261 catch
262 {
263 child.Disposing -= OnChildDisposing;
264 _children.Remove(child);
265 _localOffsetPx.Remove(child);
266 UnregisterParent(child);
267 ResetTargetWhenEmpty();
268 throw;
269 }
270
271 return this;
272 }

415 public RectangleF GetDrawLocationScreen(View view)
416 {
417 ArgumentNullException.ThrowIfNull(view);
418
419 if (Mode == DirectDrawingMode.View && !ReferenceEquals(View, view))
420 {
421 return RectangleF.Empty;
422 }
423
424 float minX = float.MaxValue;
425 float minY = float.MaxValue;
426 float maxX = float.MinValue;
427 float maxY = float.MinValue;
428
429 foreach (IDirectCompositeChild child in _children)
430 {
431 if (!child.Visible)
432 continue;
433
434 RectangleF bounds = child.GetDrawLocationScreen(view);
435
436 if (bounds.IsEmpty)
437 continue;
438
439 minX = Math.Min(minX, bounds.Left);
440 minY = Math.Min(minY, bounds.Top);
441 maxX = Math.Max(maxX, bounds.Right);
442 maxY = Math.Max(maxY, bounds.Bottom);
443 }
444
445 if (minX == float.MaxValue)
446 return RectangleF.Empty;
447
448 return RectangleF.FromLTRB(
449 (float)Math.Floor(minX),
450 (float)Math.Floor(minY),
451 (float)Math.Ceiling(maxX),
452 (float)Math.Ceiling(maxY));

574 {

577 float maxX = float.MinValue;
578 float maxY = float.MinValue;

606 private void ValidateChildIdentity(IDirectCompositeChild child)
607 {
608 if (!ReferenceEquals(child.RenderSurfaceHost, RenderSurfaceHost))
609 {
610 throw new ArgumentException("The child must belong to the same RenderSurfaceHost as the composite.", nameof(child));
611 }
612
613 if (child.Mode != Mode)
614 {
615 throw new ArgumentException($"The child drawing mode '{child.Mode}' does not match the composite drawing mode '{Mode}'.", nameof(child));
616 }
617 }

619 private void ValidateAndAssignTarget(IDirectCompositeChild child)
620 {
621 if (Mode == DirectDrawingMode.SceneLayer)
622 {
623 SceneLayer childLayer = child.SceneLayer ?? throw new ArgumentException("A scene-layer child must reference a SceneLayer.", nameof(child));
624
625 if (SceneLayer is null)
626 {
627 SceneLayer = childLayer;
628 }
629 else if (!ReferenceEquals(SceneLayer, childLayer))
630 {
631 throw new ArgumentException("All scene-layer children in a composite must reference the same SceneLayer.", nameof(child));
632 }
633
634 return;
635 }
636
637 if (child.View is null)
638 {
639 if (child is DirectComposite compositeChild &&
640 compositeChild.Children.Count == 0)
641 {
642 throw new ArgumentException(
643 "An empty composite cannot be added in view mode because it has no View yet. Add at least one descendant first.",
644 nameof(child));
645 }
646
647 throw new ArgumentException("A view child must reference a View.", nameof(child));
648 }
649
650 View childView = child.View;
651 if (View is null)
652 {
653 View = childView;
654 }
655 else if (!ReferenceEquals(View, childView))
656 {
657 throw new ArgumentException("All view children in a composite must reference the same View.", nameof(child));
658 }
659 }

661 private void ValidateParentRelationship(IDirectCompositeChild child)
662 {
663 lock (_parentSyncRoot)
664 {
665 if (_parents.TryGetValue(child, out DirectComposite? existingParent))
666 {
667 throw new InvalidOperationException($"The child already belongs to composite '{existingParent.Nickname}'.");
668 }
669
670 if (child is not DirectComposite childComposite)
671 return;
672
673 DirectComposite? ancestor = this;
674
675 while (ancestor is not null)
676 {
677 if (ReferenceEquals(ancestor, childComposite))
678 {
679 throw new InvalidOperationException("Adding the child would create a recursive composite cycle.");
680 }
681
682 _parents.TryGetValue(ancestor, out ancestor);
683 }
684 }
685 }

687 private void RegisterParent(IDirectCompositeChild child)
688 {
689 lock (_parentSyncRoot)
690 _parents.Add(child, this);
691 }

693 private void UnregisterParent(IDirectCompositeChild child)
694 {
695 lock (_parentSyncRoot)
696 {
697 if (_parents.TryGetValue(child, out DirectComposite? parent) &&
698 ReferenceEquals(parent, this))
699 {
700 _parents.Remove(child);
701 }
702 }
