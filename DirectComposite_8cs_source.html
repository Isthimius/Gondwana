Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

227 public DirectComposite Add(
228 IDirectCompositeChild child,
229 bool keepCurrentOffset = true,
230 Vector2? explicitLocalOffsetPx = null)
231 {
232 ObjectDisposedException.ThrowIf(_disposed, this);
233 ArgumentNullException.ThrowIfNull(child);
234
235 if (_children.Contains(child))
236 return this;
237
238 ValidateChildIdentity(child);
239 ValidateParentRelationship(child);
240 ValidateAndAssignTarget(child);
241
242 Vector2 offset = keepCurrentOffset
243 ? child.GetPosition() - new Vector2(_anchor.X, _anchor.Y)
244 : explicitLocalOffsetPx ?? Vector2.Zero;
245
246 RegisterParent(child);
247
248 try
249 {
250 _children.Add(child);
251 _localOffsetPx[child] = offset;
252 child.Disposing += OnChildDisposing;
253
254 child.SetPosition(
255 new Vector2(_anchor.X, _anchor.Y) + offset);
256 }
257 catch
258 {
259 child.Disposing -= OnChildDisposing;
260 _children.Remove(child);
261 _localOffsetPx.Remove(child);
262 UnregisterParent(child);
263 ResetTargetWhenEmpty();
264 throw;
265 }
266
267 return this;
268 }

585 {
586 if (Mode == DirectDrawingMode.SceneLayer)

640 private void ValidateParentRelationship(IDirectCompositeChild child)
641 {
642 lock (_parentSyncRoot)
643 {
644 if (_parents.TryGetValue(
645 child,
646 out DirectComposite? existingParent))
647 {
648 throw new InvalidOperationException(
649 $"The child already belongs to composite '{existingParent.Nickname}'.");
650 }
651
652 if (child is not DirectComposite childComposite)
653 return;
654
655 DirectComposite? ancestor = this;
656
657 while (ancestor is not null)
658 {
659 if (ReferenceEquals(
660 ancestor,
661 childComposite))
662 {
663 throw new InvalidOperationException(
664 "Adding the child would create a recursive composite cycle.");
665 }
666
667 _parents.TryGetValue(
668 ancestor,
669 out ancestor);
670 }
671 }
672 }
