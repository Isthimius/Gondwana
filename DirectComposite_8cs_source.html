Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

344 public DirectComposite Add(
345 DirectDrawingMovableBase child,
346 bool keepCurrentOffset = true,
347 Vector2? explicitLocalOffsetPx = null)
348 {
349 ArgumentNullException.ThrowIfNull(child);
350
351 if (!ReferenceEquals(child.RenderSurfaceHost, RenderSurfaceHost))
352 {
353 throw new ArgumentException(
354 "The child must belong to the same RenderSurfaceHost as the composite.",
355 nameof(child));
356 }
357
358 if (child.Mode != Mode)
359 {
360 throw new ArgumentException(
361 $"The child drawing mode '{child.Mode}' does not match " +
362 $"the composite drawing mode '{Mode}'.",
363 nameof(child));
364 }
365
366 ValidateAndAssignTarget(child);
367
368 if (_children.Contains(child))
369 return this;
370
371 Vector2 offset = keepCurrentOffset
372 ? child.GetPosition() - new Vector2(_anchor.X, _anchor.Y)
373 : explicitLocalOffsetPx ?? Vector2.Zero;
374
375 _children.Add(child);
376 _localOffsetPx[child] = offset;
377 child.Disposing += OnChildDisposing;
378
379 child.SetPosition(
380 new Vector2(_anchor.X, _anchor.Y) + offset);
381
382 return this;
383 }
384

391 {
392 ArgumentNullException.ThrowIfNull(child);
393
394 if (_children.Remove(child))

396 child.Disposing -= OnChildDisposing;
397 _localOffsetPx.Remove(child);
398 ResetTargetWhenEmpty();
399 }
400
401 return this;
402 }

585 private void ValidateAndAssignTarget(
586 DirectDrawingMovableBase child)
587 {
588 if (Mode == DirectDrawingMode.SceneLayer)
589 {
590 SceneLayer childLayer = child.SceneLayer
591 ?? throw new ArgumentException(
592 "A scene-layer child must reference a SceneLayer.",
593 nameof(child));
594
595 if (SceneLayer is null)
596 {
597 SceneLayer = childLayer;
598 }
599 else if (!ReferenceEquals(SceneLayer, childLayer))
600 {
601 throw new ArgumentException(
602 "All scene-layer children in a composite must reference the same SceneLayer.",
603 nameof(child));
604 }
605
606 return;
607 }
608
609 View childView = child.View
610 ?? throw new ArgumentException(
611 "A view child must reference a View.",
612 nameof(child));
613
614 if (View is null)
615 {
616 View = childView;
617 }
618 else if (!ReferenceEquals(View, childView))
619 {
620 throw new ArgumentException(
621 "All view children in a composite must reference the same View.",
622 nameof(child));
623 }
624 }
