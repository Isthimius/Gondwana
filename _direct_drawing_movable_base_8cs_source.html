Gondwana Game Engine @PROJECT_VERSION@
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

246 public override void Update(long tick)
247 {
248 if (tick <= _lastTick)
249 return;
250
251 // 1) clamp giant stalls (alt-tab, debugger break, GC, etc.)
252 const float MaxFrameDt = 1f / 15f; // ~66ms
253 float dt = MathF.Min(HighResTimer.GetDuration(_lastTick, tick), MaxFrameDt);
254
255 // 2) accumulate time
256 _accum += dt;
257
258 int steps = 0;
259 while (_accum >= _fixedDt && steps < _maxSubsteps)
260 {
261 // Always integrate at the fixed step (not dt!)
262 Movement.AdvanceMovement(_fixedDt);
263
264 _accum -= _fixedDt;
265 steps++;
266 }
267
268 // 3) if we hit the cap, drop remainder so we don't spiral
269 if (steps == _maxSubsteps)
270 _accum = 0f;
271
272 base.Update(tick);
273 }
