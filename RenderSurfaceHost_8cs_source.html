Gondwana Game Engine v2.2.4
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

196 internal override void RenderToBackbuffer(long tick)
197 {
198 RenderBackbufferBegin?.Invoke();
199
200 // For GL-thread-rendered backbuffers (GpuBackbuffer), the RefreshQueue mechanism is
201 // unreliable: AddWorldRect and ClearRefreshQueue both post to the engine thread, so
202 // enqueued rects are not present when CollectDirtyScreenArea runs, and posted clears
203 // can silently discard dirty rects that were added after the last GL frame.
204 // Instead, always re-render the entire surface each GL paint callback.
205 if (Backbuffer.IsGlThreadRendered)
206 {
207 RenderToBackbufferGpuFull(tick);
208 RenderBackbufferEnd?.Invoke();
209 return;
210 }
211
212 // 0) If there are no visible SceneLayers, just clear and publish the full frame.
213 if (Scene.CountOfVisibleLayers == 0)
214 {
215 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
216 Scene.FullRefreshNeeded = false;
217 }
218 else
219 {
220 // 1) Handle full scene refresh once (camera moved, zoom changed, etc.): clear and mark all layers as dirty.
221 // This already clears the whole backbuffer and enqueues a full rect per layer.
222 if (Scene.FullRefreshNeeded)
223 {
224 // this will mark the Scene.IsDirty flag as true
225 EnqueueFullSceneRefresh();
226 }
227 else
228 {
229 // scene is not dirty, this frame is done...
230 if (!Scene.IsDirty)
231 {
232 RenderBackbufferNoOp?.Invoke();
233 return;
234 }
235 }
236 }
237
238 // 2) Render all views to Backbuffer. Draw layers back -> front (ascending Z).
239 foreach (var view in ViewManager.Views)
240 {
241 RenderContext.Push(view, tick);
242
243 try
244 {
245 // 2.1) Force-refresh all DirectDrawings that overlay this view
246 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
247 foreach (var overlay in overlays)
248 {
249 overlay.ForceRefresh();
250 }
251
252 // 2.2) identify dirty SCREEN areas across all layers for this view
253 var dirtyScreenRects = CollectDirtyScreenArea(view);
254
255 // 2.3) Clip to this view's viewport
256 var vp = view.Viewport.TargetRectPx;
257
258 // clip inclusive to current viewport
259 Backbuffer.Canvas.Save();
260 Backbuffer.Canvas.ResetMatrix();
261 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
262
263 // clip exclusive to higher Z-order views
264 foreach (var blocker in ViewManager.GetViewsAbove(view))
265 {
266 var overlap = Rectangle.Intersect(vp, blocker.Viewport.TargetRectPx);
267 if (!overlap.IsEmpty)
268 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
269 }
270
271 // 2.4) Pre-clear dirty areas on backbuffer to Backbuffer.ClearColor
272 PreclearScreenAreas(view, dirtyScreenRects);
273
274 // 2.5) Render each visible layer's dirty regions for this view
275 // this will draw SceneLayerTiles, Sprites, and SceneLayer-based DirectDrawings
276 var sceneLayers = Scene.VisibleSceneLayers;
277
278 for (int i = 0; i < sceneLayers.Count; i++)
279 {
280 var layer = sceneLayers[i];
281 RenderLayerDirtyRegions(view, layer);
282 }
283
284 // 2.6) draw all View-based DirectDrawings for this view
285 for (int i = 0; i < overlays.Count; i++)
286 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
287
288 // 2.7) Restore from viewport clip
289 Backbuffer.Canvas.Restore();
290 }
291 finally
292 {
293 RenderContext.Pop();
294 }
295 }
296
297 // 3) Clear layer queues now that we've consumed them (avoids re-drawing same tiles next frame)
298 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
299 Scene.VisibleSceneLayers[i].RefreshQueue.ClearRefreshQueue();
300
301 Scene.FullRefreshNeeded = false;
302
303 RenderBackbufferEnd?.Invoke();
304 }

336 private void RenderToBackbufferGpuFull(long tick)
337 {
338 if (Scene.CountOfVisibleLayers == 0)
339 {
340 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
341 Scene.FullRefreshNeeded = false;
342 return;
343 }
344
345 foreach (var view in ViewManager.Views)
346 {
347 RenderContext.Push(view, tick);
348
349 try
350 {
351 // 1) Force-refresh all DirectDrawings that overlay this view.
352 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
353 foreach (var overlay in overlays)
354 overlay.ForceRefresh();
355
356 var vp = view.Viewport.TargetRectPx;
357
358 // 2) Clip to this view's viewport, excluding areas covered by higher Z-order views.
359 Backbuffer.Canvas.Save();
360 Backbuffer.Canvas.ResetMatrix();
361 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
362
363 foreach (var blocker in ViewManager.GetViewsAbove(view))
364 {
365 var overlap = Rectangle.Intersect(vp, blocker.Viewport.TargetRectPx);
366 if (!overlap.IsEmpty)
367 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
368 }
369
370 // 3) Clear the full viewport.
371 Backbuffer.ClearRect(vp);
372
373 // 4) Render every visible layer for the full viewport extent (layers are drawn
374 // back-to-front by ascending Z-order, which VisibleSceneLayers already provides).
375 var sceneLayers = Scene.VisibleSceneLayers;
376
377 for (int i = 0; i < sceneLayers.Count; i++)
378 {
379 var layer = sceneLayers[i];
380
381 // Compute the world-space rect visible through this viewport for this layer,
382 // expanded by one tile in each direction to cover boundary rounding.
383 var layerWorldRectF = view.ScreenRectToWorldRect(layer, vp);
384 layerWorldRectF.Inflate(layer.TileWidth, layer.TileHeight);
385 var layerWorldRect = layerWorldRectF.ToPixelAlignedRect();
386
387 var drawables = layer.GetDrawablesInWorldRect(layerWorldRect);
388 Backbuffer.DrawDrawables(view, drawables, vp);
389 }
390
391 // 5) Render view-based DirectDrawings on top.
392 for (int i = 0; i < overlays.Count; i++)
393 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
394
395 Backbuffer.Canvas.Restore();
396 }
397 finally
398 {
399 RenderContext.Pop();
400 }
401 }
402
403 Scene.FullRefreshNeeded = false;
404 }

408 private void EnqueueFullSceneRefresh()
409 {
410 foreach (var view in ViewManager.Views)
411 {
412 var viewport = view.Viewport;
413 var screenRect = viewport.TargetRectPx;
414
415 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
416 {
417 var layer = Scene.VisibleSceneLayers[i];
418
419 // 1) Compute layer-specific visible world rect
420 var layerWorldRect = view.ScreenRectToWorldRect(layer, screenRect);
421
422 // 2) Expand world rect by **one full tile** in all directions.
423 // This compensates for fractional-layer motion (parallax),
424 // camera motion, and tile boundaries.
425 int expandX = layer.TileWidth;
426 int expandY = layer.TileHeight;
427
428 layerWorldRect.Inflate(expandX, expandY);
429
430 // 3) Round to ints and enqueue
431 var worldRectInt = layerWorldRect.ToPixelAlignedRect();
432
433 layer.RefreshQueue.AddWorldRect(worldRectInt);
434 }
435 }
436 }

438 private List<Rectangle> CollectDirtyScreenArea(View view)
439 {
440 var dirty = new List<Rectangle>(64);
441 var viewportRect = view.Viewport.TargetRectPx;
442
443 foreach (var sceneLayer in Scene.VisibleSceneLayers)
444 {
445 var refreshQueue = sceneLayer.RefreshQueue;
446 if (!refreshQueue.IsDirty)
447 continue;
448
449 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
450 {
451 var screenRectF = view.WorldRectToScreenRect(sceneLayer, worldRect);
452 var rect = Rectangle.Intersect(
453 screenRectF.ToPixelAlignedRect(),
454 viewportRect);
455
456 if (rect.IsEmpty)
457 continue;
458
459 dirty.AddDeduped(rect);
460 }
461 }
462
463 return dirty;
464 }

466 private void PreclearScreenAreas(View view, List<Rectangle> screenRects)
467 {
468 if (Backbuffer is null || screenRects is null || screenRects.Count == 0)
469 return;
470
471 foreach (var screenRect in screenRects)
472 {
473 var screenRectViewport = Rectangle.Intersect(screenRect, view.Viewport.TargetRectPx);
474
475 if (screenRectViewport.IsEmpty || screenRectViewport.Width <= 0 || screenRectViewport.Height <= 0)
476 continue;
477
478 // Clear just this patch (overwrite with Backbuffer.ClearColor)
479 Backbuffer.ClearRect(screenRectViewport);
480
481 EnqueueForOverlappingSceneLayers(view, screenRectViewport);
482 }
483 }
484
485 private void EnqueueForOverlappingSceneLayers(View sourceView, Rectangle clearedScreenRect)
486 {
487 var overlap = Rectangle.Intersect(clearedScreenRect, sourceView.Viewport.TargetRectPx);
488
489 if (overlap.IsEmpty)
490 return;
491
492 foreach (var sceneLayer in Scene.VisibleSceneLayers)
493 {
494 var worldRectF = sourceView.ScreenRectToWorldRect(sceneLayer, overlap);
495 var worldRect = worldRectF.ToPixelAlignedRect();
496
497 if (!worldRect.IsEmpty)
498 sceneLayer.RefreshQueue.AddWorldRect(worldRect);
499 }
500 }

502 private void RenderLayerDirtyRegions(View view, SceneLayer layer)
503 {
504 var refreshQueue = layer.RefreshQueue;
505
506 // if this layer has no dirty regions and we are not forcing a full redraw, skip it.
507 if (!refreshQueue.IsDirty)
508 return;
509
510 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
511 {
512 // draw tiles/sprites/direct drawings in this world rect
513 var drawables = layer.GetDrawablesInWorldRect(worldRect);
514
515 // project world → screen for adapter dirty
516 var screenRect = view.WorldRectToScreenRect(layer, worldRect).ToPixelAlignedRect();
517
518 Backbuffer.DrawDrawables(view, drawables, screenRect);
519 }
520 }

606 private void OnRenderSurfaceAdapterResized(RenderSurfaceAdapterResizedEventArgs args)
607 {
608 if (Scene != null)
609 Scene.FullRefreshNeeded = true; // full redraw next frame
610
611 _backbuffer?.RequestResize(args.NewWidth, args.NewHeight); // UI thread → request only
612
613 float scaleX = (float)args.NewWidth / args.OldWidth;
614 float scaleY = (float)args.NewHeight / args.OldHeight;
615
616 // resize each View proportionally
617 foreach (var view in ViewManager.Views)
618 {
619 var old = view.Viewport.TargetRectPx;
620
621 int newLeft = (int)Math.Round(old.Left * scaleX);
622 int newTop = (int)Math.Round(old.Top * scaleY);
623 int newWidth = (int)Math.Round(old.Width * scaleX);
624 int newHeight = (int)Math.Round(old.Height * scaleY);
625
626 view.Viewport.TargetRectPx = new Rectangle(
627 newLeft, newTop, newWidth, newHeight);
628 }
629 }

645 {

659
660 // Post to UI thread
