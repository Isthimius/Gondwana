Gondwana Game Engine v2.4.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

234 internal override void RenderToBackbuffer(long tick)
235 {
236 RenderBackbufferBegin?.Invoke();
237
238 // For GL-thread-rendered backbuffers (GpuBackbuffer), the RefreshQueue mechanism is
239 // unreliable: AddWorldRect and ClearRefreshQueue both post to the engine thread, so
240 // enqueued rects are not present when CollectDirtyScreenArea runs, and posted clears
241 // can silently discard dirty rects that were added after the last GL frame.
242 // Instead, always re-render the entire surface each GL paint callback.
243 if (Backbuffer.IsGlThreadRendered)
244 {
245 RenderToBackbufferGpuFull(tick);
246 RenderBackbufferEnd?.Invoke();
247 return;
248 }
249
250 if (ViewManager.Views.Count == 0)
251 {
252 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
253 Scene.FullRefreshNeeded = false;
254 RenderBackbufferEnd?.Invoke();
255 return;
256 }
257
258 // 0) If there are no visible SceneLayers, just clear and publish the full frame.
259 if (Scene.CountOfVisibleLayers == 0)
260 {
261 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
262 Scene.FullRefreshNeeded = false;
263 }
264 else
265 {
266 // 1) Handle full scene refresh once (camera moved, zoom changed, etc.): clear and mark all layers as dirty.
267 // This already clears the whole backbuffer and enqueues a full rect per layer.
268 if (Scene.FullRefreshNeeded)
269 {
270 // this will mark the Scene.IsDirty flag as true
271 EnqueueFullSceneRefresh();
272 }
273 else
274 {
275 // scene is not dirty, this frame is done...
276 if (!Scene.IsDirty)
277 {
278 RenderBackbufferNoOp?.Invoke();
279 return;
280 }
281 }
282 }
283
284 // 2) Render all views to Backbuffer. Draw layers back -> front (ascending Z).
285 foreach (var view in ViewManager.Views)
286 {
287 RenderContext.Push(view, tick);
288
289 try
290 {
291 // 2.1) Force-refresh all DirectDrawings that overlay this view
292 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
293 foreach (var overlay in overlays)
294 {
295 overlay.ForceRefresh();
296 }
297
298 // 2.2) identify dirty SCREEN areas across all layers for this view
299 var dirtyScreenRects = CollectDirtyScreenArea(view);
300
301 // 2.3) Clip to this view's viewport
302 var vp = view.Viewport.TargetRectPx;
303
304 // clip inclusive to current viewport
305 Backbuffer.Canvas.Save();
306 Backbuffer.Canvas.ResetMatrix();
307 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
308
309 // clip exclusive to higher Z-order views
310 foreach (var blocker in ViewManager.GetViewsAbove(view))
311 {
312 var overlap = Rectangle.Intersect(vp, blocker.Viewport.TargetRectPx);
313 if (!overlap.IsEmpty)
314 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
315 }
316
317 // 2.4) Pre-clear dirty areas on backbuffer to Backbuffer.ClearColor
318 PreclearScreenAreas(view, dirtyScreenRects);
319
320 // 2.5) Render each visible layer's dirty regions for this view
321 // this will draw SceneLayerTiles, Sprites, and SceneLayer-based DirectDrawings
322 var sceneLayers = Scene.VisibleSceneLayers;
323
324 for (int i = 0; i < sceneLayers.Count; i++)
325 {
326 var layer = sceneLayers[i];
327 RenderLayerDirtyRegions(view, layer);
328 }
329
330 // 2.6) draw all View-based DirectDrawings for this view
331 for (int i = 0; i < overlays.Count; i++)
332 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
333
334 // 2.7) Restore from viewport clip
335 Backbuffer.Canvas.Restore();
336 }
337 finally
338 {
339 RenderContext.Pop();
340 }
341 }
342
343 // 3) Clear layer queues now that we've consumed them (avoids re-drawing same tiles next frame)
344 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
345 Scene.VisibleSceneLayers[i].RefreshQueue.ClearRefreshQueue();
346
347 Scene.FullRefreshNeeded = false;
348
349 // Notify subscribers that all scene content has been drawn and the canvas is ready for
350 // post-scene effects. Fires before RenderBackbufferEnd so subscribers still have a
351 // chance to draw into the canvas before it is presented to the adapter.
352 InvokePostSceneCanvasHooks();
353
354 RenderBackbufferEnd?.Invoke();
355 }

387 private void RenderToBackbufferGpuFull(long tick)
388 {
389 // When there are no views at all, clear the whole surface and bail.
390 // If there ARE views but no scene layers, we fall through so view-mode
391 // DirectDrawings (e.g. a splash screen overlay) are still rendered.
392 if (ViewManager.Views.Count == 0)
393 {
394 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
395 Scene.FullRefreshNeeded = false;
396 return;
397 }
398
399 foreach (var view in ViewManager.Views)
400 {
401 RenderContext.Push(view, tick);
402
403 try
404 {
405 // 1) Force-refresh all DirectDrawings that overlay this view.
406 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
407 foreach (var overlay in overlays)
408 overlay.ForceRefresh();
409
410 var vp = view.Viewport.TargetRectPx;
411
412 // 2) Clip to this view's viewport, excluding areas covered by higher Z-order views.
413 Backbuffer.Canvas.Save();
414 Backbuffer.Canvas.ResetMatrix();
415 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
416
417 foreach (var blocker in ViewManager.GetViewsAbove(view))
418 {
419 var overlap = Rectangle.Intersect(vp, blocker.Viewport.TargetRectPx);
420 if (!overlap.IsEmpty)
421 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
422 }
423
424 // 3) Clear the full viewport.
425 Backbuffer.ClearRect(vp);
426
427 // 4) Render every visible layer for the full viewport extent (layers are drawn
428 // back-to-front by ascending Z-order, which VisibleSceneLayers already provides).
429 var sceneLayers = Scene.VisibleSceneLayers;
430
431 for (int i = 0; i < sceneLayers.Count; i++)
432 {
433 var layer = sceneLayers[i];
434
435 // Compute the world-space rect visible through this viewport for this layer,
436 // expanded by one tile in each direction to cover boundary rounding.
437 var layerWorldRectF = view.ScreenRectToWorldRect(layer, vp);
438 layerWorldRectF.Inflate(layer.TileWidth, layer.TileHeight);
439 var layerWorldRect = layerWorldRectF.ToPixelAlignedRect();
440
441 var drawables = layer.GetDrawablesInWorldRect(layerWorldRect);
442 Backbuffer.DrawDrawables(view, drawables, vp);
443 }
444
445 // 5) Render view-based DirectDrawings on top.
446 for (int i = 0; i < overlays.Count; i++)
447 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
448
449 Backbuffer.Canvas.Restore();
450 }
451 finally
452 {
453 RenderContext.Pop();
454 }
455 }
456
457 Scene.FullRefreshNeeded = false;
458
459 // Notify subscribers that all scene content has been drawn and the canvas is ready for
460 // post-scene effects. For GPU surfaces this runs on the GL thread while GRContext is
461 // current, so subscribers may safely issue Skia GPU draw calls.
462 InvokePostSceneCanvasHooks();
463 }

467 private void EnqueueFullSceneRefresh()
468 {
469 foreach (var view in ViewManager.Views)
470 {
471 var viewport = view.Viewport;
472 var screenRect = viewport.TargetRectPx;
473
474 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
475 {
476 var layer = Scene.VisibleSceneLayers[i];
477
478 // 1) Compute layer-specific visible world rect
479 var layerWorldRect = view.ScreenRectToWorldRect(layer, screenRect);
480
481 // 2) Expand world rect by **one full tile** in all directions.
482 // This compensates for fractional-layer motion (parallax),
483 // camera motion, and tile boundaries.
484 int expandX = layer.TileWidth;
485 int expandY = layer.TileHeight;
486
487 layerWorldRect.Inflate(expandX, expandY);
488
489 // 3) Round to ints and enqueue
490 var worldRectInt = layerWorldRect.ToPixelAlignedRect();
491
492 layer.RefreshQueue.AddWorldRect(worldRectInt);
493 }
494 }
495 }

497 private List<Rectangle> CollectDirtyScreenArea(View view)
498 {
499 var dirty = new List<Rectangle>(64);
500 var viewportRect = view.Viewport.TargetRectPx;
501
502 foreach (var sceneLayer in Scene.VisibleSceneLayers)
503 {
504 var refreshQueue = sceneLayer.RefreshQueue;
505 if (!refreshQueue.IsDirty)
506 continue;
507
508 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
509 {
510 var screenRectF = view.WorldRectToScreenRect(sceneLayer, worldRect);
511 var rect = Rectangle.Intersect(
512 screenRectF.ToPixelAlignedRect(),
513 viewportRect);
514
515 if (rect.IsEmpty)
516 continue;
517
518 dirty.AddDeduped(rect);
519 }
520 }
521
522 return dirty;
523 }

525 private void PreclearScreenAreas(View view, List<Rectangle> screenRects)
526 {
527 if (Backbuffer is null || screenRects is null || screenRects.Count == 0)
528 return;
529
530 foreach (var screenRect in screenRects)
531 {
532 var screenRectViewport = Rectangle.Intersect(screenRect, view.Viewport.TargetRectPx);
533
534 if (screenRectViewport.IsEmpty || screenRectViewport.Width <= 0 || screenRectViewport.Height <= 0)
535 continue;
536
537 // Clear just this patch (overwrite with Backbuffer.ClearColor)
538 Backbuffer.ClearRect(screenRectViewport);
539
540 EnqueueForOverlappingSceneLayers(view, screenRectViewport);
541 }
542 }
543
544 private void EnqueueForOverlappingSceneLayers(View sourceView, Rectangle clearedScreenRect)
545 {
546 var overlap = Rectangle.Intersect(clearedScreenRect, sourceView.Viewport.TargetRectPx);
547
548 if (overlap.IsEmpty)
549 return;
550
551 foreach (var sceneLayer in Scene.VisibleSceneLayers)
552 {
553 var worldRectF = sourceView.ScreenRectToWorldRect(sceneLayer, overlap);
554 var worldRect = worldRectF.ToPixelAlignedRect();
555
556 if (!worldRect.IsEmpty)
557 sceneLayer.RefreshQueue.AddWorldRect(worldRect);
558 }
559 }

561 private void RenderLayerDirtyRegions(View view, SceneLayer layer)
562 {
563 var refreshQueue = layer.RefreshQueue;
564
565 // if this layer has no dirty regions and we are not forcing a full redraw, skip it.
566 if (!refreshQueue.IsDirty)
567 return;
568
569 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
570 {
571 // draw tiles/sprites/direct drawings in this world rect
572 var drawables = layer.GetDrawablesInWorldRect(worldRect);
573
574 // project world → screen for adapter dirty
575 var screenRect = view.WorldRectToScreenRect(layer, worldRect).ToPixelAlignedRect();
576
577 Backbuffer.DrawDrawables(view, drawables, screenRect);
578 }
579 }

665 private void OnRenderSurfaceAdapterResized(RenderSurfaceAdapterResizedEventArgs args)
666 {
667 if (Scene != null)
668 Scene.FullRefreshNeeded = true; // full redraw next frame
669
670 _backbuffer?.RequestResize(args.NewWidth, args.NewHeight); // UI thread → request only
671
672 float scaleX = (float)args.NewWidth / args.OldWidth;
673 float scaleY = (float)args.NewHeight / args.OldHeight;
674
675 // resize each View proportionally
676 foreach (var view in ViewManager.Views)
677 {
678 var old = view.Viewport.TargetRectPx;
679
680 int newLeft = (int)Math.Round(old.Left * scaleX);
681 int newTop = (int)Math.Round(old.Top * scaleY);
682 int newWidth = (int)Math.Round(old.Width * scaleX);
683 int newHeight = (int)Math.Round(old.Height * scaleY);
684
685 view.Viewport.TargetRectPx = new Rectangle(
686 newLeft, newTop, newWidth, newHeight);
687 }
688 }
