Gondwana Game Engine v2.4.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

208 public static void ApplyAlphaMask(this SKBitmap bitmap, SKColor targetColor, byte tolerance = 5)
209 {
210 if (bitmap == null || bitmap.IsEmpty)
211 throw new ArgumentException("Invalid bitmap.");
212
213 int width = bitmap.Width;
214 int height = bitmap.Height;
215

240 public static SKBitmap PremultiplyAlpha(this SKBitmap bitmap)
241 {
242 if (bitmap == null || bitmap.IsEmpty)
243 throw new ArgumentException("Invalid bitmap.");
244
245 if (bitmap.AlphaType == SKAlphaType.Premul)
246 return bitmap;
247
248 var info = new SKImageInfo(bitmap.Width, bitmap.Height, bitmap.ColorType, SKAlphaType.Premul);
249 var premulBitmap = new SKBitmap(info);
250
251 int width = bitmap.Width;
252 int height = bitmap.Height;
253
254 for (int y = 0; y < height; y++)
255 {
256 for (int x = 0; x < width; x++)
257 {
258 var c = bitmap.GetPixel(x, y);
259 byte a = c.Alpha;
260
261 byte r = (byte)((c.Red * a + 127) / 255);
262 byte g = (byte)((c.Green * a + 127) / 255);
263 byte b = (byte)((c.Blue * a + 127) / 255);
264
265 premulBitmap.SetPixel(x, y, new SKColor(r, g, b, a));
266 }
267 }
268
269 premulBitmap.NotifyPixelsChanged();
270 return premulBitmap;
271 }
