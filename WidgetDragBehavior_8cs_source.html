Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

80 Func<Vector2, Vector2> constrainPosition,
81 Action<WidgetDragEventArgs> dragStarted,
82 Action<WidgetDragEventArgs> dragged)
83 {
84 if (!_isPointerDown || args.PointerId != _dragPointerId)
85 return;
86
87 Vector2 totalScreenDeltaPx = GetTotalScreenDelta(args.ScreenPositionPx);
88
89 if (!IsDragging)

92 if (totalScreenDeltaPx.LengthSquared() < thresholdSquared)
93 return;
94
95 IsDragging = true;
96 _suppressNextPointerClick = true;
97
98 dragStarted(CreateDragEventArgs(args));
99
100 // The callback may cancel or hide the widget, which resets this state.
101 if (!IsDragging)
102 return;
103 }
104
105 ApplyDragPosition(totalScreenDeltaPx, _dragView ?? args.View, convertDelta, constrainPosition);
106
107 dragged(CreateDragEventArgs(args));
108 }

110 internal void ProcessPointerUp(WidgetPointerEventArgs args,
111 Func<Vector2, View, Vector2> convertDelta,
112 Func<Vector2, Vector2> constrainPosition,
113 Action<WidgetDragEventArgs> dragEnded)
114 {
115 if (!_isPointerDown || !IsMatchingRelease(args))
116 return;
117
118 WidgetDragEventArgs? dragArgs = null;
119
120 if (IsDragging)
121 {
122 Vector2 totalScreenDeltaPx = GetTotalScreenDelta(args.ScreenPositionPx);
123 ApplyDragPosition(totalScreenDeltaPx, _dragView ?? args.View, convertDelta, constrainPosition);
124
125 dragArgs = CreateDragEventArgs(args);
126 }
127
128 bool completedDrag = IsDragging;
129
130 ResetPointerState();
131
132 if (completedDrag && dragArgs is not null)
133 {
134 _suppressNextPointerClick = true;
135 dragEnded(dragArgs);
136 }
137 }
