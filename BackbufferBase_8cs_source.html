Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

311 internal void DrawDrawables(View view, IEnumerable<IDrawable> drawables, Rectangle clipRect)
312 {
313 Canvas.Save();
314 Canvas.ClipRect(clipRect.ToSKRect());
315
316 var tiles = new List<Tile>();
317
318 foreach (var drawable in drawables)
319 {
320 if (!drawable.Visible)
321 continue;
322
323 var destRectScreen = drawable.GetDrawLocationScreen(view);
324 if (drawable is Sprite sprite)
325 {
326 destRectScreen = sprite.ApplyJiggleToDestRect(destRectScreen);
327 }
328 drawable.Draw(this, destRectScreen);
329
330 RectangleF visualBoundsScreen = drawable is Sprite drawnSprite
331 ? drawnSprite.GetVisualBoundsScreen(destRectScreen)
332 : destRectScreen;
333
334 AddToBackbufferDirtyRectangle(visualBoundsScreen.ToPixelAlignedRect());
335
336 if (drawable is Tile tile)
337 tiles.Add(tile);
338 }
339
340 PostDrawTiles(view, tiles);
341
342 Canvas.Restore();
343 }

345 private void PostDrawTiles(View view, List<Tile> tiles)
346 {
347 foreach (var tile in tiles)
348 {
349 // WORLD -> SCREEN conversion
350 var worldPts = tile.OutlinePointsWorld;
351 var ptsScreen = new SKPoint[worldPts.Length];
352
353 for (int i = 0; i < worldPts.Length; i++)
354 {
355 var p = worldPts[i];
356 var sp = view.WorldPxToScreenPx(
357 tile.SceneLayer,
358 new PointF(p.X, p.Y)
359 );
360
361 ptsScreen[i] = new SKPoint(sp.X, sp.Y);
362 }
363
364 // close polygon when needed
365 static SKPoint[] Enclose(SKPoint[] pts)
366 {
367 if (pts.Length == 0) return pts;
368 var arr = new SKPoint[pts.Length + 1];
369 Array.Copy(pts, arr, pts.Length);
370 arr[^1] = pts[0];
371 return arr;
372 }
373
374 if (tile.EnableFog)
375 {
376 using var path = new SKPath();
377 path.AddPoly(ptsScreen, close: true);
378 Canvas.DrawPath(path, FogPaint);
379 }
380
381 if (tile.SceneLayer.ShowGridLines && tile.Visible && tile.IsPositionFixed)
382 Canvas.DrawPoints(SKPointMode.Polygon, Enclose(ptsScreen), GridLinePaint);
383
384 if (tile.SceneLayer.ShowCollisionBoxes && tile.Visible && tile.CollisionsEnabled)
385 {
386 var colRectScreen = tile.GetCollisionAreaScreen(view).ToSKRect();
387 Canvas.DrawRect(colRectScreen, CollisionBoxPaint);
388 }
389 }
390 }
