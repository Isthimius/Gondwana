Gondwana Game Engine v2.5.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

74 protected override void OnOpenGlInit(GlInterface gl)
75 {
76 // Create a SkiaSharp GRContext from the current OpenGL context.
77 // GRGlInterface.Create() auto-detects the current platform GL/EGL/WGL context
78 // (which Avalonia has made current before this callback) and loads function
79 // pointers from it without requiring any explicit proc-address lookup delegate.
80 _glInterface = GRGlInterface.Create();
81 if (_glInterface == null) return;
82 _grContext = GRContext.CreateGl(_glInterface);
83
84 if (_grContext == null) return;
85
86 var scaling = VisualRoot?.RenderScaling ?? 1.0;
87 var physW = Math.Max(1, (int)Math.Round(Bounds.Width * scaling));
88 var physH = Math.Max(1, (int)Math.Round(Bounds.Height * scaling));
89
90 _lastPhysW = physW;
91 _lastPhysH = physH;
92
93 Adapter.UpdateDimensions(physW, physH);
94 _gpuBackbuffer?.Initialize(_grContext, physW, physH);
95 }

98 protected override void OnOpenGlRender(GlInterface gl, int fb)
99 {
100 if (_grContext == null || _gpuBackbuffer == null) return;
101
102 // Notify SkiaSharp that external GL code (Avalonia's compositor) may have
103 // modified the GL state since the last Skia draw call.
104 _grContext.ResetContext();
105
106 var scaling = VisualRoot?.RenderScaling ?? 1.0;
107 var physW = Math.Max(1, (int)Math.Round(Bounds.Width * scaling));
108 var physH = Math.Max(1, (int)Math.Round(Bounds.Height * scaling));
109
110 // Reinitialise the GPU surface when the control is resized.
111 if (physW != _lastPhysW || physH != _lastPhysH)
112 {
113 _lastPhysW = physW;
114 _lastPhysH = physH;
115 Adapter.UpdateDimensions(physW, physH);
116 _gpuBackbuffer.Initialize(_grContext, physW, physH);
117 }
118
119 // Wrap the Avalonia-provided framebuffer in a SkiaSharp surface for compositing.
120 // The framebuffer uses OpenGL bottom-left origin, RGBA8 color format.
121 var fbInfo = new GRGlFramebufferInfo((uint)fb, GlRgba8);
122 using var renderTarget = new GRBackendRenderTarget(physW, physH, sampleCount: 0, stencilBits: 0, fbInfo);
123 using var fbSurface = SKSurface.Create(_grContext, renderTarget, GRSurfaceOrigin.BottomLeft, SKColorType.Rgba8888);
124
125 if (fbSurface == null) return;
126
127 var canvas = fbSurface.Canvas;
128
129 // Render the scene entirely on the GL thread and blit the result.
130 using var img = Host.GlRenderAndSnapshot();
131 if (img != null)
132 {
133 canvas.DrawImage(img, SKRect.Create(0, 0, physW, physH));
134 }
135 else
136 {
137 canvas.Clear(Adapter.ClearColor);
138 }
139
140 _grContext.Flush();
141 _gpuBackbuffer.RecordFrame();
142 }
