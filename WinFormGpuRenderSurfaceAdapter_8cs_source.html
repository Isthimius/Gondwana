Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

130 public void SetHost(RenderSurfaceHostBase host)
131 {

133
134 // Cache the GpuBackbuffer so OnPaintSurface can read its settings.
135 _gpuBackbuffer = _host.Backbuffer as GpuBackbuffer;
136
137 // Invalidate the GL control on the UI thread after every engine foreground cycle.
138 // Use _pendingInvalidate to ensure at most one Post(Invalidate) is queued: when the engine
139 // cycle rate exceeds the GPU render rate, excess calls are dropped rather than piling
140 // up in the message queue. The flag is cleared at the start of each paint.
141 _afterFrameRenderHandler = () =>
142 {
143 if (!_glControl.IsDisposed
144 && Interlocked.CompareExchange(ref _pendingInvalidate, 1, 0) == 0)
145 Engine.Instance.UiDispatcher!.Post(_glControl.Invalidate);
146 };
147 Engine.Instance.AfterFrameRender += _afterFrameRenderHandler;
148 }

159 private void OnPaintSurface(object? sender, SKPaintGLSurfaceEventArgs e) {
160 // Clear the pending-invalidate flag so the next AfterFrameRender can queue a new one.
161 Interlocked.Exchange(ref _pendingInvalidate, 0);
162
163 // Lazily sync VSync → GLControl.VSync whenever the backbuffer value changes.
164 if (_gpuBackbuffer != null)
165 {
166 bool desiredVSync = _gpuBackbuffer.VSync;
167 if (_appliedVSync != desiredVSync)
168 {
169 _glControl.VSync = desiredVSync;
170 _appliedVSync = desiredVSync;
171 }
172 }
173
174 // Capture/refresh the GRContext so callers can wire the backbuffer to the same one.
175 GrContext ??= _glControl.GRContext;
176
177 if (GrContext != null)
178 {
179 if (!_grContextReady)
180 {
181 // First time the context is available: clear any resize that happened before the
182 // first paint (the Initialize call will use the current adapter dimensions).
183 _grContextReady = true;
184 Interlocked.Exchange(ref _pendingResize, 0);
185 GrContextFirstAvailable?.Invoke(GrContext);
186 }
187 else if (Interlocked.Exchange(ref _pendingResize, 0) == 1)
188 {
189 // Subsequent resize: reinitialize the backbuffer's GPU resources on the GL thread.
190 // WinForms controls can report zero size while minimized or before layout completes;
191 // suppress invalid resize requests so downstream GPU initialization is not attempted
192 // with non-positive dimensions.
193 var width = Width;
194 var height = Height;
195 if (width > 0 && height > 0)
196 ResizeRequested?.Invoke(GrContext, width, height);
197 }
198 }
199
200 var canvas = e.Surface.Canvas;
201
202 // Render + blit entirely on the GL thread.
203 // GlRenderAndSnapshot drives RenderToBackbuffer on the GPU surface then returns a
204 // lightweight GPU-backed snapshot. Both the snapshot texture and e.Surface share the
205 // same GRContext, so DrawImage is a zero-copy GPU blit.
206 if (_host != null)
207 {
208 using var img = _host.GlRenderAndSnapshot();
209 if (img != null)
210 {
211 // DrawImage covers the full render target, so no pre-clear is needed.
212 // Clearing the window surface before blitting would cause a black flash
213 // that is visible when VSync is off (the monitor may scan between the
214 // clear and the blit, seeing the cleared back buffer).
215 var dst = SKRect.Create(0, 0,
216 e.BackendRenderTarget.Width,
217 e.BackendRenderTarget.Height);
218 canvas.DrawImage(img, dst);
219 }
220 else
221 {
222 canvas.Clear(ClearColor);
223 }
224 }
225 else
226 {
227 canvas.Clear(ClearColor);
228 }
229
230 // Optional: flush to ensure work is queued to GPU before we hand new images next frame
231 _glControl.GRContext?.Flush();
232
233 // Record the completed frame so the engine's CPS sampler can compute actual GPU FPS.
234 _gpuBackbuffer?.RecordFrame();
235 }
