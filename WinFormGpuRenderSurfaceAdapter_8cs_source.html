Gondwana Game Engine v2.1.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

59 public WinFormGpuRenderSurfaceAdapter(SKGLControl gl)
60 : base(gl.Width, gl.Height)
61 {
62 _glControl = gl;
63
64 // Store the resize handler in a field so it can be unsubscribed in Dispose().
65 // WinForms can transiently report 0×0 during minimize/layout; avoid committing that size
66 // so downstream resize code (which divides by the previous width/height) never observes a
67 // zero denominator. We skip both the dimension update and the pending-resize flag when the
68 // control reports non-positive dimensions.
69 _resizeHandler = (_, _) =>
70 {
71 var width = _glControl.Width;
72 var height = _glControl.Height;
73
74 if (width > 0 && height > 0)
75 {
76 SetDestinationSize(width, height);
77 Interlocked.Exchange(ref _pendingResize, 1);
78 }
79 };
80
81 // Wire events
82 _glControl.PaintSurface += OnPaintSurface;
83 _glControl.Resize += _resizeHandler;
84
85 // On modern SkiaSharp, SKGLControl exposes GRContext; otherwise capture it in the first paint.
86 GrContext = _glControl.GRContext; // may be null until first paint; we also set in OnPaintSurface
87 }

96 public override void Present(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)
97 {
98 // If the control can't paint, dispose immediately to avoid a leak.
99 if (_glControl.IsDisposed || !_glControl.IsHandleCreated)

102 return;

107 _currentImage = bufferImage;

112 _destRect = destRect;

116 }

118 private static SKRect ToRect(SKRectI r) => new SKRect(r.Left, r.Top, r.Right, r.Bottom);

120 private void OnPaintSurface(object? sender, SKPaintGLSurfaceEventArgs e)
121 {
122 // Capture/refresh the GRContext so callers can wire the backbuffer to the same one.
123 GrContext ??= _glControl.GRContext;
124
125 if (GrContext != null)
126 {
127 if (!_grContextReady)
128 {
129 // First time the context is available: clear any resize that happened before the
130 // first paint (the Initialize call will use the current adapter dimensions).
131 _grContextReady = true;
132 Interlocked.Exchange(ref _pendingResize, 0);
133 GrContextFirstAvailable?.Invoke(GrContext);
134 }
135 else if (Interlocked.Exchange(ref _pendingResize, 0) == 1)
136 {
137 // Subsequent resize: reinitialize the backbuffer's GPU resources on the GL thread.
138 // WinForms controls can report zero size while minimized or before layout completes;
139 // suppress invalid resize requests so downstream GPU initialization is not attempted
140 // with non-positive dimensions.
141 var width = Width;
142 var height = Height;
143 if (width > 0 && height > 0)
144 ResizeRequested?.Invoke(GrContext, width, height);
145 }
146 }
147
148 var canvas = e.Surface.Canvas;
149
150 // Clear with your configured clear color
151 canvas.Clear(ClearColor);
152
153 var img = _currentImage;
154 if (img != null)
155 {
156 // IMPORTANT: For zero-copy, img MUST belong to this same GRContext.
157 // If you hand us a raster image, Skia will upload it each frame (works, but costs bandwidth).
158 var imgBoundsI = new SKRectI(0, 0, img.Width, img.Height);
159 var srcI = SKRectI.Intersect(_sourceRect, imgBoundsI);
160
161 if (!srcI.IsEmpty)
162 {
163 var src = ToRect(srcI);
164 var dest = _destRect;
165
166 var bounds = SKRect.Create(e.BackendRenderTarget.Width, e.BackendRenderTarget.Height);
167 if (!bounds.Contains(dest))
168 dest = SKRect.Intersect(dest, bounds);
169
170 canvas.DrawImage(img, src, dest);
171 }
172 }
173
174 // Safe to free the previously drawn GPU image NOW (on GL thread)
175 _prevToDispose?.Dispose();
176 _prevToDispose = null;
177
178 // Optional: flush to ensure work is queued to GPU before we hand new images next frame
179 _glControl.GRContext?.Flush();
180 }
181
185 public void Dispose()
186 {
187 if (!_glControl.IsDisposed)
188 {
189 _glControl.PaintSurface -= OnPaintSurface;
190 _glControl.Resize -= _resizeHandler;
191 }
192
193 // These MUST be disposed on the GL thread; do a last paint if needed.
194 // If you're disposing off the UI thread at shutdown, it's usually fine
195 // because the context is being torn down; still, we null them here.
196 _prevToDispose?.Dispose();
197 _currentImage?.Dispose();
198 _prevToDispose = null;
199 _currentImage = null;
200 }
