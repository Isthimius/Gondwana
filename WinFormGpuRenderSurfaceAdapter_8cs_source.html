Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

60 public override void Present(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)
61 {
62 // If the control can't paint, dispose immediately to avoid a leak.
63 if (_glControl.IsDisposed || !_glControl.IsHandleCreated)
64 {
65 bufferImage.Dispose();
66 return;
67 }
68
69 // Swap
70 var old = _currentImage;
71 _currentImage = bufferImage;
72 if (!ReferenceEquals(old, _currentImage) && old is not null)
73 _prevToDispose = old;
74
75 _sourceRect = bufferRect;
76 _destRect = destRect;
77
78 // Kick the GL paint; this runs on the GL thread/context
79 _glControl.Invalidate();
80 }

82 private static SKRect ToRect(SKRectI r) => new SKRect(r.Left, r.Top, r.Right, r.Bottom);

84 private void OnPaintSurface(object? sender, SKPaintGLSurfaceEventArgs e)
85 {
86 // Capture/refresh the GRContext so callers can wire the backbuffer to the same one.
87 GrContext ??= _glControl.GRContext;
88
89 var canvas = e.Surface.Canvas;
90
91 // Clear with your configured clear color
92 canvas.Clear(ClearColor);
93
94 var img = _currentImage;
95 if (img != null)
96 {
97 // IMPORTANT: For zero-copy, img MUST belong to this same GRContext.
98 // If you hand us a raster image, Skia will upload it each frame (works, but costs bandwidth).
99 var imgBoundsI = new SKRectI(0, 0, img.Width, img.Height);
100 var srcI = SKRectI.Intersect(_sourceRect, imgBoundsI);
101
102 if (!srcI.IsEmpty)
103 {
104 var src = ToRect(srcI);
105 var dest = _destRect;
106
107 var bounds = SKRect.Create(e.BackendRenderTarget.Width, e.BackendRenderTarget.Height);
108 if (!bounds.Contains(dest))
109 dest = SKRect.Intersect(dest, bounds);
110
111 canvas.DrawImage(img, src, dest);
112 }
113 }
114
115 // Safe to free the previously drawn GPU image NOW (on GL thread)
116 _prevToDispose?.Dispose();
117 _prevToDispose = null;
118
119 // Optional: flush to ensure work is queued to GPU before we hand new images next frame
120 _glControl.GRContext?.Flush();
121 }
122
126 public void Dispose()
127 {
128 if (!_glControl.IsDisposed) _glControl.PaintSurface -= OnPaintSurface;
129
130 // These MUST be disposed on the GL thread; do a last paint if needed.
131 // If you're disposing off the UI thread at shutdown, it's usually fine
132 // because the context is being torn down; still, we null them here.
133 _prevToDispose?.Dispose();
134 _currentImage?.Dispose();
135 _prevToDispose = null;
136 _currentImage = null;
137 }
