Gondwana Game Engine v2.2.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

66 internal void PollForEvents(long tick)
67 {
68 if (PauseAllKeyEvents)
69 return;
70
71 var adapter = Adapter;
72 if (adapter is null)
73 return;
74
75 // Apply any pending adds/removes on the engine thread.
76 while (_pendingOps.TryDequeue(out var op))
77 op();
78
79 var mods = adapter.CurrentKeyboardModifiers;
80
81 foreach (var kvp in _keyConfigs)
82 {
83 int keyCode = kvp.Key;
84 var config = kvp.Value;
85
86 bool currentlyPressed = adapter.IsDown(keyCode);
87 bool previouslyPressed = _previousPressed.TryGetValue(keyCode, out var prev) && prev;
88
89 // Transition: not pressed -> pressed
90 if (currentlyPressed && !previouslyPressed)
91 {
92 _previousPressed[keyCode] = true;
93 config._lastEventTick = tick;
94
95 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Pressed));
96 continue;
97 }
98
99 // Still pressed -> maybe repeated (throttled)
100 if (currentlyPressed && previouslyPressed)
101 {
102 if (!config.IsPaused && config.ReadyForNextEvent(tick))
103 {
104 config._lastEventTick = tick;
105 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Repeated));
106 }
107 continue;
108 }
109
110 // Transition: pressed -> not pressed
111 if (!currentlyPressed && previouslyPressed)
112 {
113 _previousPressed[keyCode] = false;
114 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Released));
115 }
116
117 // not pressed && not previously pressed => nothing
118 }
119 }
