Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

199 internal static void RaiseTimerEvents(TimerType type, long engineTick)
200 {
201 var expired = new List<string>();
202
203 foreach (var (key, timer) in _timers.ToArray())
204 {
205 // checking this TimerType (i.e., PreCycle or PostCycle)
206 if (timer.Type != type) continue;
207
208 // "push" time forward for Paused Timer instances
209 if (PausedAll || timer.Paused)
210 {
211 timer._lastEventTick += (engineTick - timer._lastEventTick);
212 continue;
213 }
214
215 // check if Timer.Length time has passed
216 while (engineTick - timer._lastEventTick >= timer.Length)
217 {
218 // save the time this event was scheduled to run
219 // might be different from current system tick, but storing this value
220 // will ensure that a lag in repeating timer events does not
221 // accumulate over time
222 timer._lastEventTick += timer.Length;
223 timer.RaiseTickEvent();
224
225 // check for any expired timers
226 if (timer.Cycles == TimerCycles.Once)
227 expired.Add(key);
228 }
229 }
230
231 foreach (var key in expired)
232 _timers.TryRemove(key, out _);
233 }
