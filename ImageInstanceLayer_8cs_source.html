Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

350 public override void Update(long tick)
351 {
352 if (tick <= _lastTick)
353 return;
354
355 float dt = 0f;
356
357 if (_instanceLastTick > 0)
358 dt = HighResTimer.GetDuration(_instanceLastTick, tick);
359
360 _instanceLastTick = tick;
361
362 if (dt <= 0f)
363 {
364 base.Update(tick);
365 return;
366 }
367
368 var bounds = Mode == DirectDrawingMode.SceneLayer
369 ? WorldBounds
370 : ScreenBounds;
371
372 for (int i = 0; i < Instances.Count; i++)
373 {
374 var instance = Instances[i];
375 var oldRefreshBounds = GetRefreshBounds(instance);
376
377 instance.Bounds = new RectangleF(
378 instance.Bounds.X + (instance.VelocityX * dt),
379 instance.Bounds.Y + (instance.VelocityY * dt),
380 instance.Bounds.Width,
381 instance.Bounds.Height);
382
383 instance.Rotation += instance.AngularVelocity * dt;
384
385 UpdateInstance?.Invoke(instance, dt);
386
387 if (ShouldRecycle is not null &&
388 RecycleInstance is not null &&
389 ShouldRecycle(instance, bounds))
390 {
391 var recycled = RecycleInstance(instance, bounds, _rng);
392 Instances[i] = recycled;
393
394 AddRefreshRect(oldRefreshBounds);
395 AddRefreshRect(GetRefreshBounds(recycled));
396 }
397 else
398 {
399 AddRefreshRect(oldRefreshBounds);
400 AddRefreshRect(GetRefreshBounds(instance));
401 }
402 }
403
404 base.Update(tick);
405 }

410 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
411 {
412 var canvas = backbuffer.Canvas;
413
414 Rectangle srcBounds = Mode == DirectDrawingMode.SceneLayer
415 ? WorldBounds
416 : ScreenBounds;
417
418 float sx = srcBounds.Width > 0
419 ? destRectScreen.Width / srcBounds.Width
420 : 1f;
421
422 float sy = srcBounds.Height > 0
423 ? destRectScreen.Height / srcBounds.Height
424 : 1f;
425
426 canvas.Save();
427 canvas.ClipRect(destRectScreen.ToSKRect());
428
429 foreach (var instance in Instances)
430 {
431 _paint.Color = instance.Tint;
432
433 var dst = new SKRect(
434 destRectScreen.Left + ((instance.Bounds.Left - srcBounds.Left) * sx),
435 destRectScreen.Top + ((instance.Bounds.Top - srcBounds.Top) * sy),
436 destRectScreen.Left + ((instance.Bounds.Right - srcBounds.Left) * sx),
437 destRectScreen.Top + ((instance.Bounds.Bottom - srcBounds.Top) * sy));
438
439 if (instance.Rotation != 0f)
440 {
441 float cx = (dst.Left + dst.Right) * 0.5f;
442 float cy = (dst.Top + dst.Bottom) * 0.5f;
443
444 canvas.Save();
445 canvas.RotateDegrees(instance.Rotation, cx, cy);
446 canvas.DrawBitmap(instance.Bitmap, dst, _paint);
447 canvas.Restore();
448 }
449 else
450 {
451 canvas.DrawBitmap(instance.Bitmap, dst, _paint);
452 }
453 }
454

473 private void AddRefreshRect(RectangleF rect)
474 {
475 const float pad = 4f;
476
477 var expanded = Rectangle.Ceiling(RectangleF.Inflate(rect, pad, pad));
478
479 if (Mode == DirectDrawingMode.SceneLayer)
480 {
481 SceneLayer!.RefreshQueue.AddWorldRect(expanded);
482 return;
483 }
484
485 foreach (var sceneLayer in RenderSurfaceHost.Scene.SceneLayers)
486 sceneLayer.RefreshQueue.AddViewScreenRect(View!, sceneLayer, expanded);
487 }

494 private static RectangleF GetRefreshBounds(ImageInstance instance)
495 {
496 if (instance.Rotation == 0f && instance.AngularVelocity == 0f)
497 return instance.Bounds;
498
499 float cx = instance.Bounds.Left + (instance.Bounds.Width * 0.5f);
500 float cy = instance.Bounds.Top + (instance.Bounds.Height * 0.5f);
501
502 float halfWidth = instance.Bounds.Width * 0.5f;
503 float halfHeight = instance.Bounds.Height * 0.5f;
504 float halfDiagonal = MathF.Sqrt((halfWidth * halfWidth) + (halfHeight * halfHeight));
505
506 return new RectangleF(
507 cx - halfDiagonal,
508 cy - halfDiagonal,
509 halfDiagonal * 2f,
510 halfDiagonal * 2f);
511 }
