Gondwana Game Engine v2.4.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

294 public override void Update(long tick)
295 {
296 if (tick <= _lastTick)
297 return;
298
299 float dt = 0f;
300
301 if (_instanceLastTick > 0)
302 dt = HighResTimer.GetDuration(_instanceLastTick, tick);
303
304 _instanceLastTick = tick;
305
306 if (dt <= 0f)
307 {
308 base.Update(tick);
309 return;
310 }
311
312 var bounds = ScreenBounds;
313
314 for (int i = 0; i < Instances.Count; i++)
315 {
316 var instance = Instances[i];
317 var oldRefreshBounds = GetRefreshBounds(instance);
318
319 instance.Bounds = new RectangleF(
320 instance.Bounds.X + (instance.VelocityX * dt),
321 instance.Bounds.Y + (instance.VelocityY * dt),
322 instance.Bounds.Width,
323 instance.Bounds.Height);
324
325 instance.Rotation += instance.AngularVelocity * dt;
326
327 UpdateInstance?.Invoke(instance, dt);
328
329 if (ShouldRecycle is not null &&
330 RecycleInstance is not null &&
331 ShouldRecycle(instance, bounds))
332 {
333 var recycled = RecycleInstance(instance, bounds, _rng);
334 Instances[i] = recycled;
335
336 AddRefreshRect(oldRefreshBounds);
337 AddRefreshRect(GetRefreshBounds(recycled));
338 }
339 else
340 {
341 AddRefreshRect(oldRefreshBounds);
342 AddRefreshRect(GetRefreshBounds(instance));
343 }
344 }
345
346 base.Update(tick);
347 }

418 private static RectangleF GetRefreshBounds(ImageInstance instance)
419 {
420 if (instance.Rotation == 0f && instance.AngularVelocity == 0f)
421 return instance.Bounds;
422
423 float cx = instance.Bounds.Left + (instance.Bounds.Width * 0.5f);
424 float cy = instance.Bounds.Top + (instance.Bounds.Height * 0.5f);
425
426 float halfWidth = instance.Bounds.Width * 0.5f;
427 float halfHeight = instance.Bounds.Height * 0.5f;
428 float halfDiagonal = MathF.Sqrt((halfWidth * halfWidth) + (halfHeight * halfHeight));
429
430 return new RectangleF(
431 cx - halfDiagonal,
432 cy - halfDiagonal,
433 halfDiagonal * 2f,
434 halfDiagonal * 2f);
435 }
