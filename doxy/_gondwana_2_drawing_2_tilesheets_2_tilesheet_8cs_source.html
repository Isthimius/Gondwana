212 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
213 {
214 if (SkBitmap == null || SkBitmap.IsEmpty)
215 throw new ArgumentException("Invalid bitmap.");
216
217 MaskColor = maskColor;
218 MaskTolerance = tolerance;
219 Premultiplied = true;
220
221 var targetColor = maskColor ?? SKColors.White;
222
223 // ensure the bitmap actually supports alpha
224 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
225 {
226 var info = new SKImageInfo(
227 SkBitmap.Width,
228 SkBitmap.Height,
229 SkBitmap.Info.ColorType,
230 SKAlphaType.Premul
231 );
232
233 var withAlpha = new SKBitmap(info);
234 using (var canvas = new SKCanvas(withAlpha))
235 {
236 canvas.DrawBitmap(SkBitmap, 0, 0);
237 }
238
239 SkBitmap.Dispose();
240 SkBitmap = withAlpha;
241 }
242
243 SkBitmapOriginal = SkBitmap.Copy();
244
245 SkiaHelper.ApplyAlphaMask(SkBitmap, targetColor, tolerance);
246 SkBitmap = SkiaHelper.PremultiplyAlpha(SkBitmap);
247
248 BuildTileCache();
249 }

278 private void BuildTileCache()
279 {
280 ClearCache();
281
282 if (TileSize.Width <= 0 || TileSize.Height <= 0)
283 return;
284
285 int xTiles = (SkBitmap.Width - InitialOffsetX + XPixelsBetweenTiles) / (_tileSize.Width + XPixelsBetweenTiles);
286 int yTiles = (SkBitmap.Height - InitialOffsetY + YPixelsBetweenTiles) / (_tileSize.Height + YPixelsBetweenTiles);
287
288 _tileCache = new TilesheetSlice?[xTiles, yTiles];
289
290 for (int y = 0; y < yTiles; y++)
291 {
292 for (int x = 0; x < xTiles; x++)
293 {
294 var srcRect = GetTileBounds(x, y);
295 if (!SkBitmap.Info.Rect.Contains(srcRect.ToSKRectI()))
296 continue;
297
298 var srcInfo = SkBitmap.Info;
299
300 // IMPORTANT: preserve alpha + color type from the masked tilesheet bitmap
301 var sliceInfo = new SKImageInfo(
302 _tileSize.Width,
303 _tileSize.Height,
304 srcInfo.ColorType,
305 srcInfo.AlphaType
306 );
307
308 var bmp = new SKBitmap(sliceInfo);
309
310 // Optional but nice: ensure any untouched pixels are transparent
311 bmp.Erase(SKColors.Transparent);
312
313 if (SkBitmap.ExtractSubset(bmp, srcRect.ToSKRectI()))
314 {
315 var img = SKImage.FromBitmap(bmp);
316 _tileCache[x, y] = new TilesheetSlice(bmp, img);
317 }
318 else
319 {
320 bmp.Dispose();
321 }
322 }
323 }
324 }

440 private void Dispose(bool disposing)
441 {
442 if (_disposed) return;
443 _disposed = true;
444
445 if (disposing)
446 {
447 // unregister from registry
448 TilesheetRegistry.Instance.Remove(_name, this, dispose: false);
449
450 // clean up tile cache
451 if (_tileCache != null)
452 {
453 for (int x = 0; x < _tileCache.GetLength(0); x++)
454 {
455 for (int y = 0; y < _tileCache.GetLength(1); y++)
456 {
457 _tileCache[x, y]?.Bitmap?.Dispose();
458 _tileCache[x, y]?.Image?.Dispose();
459 }
460 }
461 _tileCache = null;
462 }
463
464 // dispose the main bitmaps
465 SkBitmap?.Dispose();
466 SkBitmapOriginal?.Dispose();
467
468 try
469 {
470 Disposed?.Invoke(this, new TilesheetDisposedEventArgs(this));
471 }
472 catch (Exception ex)
473 {
474 Engine.Logger.LogError(ex, "Error during Tilesheet Disposed event handling.");
475 }
476
477 // break delegate references
478 Disposed = null;
479 }
