52 internal void PollForEvents(long tick)
53 {
54 if (PauseAllKeyEvents)
55 return;
56
57 var adapter = Adapter;
58 if (adapter is null)
59 return;
60
61 // Apply any pending adds/removes on the engine thread.
62 while (_pendingOps.TryDequeue(out var op))
63 op();
64
65 var mods = adapter.CurrentKeyboardModifiers;
66
67 foreach (var kvp in _keyConfigs)
68 {
69 int keyCode = kvp.Key;
70 var config = kvp.Value;
71
72 bool currentlyPressed = adapter.IsDown(keyCode);
73 bool previouslyPressed = _previousPressed.TryGetValue(keyCode, out var prev) && prev;
74
75 // Transition: not pressed -> pressed
76 if (currentlyPressed && !previouslyPressed)
77 {
78 _previousPressed[keyCode] = true;
79 config._lastEventTick = tick;
80
81 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Pressed));
82 continue;
83 }
84
85 // Still pressed -> maybe repeated (throttled)
86 if (currentlyPressed && previouslyPressed)
87 {
88 if (!config.IsPaused && config.ReadyForNextEvent(tick))
89 {
90 config._lastEventTick = tick;
91 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Repeated));
92 }
93 continue;
94 }
95
96 // Transition: pressed -> not pressed
97 if (!currentlyPressed && previouslyPressed)
98 {
99 _previousPressed[keyCode] = false;
100 KeyDown?.Invoke(new KeyDownEventArgs(config, mods, KeyAction.Released));
101 }
102
103 // not pressed && not previously pressed => nothing
104 }
105 }
