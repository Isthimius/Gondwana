83 internal override void DrawRefreshQueueToBackbuffer(long tick)
84 {
85 // 0) If there are no visible SceneLayers, just clear and publish the full frame.
86 if (Scene.CountOfVisibleLayers == 0)
87 {
88 Backbuffer.Canvas.Clear(Backbuffer.ClearColor);
89 Backbuffer.MarkFullDirty();
90
91 Scene.FullRefreshNeeded = false;
92
93 _lastTick = tick;
94 return;
95 }
96
97 // 1) find total real seconds passed since last background loop
98 var deltaSeconds = HighResTimer.GetDuration(_lastTick, tick);
99
100 // 2) Handle full scene refresh once (camera moved, zoom changed, etc.): clear and mark all layers as dirty.
101 // This already clears the whole backbuffer and enqueues a full rect per layer.
102 bool forceFullRedraw = Scene.FullRefreshNeeded;
103
104 if (forceFullRedraw)
105 EnqueueFullSceneRefresh();
106
107 // 3) Fast “no work” probe
108 if (!forceFullRedraw && !Scene.IsDirty)
109 {
110 _lastTick = tick; // keep deltaSeconds sane next frame
111 return;
112 }
113
114 // 4.5) if doing a partial redraw, identify and clear any rects dirty from DirectDrawing overlays or dirty Tiles
115 if (!forceFullRedraw)
116 {
117 var dirtyScreenRects = CollectDirtyScreenArea();
118 PreclearScreenAreas(dirtyScreenRects);
119 }
120
121 // 5) Render all views to Backbuffer. Draw layers back -> front (ascending Z).
122 ViewRenderer.Render(deltaSeconds, Scene,
123 (view, layer) => RenderLayerDirtyRegions(view, layer, forceFullRedraw));
124
125 // 5.5) View-mode DirectDrawings pass (screen-space, on top of everything)
126 // IMPORTANT: View-mode assumes origin is screen (0,0), so reset any camera/parallax transforms.
127 var canvas = Backbuffer.Canvas;
128
129 canvas.Save();
130 canvas.ResetMatrix();
131
132 foreach (var view in ViewRenderer.Views)
133 {
134 var vp = view.Viewport.TargetRectPx;
135
136 canvas.Save();
137 canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
138
139 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
140 for (int i = 0; i < overlays.Count; i++)
141 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
142
143 canvas.Restore();
144 }
145
146 canvas.Restore();
147
148 // 6) Preserve any pre-existing dirty (e.g., set earlier this frame) — union, don’t replace.
149 // If this was a full redraw, mark the entire backbuffer as dirty so the adapter blits all of it.
150 if (forceFullRedraw)
151 Backbuffer.MarkFullDirty();
152
153 // 7) Clear layer queues now that we’ve consumed them (avoids re-drawing same tiles next frame)
154 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
155 Scene.VisibleSceneLayers[i].RefreshQueue.ClearRefreshQueue();
156
157 Scene.FullRefreshNeeded = false;
158 _lastTick = tick;
159 }

163 private void EnqueueFullSceneRefresh()
164 {
165 if (Backbuffer is null)
166 return;
167
168 Backbuffer.Canvas.Clear(Backbuffer.ClearColor);
169
170 foreach (var view in ViewRenderer.Views)
171 {
172 var viewport = view.Viewport;
173 var screenRect = viewport.TargetRectPx;
174
175 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
176 {
177 var layer = Scene.VisibleSceneLayers[i];
178
179 // 1) Compute layer-specific visible world rect
180 var layerWorldRect = view.ScreenRectToWorldRect(layer, screenRect);
181
182 // 2) Expand world rect by **one full tile** in all directions.
183 // This compensates for fractional-layer motion (parallax),
184 // camera motion, and tile boundaries.
185 int expandX = layer.SceneLayerTileWidth;
186 int expandY = layer.SceneLayerTileHeight;
187
188 layerWorldRect.Inflate(expandX, expandY);
189
190 // 3) Round to ints and enqueue
191 var worldRectInt = layerWorldRect.ToPixelAlignedRect();
192
193 layer.RefreshQueue.AddWorldRect(worldRectInt);
194 }
195 }
196 }

198 private List<Rectangle> CollectDirtyScreenArea()
199 {
200 var dirty = new List<Rectangle>(64);
201
202 foreach (var view in ViewRenderer.Views)
203 {
204 var viewportRect = view.Viewport.TargetRectPx;
205
206 foreach (var sceneLayer in Scene.VisibleSceneLayers)
207 {
208 var refreshQueue = sceneLayer.RefreshQueue;
209 if (!refreshQueue.IsDirty)
210 continue;
211
212 foreach (var worldRect in refreshQueue.WorldRects)
213 {
214 var screenRectF = view.WorldRectToScreenRect(sceneLayer, worldRect);
215 var rect = Rectangle.Intersect(
216 screenRectF.ToPixelAlignedRect(),
217 viewportRect);
218
219 if (rect.IsEmpty)
220 continue;
221
222 AddDeduped(rect, dirty);
223 }
224 }
225 }
226
227 // HACK: re-add dirty rectangles to all RefreshQueues
228 foreach (var rect in dirty)
229 {
230 foreach (var view in ViewRenderer.Views)
231 {
232 var viewportRect = view.Viewport.TargetRectPx;
233 if (!viewportRect.IntersectsWith(rect))
234 continue;
235 foreach (var sceneLayer in Scene.VisibleSceneLayers)
236 {
237 var worldRectF = view.ScreenRectToWorldRect(sceneLayer, rect);
238 var worldRect = worldRectF.ToPixelAlignedRect();
239 sceneLayer.RefreshQueue.AddWorldRect(worldRect);
240 }
241 }
242 }
243
244 return dirty;
245 }

247 private static void AddDeduped(Rectangle rect, List<Rectangle> list)
248 {
249 // If an existing rect fully contains this one, skip it
250 for (int i = 0; i < list.Count; i++)
251 {
252 if (list[i].Contains(rect))
253 return;
254 }
255
256 // Merge with any overlapping rects
257 for (int i = list.Count - 1; i >= 0; i--)
258 {
259 if (rect.IntersectsWith(list[i]))
260 {
261 rect = Rectangle.Union(rect, list[i]);
262 list.RemoveAt(i);
263 }
264 }
265
266 list.Add(rect);
267 }

269 private void PreclearScreenAreas(List<Rectangle> screenRects)
270 {
271 if (Backbuffer is null || screenRects is null || screenRects.Count == 0)
272 return;
273
274 var canvas = Backbuffer.Canvas;
275
276 // Screen-pixel space
277 canvas.Save();
278 canvas.ResetMatrix();
279
280 foreach (var screenRect in screenRects)
281 {
282 if (screenRect.IsEmpty || screenRect.Width <= 0 || screenRect.Height <= 0)
283 continue;
284
285 // Clear just this patch (overwrite with Backbuffer.ClearColor)
286 Backbuffer.ClearRect(screenRect);
287
288 // Ensure adapter blits it; mark Backbuffer dirty
289 screenRect.Inflate(1, 1);
290 Backbuffer.AddToDirtyRectangle(screenRect);
291 }
292
293 canvas.Restore();
294 }

296 private void RenderLayerDirtyRegions(View view, SceneLayer layer, bool forceFullRedraw)
297 {
298 var refreshQueue = layer.RefreshQueue;
299
300 // if this layer has no dirty regions and we are not forcing a full redraw, skip it.
301 if (!forceFullRedraw && !refreshQueue.IsDirty)
302 return;
303
304 foreach (var worldRect in refreshQueue.WorldRects)
305 {
306 // 1) project world → screen for adapter dirty
307 var screenRect = Rectangle.Intersect(
308 view.WorldRectToScreenRect(layer, worldRect).ToPixelAlignedRect(),
309 view.Viewport.TargetRectPx);
310
311 if (screenRect.Width <= 0 || screenRect.Height <= 0)
312 continue;
313
314 // 2) mark adapter dirty (screen-space)
315 Backbuffer.AddToDirtyRectangle(screenRect);
316
317 // 3) clip and redraw tiles (world-space)
318 var canvas = Backbuffer.Canvas;
319
320 canvas.Save();
321 canvas.ClipRect(screenRect.ToSKRect());
322
323 var drawables = layer.GetDrawablesInWorldRect(worldRect);
324 Backbuffer.DrawDrawables(view, drawables);
325
326 canvas.Restore();
327 }

391 private void OnRenderSurfaceAdapterResized(RenderSurfaceAdapterResizedEventArgs args)
392 {
393 if (Scene != null)
394 Scene.FullRefreshNeeded = true; // full redraw next frame
395
396 _backbuffer?.RequestResize(args.NewWidth, args.NewHeight); // UI thread → request only
397
398 float scaleX = (float)args.NewWidth / args.OldWidth;
399 float scaleY = (float)args.NewHeight / args.OldHeight;
400
401 // resize each View proportionally
402 foreach (var view in ViewRenderer.Views)
403 {
404 var old = view.Viewport.TargetRectPx;
405
406 int newLeft = (int)Math.Round(old.Left * scaleX);
407 int newTop = (int)Math.Round(old.Top * scaleY);
408 int newWidth = (int)Math.Round(old.Width * scaleX);
409 int newHeight = (int)Math.Round(old.Height * scaleY);
410
411 view.Viewport.TargetRectPx = new Rectangle(
412 newLeft, newTop, newWidth, newHeight);
413 }
