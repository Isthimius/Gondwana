122 public void SetPosition(Vector2 pos)
123 {
124 // old and new coordinate space positions
125 PointF oldCoord = sceneLayerCoordinates;
126 PointF newCoord = new PointF(pos.X, pos.Y);
127
128 // compute old/new draw rects in WORLD pixels
129 Rectangle oldDraw = SpriteManager.GetDrawLocation(this, _sceneLayer, oldCoord, renderSize);
130 Rectangle newDraw = SpriteManager.GetDrawLocation(this, _sceneLayer, newCoord, renderSize);
131
132 // union of old + new = full movement envelope
133 Rectangle movementWorldRect = Rectangle.Union(oldDraw, newDraw);
134 movementWorldRect.Inflate(new Size(5, 5));
135
136 // commit the move
137 sceneLayerCoordinates = newCoord;
138
139 // enqueue ONE world-space dirty rect for the whole movement
140 _sceneLayer.RefreshQueue.AddWorldRect(movementWorldRect);
141
142 // raise the event
143 SpriteMoved?.Invoke(new SpriteMovedEventArgs(this, oldCoord, newCoord));
144 }

277 public override void Dispose()
278 {
279 GC.SuppressFinalize(this);
280
281 Disposing?.Invoke(this);
282
283 if (_sceneLayer != null)
284 {
285 // Mark the last draw region as dirty so the background under this sprite is repainted.
286 // DrawLocation should already be a world-space rectangle.
287 _sceneLayer.RefreshQueue.AddWorldRect(DrawLocationWorld);
288
289 if (_collider != null)
290 {
291 var world = _sceneLayer.Scene?.CollisionWorld;
292 if (world != null)
293 {
294 world.Unregister(_collider);
295 }
296 }
297 }
298
299 if (SpriteManager._spriteList.IndexOf(this) != -1)
300 SpriteManager._spriteList.Remove(this);
301
302 // clear the events
303 SpriteMoved = null;
304 Disposing = null;
305
306 base.Dispose();
307 }
