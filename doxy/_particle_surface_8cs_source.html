267 public override void Update(long tick)
268 {
269 if (tick <= _lastTick)
270 return;
271
272 // Compute dt from ticks (seconds)
273 float dt = 0f;
274
275 // no previous tick assume first frame, so skip dt-based updates
276 if (_particlesLastTick > 0)
277 dt = HighResTimer.GetDuration(_particlesLastTick, tick);
278
279 _particlesLastTick = tick;
280
281 // (A) per-emitter motion
282 for (int i = 0; i < Emitters.Count; i++)
283 Emitters[i].OnUpdate?.Invoke(Emitters[i], dt);
284
285 // (B) emission
286 for (int i = 0; i < Emitters.Count; i++)
287 {
288 var em = Emitters[i];
289 float rate = MathF.Max(0f, em.EmitRate * GlobalEmitScale);
290 em._accumulator += rate * dt;
291
292 int toEmit = (int)em._accumulator;
293 if (toEmit > 0) { em._accumulator -= toEmit; EmitFrom(em, toEmit); }
294 }
295
296 // (C) integrate & compact
297 int write = 0;
298 for (int i = 0; i < _alive; i++)
299 {
300 ref var p = ref _particles[i];
301
302 // integrate acceleration (including gravity)
303 p.VX += p.AX * dt;
304 p.VY += p.AY * dt;
305
306 // clamp max velocity
307 if (p.MaxVelocity > 0f)
308 {
309 // thank you, Pythagoras
310 float vx = p.VX, vy = p.VY;
311 float velocity2 = (vx * vx) + (vy * vy);
312 float max2 = p.MaxVelocity * p.MaxVelocity;
313
314 if (velocity2 > max2)
315 {
316 float inv = p.MaxVelocity / MathF.Sqrt(velocity2);
317 p.VX = vx * inv;
318 p.VY = vy * inv;
319 }
320 }
321
322 // integrate velocity
323 p.X += p.VX * dt;
324 p.Y += p.VY * dt;
325
326 // to every season...
327 p.Rotation += p.AngularVel * dt;
328 p.Life -= dt;
329
330 // cull if out of bounds (with margin)
331 bool isInView = p.X >= ScreenBounds.Left - CullingMarginX
332 && p.X <= ScreenBounds.Right + CullingMarginX
333 && p.Y >= ScreenBounds.Top - CullingMarginY
334 && p.Y <= ScreenBounds.Bottom + CullingMarginY;
335
336 if (p.Life > 0 && isInView)
337 _particles[write++] = p;
338 }
339 _alive = write;
340
341 ForceRefresh();
342
343 base.Update(tick);
344 }

373 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
374 {
375 var canvas = backbuffer.Canvas;
376
377 // e.g., default blend; you may switch per-emitter later if you add BlendMode there
378 _paint.BlendMode = SKBlendMode.Plus;
379
380 for (int i = 0; i < _alive; i++)
381 {
382 ref var p = ref _particles[i];
383
384 // life-based fade
385 float t = 1f - (p.Life / p.MaxLife);
386 byte a = (byte)(255 * (1f - t));
387
388 // choose tint: emitter override if set, otherwise current global
389 var tint = p.Tint ?? GlobalColorTint;
390
391 // apply global tint
392 _paint.Color = ApplyTint(p.Color, a, tint);
393
394 if (p.ParticleSprite is null)
395 {
396 // circle primitive
397 float size = p.Size * (1f + 0.5f * t);
398 canvas.DrawCircle(p.X, p.Y, size, _paint);
399 }
400 else
401 {
402 // textured quad
403 float s = p.Size;
404 var dst = new SKRect(p.X - s * 0.5f, p.Y - s * 0.5f,
405 p.X + s * 0.5f, p.Y + s * 0.5f);
406
407 canvas.Save();
408 canvas.RotateDegrees(p.Rotation, p.X, p.Y);
409 canvas.DrawBitmap(p.ParticleSprite, dst, _paint);
410 canvas.Restore();
411 }
412 }

415 private void EmitFrom(ParticleEmitter em, int count)
416 {
417 for (int i = 0; i < count && _alive < _particles.Length; i++)
418 {
419 ref var p = ref _particles[_alive++];
420
421 // Position
422 if (em.JitterX == 0f && em.JitterY == 0f)
423 {
424 p.X = em.Position.X;
425 p.Y = em.Position.Y;
426 }
427 else
428 {
429 (float dx, float dy) = SampleSpawnOffset(em);
430 p.X = em.Position.X + dx;
431 p.Y = em.Position.Y + dy;
432 }
433
434 // Velocity
435 p.VX = NextRange(em.VelocityRangeX.Min, em.VelocityRangeX.Max);
436 p.VY = NextRange(em.VelocityRangeY.Min, em.VelocityRangeY.Max);
437
438 // Acceleration (emitter override or surface default)
439 p.AX = em.GravityX ?? this.GravityX;
440 p.AY = em.GravityY ?? this.GravityY;
441
442 // Life/Size
443 p.MaxLife = p.Life = NextRange(em.LifeRange.Min, em.LifeRange.Max);
444 p.Size = NextRange(em.SizeRange.Min, em.SizeRange.Max);
445
446 // Color & spin
447 p.Color = em.Color;
448 p.Rotation = NextRange(0f, 360f);
449 p.AngularVel = NextRange(-180f, 180f);
450
451 // choose sprite once; no emitter lookups in hot loop
452 p.ParticleSprite = em.ParticleSprite ?? this._particleSprite;
453
454 // tint override
455 p.Tint = em.Tint; // null means "use global at render"
456
457 // max speed clamp
458 p.MaxVelocity = em.MaxVelocity ?? 0f; // 0 = no clamp
459
460 // User hook for last-mile per-particle tweaks
461 em.OnSpawn?.Invoke(ref p);
462 }
463 }

467 private (float dx, float dy) SampleSpawnOffset(ParticleEmitter em)
468 {
469 float jx = MathF.Abs(em.JitterX);
470 float jy = MathF.Abs(em.JitterY);
471
472 // Handle "line" cases cleanly (one axis zero)
473 if (jx == 0f) return (0f, NextRange(-jy, jy));
474 if (jy == 0f) return (NextRange(-jx, jx), 0f);
475
476 switch (em.SpawnDistribution)
477 {
478 default:
479 case ParticleSpawnDistribution.Rectangle:
480 return (NextRange(-jx, jx), NextRange(-jy, jy));
481
482 case ParticleSpawnDistribution.Ellipse:
483 {
484 // Uniform over area of ellipse
485 float a = NextRange(0f, MathF.Tau);
486 float r = MathF.Sqrt(NextRange(0f, 1f)); // sqrt => uniform area
487 return (MathF.Cos(a) * r * jx, MathF.Sin(a) * r * jy);
488 }
489
490 case ParticleSpawnDistribution.Ring:
491 {
492 // Uniform over area of annulus (ellipse-scaled)
493 float inner = Math.Clamp(em.RingInnerRadius01, 0f, 1f);
494 float a = NextRange(0f, MathF.Tau);
495
496 // area-uniform annulus: r = sqrt(inner^2 + u*(1-inner^2))
497 float u = NextRange(0f, 1f);
498 float r = MathF.Sqrt(inner * inner + u * (1f - inner * inner));
499
500 return (MathF.Cos(a) * r * jx, MathF.Sin(a) * r * jy);
501 }
502
503 case ParticleSpawnDistribution.Gaussian:
504 {
505 // Center-weighted, then clamped to ellipse boundary
506 float sd = Math.Clamp(em.GaussianStdDev01, 1e-4f, 10f);
507
508 float dx = NextGaussian() * (jx * sd);
509 float dy = NextGaussian() * (jy * sd);
510
511 // Clamp to ellipse: (dx/jx)^2 + (dy/jy)^2 <= 1
512 float nx = dx / jx;
513 float ny = dy / jy;
514 float q = nx * nx + ny * ny;
515
516 if (q > 1f)
517 {
518 float s = 1f / MathF.Sqrt(q);
519 dx *= s;
520 dy *= s;
521 }
522
523 return (dx, dy);
524 }
525 }
526 }

528 private float NextGaussian()
529 {
530 // Box–Muller transform (standard normal)
531 // Guard u1 from 0 to avoid log(0)
532 double u1 = 1.0 - _rng.NextDouble();
533 double u2 = 1.0 - _rng.NextDouble();
534 double mag = Math.Sqrt(-2.0 * Math.Log(u1));
535 double z0 = mag * Math.Cos(Math.Tau * u2);
536 return (float)z0;
537 }

539 // Fast, branch-free tint (multiplies RGB by tint)
540 // alpha = lifeAlpha * globalAlpha). Assumes particle base alpha = 255.
541 [MethodImpl(MethodImplOptions.AggressiveInlining)]
542 private SKColor ApplyTint(SKColor c, byte lifeAlpha, SKColor tint)
543 {
544 int r = (c.Red * tint.Red) / 255;
545 int g = (c.Green * tint.Green) / 255;
546 int b = (c.Blue * tint.Blue) / 255;
547 int a = (lifeAlpha * tint.Alpha) / 255;
548
549 return new SKColor((byte)r, (byte)g, (byte)b, (byte)a);
550 }
