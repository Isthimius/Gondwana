175 private void MouseEventPoller_MouseEvent(Gondwana.Input.Mouse.MouseEventArgs args)
176 {
177 var view = RenderSurface.Host.ViewRenderer.Views[0];
178 var layer = Scene.SceneLayers[0];
179
180 var screenPos = args.CurrentPosition;
181
182 // 1) screen → world (via View)
183 var worldFromScreen = view.ScreenPxToWorldPx(layer, screenPos);
184
185 // 2) world → grid (via View wrapper, which calls SceneLayer internally)
186 var gridFromScreen = view.ScreenPxToGrid(layer, screenPos);
187
188 // 3) grid → world (via SceneLayer wrapper)
189 var worldFromGrid = layer.GridToWorldPx(gridFromScreen);
190
191 // 4) world → screen (via View)
192 var screenFromGrid = view.WorldPxToScreenPx(layer, worldFromGrid);
193 }
