77 private void DisplayAsset()
78 {
79 PictureBox picBox = null;
80
81 switch (_type)
82 {
83 case EngineResourceFileTypes.Bitmap:
84 btnBackground.Visible = true;
85 btnMask.Visible = true;
86 axWMP.Visible = false;
87
88 picBox = new PictureBox();
89 picBox.SizeMode = PictureBoxSizeMode.AutoSize;
90 picBox.Image = new Bitmap(_stream);
91 picBox.MouseClick += picBox_MouseClick;
92
93 this.panel1.Controls.Add(picBox);
94 break;
95
96 case EngineResourceFileTypes.Audio:
97 case EngineResourceFileTypes.Video:
98 btnBackground.Visible = false;
99 btnMask.Visible = false;
100 panel1.Dock = DockStyle.Fill;
101
102 if (!string.IsNullOrWhiteSpace(_file))
103 {
104 axWMP.settings.autoStart = false;
105 axWMP.Visible = true;
106 axWMP.URL = _file;
107 axWMP.Dock = DockStyle.Fill;
108 }
109
110 break;
111
112 case EngineResourceFileTypes.Cursor:
113 btnBackground.Visible = false;
114 btnMask.Visible = false;
115 axWMP.Visible = false;
116
117 picBox = new PictureBox();
118 picBox.SizeMode = PictureBoxSizeMode.AutoSize;
119 picBox.BorderStyle = BorderStyle.Fixed3D;
120
121 var cursor = new Cursor(_file);
122 var bmp = new Bitmap(cursor.Size.Width, cursor.Size.Height);
123 using (var graphics = Graphics.FromImage(bmp))
124 cursor.Draw(graphics, new Rectangle(new Point(), cursor.Size));
125
126 picBox.Image = bmp;
127 picBox.MouseClick += picBox_MouseClick;
128
129 this.panel1.Controls.Add(picBox);
130
131 break;
132
133 case EngineResourceFileTypes.Misc:
134 btnBackground.Text = "Open File";
135 btnBackground.Visible = true;
136 btnMask.Visible = false;
137 axWMP.Visible = false;
138
139 break;
140
141 default:
142 btnBackground.Visible = false;
143 btnMask.Visible = false;
144 axWMP.Visible = false;
145
146 break;
147 }
