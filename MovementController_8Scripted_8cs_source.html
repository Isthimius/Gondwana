Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

381 private bool AdvanceTween(float dt)
382 {
383 if (_state.Script.DurationSec <= 0f)
384 {
385 _mover.SetPosition(_state.Script.Target);
386
387 _state.Acceleration = Vector2.Zero;
388 _state.Velocity = Vector2.Zero;
389
390 Step(0f);
391 CompleteScript();
392
393 return true;
394 }
395
396 _state.Script.ElapsedSec += MathF.Max(0f, dt);
397
398 float t = Math.Clamp(
399 _state.Script.ElapsedSec / _state.Script.DurationSec,
400 0f,
401 1f);
402
403 if (_state.Script.Easing is not null)
404 {
405 t = Math.Clamp(
406 _state.Script.Easing(t),
407 0f,
408 1f);
409 }
410
411 var pos = Vector2.Lerp(
412 _state.Script.Origin,
413 _state.Script.Target,
414 t);
415
416 _mover.SetPosition(pos);
417
418 _state.Acceleration = Vector2.Zero;
419 _state.Velocity = Vector2.Zero;
420
421 Step(0f);
422
423 var snapDistanceSquared =
424 _state.Script.SnapEpsilon *
425 _state.Script.SnapEpsilon;
426
427 bool reachedTarget =
428 Vector2.DistanceSquared(
429 pos,
430 _state.Script.Target) <= snapDistanceSquared;
431
432 if (reachedTarget || t >= 1f)
433 {
434 _mover.SetPosition(_state.Script.Target);
435
436 _state.Acceleration = Vector2.Zero;
437 _state.Velocity = Vector2.Zero;
438
439 Step(0f);
440 CompleteScript();
441 }
442
443 return true;
444 }

446 private bool AdvanceToward(float dt)
447 {
448 var current = _mover.GetPosition();
449 var to = _state.Script.Target - current;
450 var distance = to.Length();
451
452 if (distance <= _state.Script.SnapEpsilon ||
453 _state.Script.SpeedPerSec <= 0f ||
454 dt <= 0f)
455 {
456 _mover.SetPosition(_state.Script.Target);
457
458 _state.Velocity = Vector2.Zero;
459 _state.Acceleration = Vector2.Zero;
460
461 Step(0f);
462 CompleteScript();
463
464 return true;
465 }
466
467 var stepLength =
468 _state.Script.SpeedPerSec * dt;
469
470 if (stepLength >= distance)
471 {
472 _mover.SetPosition(_state.Script.Target);
473
474 _state.Velocity = Vector2.Zero;
475 _state.Acceleration = Vector2.Zero;
476
477 Step(0f);
478 CompleteScript();
479
480 return true;
481 }
482
483 var direction = to / distance;
484
485 _state.Acceleration = Vector2.Zero;
486 _state.Velocity =
487 direction *
488 _state.Script.SpeedPerSec;
489
490 Step(dt);
491
492 return true;
493 }
