Gondwana Game Engine v2.2.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

162 private bool AdvanceScripted(float dt)
163 {
164 switch (_state.Script.Type)
165 {
166 case MovementScriptType.None:
167 return false;
168
169 case MovementScriptType.TweenTo:
170 return AdvanceTween(dt);
171
172 case MovementScriptType.Toward:
173 return AdvanceToward(dt);
174
175 default:
176 return false;
177 }

180 private bool AdvanceTween(float dt)
181 {
182 if (_state.Script.DurationSec <= 0f)
183 {
184 _mover.SetPosition(_state.Script.Target);
185 Step(0f);
186 _state.Script = default;
187
188 return true;
189 }
190
191 _state.Script.ElapsedSec += MathF.Max(0f, dt);
192 float t = Math.Clamp(_state.Script.ElapsedSec / _state.Script.DurationSec, 0f, 1f);
193
194 if (_state.Script.Easing is not null)
195 t = Math.Clamp(_state.Script.Easing(t), 0f, 1f);
196
197 var pos = Vector2.Lerp(_state.Script.Origin, _state.Script.Target, t);
198 _mover.SetPosition(pos);
199 _state.Acceleration = Vector2.Zero;
200 _state.Velocity = Vector2.Zero;
201 Step(0f);
202
203 if (Vector2.DistanceSquared(pos, _state.Script.Target) <= _state.Script.SnapEpsilon * _state.Script.SnapEpsilon || t >= 1f)
204 {
205 _mover.SetPosition(_state.Script.Target);
206 Step(0f);
207
208 ScriptedMovementStopped?.Invoke(_state.Script);
209 _state.Script = default;
210 }
211
212 return true;
213 }

215 private bool AdvanceToward(float dt)
216 {
217 var current = _mover.GetPosition();
218 var to = _state.Script.Target - current;
219 var dist = to.Length();
220
221 if (dist <= _state.Script.SnapEpsilon || _state.Script.SpeedPerSec <= 0f || dt <= 0f)
222 {
223 _mover.SetPosition(_state.Script.Target);
224 _state.Velocity = Vector2.Zero;
225 _state.Acceleration = Vector2.Zero;
226 Step(0f);
227
228 ScriptedMovementStopped?.Invoke(_state.Script);
229 _state.Script = default;
230
231 return true;
232 }
233
234 var stepLen = _state.Script.SpeedPerSec * dt;
235 if (stepLen >= dist)
236 {
237 _mover.SetPosition(_state.Script.Target);
238 _state.Velocity = Vector2.Zero;
239 _state.Acceleration = Vector2.Zero;
240 Step(0f);
241
242 ScriptedMovementStopped?.Invoke(_state.Script);
243 _state.Script = default;
244
245 return true;
246 }
247
248 var dir = to / dist;
249 _state.Acceleration = Vector2.Zero;
250 _state.Velocity = dir * _state.Script.SpeedPerSec;
251 Step(dt);
252
253 return true;
254 }
