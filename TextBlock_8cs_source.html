Gondwana Game Engine v2.1.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

524 public override void Update(long tick)
525 {
526 if (tick == _lastTick)
527 return;
528
529 // --- PULSE TIMING (independent) ---
530 float dtPulse = 0f;
531 if (_pulseLastTick.HasValue)
532 dtPulse = HighResTimer.GetDuration(_pulseLastTick.Value, tick);
533
534 _pulseLastTick = tick;
535
536 if (dtPulse > 0f && dtPulse < 1f)
537 _timeSec += dtPulse;
538
539 if (_pulseTextEnabled)
540 {
541 float t = PulseT(_timeSec, _pulsePeriodSec, _pulseWave);
542 var c = LerpColor(_pulseFrom, _pulseTo, t);
543 if (c != _resolvedForeColor)
544 {
545 _resolvedForeColor = c;
546 ForceRefresh();
547 }
548 }
549 else
550 {
551 // keep resolved color at base if not pulsing
552 if (_resolvedForeColor != _foreColor)
553 {
554 _resolvedForeColor = _foreColor;
555 ForceRefresh();
556 }
557 }
558
559 // --- REVEAL TIMING (independent and clamped) ---
560 float dtReveal = 0f;
561 if (_revealLastTick == 0)
562 {
563 _revealLastTick = tick; // warm up: first frame uses 0 dt
564 }
565 else
566 {
567 dtReveal = HighResTimer.GetDuration(_revealLastTick, tick);
568 _revealLastTick = tick;
569
570 // Clamp to avoid first-frame "time warp" completing the whole text
571 if (dtReveal > 0.1f) dtReveal = 0.1f; // ~100ms/frame cap
572 if (dtReveal < 0f) dtReveal = 0f;
573 }
574
575 // --- Typewriter / word reveal ---
576 if (_textRevealMode == TextRevealMode.CharactersPerSecond)
577 {
578 _revealAccum += dtReveal;
579 if (_revealRate > 0f)
580 {
581 int step = (int)(_revealAccum * _revealRate);
582 if (step > 0)
583 {
584 _revealAccum -= step / _revealRate;
585
586 int old = _revealCharCount;
587 _revealCharCount = Math.Min(_revealCharCount + step, _revealTargetCharCount);
588
589 // Punctuation pauses for newly revealed chars
590 for (int i = old; i < _revealCharCount; i++)
591 _revealAccum -= PauseFor(_text[i]);
592
593 // Mark dirty + redraw
594 ForceRefresh();
595
596 // Fire TextRevealed with the cumulative revealed text
597 if (_revealCharCount > old)
598 {
599 var revealed = _text.Substring(0, _revealCharCount);
600 TextRevealed?.Invoke(revealed);
601 }
602
603 // If we've reached the target, fire completion (TextRevealed already sent above)
604 if (_revealCharCount >= _revealTargetCharCount)
605 {
606 _textRevealMode = TextRevealMode.None;
607 // Ensure ordering: revealed then complete
608 TextRevealComplete?.Invoke(_text);
609 }
610 }
611 }
612 }
613 else if (_textRevealMode == TextRevealMode.WordsPerSecond && _wordEndCharIndexes is not null)
614 {
615 _revealAccum += dtReveal;
616 if (_revealRate > 0f)
617 {
618 int step = (int)(_revealAccum * _revealRate);
619 if (step > 0)
620 {
621 _revealAccum -= step / _revealRate;
622
623 int oldCount = _revealCharCount;
624
625 _wordIndex = Math.Min(_wordIndex + step, _wordEndCharIndexes.Length);
626 _revealCharCount = _wordIndex == 0 ? 0 : _wordEndCharIndexes[_wordIndex - 1];
627
628 // Simple frontier pause (cheap). Loop oldCount.._revealCharCount-1 if you want per-char.
629 if (_revealCharCount > 0)
630 _revealAccum -= PauseFor(_text[_revealCharCount - 1]);
631
632 ForceRefresh();
633
634 // Fire TextRevealed with the cumulative revealed text
635 if (_revealCharCount > oldCount)
636 {
637 var revealed = _text.Substring(0, _revealCharCount);
638 TextRevealed?.Invoke(revealed);
639 }
640
641 if (_wordIndex >= _wordEndCharIndexes.Length)
642 {
643 _textRevealMode = TextRevealMode.None;
644 // Ensure ordering: revealed then complete
645 TextRevealComplete?.Invoke(_text);
646 }
647 }
648 }
649 }
650
651 // Let the base advance fade tween and set _lastTick
652 base.Update(tick);
653 }

655 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
656 {
657 var canvas = backbuffer.Canvas;
658 var rect = destRectScreen.ToSKRect();
659
660 float zoom;
661
662 if (Mode == DirectDrawingMode.SceneLayer)
663 {
664 // SceneLayer-mode has View == null by design; use the ambient render context.
665 var contextZoom = RenderContext.Current?.ViewportZoom ?? 1f;
666 zoom = (contextZoom > 0f)
667 ? (1f / contextZoom)
668 : 1f;
669 }
670 else
671 {
672 // View-mode is screen/UI; do not compensate for camera zoom.
673 zoom = 1f;
674 }
675
676 // Scale "pixel-like" adornments with zoom so text behaves like other world-space drawables.
677 float hPad = HorizontalPadding * zoom;
678 float vPad = VerticalPadding * zoom;
679 float shadowDx = _shadowDx * zoom;
680 float shadowDy = _shadowDy * zoom;
681 float shadowBlur = _shadowBlurSigma * zoom;
682 float outlineWidth = 1.5f * zoom;
683
684 // Background
685 if (_backColor.Alpha != 0)
686 {
687 using var bg = new SKPaint { Color = _backColor };
688 canvas.DrawRect(rect, bg);
689 }
690
691 _typeface ??= SKTypeface.Default;
692
693 // Build a paint we can reuse for layout + draw (TEXT SIZE IS IN SCREEN PIXELS)
694 using var paint = new SKPaint
695 {
696 Typeface = _typeface ?? SKTypeface.Default,
697 TextSize = _fontSize * zoom,
698 Color = _resolvedForeColor,
699 IsAntialias = true,
700 IsStroke = false,
701 TextAlign = _hAlign
702 };
703
704 // Auto-shrink: reflow until it fits height (if min size provided)
705 float fontSize = _fontSize * zoom;
706 float minFontSize = _minFontSize.HasValue ? _minFontSize.Value * zoom : 0f;
707
708 float innerW = Math.Max(0, rect.Width - hPad * 2f);
709 float innerH = Math.Max(0, rect.Height - vPad * 2f);
710
711 while (true)
712 {
713 paint.TextSize = fontSize;
714
715 // Rebuild when flagged, and also whenever zoom is in play so wrap/line-height stay correct.
716 if (_layoutDirty || zoom != 1f)
717 RebuildLayout(paint, innerW);
718
719 int drawableLines = _maxLines.HasValue ? Math.Min(_lines.Count, _maxLines.Value) : _lines.Count;
720 float totalH = drawableLines * _lineHeight;
721
722 bool exceedsHeight = totalH > innerH;
723 bool exceedsWidth = _lines.Any(line => paint.MeasureText(line) > innerW);
724
725 if (_minFontSize.HasValue &&
726 (exceedsHeight || exceedsWidth) &&
727 fontSize > minFontSize)
728 {
729 fontSize = Math.Max(minFontSize, fontSize - 1f);
730 _layoutDirty = true;
731
732 if (fontSize <= minFontSize)
733 break;
734
735 continue;
736 }
737
738 break;
739 }
740
741 // Determine how many characters are currently visible (content-driven reveal)
742 int visibleChars = _textRevealMode != TextRevealMode.None
743 ? Math.Clamp(_revealCharCount, 0, _text.Length)
744 : _text.Length;
745
746 // Build the set of lines to draw from the already-laid-out _lines,
747 // truncating at 'visibleChars' so wrapping and alignment still work.
748 List<string> drawLines = new(_lines.Count);
749 if (visibleChars <= 0)
750 {
751 // nothing to show
752 }
753 else if (visibleChars >= _text.Length)
754 {
755 drawLines.AddRange(_lines);
756 }
757 else
758 {
759 int remaining = visibleChars;
760 foreach (var ln in _lines)
761 {
762 if (remaining <= 0)
763 break;
764
765 if (ln.Length <= remaining)
766 {
767 drawLines.Add(ln);
768 remaining -= ln.Length;
769 }
770 else
771 {
772 drawLines.Add(ln.Substring(0, remaining));
773 remaining = 0;
774 }
775 }
776 }
777
778 bool wasTruncatedByMaxLines = _maxLines.HasValue && drawLines.Count > _maxLines.Value;
779
780 // Apply max-lines cap at draw time
781 int linesToDraw = _maxLines.HasValue ? Math.Min(drawLines.Count, _maxLines.Value) : drawLines.Count;
782
783 if (linesToDraw > 0 && wasTruncatedByMaxLines)
784 {
785 drawLines[linesToDraw - 1] = FitLineWithEllipsis(paint, drawLines[linesToDraw - 1], innerW);
786 }
787
788 // Vertical start (Skia draws at baseline, so apply ascent shift)
789 var fm = paint.FontMetrics;
790 float baselineShift = -fm.Ascent;
791
792 float contentH = linesToDraw * _lineHeight;
793 float yStart = _vAlign switch
794 {
795 VerticalAlign.Center => rect.Top + vPad + Math.Max(0, (innerH - contentH) * 0.5f),
796 VerticalAlign.Bottom => rect.Bottom - vPad - contentH,
797 _ => rect.Top + vPad
798 };
799
800 // Horizontal anchor per line
801 float xAnchorLeft = rect.Left + hPad;
802 float xAnchorCenter = rect.MidX;
803 float xAnchorRight = rect.Right - hPad;
804
805 canvas.Save();
806 canvas.ClipRect(rect);
807
808 float y = yStart;
809 for (int i = 0; i < linesToDraw; i++)
810 {
811 var line = drawLines[i];
812 float x = _hAlign switch
813 {
814 SKTextAlign.Center => xAnchorCenter,
815 SKTextAlign.Right => xAnchorRight,
816 _ => xAnchorLeft
817 };
818
819 if (_useShadow)
820 {
821 using var shadow = paint.Clone();
822 shadow.IsStroke = false;
823 shadow.MaskFilter = shadowBlur > 0f
824 ? SKMaskFilter.CreateBlur(SKBlurStyle.Normal, shadowBlur)
825 : null;
826 shadow.Color = new SKColor(0, 0, 0, _shadowAlpha);
827
828 // manually offset
829 canvas.DrawText(line, x + shadowDx, y + baselineShift + shadowDy, shadow);
830 }
831
832 if (_useOutline)
833 {
834 using var outline = paint.Clone();
835 outline.IsStroke = true;
836 outline.StrokeWidth = outlineWidth;
837 outline.Color = SKColors.Black;
838 canvas.DrawText(line, x, y + baselineShift, outline);
839 }
840
841 canvas.DrawText(line, x, y + baselineShift, paint);
842 y += _lineHeight;
843
844 // safety clip
845 if (y > rect.Bottom)
846 break;
847 }
848
849 canvas.Restore();
850 }

852 private void RebuildLayout(SKPaint paint, float maxWidth)
853 {
854 _lines.Clear();
855
856 var fm = paint.FontMetrics;
857 _lineHeight = ((fm.Descent - fm.Ascent) + fm.Leading) * LineSpacingMultiplier;
858
859 var paragraphs = _text.Replace("\r\n", "\n").Split('\n');
860
861 foreach (var para in paragraphs)
862 {
863 if (string.IsNullOrEmpty(para))
864 {
865 _lines.Add(string.Empty);
866 continue;
867 }
868
869 if (!_wrapText)
870 {
871 _lines.Add(para);
872 continue;
873 }
874
875 var words = para.Split(' ', StringSplitOptions.None);
876 var current = string.Empty;
877
878 foreach (var word in words)
879 {
880 string candidate = string.IsNullOrEmpty(current) ? word : current + " " + word;
881 float candidateWidth = paint.MeasureText(candidate);
882
883 if (candidateWidth <= maxWidth)
884 {
885 current = candidate;
886 continue;
887 }
888
889 // If current already has content, commit it first and retry this word on a new line.
890 if (!string.IsNullOrEmpty(current))
891 {
892 _lines.Add(current);
893 current = string.Empty;
894 }
895
896 // If the single word fits on its own line, use it.
897 if (paint.MeasureText(word) <= maxWidth)
898 {
899 current = word;
900 continue;
901 }
902
903 // Fallback: hard-break an overlong word by character.
904 var chunk = string.Empty;
905 foreach (char ch in word)
906 {
907 string next = chunk + ch;
908 if (paint.MeasureText(next) <= maxWidth || string.IsNullOrEmpty(chunk))
909 {
910 chunk = next;
911 }
912 else
913 {
914 _lines.Add(chunk);
915 chunk = ch.ToString();
916 }
917 }
918
919 if (!string.IsNullOrEmpty(chunk))
920 current = chunk;
921 }
922
923 if (!string.IsNullOrEmpty(current))
924 _lines.Add(current);
925 }
926
927 _layoutDirty = false;
928 }

930 [MethodImpl(MethodImplOptions.AggressiveInlining)]
931 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
932 {
933 float phase = (timeSec / periodSec) % 1f;
934 return wave == PulseWave.Sine
935 ? 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f)) // [0..1]
936 : (phase < 0.5f ? phase * 2f : 1f - (phase - 0.5f) * 2f); // triangle
937 }

939 [MethodImpl(MethodImplOptions.AggressiveInlining)]
940 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
941 {
942 t01 = Math.Clamp(t01, 0f, 1f);
943 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
944 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
945 byte bl = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
946 byte al = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
947 return new SKColor(r, g, bl, al);
948 }

950 private static string FitLineWithEllipsis(SKPaint paint, string text, float maxWidth)
951 {
952 const string ellipsis = "...";
953
954 if (string.IsNullOrEmpty(text))
955 return text;
956
957 if (paint.MeasureText(text) <= maxWidth)
958 return text;
959
960 float ellipsisWidth = paint.MeasureText(ellipsis);
961 if (ellipsisWidth > maxWidth)
962 return string.Empty;
963
964 int length = text.Length;
965 while (length > 0)
966 {
967 string candidate = text.Substring(0, length).TrimEnd() + ellipsis;
968 if (paint.MeasureText(candidate) <= maxWidth)
969 return candidate;
970
971 length--;
972 }
973
974 return ellipsisWidth <= maxWidth ? ellipsis : string.Empty;
975 }
