39 public PointF GetSceneLayerCoordinatesAtPixel(SceneLayer sceneLayer, PointF pixelPt)
40 {
41 int W = sceneLayer.TileWidth;
42 int H = sceneLayer.TileHeight;
43
44 float originX = sceneLayer.OriginPx.X;
45 float originY = sceneLayer.OriginPx.Y;
46
47 // --- 1) Coarse column estimate using centers, not box top-left ---
48
49 // Center of column 0 is at originX + W/2
50 float dxFromCol0Center = pixelPt.X - (-originX + W * 0.5f);
51
52 // Columns advance by 0.75 * W horizontally
53 float colF = dxFromCol0Center / (W * 0.75f);
54 int approxCol = (int)Math.Round(colF);
55
56 // --- 2) Coarse row estimate using centers and parity ---
57
58 // Center Y of row 0 in this column:
59 // even col: originY + H/2
60 // odd col: originY + H/2 + H/2 = originY + H
61 float baseCenterY =
62 -originY + H * 0.5f
63 + ((approxCol & 1) == 0 ? 0f : H * 0.5f);
64
65 float dyFromRow0Center = pixelPt.Y - baseCenterY;
66 float rowF = dyFromRow0Center / H;
67 int approxRow = (int)Math.Round(rowF);
68
69 var best = new Point(approxCol, approxRow);
70 float bestDist = float.MaxValue;
71
72 // --- 3) Refine via polygon hit-test + nearest center ---
73
74 foreach (var cand in NeighborsFlatTop(approxCol, approxRow, includeSelf: true))
75 {
76 var poly = HexPolygonFlatTop(sceneLayer, cand.X, cand.Y, includeOverhang: false);
77
78 var pInt = new Point((int)Math.Round(pixelPt.X), (int)Math.Round(pixelPt.Y));
79 if (PointInPolygon(poly, pInt))
80 return new PointF(cand.X, cand.Y);
81
82 // Center of candidate hex
83 float cx = -originX + cand.X * (W * 0.75f) + W / 2f;
84 float cy = -originY + cand.Y * (float)H
85 + ((cand.X & 1) == 0 ? 0 : H / 2) + H / 2f;
86
87 float dx = cx - pixelPt.X;
88 float dy = cy - pixelPt.Y;
89 float d = dx * dx + dy * dy;
90
91 if (d < bestDist)
92 {
93 bestDist = d;
94 best = cand;
95 }
96 }
97
98 return new PointF(best.X, best.Y);
99 }

108 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(SceneLayer sceneLayer, Rectangle worldPixelRange, bool includeOverhang)
109 {
110 var result = new List<SceneLayerTile>();
111 int W = sceneLayer.TileWidth;
112 int H = sceneLayer.TileHeight;
113
114 int originX = sceneLayer.OriginPx.X;
115 int originY = sceneLayer.OriginPx.Y;
116
117 int minCol = (int)Math.Floor((worldPixelRange.Left + originX) / (W * 0.75f)) - 2;
118 int maxCol = (int)Math.Ceiling((worldPixelRange.Right + originX) / (W * 0.75f)) + 2;
119
120 for (int col = minCol; col <= maxCol; col++)
121 {
122 int yOffset = ((col & 1) == 0 ? 0 : H / 2);
123 int minRow = (int)Math.Floor((worldPixelRange.Top + originY - yOffset) / (float)H) - 2;
124 int maxRow = (int)Math.Ceiling((worldPixelRange.Bottom + originY - yOffset) / (float)H) + 2;
125
126 for (int row = minRow; row <= maxRow; row++)
127 {
128 var gp = sceneLayer[col, row];
129 if (gp == null) continue;
130 var r = GetPixelRangeForTile(gp, includeOverhang);
131 if (r.IntersectsWith(worldPixelRange))
132 result.Add(gp);
133 }
134 }
135
136 return result;
137 }

226 private static IEnumerable<Point> NeighborsFlatTop(int col, int row, bool includeSelf)
227 {
228 if (includeSelf) yield return new Point(col, row);
229 bool even = (col & 1) == 0;
230 yield return new Point(col + 1, row);
231 yield return new Point(col - 1, row);
232 yield return new Point(col + 1, row + (even ? 0 : 1));
233 yield return new Point(col + 1, row - (even ? 1 : 0));
234 yield return new Point(col - 1, row + (even ? 0 : 1));
235 yield return new Point(col - 1, row - (even ? 1 : 0));

238 private static Point[] HexPolygonFlatTop(SceneLayer sceneLayer, int col, int row, bool includeOverhang)
239 {
240 int W = sceneLayer.TileWidth;
241 int H = sceneLayer.TileHeight;
242
243 int originX = sceneLayer.OriginPx.X;
244 int originY = sceneLayer.OriginPx.Y;
245
246 var p = new Point(
247 -originX + (int)Math.Floor(col * (W * 0.75f)),
248 -originY + row * H + ((col & 1) == 0 ? 0 : H / 2));
249
250 var rect = new Rectangle(p.X, p.Y, W, H);
251 var ohRect = TileBounds.ApplyOverhang(
252 rect,
253 includeOverhang ? new Overhang(0, 0, 0, 0) : Overhang.None,
254 includeOverhang); // polygon uses base box; overhang impacts range, not shape
255
256 int x = ohRect.X;
257 int y = ohRect.Y;
258
259 // Flat-top vertices from bounding box
260 return new[]
261 {
262 new Point(x + W/4, y),
263 new Point(x + 3*W/4, y),
264 new Point(x + W, y + H/2),
265 new Point(x + 3*W/4, y + H),
266 new Point(x + W/4, y + H),
267 new Point(x, y + H/2)
268 };
269 }

271 private static bool PointInPolygon(Point[] poly, Point p)
272 {
273 bool inside = false; int j = poly.Length - 1;
274 for (int i = 0; i < poly.Length; i++)
275 {
276 bool intersect = ((poly[i].Y > p.Y) != (poly[j].Y > p.Y)) &&
277 (p.X < (poly[j].X - poly[i].X) * (p.Y - poly[i].Y) / (float)(poly[j].Y - poly[i].Y) + poly[i].X);
278 if (intersect) inside = !inside; j = i;
279 }
280 return inside;
281 }
