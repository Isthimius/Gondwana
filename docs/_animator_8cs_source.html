Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

161 internal void CycleAnimation(long currentTick)
162 {
163 if (CurrentCycle is null)
164 return;
165
166 // if throttle is 0, stop the animation so loop doesn't hang
167 if (CurrentCycle._throttle == 0)
168 {
169 StopAnimation();
170 return;
171 }
172
173 //if "throttle" time has passed
174 while (currentTick >= LastTick + CurrentCycle._throttle)
175 {
176 // capture the LastTick value
177 LastTick += CurrentCycle._throttle;
178
179 // do not change image if animation is paused
180 if (parent.PauseAnimation)
181 return;
182
183 // cycle the frame and set the sprite image; this will add it to RefreshQueue
184 parent.CurrentFrame = CurrentCycle.Sequence.AdvanceFrame();
185
186 // raise the event
187 Cycled?.Invoke(new AnimatorEventArgs(parent, this));
188
189 // if terminating cycle is done, stop the animation
190 if (CurrentCycle.Sequence.CycleFinished)
191 StopAnimation();
192 }
