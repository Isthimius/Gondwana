Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

525 public override void Update(long tick)
526 {
527 if (tick == _lastTick)
528 return;
529
530 // --- PULSE TIMING (independent) ---
531 float dtPulse = 0f;
532 if (_pulseLastTick.HasValue)
533 dtPulse = HighResTimer.GetDuration(_pulseLastTick.Value, tick);
534
535 _pulseLastTick = tick;
536
537 if (dtPulse > 0f && dtPulse < 1f)
538 _timeSec += dtPulse;
539
540 if (_pulseTextEnabled)
541 {
542 float t = PulseT(_timeSec, _pulsePeriodSec, _pulseWave);
543 var c = LerpColor(_pulseFrom, _pulseTo, t);
544 if (c != _resolvedForeColor)
545 {
546 _resolvedForeColor = c;
547 ForceRefresh();
548 }
549 }
550 else
551 {
552 // keep resolved color at base if not pulsing
553 if (_resolvedForeColor != _foreColor)
554 {
555 _resolvedForeColor = _foreColor;
556 ForceRefresh();
557 }
558 }
559
560 // --- REVEAL TIMING (independent and clamped) ---
561 float dtReveal = 0f;
562 if (_revealLastTick == 0)
563 {
564 _revealLastTick = tick; // warm up: first frame uses 0 dt
565 }
566 else
567 {
568 dtReveal = HighResTimer.GetDuration(_revealLastTick, tick);
569 _revealLastTick = tick;
570
571 // Clamp to avoid first-frame "time warp" completing the whole text
572 if (dtReveal > 0.1f) dtReveal = 0.1f; // ~100ms/frame cap
573 if (dtReveal < 0f) dtReveal = 0f;
574 }
575
576 // --- Typewriter / word reveal ---
577 if (_textRevealMode == TextRevealMode.CharactersPerSecond)
578 {
579 _revealAccum += dtReveal;
580 if (_revealRate > 0f)
581 {
582 int step = (int)(_revealAccum * _revealRate);
583 if (step > 0)
584 {
585 _revealAccum -= step / _revealRate;
586
587 int old = _revealCharCount;
588 _revealCharCount = Math.Min(_revealCharCount + step, _revealTargetCharCount);
589
590 // Punctuation pauses for newly revealed chars
591 for (int i = old; i < _revealCharCount; i++)
592 _revealAccum -= PauseFor(_text[i]);
593
594 // Mark dirty + redraw
595 ForceRefresh();
596
597 // Fire TextRevealed with the cumulative revealed text
598 if (_revealCharCount > old)
599 {
600 var revealed = _text.Substring(0, _revealCharCount);
601 TextRevealed?.Invoke(revealed);
602 }
603
604 // If we've reached the target, fire completion (TextRevealed already sent above)
605 if (_revealCharCount >= _revealTargetCharCount)
606 {
607 _textRevealMode = TextRevealMode.None;
608 // Ensure ordering: revealed then complete
609 TextRevealComplete?.Invoke(_text);
610 }
611 }
612 }
613 }
614 else if (_textRevealMode == TextRevealMode.WordsPerSecond && _wordEndCharIndexes is not null)
615 {
616 _revealAccum += dtReveal;
617 if (_revealRate > 0f)
618 {
619 int step = (int)(_revealAccum * _revealRate);
620 if (step > 0)
621 {
622 _revealAccum -= step / _revealRate;
623
624 int oldCount = _revealCharCount;
625
626 _wordIndex = Math.Min(_wordIndex + step, _wordEndCharIndexes.Length);
627 _revealCharCount = _wordIndex == 0 ? 0 : _wordEndCharIndexes[_wordIndex - 1];
628
629 // Simple frontier pause (cheap). Loop oldCount.._revealCharCount-1 if you want per-char.
630 if (_revealCharCount > 0)
631 _revealAccum -= PauseFor(_text[_revealCharCount - 1]);
632
633 ForceRefresh();
634
635 // Fire TextRevealed with the cumulative revealed text
636 if (_revealCharCount > oldCount)
637 {
638 var revealed = _text.Substring(0, _revealCharCount);
639 TextRevealed?.Invoke(revealed);
640 }
641
642 if (_wordIndex >= _wordEndCharIndexes.Length)
643 {
644 _textRevealMode = TextRevealMode.None;
645 // Ensure ordering: revealed then complete
646 TextRevealComplete?.Invoke(_text);
647 }
648 }
649 }
650 }
651
652 // Let the base advance fade tween and set _lastTick
653 base.Update(tick);
654 }

656 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
657 {
658 var canvas = backbuffer.Canvas;
659 var rect = destRectScreen.ToSKRect();
660
661 float zoom;
662
663 if (Mode == DirectDrawingMode.SceneLayer)
664 {
665 // SceneLayer-mode has View == null by design; use the ambient render context.
666 var contextZoom = RenderContext.Current?.ViewportZoom ?? 1f;
667 zoom = (contextZoom > 0f)
668 ? (1f / contextZoom)
669 : 1f;
670 }
671 else
672 {
673 // View-mode is screen/UI; do not compensate for camera zoom.
674 zoom = 1f;
675 }
676
677 // Scale "pixel-like" adornments with zoom so text behaves like other world-space drawables.
678 float hPad = HorizontalPadding * zoom;
679 float vPad = VerticalPadding * zoom;
680 float shadowDx = _shadowDx * zoom;
681 float shadowDy = _shadowDy * zoom;
682 float shadowBlur = _shadowBlurSigma * zoom;
683 float outlineWidth = 1.5f * zoom;
684
685 // Background
686 if (_backColor.Alpha != 0)
687 {
688 using var bg = new SKPaint { Color = _backColor };
689 canvas.DrawRect(rect, bg);
690 }
691
692 _typeface ??= SKTypeface.Default;
693
694 // Build a paint we can reuse for layout + draw (TEXT SIZE IS IN SCREEN PIXELS)
695 using var paint = new SKPaint
696 {
697 Typeface = _typeface ?? SKTypeface.Default,
698 TextSize = _fontSize * zoom,
699 Color = _resolvedForeColor,
700 IsAntialias = true,
701 IsStroke = false,
702 TextAlign = _hAlign
703 };
704
705 // Auto-shrink: reflow until it fits height (if min size provided)
706 float fontSize = _fontSize * zoom;
707 float minFontSize = _minFontSize.HasValue ? _minFontSize.Value * zoom : 0f;
708
709 float innerW = Math.Max(0, rect.Width - hPad * 2f);
710 float innerH = Math.Max(0, rect.Height - vPad * 2f);
711
712 while (true)
713 {
714 paint.TextSize = fontSize;
715
716 // Rebuild when flagged, and also whenever zoom is in play so wrap/line-height stay correct.
717 if (_layoutDirty || zoom != 1f)
718 RebuildLayout(paint, innerW);
719
720 int drawableLines = _maxLines.HasValue ? Math.Min(_lines.Count, _maxLines.Value) : _lines.Count;
721 float totalH = drawableLines * _lineHeight;
722
723 bool exceedsHeight = totalH > innerH;
724 bool exceedsWidth = _lines.Any(line => paint.MeasureText(line) > innerW);
725
726 if (_minFontSize.HasValue &&
727 (exceedsHeight || exceedsWidth) &&
728 fontSize > minFontSize)
729 {
730 fontSize = Math.Max(minFontSize, fontSize - 1f);
731 _layoutDirty = true;
732
733 if (fontSize <= minFontSize)
734 break;
735
736 continue;
737 }
738
739 break;
740 }
741
742 // Determine how many characters are currently visible (content-driven reveal)
743 int visibleChars = _textRevealMode != TextRevealMode.None
744 ? Math.Clamp(_revealCharCount, 0, _text.Length)
745 : _text.Length;
746
747 // Build the set of lines to draw from the already-laid-out _lines,
748 // truncating at 'visibleChars' so wrapping and alignment still work.
749 List<string> drawLines = new(_lines.Count);
750 if (visibleChars <= 0)
751 {
752 // nothing to show
753 }
754 else if (visibleChars >= _text.Length)
755 {
756 drawLines.AddRange(_lines);
757 }
758 else
759 {
760 int remaining = visibleChars;
761 foreach (var ln in _lines)
762 {
763 if (remaining <= 0)
764 break;
765
766 if (ln.Length <= remaining)
767 {
768 drawLines.Add(ln);
769 remaining -= ln.Length;
770 }
771 else
772 {
773 drawLines.Add(ln.Substring(0, remaining));
774 remaining = 0;
775 }
776 }
777 }
778
779 bool wasTruncatedByMaxLines = _maxLines.HasValue && drawLines.Count > _maxLines.Value;
780
781 // Apply max-lines cap at draw time
782 int linesToDraw = _maxLines.HasValue ? Math.Min(drawLines.Count, _maxLines.Value) : drawLines.Count;
783
784 if (linesToDraw > 0 && wasTruncatedByMaxLines)
785 {
786 drawLines[linesToDraw - 1] = FitLineWithEllipsis(paint, drawLines[linesToDraw - 1], innerW);
787 }
788
789 // Vertical start (Skia draws at baseline, so apply ascent shift)
790 var fm = paint.FontMetrics;
791 float baselineShift = -fm.Ascent;
792
793 float contentH = linesToDraw * _lineHeight;
794 float yStart = _vAlign switch
795 {
796 VerticalAlign.Center => rect.Top + vPad + Math.Max(0, (innerH - contentH) * 0.5f),
797 VerticalAlign.Bottom => rect.Bottom - vPad - contentH,
798 _ => rect.Top + vPad
799 };
800
801 // Horizontal anchor per line
802 float xAnchorLeft = rect.Left + hPad;
803 float xAnchorCenter = rect.MidX;
804 float xAnchorRight = rect.Right - hPad;
805
806 canvas.Save();
807 canvas.ClipRect(rect);
808
809 float y = yStart;
810 for (int i = 0; i < linesToDraw; i++)
811 {
812 var line = drawLines[i];
813 float x = _hAlign switch
814 {
815 SKTextAlign.Center => xAnchorCenter,
816 SKTextAlign.Right => xAnchorRight,
817 _ => xAnchorLeft
818 };
819
820 if (_useShadow)
821 {
822 using var shadow = paint.Clone();
823 shadow.IsStroke = false;
824 shadow.MaskFilter = shadowBlur > 0f
825 ? SKMaskFilter.CreateBlur(SKBlurStyle.Normal, shadowBlur)
826 : null;
827 shadow.Color = new SKColor(0, 0, 0, _shadowAlpha);
828
829 // manually offset
830 canvas.DrawText(line, x + shadowDx, y + baselineShift + shadowDy, shadow);
831 }
832
833 if (_useOutline)
834 {
835 using var outline = paint.Clone();
836 outline.IsStroke = true;
837 outline.StrokeWidth = outlineWidth;
838 outline.Color = SKColors.Black;
839 canvas.DrawText(line, x, y + baselineShift, outline);
840 }
841
842 canvas.DrawText(line, x, y + baselineShift, paint);
843 y += _lineHeight;
844
845 // safety clip
846 if (y > rect.Bottom)
847 break;
848 }
849
850 canvas.Restore();
851 }

853 private void RebuildLayout(SKPaint paint, float maxWidth)
854 {
855 _lines.Clear();
856
857 var fm = paint.FontMetrics;
858 _lineHeight = ((fm.Descent - fm.Ascent) + fm.Leading) * LineSpacingMultiplier;
859
860 var paragraphs = _text.Replace("\r\n", "\n").Split('\n');
861
862 foreach (var para in paragraphs)
863 {
864 if (string.IsNullOrEmpty(para))
865 {
866 _lines.Add(string.Empty);
867 continue;
868 }
869
870 if (!_wrapText)
871 {
872 _lines.Add(para);
873 continue;
874 }
875
876 var words = para.Split(' ', StringSplitOptions.None);
877 var current = string.Empty;
878
879 foreach (var word in words)
880 {
881 string candidate = string.IsNullOrEmpty(current) ? word : current + " " + word;
882 float candidateWidth = paint.MeasureText(candidate);
883
884 if (candidateWidth <= maxWidth)
885 {
886 current = candidate;
887 continue;
888 }
889
890 // If current already has content, commit it first and retry this word on a new line.
891 if (!string.IsNullOrEmpty(current))
892 {
893 _lines.Add(current);
894 current = string.Empty;
895 }
896
897 // If the single word fits on its own line, use it.
898 if (paint.MeasureText(word) <= maxWidth)
899 {
900 current = word;
901 continue;
902 }
903
904 // Fallback: hard-break an overlong word by character.
905 var chunk = string.Empty;
906 foreach (char ch in word)
907 {
908 string next = chunk + ch;
909 if (paint.MeasureText(next) <= maxWidth || string.IsNullOrEmpty(chunk))
910 {
911 chunk = next;
912 }
913 else
914 {
915 _lines.Add(chunk);
916 chunk = ch.ToString();
917 }
918 }
919
920 if (!string.IsNullOrEmpty(chunk))
921 current = chunk;
922 }
923
924 if (!string.IsNullOrEmpty(current))
925 _lines.Add(current);
926 }
927
928 _layoutDirty = false;
929 }

931 [MethodImpl(MethodImplOptions.AggressiveInlining)]
932 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
933 {
934 float phase = (timeSec / periodSec) % 1f;
935 return wave == PulseWave.Sine
936 ? 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f)) // [0..1]
937 : (phase < 0.5f ? phase * 2f : 1f - (phase - 0.5f) * 2f); // triangle
938 }

940 [MethodImpl(MethodImplOptions.AggressiveInlining)]
941 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
942 {
943 t01 = Math.Clamp(t01, 0f, 1f);
944 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
945 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
946 byte bl = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
947 byte al = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
948 return new SKColor(r, g, bl, al);
949 }

951 private static string FitLineWithEllipsis(SKPaint paint, string text, float maxWidth)
952 {
953 const string ellipsis = "...";
954
955 if (string.IsNullOrEmpty(text))
956 return text;
957
958 if (paint.MeasureText(text) <= maxWidth)
959 return text;
960
961 float ellipsisWidth = paint.MeasureText(ellipsis);
962 if (ellipsisWidth > maxWidth)
963 return string.Empty;
964
965 int length = text.Length;
966 while (length > 0)
967 {
968 string candidate = text.Substring(0, length).TrimEnd() + ellipsis;
969 if (paint.MeasureText(candidate) <= maxWidth)
970 return candidate;
971
972 length--;
973 }
974
975 return ellipsisWidth <= maxWidth ? ellipsis : string.Empty;
976 }
