298 private void KeyboardEventPoller_KeyDown(Input.Keyboard.KeyDownEventArgs args)
299 {
300 var camera = RenderSurface.Host.ViewRenderer.Views[0].Camera;
301 var curPos = camera.PositionPx;
302 var sprite = SpriteManager.GetSpriteByID("rooster_1");
303
304 // Parse the received key string into the Keys enum (case-insensitive)
305 if (!Enum.TryParse<Keys>(args.KeyConfig.Key, ignoreCase: true, out var key))
306 {
307 // If parsing fails, ignore — preserves existing behavior for any non-standard strings
308 return;
309 }
310
311 switch (key)
312 {
313 case Keys.W:
314 camera.PanToOverDuration(new PointF(curPos.X, curPos.Y - 100), 1.5f);
315 break;
316 case Keys.A:
317 camera.PanToOverDuration(new PointF(curPos.X - 100, curPos.Y), 1.5f);
318 break;
319 case Keys.S:
320 camera.PanToOverDuration(new PointF(curPos.X, curPos.Y + 100), 1.5f);
321 break;
322 case Keys.D:
323 camera.PanToOverDuration(new PointF(curPos.X + 100, curPos.Y), 1.5f);
324 break;
325 case Keys.Right:
326 if (args.KeyAction == Input.Keyboard.KeyAction.Released)
327 sprite.Movement.SetAcceleration(new Vector2(0, 0));
328 else
329 sprite.Movement.SetAcceleration(new Vector2(2f, 0));
330
331 sprite.Movement.SetLinearDamping(0.3f);
332
333 break;
334 case Keys.Left:
335 if (args.KeyAction == Input.Keyboard.KeyAction.Released)
336 sprite.Movement.SetAcceleration(new Vector2(0, 0));
337 else
338 sprite.Movement.SetAcceleration(new Vector2(-2f, 0));
339
340 break;
341 case Keys.Up:
342 if (args.KeyAction == Input.Keyboard.KeyAction.Released)
343 sprite.Movement.SetAcceleration(new Vector2(0, 0));
344 else
345 sprite.Movement.SetAcceleration(new Vector2(0, -2f));
346
347 sprite.Movement.SetLinearDamping(0.3f);
348
349 break;
350 case Keys.Down:
351 if (args.KeyAction == Input.Keyboard.KeyAction.Released)
352 sprite.Movement.SetAcceleration(new Vector2(0, 0));
353 else
354 sprite.Movement.SetAcceleration(new Vector2(0, 2f));
355
356 break;
357 case Keys.PageUp:
358 sprite.ScaleBy(1.1f, 0.15f);
359 break;
360 case Keys.PageDown:
361 sprite.ScaleBy(0.9f, 0.15f);
362 break;
363 default:
364 break;
365 }

375 private void MouseEventPoller_MouseEvent(Input.Mouse.MouseEventArgs args)
376 {
377 var view = RenderSurface.Host.ViewRenderer.Views[0];
378 var layer = Scene.SceneLayers[0];
379
380 var screenPos = args.CurrentPosition;
381
382 var worldPx = view.ScreenPxToWorldPx(layer, screenPos);
383 var screenPx = view.WorldPxToScreenPx(layer, worldPx);
384 //Engine.Logger.LogTrace($"mouse={screenPos} roundtrip={s} cam={view.Camera.PositionPx} zoom={view.Viewport.Zoom} p={layer.Parallax}");
385 Engine.Logger.LogTrace($"\r\nscreen1 = {screenPos} \r\nworld = {worldPx} \r\nscreen2 = {screenPx}\r\n");
386
387 // 1) screen → world (via View)
388 var worldFromScreen = view.ScreenPxToWorldPx(layer, screenPos);
389
390 // 2) world → grid (via View wrapper, which calls SceneLayer internally)
391 var gridFromScreen = view.ScreenPxToGrid(layer, screenPos);
392
393 // 3) grid → world (via SceneLayer wrapper)
394 var worldFromGrid = layer.GridToWorldPx(gridFromScreen);
395
396 // 4) world → screen (via View)
397 var screenFromGrid = view.WorldPxToScreenPx(layer, worldFromGrid);
398
399 var dx = screenFromGrid.X - screenPos.X;
400 var dy = screenFromGrid.Y - screenPos.Y;
401
402 // Existing HUD text
403 var cameraPos = view.Camera.PositionPx;
404 var message =
405 $"Mouse Pos (screen): {screenPos.X}, {screenPos.Y}\n" +
406 $"World Pos (px): {worldFromScreen.X:F1}, {worldFromScreen.Y:F1}\n" +
407 $"Grid coordinates: {gridFromScreen.X}, {gridFromScreen.Y}\n" +
408 $"Camera Pos: (px): {cameraPos.X}, {cameraPos.Y}";
409 _textBlockMouse?.SetText(message);
410
411 // Highlight logic, unchanged
412 foreach (SceneLayerTile tile in layer)
413 tile.EnableFog = false;
414
415 var pickedTile = layer[gridFromScreen];
416 if (pickedTile is not null)
417 pickedTile.EnableFog = true;
418
419 ScrollWheelZoom(args, view, layer);
420
421 if (args.ButtonStates[Input.Mouse.MouseButton.Left].JustPressed)
422 {
423 var pos = args.CurrentPosition;
424 _clickEmitter.Position = new PointF(pos.X, pos.Y);
425 _particleSurface.Burst(_clickEmitter, 80);
426 }
427 }
