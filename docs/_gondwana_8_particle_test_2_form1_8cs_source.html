153 // helper for hue variation
154 private static SKColor HsvToColor(float h, float s, float v)
155 {
156 h %= 360f;
157 float c = v * s;
158 float x = c * (1 - Math.Abs((h / 60f) % 2 - 1));
159 float m = v - c;
160
161 float r = 0, g = 0, b = 0;
162 if (h < 60) (r, g, b) = (c, x, 0);
163 else if (h < 120) (r, g, b) = (x, c, 0);
164 else if (h < 180) (r, g, b) = (0, c, x);
165 else if (h < 240) (r, g, b) = (0, x, c);
166 else if (h < 300) (r, g, b) = (x, 0, c);
167 else (r, g, b) = (c, 0, x);
168
169 return new SKColor(
170 (byte)((r + m) * 255),
171 (byte)((g + m) * 255),
172 (byte)((b + m) * 255),
173 255);
174 }
