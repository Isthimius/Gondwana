42 public DirectComposite Add(DirectDrawingMovableBase child, bool keepCurrentOffset = true, Vector2? explicitLocalOffsetPx = null)
43 {
44 if (child is null)
45 throw new ArgumentNullException(nameof(child));
46
47 if (child.RenderSurfaceHost != RenderSurfaceHost)
48 throw new ArgumentException("Child's RenderSurfaceHost must match the Composite's RenderSurfaceHost.", nameof(child));
49
50 if (_children.Contains(child))
51 return this;
52
53 _children.Add(child);
54 child.Disposing += OnChildDisposing;
55
56 Vector2 offset = keepCurrentOffset
57 ? new Vector2(child.ScreenBounds.X - _anchor.X, child.ScreenBounds.Y - _anchor.Y)
58 : (explicitLocalOffsetPx ?? Vector2.Zero);
59
60 _localOffsetPx[child] = offset;
61
62 // Normalize immediately
63 child.SetPosition(new Vector2(_anchor.X, _anchor.Y) + offset);
64 return this;
65 }

184 minY = b.Top;

187 maxX = b.Right;
188
189 if (b.Bottom > maxY)
190 maxY = b.Bottom;
191 }
192
193 if (minX == float.MaxValue)
194 return RectangleF.Empty;
195
196 return RectangleF.FromLTRB(
197 (int)Math.Floor(minX),
198 (int)Math.Floor(minY),
199 (int)Math.Ceiling(maxX),
