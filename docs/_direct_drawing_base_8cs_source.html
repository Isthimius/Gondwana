187 public void Draw(BackbufferBase backbuffer, RectangleF destRectScreen)
188 {
189 if (!Visible)
190 return;
191
192 var canvas = backbuffer.Canvas;
193
194 // Compute reveal clip rect (screen pixel space) from bounds
195 bool useClip = _revealT < 0.999f;
196
197 if (useClip)
198 {
199 var r = new SKRect(destRectScreen.Left, destRectScreen.Top, destRectScreen.Right, destRectScreen.Bottom);
200
201 SKRect clipRect = _revealDir switch
202 {
203 RevealDirection.LeftToRight => new SKRect(r.Left, r.Top, r.Left + r.Width * _revealT, r.Bottom),
204 RevealDirection.RightToLeft => new SKRect(r.Right - r.Width * _revealT, r.Top, r.Right, r.Bottom),
205 RevealDirection.TopToBottom => new SKRect(r.Left, r.Top, r.Right, r.Top + r.Height * _revealT),
206 RevealDirection.BottomToTop => new SKRect(r.Left, r.Bottom - r.Height * _revealT, r.Right, r.Bottom),
207 _ => r
208 };
209
210 // Early-out if reveal window is empty
211 if (clipRect.Width <= 0f || clipRect.Height <= 0f)
212 return;
213
214 // Outer save owns the clip lifetime
215 canvas.Save();
216
217 // Capture current matrix
218 var m = canvas.TotalMatrix;
219
220 canvas.ResetMatrix();
221 canvas.ClipRect(clipRect, SKClipOperation.Intersect, antialias: false);
222
223 // Put the prior matrix back so Draw() sees the same transform state
224 canvas.SetMatrix(m);
225 }
226
227 if (_opacity >= 0.999f)
228 {
229 OnDraw(backbuffer, destRectScreen);
230 }
231 else
232 {
233 using var layerPaint = new SKPaint
234 {
235 Color = new SKColor(255, 255, 255, (byte)(_opacity * 255))
236 };
237
238 canvas.SaveLayer(destRectScreen.ToSKRect(), layerPaint);
239 OnDraw(backbuffer, destRectScreen);
240 canvas.Restore(); // end SaveLayer
241 }
242
243 if (useClip)
244 canvas.Restore(); // end outer clip Save
245 }

342 protected internal void ForceRefresh()
343 {
344 switch (Mode)
345 {
346 case DirectDrawingMode.SceneLayer:
347 // bounds is WORLD-space
348 // there will only be one SceneLayer per DirectDrawing in this mode
349 SceneLayer!.RefreshQueue.AddWorldRect(_worldBounds);
350 break;
351
352 case DirectDrawingMode.View:
353 // bounds is SCREEN-space
354 // need to cycle through all SceneLayers for the View to which this DirectDrawing belongs
355 foreach (var sceneLayer in RenderSurfaceHost.Scene.SceneLayers)
356 {
357 sceneLayer.RefreshQueue.AddViewScreenRect(View!, sceneLayer, _screenBounds);
358 }
359 break;
360
361 default:
362 break;
363 }

371 public virtual void Update(long tick)
372 {
373 if (tick <= _lastTick)
374 return;
375
376 // Advance fade tween
377 if (_isFading)
378 {
379 float dt = HighResTimer.GetDuration(_lastTick, tick);
380
381 _fadeElapsedSec += dt;
382 float timeElapsed = Math.Clamp(_fadeElapsedSec / _fadeDurationSec, 0f, 1f);
383 // linear; swap in easing if you like
384 _opacity = _fadeFrom + (_fadeTo - _fadeFrom) * timeElapsed;
385
386 if (HideWhenFullyTransparent)
387 Visible = _opacity > 0f; // hides when hit zero
388
389 ForceRefresh();
390
391 if (timeElapsed >= 1f)
392 {
393 _isFading = false;
394 FadeToCompleted?.Invoke(this, this);
395 }
396 }
397
398 // Advance reveal tween
399 if (_revealAnimating)
400 {
401 float dt = HighResTimer.GetDuration(_lastTick, tick);
402 _revealElapsedSec = Math.Min(_revealElapsedSec + dt, _revealDurationSec);
403
404 float u = _revealElapsedSec / _revealDurationSec;
405 _revealT = (_revealEasing is null ? u : _revealEasing(u));
406 _revealT = _revealStart + (_revealTarget - _revealStart) * _revealT;
407
408 ForceRefresh();
409
410 if (_revealElapsedSec >= _revealDurationSec)
411 _revealAnimating = false;
412 }
413
414 _lastTick = tick;
415 }
