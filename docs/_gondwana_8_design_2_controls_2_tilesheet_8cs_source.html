85 private void DrawBmpWithGridLines()
86 {
87 picSource.Image = (Bitmap)_sheet.Bmp.Clone();
88
89 if ((_sheet.TileSize.Width <= 0) || (_sheet.TileSize.Height <= 0))
90 return;
91
92 Color backColor = Gondwana.Common.Drawing.Tilesheet.InferBackgroundColor(_sheet.Bmp);
93 Color penColor = Color.FromArgb(RGBMAX - backColor.R, RGBMAX - backColor.G, RGBMAX - backColor.B);
94
95 int w = picSource.Image.Width;
96 int h = picSource.Image.Height;
97 int x = _sheet.InitialOffsetX;
98 int y = _sheet.InitialOffsetY;
99
100 using (var graphics = Graphics.FromImage(picSource.Image))
101 {
102 using (var pen = new Pen(penColor))
103 using (var dashPen = new Pen(penColor))
104 {
105 dashPen.DashStyle = DashStyle.Dot;
106
107 while (x <= w)
108 {
109 // draw left border
110 graphics.DrawLine(pen, x, 0, x, h);
111
112 // draw right border
113 x += _sheet.TileSize.Width;
114 graphics.DrawLine(pen, x, 0, x, h);
115
116 // shift to the right as necessary for next line
117 x += _sheet.XPixelsBetweenTiles;
118 }
119
120 while (y <= h)
121 {
122 // draw top border
123 graphics.DrawLine(pen, 0, y, w, y);
124
125 // draw dotted line to mark "overlapping" pixels if necessary
126 if (_sheet.ExtraTopSpace > 0)
127 graphics.DrawLine(dashPen, 0, y + _sheet.ExtraTopSpace, w, y + _sheet.ExtraTopSpace);
128
129 // draw bottom border
130 y += _sheet.TileSize.Height;
131 graphics.DrawLine(pen, 0, y, w, y);
132
133
134 // shift down as necessary for next line
135 y += _sheet.YPixelsBetweenTiles;
136 }
137 }
138 }

141 private void txtName_Leave(object sender, EventArgs e)
142 {
143 if (txtName.Text == _sheet.Name)
144 return;
145
146 if (string.IsNullOrWhiteSpace(txtName.Text))
147 {
148 txtName.Text = _sheet.Name;
149 return;
150 }
151
152 if (Gondwana.Common.Drawing.Tilesheet.AllTilesheets.Find(x => x.Name == txtName.Text) != null)
153 {
154 var msg = "Another Tilesheet with this Name already exists. Changing the existing "
155 + "Tilesheet Name to this value will permanently replace the other Tilesheet. "
156 + "Is this okay?";
157
158 if (MessageBox.Show(msg, "Confirm Change",
159 MessageBoxButtons.OKCancel,
160 MessageBoxIcon.Question,
161 MessageBoxDefaultButton.Button2) != DialogResult.OK)
162 {
163 txtName.Text = _sheet.Name;
164 return;
165 }
166
167 ((ScriptDesigner)this.ParentForm).treeSelect.Nodes["ndFile"].Nodes["ndTilesheets"].Nodes[txtName.Text].Remove();
168 }
169
170 string oldName = _sheet.Name;
171 _sheet.Name = txtName.Text;
172
173 var node = ((ScriptDesigner)this.ParentForm).treeSelect.Nodes["ndFile"].Nodes["ndTilesheets"].Nodes[oldName];
174 node.Name = _sheet.Name;
175 node.Text = _sheet.Name;
176
177 Program.State.IsDirty = true;
178 }
