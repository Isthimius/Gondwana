48 public void ZoomAroundScreenPoint(SceneLayer layer, PointF screenPoint, float targetZoom, float durationSeconds)
49 {
50 if (layer is null)
51 throw new ArgumentNullException(nameof(layer));
52
53 // TODO: Make min/max zoom configurable per-viewport?
54 float minZoom = 0.1f;
55 float maxZoom = 8f;
56 targetZoom = Math.Clamp(targetZoom, minZoom, maxZoom);
57
58 // World under cursor BEFORE zoom changes
59 var worldUnderCursor = ScreenPxToWorldPx(layer, screenPoint);
60
61 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
62 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
63
64 float parallax = layer.Parallax;
65 if (Math.Abs(parallax) < 1e-6f)
66 parallax = 1f;
67
68 float localX = screenPoint.X - offsetX;
69 float localY = screenPoint.Y - offsetY;
70
71 // screen = offset + (world - camera*p) / zoom
72 // camera = (world - local*zoom) / p
73 float camTargetX = (worldUnderCursor.X - (localX * targetZoom)) / parallax;
74 float camTargetY = (worldUnderCursor.Y - (localY * targetZoom)) / parallax;
75
76 var cameraTargetUL = new PointF(camTargetX, camTargetY);
77
78 if (durationSeconds <= 0f)
79 {
80 Viewport.SnapZoom(targetZoom);
81 Camera.SnapTo(cameraTargetUL);
82 }
83 else
84 {
85 Viewport.ZoomToOverDuration(targetZoom, durationSeconds);
86 Camera.PanToOverDuration(cameraTargetUL, durationSeconds);
87 }
88 }

145 public RectangleF WorldRectToScreenRect(SceneLayer layer, RectangleF worldRect)
146 {
147 if (layer is null)
148 throw new ArgumentNullException(nameof(layer));
149
150 float zoom = (Viewport.Zoom <= 0f ? 1f : Viewport.Zoom);
151 float inverseZoom = 1f / zoom;
152
153 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
154 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
155
156 float parallax = layer.Parallax;
157
158 // screen = offset + (world - camera * p) / zoom
159 float localLeft = worldRect.Left - (Camera.PositionPx.X * parallax);
160 float localTop = worldRect.Top - (Camera.PositionPx.Y * parallax);
161
162 float scaledLeft = localLeft * inverseZoom;
163 float scaledTop = localTop * inverseZoom;
164 float scaledWidth = worldRect.Width * inverseZoom;
165 float scaledHeight = worldRect.Height * inverseZoom;
166
167 float screenLeft = offsetX + scaledLeft;
168 float screenTop = offsetY + scaledTop;
169
170 return new RectangleF(screenLeft, screenTop, scaledWidth, scaledHeight);
171 }

181 public RectangleF ScreenRectToWorldRect(SceneLayer layer, RectangleF screenRect)
182 {
183 if (layer is null)
184 throw new ArgumentNullException(nameof(layer));
185
186 float zoom = (Viewport.Zoom <= 0f ? 1f : Viewport.Zoom);
187
188 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
189 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
190
191 float parallax = layer.Parallax;
192
193 // (screen - offset)
194 float localLeft = screenRect.Left - offsetX;
195 float localTop = screenRect.Top - offsetY;
196
197 // world = camera*parallax + local * zoom
198 float worldLeft = (Camera.PositionPx.X * parallax) + (localLeft * zoom);
199 float worldTop = (Camera.PositionPx.Y * parallax) + (localTop * zoom);
200
201 float worldWidth = screenRect.Width * zoom;
202 float worldHeight = screenRect.Height * zoom;
203
204 return new RectangleF(worldLeft, worldTop, worldWidth, worldHeight);
