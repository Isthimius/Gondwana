35 private static void ResolveForSprite(Sprite sprite, CollisionWorld world)
36 {
37 var layer = sprite.SceneLayer;
38 if (layer == null)
39 return;
40
41 // Start from the sprite’s current collision rect in pixel space.
42 Rectangle rect = sprite.CollisionArea;
43
44 // Build AABB for broad-phase query.
45 var aabb = Aabb.FromRectangle(rect);
46
47 // For now, treat the sprite’s “layerMask” as 1 and it collides with everything (~0).
48 // Later you can pass real masks from the sprite’s collider.
49 const int spriteLayerMask = 1;
50 const int spriteCollidesWithMask = ~0;
51
52 world.QueryAabb(aabb, spriteLayerMask, spriteCollidesWithMask, _queryResults);
53
54 foreach (var collider in _queryResults)
55 {
56 // Don’t collide with yourself
57 if (ReferenceEquals(collider.Owner, sprite))
58 continue;
59
60 // Only care about Tiles on the same SceneLayer
61 if (collider.Owner is not Tile tile)
62 continue;
63
64 if (tile.SceneLayer != layer)
65 continue;
66
67 var other = collider.BoundsWorldPx.ToRectangle();
68
69 if (!rect.IntersectsWith(other))
70 continue;
71
72 // Compute intersection
73 Rectangle overlap = Rectangle.Intersect(rect, other);
74 if (overlap.IsEmpty)
75 continue;
76
77 // Centers for deciding push direction
78 float centerX = rect.Left + rect.Width * 0.5f;
79 float centerY = rect.Top + rect.Height * 0.5f;
80
81 float otherCenterX = other.Left + other.Width * 0.5f;
82 float otherCenterY = other.Top + other.Height * 0.5f;
83
84 // Minimum Translation Vector: push along the smallest axis of overlap.
85 int dx = 0;
86 int dy = 0;
87
88 if (overlap.Width < overlap.Height)
89 {
90 // Push horizontally
91 dx = (centerX < otherCenterX) ? -overlap.Width : overlap.Width;
92 }
93 else
94 {
95 // Push vertically
96 dy = (centerY < otherCenterY) ? -overlap.Height : overlap.Height;
97 }
98
99 rect.X += dx;
100 rect.Y += dy;
101 }
102
103 // If rect changed, map back to grid coordinates and update the sprite.
104 if (rect != sprite.CollisionArea)
105 {
106 var sceneCoord = SpriteManager.GridCoordinates(sprite, layer, rect);
107 sprite.SetPosition(new System.Numerics.Vector2(sceneCoord.X, sceneCoord.Y));
108 }
109 }
