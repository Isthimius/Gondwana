Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

49 private void ResolveForMover(ICollider mover, ICollisionMovableEntity movableOwner)
50 {
51 // Start from mover's current collision rect in world pixel space.
52 Rectangle rect = mover.Owner.CollisionArea;
53
54 // Broad-phase query based on current rect.
55 var aabb = Aabb.FromRectangle(rect);
56
57 _world.QueryAabb(aabb, mover.CollisionGroup, mover.CollidesWith, _queryResults, ignore: mover);
58
59 int totalDx = 0;
60 int totalDy = 0;
61
62 foreach (var otherCollider in _queryResults)
63 {
64 var otherRect = otherCollider.BoundsWorldPx.ToRectangle();
65
66 if (!rect.IntersectsWith(otherRect))
67 continue;
68
69 Rectangle overlap = Rectangle.Intersect(rect, otherRect);
70 if (overlap.IsEmpty)
71 continue;
72
73 // Trigger collisions: report only, no push-out.
74 if (mover.ResponseType == CollisionResponseType.Trigger || otherCollider.ResponseType == CollisionResponseType.Trigger)
75 {
76 TriggerOverlap?.Invoke(mover, otherCollider, overlap);
77 continue;
78 }
79
80 SolidOverlap?.Invoke(mover, otherCollider, overlap);
81
82 // Centers for deciding push direction
83 float centerX = rect.Left + rect.Width * 0.5f;
84 float centerY = rect.Top + rect.Height * 0.5f;
85
86 float otherCenterX = otherRect.Left + otherRect.Width * 0.5f;
87 float otherCenterY = otherRect.Top + otherRect.Height * 0.5f;
88
89 int dx = 0;
90 int dy = 0;
91
92 if (overlap.Width < overlap.Height)
93 dx = (centerX < otherCenterX) ? -overlap.Width : overlap.Width;
94 else
95 dy = (centerY < otherCenterY) ? -overlap.Height : overlap.Height;
96
97 rect.X += dx;
98 rect.Y += dy;
99
100 totalDx += dx;
101 totalDy += dy;
102 }
103

105 {

107 }
