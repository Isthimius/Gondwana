Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

75 public override void Present(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)

95 private void OnPaintSurface(object? sender, SKPaintSurfaceEventArgs e)
96 {
97 var canvas = e.Surface.Canvas;
98
99 var img = _currentImage;
100 if (img == null) return;
101
102 var srcI = SKRectI.Intersect(_sourceRect, new SKRectI(0, 0, img.Width, img.Height));
103 if (srcI.IsEmpty) return;
104
105 var destI = new SKRectI(srcI.Left, srcI.Top, srcI.Right, srcI.Bottom);
106
107 var boundsI = new SKRectI(0, 0, e.Info.Width, e.Info.Height);
108 var clippedDestI = SKRectI.Intersect(destI, boundsI);
109 if (clippedDestI.IsEmpty) return;
110
111 var dx = clippedDestI.Left - destI.Left;
112 var dy = clippedDestI.Top - destI.Top;
113 var clippedSrcI = new SKRectI(
114 srcI.Left + dx, srcI.Top + dy,
115 srcI.Left + dx + clippedDestI.Width,
116 srcI.Top + dy + clippedDestI.Height);
117
118 // clear the destination patch (overwrite)
119 canvas.DrawRect(clippedDestI, _clearPaint);
120
121 // blit the updated patch (overwrite)
122 canvas.DrawImage(img, clippedSrcI, clippedDestI, _presentPaint);
123
124 while (_toDispose.Count > 0)
125 _toDispose.Dequeue().Dispose();
126 }
