45 public override void Render(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)
46 {
47 // If the control can't paint, dispose immediately to avoid a leak.

75
76 // Clear with your configured clear color

79 var img = _currentImage;
80 if (img != null)
81 {
82 // IMPORTANT: For zero-copy, img MUST belong to this same GRContext.
83 // If you hand us a raster image, Skia will upload it each frame (works, but costs bandwidth).
84 var imgBoundsI = new SKRectI(0, 0, img.Width, img.Height);
85 var srcI = SKRectI.Intersect(_sourceRect, imgBoundsI);

93 if (!bounds.Contains(dest))

97 }
