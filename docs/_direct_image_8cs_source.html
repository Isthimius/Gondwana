Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

579 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
580 {
581 var canvas = backbuffer.Canvas;
582
583 // get SKImage to draw
584 var img = _image ?? (_bitmap != null ? SKImage.FromBitmap(_bitmap) : null);
585 if (img is null)
586 return;
587
588 // source rect (pixels in image space)
589 SKRect src = _src ?? new SKRect(0, 0, img.Width, img.Height);
590
591 // destination rect (screen space), computed by scale mode
592 SKRect dst = ComputeDestRect(destRectScreen.ToPixelAlignedRect(), src, _scale);
593
594 // apply tint/opacity via color filter
595 // combine tint with opacity (multiply in linear-ish sRGB)
596 var tint = _tint ?? SKColors.White;
597 var withAlpha = new SKColor(tint.Red, tint.Green, tint.Blue, _opacity);
598 using var cf = SKColorFilter.CreateBlendMode(withAlpha, SKBlendMode.Modulate);
599 _paint.ColorFilter = cf;
600
601 // rotation about anchor inside dst
602 if (_rotationDeg != 0f)
603 {
604 float ax = dst.Left + _anchorX * dst.Width;
605 float ay = dst.Top + _anchorY * dst.Height;
606 canvas.Save();
607 canvas.RotateDegrees(_rotationDeg, ax, ay);
608 }
609
610 // draw
611 canvas.DrawImage(img, src, dst, _paint);
612
613 if (_rotationDeg != 0f)
614 canvas.Restore();
615
616 // if we created a transient SKImage from a bitmap, dispose it now
617 if (_image is null && img is not null)
618 img.Dispose();
619 }

621 private static SKRect ComputeDestRect(Rectangle bounds, SKRect src, ScaleMode mode)
622 {
623 var dst = bounds.ToSKRect();
624
625 if (mode == ScaleMode.Stretch)
626 return dst;
627
628 if (mode == ScaleMode.Center) // center, no scale
629 {
630 float w = MathF.Min(dst.Width, src.Width);
631 float h = MathF.Min(dst.Height, src.Height);
632 float dx = dst.MidX - w * 0.5f, dy = dst.MidY - h * 0.5f;
633
634 return new SKRect(dx, dy, dx + w, dy + h);
635 }
636
637 float srcW = src.Width, srcH = src.Height;
638 float dstW = dst.Width, dstH = dst.Height;
639
640 float scaleX = dstW / srcW;
641 float scaleY = dstH / srcH;
642
643 float scale = mode switch
644 {
645 ScaleMode.Fit => MathF.Min(scaleX, scaleY),
646 ScaleMode.Fill => MathF.Max(scaleX, scaleY),
647 ScaleMode.PixelPerfect => MathF.Max(1f, MathF.Floor(MathF.Min(scaleX, scaleY))), _ => 1f
648 };
649
650 float w2 = srcW * scale, h2 = srcH * scale;
651 float x = dst.MidX - w2 * 0.5f;
652 float y = dst.MidY - h2 * 0.5f;
653
654 return new SKRect(x, y, x + w2, y + h2);
655 }
