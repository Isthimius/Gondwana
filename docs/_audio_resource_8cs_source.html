81 private ISampleProvider BuildAudioGraph(WaveStream source, float volume, float pan)
82 {
83 _pan = Math.Clamp(pan, -1f, 1f);
84 ISampleProvider baseProvider = source.ToSampleProvider();
85
86 int ch = baseProvider.WaveFormat.Channels;
87 if (ch < 1)
88 {
89 Engine.Logger.LogWarning(
90 "AudioResource {Key} has invalid channel count: {ChannelCount}", Key, ch);
91
92 // just pass through, no pan stage
93 volumeProvider = new VolumeSampleProvider(baseProvider)
94 {
95 Volume = Math.Clamp(volume, 0f, 1f)
96 };
97
98 return volumeProvider;
99 }
100
101 switch (ch)
102 {
103 case 1:
104 // MONO -> use PanningSampleProvider (expects mono, outputs stereo)
105 monoPanProvider = new PanningSampleProvider(baseProvider)
106 {
107 Pan = Math.Clamp(pan, -1f, 1f)
108 };
109 stereoPanProvider = null;
110 baseProvider = monoPanProvider; // now stereo
111 break;
112
113 case 2:
114 // STEREO -> use StereoSampleProvider for balance/pan
115 stereoPanProvider = new StereoPanSampleProvider(baseProvider);
116 ApplyStereoPan(stereoPanProvider, pan); // set L/R gains from pan
117 monoPanProvider = null;
118 baseProvider = stereoPanProvider; // stays stereo
119 break;
120
121 default:
122 // >2 CH -> pick first two channels, then treat as stereo
123 var mux = new MultiplexingSampleProvider(new[] { baseProvider }, 2);
124 mux.ConnectInputToOutput(0, 0); // channel 0 -> output L
125 mux.ConnectInputToOutput(1, 1); // channel 1 -> output R
126 baseProvider = mux;
127
128 stereoPanProvider = new StereoPanSampleProvider(baseProvider);
129 ApplyStereoPan(stereoPanProvider, pan);
130 monoPanProvider = null;
131 baseProvider = stereoPanProvider;
132 break;
133 }
134
135 // final stage: master volume
136 volumeProvider = new VolumeSampleProvider(baseProvider)
137 {
138 Volume = Math.Clamp(volume, 0f, 1f)
139 };
140
141 return volumeProvider;
142 }

348 internal void ReloadIntoManager(bool forceReload = false)
349 {
350 if (string.IsNullOrWhiteSpace(Key))
351 throw new InvalidOperationException("AudioResource has no Key and cannot be reloaded.");
352
353 var mgr = AudioResourceManager.Instance;
354
355 // If already loaded, just apply settings and bail (idempotent).
356 if (!forceReload && mgr.TryGet(Key, out var existing) && existing is not null)
357 {
358 existing.Volume = Volume;
359 existing.Pan = Pan;
360 existing.IsLooping = IsLooping;
361 return;
362 }
363
364 if (forceReload && mgr.Contains(Key))
365 mgr.Unload(Key); // safe: manager owns the live instance :contentReference[oaicite:2]{index=2}
366
367 // Load from persisted source
368 if (AssetIdentifier is not null && AssetIdentifier.IsValid)
369 {
370 using var s = AssetIdentifier.Data;
371 if (s is null)
372 throw new InvalidOperationException($"Missing asset data for {Key}.");
373
374 mgr.LoadFromStream(Key, s, SourceExtension ?? ".wav", Volume, Pan);
375 }
376 else if (!string.IsNullOrWhiteSpace(SourceFilePath))
377 {
378 mgr.LoadFromFile(Key, SourceFilePath, Volume, Pan);
379 }
380 else
381 {
382 throw new InvalidOperationException($"AudioResource '{Key}' has no persisted source.");
383 }
384
385 // Apply looping after load (LoadFromStream/File sets volume/pan during graph creation)
386 if (mgr.TryGet(Key, out var loaded) && loaded is not null)
387 loaded.IsLooping = IsLooping;
388 }

440 private static void ApplyStereoPan(StereoPanSampleProvider s, float pan)
441 {
442 pan = Math.Clamp(pan, -1f, 1f);
443 // equal-power: map [-1..1] to [0..pi/2]
444 float angle = (pan + 1f) * 0.5f * (float)(Math.PI / 2);
445 s.LeftVolume = MathF.Cos(angle);
446 s.RightVolume = MathF.Sin(angle);
447 }
