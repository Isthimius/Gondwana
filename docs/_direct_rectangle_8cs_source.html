Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

674 public override void Update(long tick)
675 {
676 if (tick <= _lastTick)
677 return;
678
679 // Compute dt from ticks (seconds)
680 float dt = 0f;
681
682 // no previous tick assume first frame, so skip dt-based updates
683 if (_pulseLastTick > 0)
684 {
685 dt = HighResTimer.GetDuration(_pulseLastTick, tick);
686 _timeSec += dt;
687 }
688
689 _pulseLastTick = tick;
690
691 bool changed = false;
692
693 if (_pulseFillEnabled)
694 {
695 float t = PulseT(_timeSec, _pulseFillPeriodSec, _pulseFillWave);
696 var c = LerpColor(_pulseFillFrom, _pulseFillTo, t);
697 if (_fillPaint.Color != c) { _fillPaint.Color = c; changed = true; }
698 }
699
700 if (_pulseBorderEnabled)
701 {
702 float t = PulseT(_timeSec, _pulseBorderPeriodSec, _pulseBorderWave);
703 var c = LerpColor(_pulseBorderFrom, _pulseBorderTo, t);
704
705 if (_strokePaint.Color != c)
706 {
707 _strokePaint.Color = c; // paint-only override
708 ForceRefresh();
709 }
710 }
711
712 if (changed)
713 {
714 _needsRebuildPaints = false; // we already set paints directly
715 ForceRefresh(); // request redraw
716 }
717
718 base.Update(tick);
719 }

748 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
749 {
750 var canvas = backbuffer.Canvas;
751
752 if (_needsRebuildPaints)
753 RebuildPaints();
754
755 var fillRect = destRectScreen.ToSKRect();
756 var strokeRect = fillRect;
757
758 bool willDrawStroke = !_isFilled || _borderColor.HasValue || _strokePaint.StrokeWidth > 0.01f;
759 float half = _strokePaint.StrokeWidth * 0.5f;
760
761 // 1) APPLY STROKE ALIGNMENT (use HALF the width; path is centered)
762 if (willDrawStroke && _strokeAlign != StrokeAlign.Center)
763 {
764 if (_strokeAlign == StrokeAlign.Inside)
765 strokeRect.Inflate(-half, -half);
766 else if (_strokeAlign == StrokeAlign.Outside)
767 strokeRect.Inflate(half, half);
768 }
769
770 // 1.5) corner radius for the stroke path, adjusted to keep the inner/outer arcs aligned to the fill
771 float strokeCornerRadius;
772 switch (_strokeAlign)
773 {
774 case StrokeAlign.Outside:
775 strokeCornerRadius = MathF.Max(0f, _cornerRadius - half);
776 break;
777 case StrokeAlign.Inside:
778 strokeCornerRadius = _cornerRadius + half;
779 break;
780 case StrokeAlign.Center:
781 default:
782 strokeCornerRadius = _cornerRadius;
783 break;
784 }
785
786 // 2) Dash for stroke only
787 _strokePaint.PathEffect = _dashPattern is { Length: > 0 }
788 ? SKPathEffect.CreateDash(_dashPattern, 0)
789 : null;
790
791 // 3) Draw fill (unmodified rect)
792 if (_isFilled)
793 {
794 if (_cornerRadius > 0)
795 canvas.DrawRoundRect(fillRect, strokeCornerRadius, strokeCornerRadius, _fillPaint);
796 else
797 canvas.DrawRect(fillRect, _fillPaint);
798 }
799
800 // 4) Draw stroke on its aligned rect (pure/opaque so it stays white)
801 if (willDrawStroke)
802 {
803 var prevBlend = _strokePaint.BlendMode;
804 var prevColor = _strokePaint.Color;
805 var prevAA = _strokePaint.IsAntialias;
806
807 _strokePaint.IsAntialias = false;
808 _strokePaint.BlendMode = SKBlendMode.Src;
809
810 // If pulsing, keep color from Update(),
811 // else use base border color (or previous).
812 var strokeColor = _pulseBorderEnabled ? _strokePaint.Color
813 : (_borderColor ?? prevColor);
814
815 _strokePaint.Color = strokeColor.WithAlpha(255);
816
817 if (_cornerRadius > 0)
818 canvas.DrawRoundRect(strokeRect, _cornerRadius, _cornerRadius, _strokePaint);
819 else
820 canvas.DrawRect(strokeRect, _strokePaint);
821
822 _strokePaint.IsAntialias = prevAA;
823 _strokePaint.BlendMode = prevBlend;
824 _strokePaint.Color = prevColor;
825 _strokePaint.StrokeJoin = SKStrokeJoin.Round;
826 _strokePaint.StrokeCap = SKStrokeCap.Round;
827 }
828 }

833 private void RebuildPaints()
834 {
835 // Ensure styles/AA set
836 _fillPaint.IsAntialias = true;
837 _fillPaint.Style = SKPaintStyle.Fill;
838
839 _strokePaint.IsAntialias = true;
840 _strokePaint.Style = SKPaintStyle.Stroke;
841
842 // Only rebuild stroke color if not pulsing border
843 if (!_pulseBorderEnabled)
844 {
845 // If border color set, use it for stroke; else match fill/base color
846 if (_borderColor.HasValue)
847 {
848 var sc = _strokePaint.Color; // preserve alpha if you want; otherwise:
849 _strokePaint.Color = _borderColor.Value;
850 }
851 else
852 {
853 _strokePaint.Color = _fillPaint.Color; // same as fill if no explicit border
854 }
855
856 _needsRebuildPaints = false;
857 }

888 [MethodImpl(MethodImplOptions.AggressiveInlining)]
889 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
890 {
891 float phase = (timeSec / periodSec) % 1f;
892 if (wave == PulseWave.Sine)
893 {
894 // (sin(2πx)+1)/2 in [0..1]
895 return 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f));
896 }
897 else // Triangle: ramp up 0..1 then down 1..0
898 {
899 return phase < 0.5f ? (phase * 2f) : (1f - ((phase - 0.5f) * 2f));
900 }
901 }
902
903 [MethodImpl(MethodImplOptions.AggressiveInlining)]
904 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
905 {
906 t01 = Math.Clamp(t01, 0f, 1f);
907
908 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
909 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
910 byte bch = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
911 byte aA = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
912
913 return new SKColor(r, g, bch, aA);
914 }
