Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

667 private void Cycle()
668 {
669 EngineDispatcher.Drain();
670
671 long tick = HighResTimer.GetCurrentTick();
672
673 DoBackgroundTasks(tick);
674
675 // if TargetFPS <= 0, render to screen unbounded
676 // otherwise, check if throttle time has passed since last tick...
677 if ((Configuration.TargetFPS <= 0)
678 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS)
679 {
680 DoForegroundTasks(tick);
681
682 // save time of this last tick; increment CPS counter
683 _lastForegroundTick = tick;
684 _netCyclesThisMeasure++;
685 }
686
687 // increment CPS counter
688 _grossCyclesThisMeasure++;
689
690 // if 0 or negative, sampling is turned off
691 if (Configuration.SamplingTimeForCPS > 0)
692 CalculateCPS(tick);
693 }
694
695 private void DoBackgroundTasks(long tick)
696 {
697 // find total real seconds passed since last background loop
698 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
699
700 BeforeBackgroundTasksExecute?.Invoke();
701
702 // raise pre-cycle timer events
703 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
704
705 // check for keyboard events
706 KeyboardEventPoller.Instance?.PollForEvents(tick);
707
708 // check for mouse events
709 MouseEventPoller.Instance?.PollForEvents(tick);
710
711 // check for gamepad events
712 GamepadEventPoller.Instance?.PollForEvents(tick);
713
714 // cycle Animator frames
715 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
716 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
717
718 // advance Sprite Movement paths
719 SpriteManager.Instance.MoveSprites(tick);
720
721 // resolve collisions after movement
722 foreach (var scene in Scenes.Scene.GetAllScenes())
723 {
724 foreach (var layer in scene.SceneLayers)
725 layer.CollisionResolver.Resolve();
726 }
727
728 // update cameras so any movement can mark RefreshNeeded = All.
729 foreach (var surface in RenderSurfaceHostRegistry.All)
730 surface.ViewManager.UpdateCameras(deltaSeconds);
731
732 AfterBackgroundTasksExecute?.Invoke();
733
734 _lastBackgroundTick = tick;
735 }
736
737 private void DoForegroundTasks(long tick)
738 {
739 // raise event
740 BeforeFrameRender?.Invoke();
741
742 // update the DirectDrawing instances' states
743 DirectDrawingManager.Instance.UpdateAll(tick);
744
745 // refresh all RenderSurfaceHost backbuffers
746 foreach (var surface in RenderSurfaceHostRegistry.All)
747 surface.RenderToBackbuffer(tick);
748
749 // render each Backbuffer to RenderSurfaceHost adapter
750 foreach (var surface in RenderSurfaceHostRegistry.All)
751 surface.PresentBackbufferToAdapter();
752
753 // update state of gamepad(s)
754 Input.GamepadManager?.Update();
755
756 // raise event
757 AfterFrameRender?.Invoke();
758
759 // raise post-cycle timer events
760 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
761 }

763 private void CalculateCPS(long tick)
764 {
765 // Has the sampling interval elapsed?
766 long elapsedTicks = tick - _lastCPSSamplingTick;
767 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
768 return;
769
770 // SNAPSHOT the counters BEFORE resetting or posting
771 long grossCycles = _grossCyclesThisMeasure;
772 long netCycles = _netCyclesThisMeasure;
773
774 // Compute using the snapshot
775 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
776 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
777 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
778
779 // Build immutable args NOW (so lambda doesn't read changing fields later)
780 var args = new CyclesPerSecondCalculatedEventArgs(
781 grossCycles,
782 netCycles,
783 grossCps,
784 netCps,
785 elapsedSec
786 );
787
788 // Post the snapshot
789 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
790
791 _grossCPS = grossCps;
792 _netFPS = netCps;
793
794 // Reset for next window
795 _lastCPSSamplingTick = tick;
796 _grossCyclesThisMeasure = 0;
797 _netCyclesThisMeasure = 0;
798 }

804 private void Dispose(bool disposing)
805 {
806 if (!IsDisposed)
807 {
808 if (disposing)
809 {
810 IsDisposing = true;
811
812 // stop the loop first so handlers don't race the cycle thread
813 try
814 {
815 Stop();
816 }
817 catch (Exception ex)
818 {
819 Logger.LogError(ex, "Unhandled exception calling Stop()");
820 }
821
822 // wait for the background loop to actually exit
823 try
824 {
825 _cycleTask?.Wait();
826 }
827 catch (Exception ex)
828 {
829 Logger.LogError(ex, "Error waiting for engine loop to exit.");
830 }
831
832 // raise Disposing on UI thread if possible; otherwise inline
833 if (UiDispatcher is not null)
834 UiDispatcher.Post(() => SafeInvoke(Disposing));
835 else
836 SafeInvoke(Disposing);
837
838 // managed cleanup...
839 Input.KeyboardEventPoller?.StopMonitoringAllKeys();
840 Input.MouseEventPoller?.StopMonitoringMouse();
841
842 if (Input.GamepadManager is not null)
843 foreach (var gamepadAdapter in Input.GamepadManager.ConnectedAdapters)
844 Input.GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
845
846 if (Configuration.LoggingMode == EngineLoggingMode.Asynchronous && Configuration.FlushAsyncLogsOnShutdown)
847 EngineLogger.StopAsyncLogging(flush: true);
848
849 Timer.ClearAll();
850 State.Clear();
851 }
852
853 // unmanaged cleanup...
854 IsDisposed = true;
855
856 if (disposing)
857 {
858 // now signal we're fully torn down
859 if (UiDispatcher is not null)
860 UiDispatcher.Post(() => SafeInvoke(Disposed));
861 else
862 SafeInvoke(Disposed);
863 }
864 }
865 }

870 catch (Exception ex)

872 // Keep disposal robust; log and continue
873 Logger.LogError(ex, "Unhandled exception in disposal event handler.");
874 }
