490 private void Cycle()
491 {
492 EngineDispatcher.Drain();
493
494 long tick = HighResTimer.GetCurrentTick();
495
496 DoBackgroundTasks(tick);
497
498 // if TargetFPS <= 0, render to screen unbounded
499 // otherwise, check if throttle time has passed since last tick...
500 if ((Configuration.TargetFPS <= 0)
501 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS)
502 {
503 DoForegroundTasks(tick);
504
505 // save time of this last tick; increment CPS counter
506 _lastForegroundTick = tick;
507 _netCyclesThisMeasure++;
508 }
509
510 // increment CPS counter
511 _grossCyclesThisMeasure++;
512
513 // if 0 or negative, sampling is turned off
514 if (Configuration.SamplingTimeForCPS > 0)
515 CalculateCPS(tick);
516 }
517
518 private void DoBackgroundTasks(long tick)
519 {
520 // find total real seconds passed since last background loop
521 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
522
523 BeforeBackgroundTasksExecute?.Invoke();
524
525 // raise pre-cycle timer events
526 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
527
528 // check for keyboard events
529 KeyboardEventPoller.Instance?.PollForEvents(tick);
530
531 // check for mouse events
532 MouseEventPoller.Instance?.PollForEvents(tick);
533
534 // check for gamepad events
535 GamepadEventPoller.Instance?.PollForEvents(tick);
536
537 // cycle Animator frames
538 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
539 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
540
541 // advance Sprite Movement paths
542 SpriteManager.MoveSprites(tick);
543
544 // resolve collisions after movement
545 foreach (var scene in Scenes.Scene.GetAllScenes())
546 CollisionResolver.ResolveSpriteTileCollisions(scene);
547
548 // update cameras so any movement can mark RefreshNeeded = All.
549 foreach (var surface in RenderSurfaceHostRegistry.All)
550 surface.ViewRenderer!.UpdateCameras(deltaSeconds);
551
552 AfterBackgroundTasksExecute?.Invoke();
553
554 _lastBackgroundTick = tick;
555 }
556
557 private void DoForegroundTasks(long tick)
558 {
559 // raise event
560 BeforeEngineCycle?.Invoke();
561
562 // refresh all RenderSurfaceHost backbuffers
563 foreach (var surface in RenderSurfaceHostRegistry.All)
564 surface.DrawRefreshQueueToBackbuffer(tick);
565
566 // update the DirectDrawing instances' states
567 DirectDrawingManager.Instance.UpdateAll(tick);
568
569 // render each Backbuffer to RenderSurfaceHost adapter
570 foreach (var surface in RenderSurfaceHostRegistry.All)
571 surface.RenderBackbufferToAdapter();
572
573 // update state of gamepad(s)
574 GamepadManager?.Update();
575
576 // raise event
577 AfterEngineCycle?.Invoke();
578
579 // raise post-cycle timer events
580 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
581 }

583 private void CalculateCPS(long tick)
584 {
585 // Has the sampling interval elapsed?
586 long elapsedTicks = tick - _lastCPSSamplingTick;
587 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks) return;
588
589 // SNAPSHOT the counters BEFORE resetting or posting
590 long grossCycles = _grossCyclesThisMeasure;
591 long netCycles = _netCyclesThisMeasure;
592
593 // Compute using the snapshot
594 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
595 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
596 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
597
598 // Build immutable args NOW (so lambda doesn’t read changing fields later)
599 var args = new CyclesPerSecondCalculatedEventArgs(
600 grossCycles,
601 netCycles,
602 grossCps,
603 netCps,
604 elapsedSec
605 );
606
607 // Post the snapshot
608 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
609
610 _grossCPS = grossCps;
611 _netFPS = netCps;
612
613 // Reset for next window
614 _lastCPSSamplingTick = tick;
615 _grossCyclesThisMeasure = 0;
616 _netCyclesThisMeasure = 0;
617 }

623 private void Dispose(bool disposing)
624 {
625 if (!IsDisposed)
626 {
627 if (disposing)
628 {
629 IsDisposing = true;
630
631 // stop the loop first so handlers don't race the cycle thread
632 try { Stop(); }
633 catch (Exception ex)
634 {
635 Logger.LogError(ex, "Unhandled exception calling Stop()");
636 }
637
638 // wait for the background loop to actually exit
639 try
640 {
641 _cycleTask?.Wait();
642 }
643 catch (Exception ex)
644 {
645 Logger.LogError(ex, "Error waiting for engine loop to exit.");
646 }
647
648 // raise Disposing on UI thread if possible; otherwise inline
649 if (UiDispatcher is not null)
650 UiDispatcher.Post(() => SafeInvoke(Disposing));
651 else
652 SafeInvoke(Disposing);
653
654 // managed cleanup...
655 KeyboardEventPoller?.StopMonitoringAllKeys();
656 MouseEventPoller?.StopMonitoringMouse();
657
658 if (GamepadManager is not null)
659 foreach (var gamepadAdapter in GamepadManager.ConnectedAdapters)
660 GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
661
662 if (Configuration.LoggingMode == EngineLoggingMode.Asynchronous && Configuration.FlushAsyncLogsOnShutdown)
663 EngineLogger.StopAsyncLogging(flush: true);
664
665 Timer.ClearAll();
666 State.Clear();
667 }
668
669 // unmanaged cleanup...
670 IsDisposed = true;
671
672 if (disposing)
673 {
674 // now signal we're fully torn down
675 if (UiDispatcher is not null)
676 UiDispatcher.Post(() => SafeInvoke(Disposed));
677 else
678 SafeInvoke(Disposed);
679 }
680 }
681 }

690 }
