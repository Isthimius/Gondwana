Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

247 internal void MoveSprites(long tick)
248 {
249 if (tick <= _lastTick)
250 return;
251
252 float duration = HighResTimer.GetDuration(_lastTick, tick);
253
254 for (int i = 0; i < _spriteList.Count; i++)
255 {
256 var sprite = _spriteList[i];
257
258 if (sprite._pendingDispose)
259 continue;
260
261 sprite.Movement.AdvanceMovement(duration);
262 sprite.AdvanceResize(duration);
263 sprite.AdvanceJiggle(duration);
264 }
265
266 SweepDisposedSprites();
267
268 _lastTick = tick;
269 }
270
271 private void SweepDisposedSprites()
272 {
273 int i = 0;
274 while (i < _spriteList.Count)
275 {
276 if (!_spriteList[i]._pendingDispose)
277 {
278 i++;
279 continue;
280 }
281
282 _spriteList[i].DisposeImmediate();
283
284 int last = _spriteList.Count - 1;
285 _spriteList[i] = _spriteList[last];
286 _spriteList.RemoveAt(last);
287 }
288 }
