Gondwana Game Engine @PROJECT_VERSION@
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

76 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(SceneLayer sceneLayer, Rectangle worldPixelRange, bool includeOverhang)
77 {
78 var result = new List<SceneLayerTile>();
79 WH(sceneLayer, out int W, out int H, out float halfW, out float halfH);
80
81 // Corner → coarse grid bounds (continuous)
82 var ul = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Left, worldPixelRange.Top));
83 var ur = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Right, worldPixelRange.Top));
84 var ll = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Left, worldPixelRange.Bottom));
85 var lr = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Right, worldPixelRange.Bottom));
86
87 int minX = (int)System.Math.Floor(System.Math.Min(System.Math.Min(ul.X, ur.X), System.Math.Min(ll.X, lr.X))) - 2;
88 int maxX = (int)System.Math.Ceiling(System.Math.Max(System.Math.Max(ul.X, ur.X), System.Math.Max(ll.X, lr.X))) + 2;
89 int minY = (int)System.Math.Floor(System.Math.Min(System.Math.Min(ul.Y, ur.Y), System.Math.Min(ll.Y, lr.Y))) - 2;
90 int maxY = (int)System.Math.Ceiling(System.Math.Max(System.Math.Max(ul.Y, ur.Y), System.Math.Max(ll.Y, lr.Y))) + 2;
91
92 // Clamp to layer bounds (no wrapping)
93 int cols = sceneLayer.GridColumnCount;
94 int rows = sceneLayer.GridRowCount;
95
96 int xStart = System.Math.Max(minX, 0);
97 int xEnd = System.Math.Min(maxX, cols - 1);
98 int yStart = System.Math.Max(minY, 0);
99 int yEnd = System.Math.Min(maxY, rows - 1);
100 if (xStart > xEnd || yStart > yEnd) return result;
101
102 for (int y = yStart; y <= yEnd; y++)
103 {
104 for (int x = xStart; x <= xEnd; x++)
105 {
106 var gp = sceneLayer[x, y];
107 if (gp == null) continue;
108
109 var r = GetPixelRangeForTile(gp, includeOverhang);
110 if (r.IntersectsWith(worldPixelRange)) result.Add(gp);
111 }
112 }
113 return result;
114 }

122 public Rectangle GetPixelRangeForTile(Tile tile, bool includeOverhang)
123 {
124 WH(tile.SceneLayer, out int W, out int H, out float halfW, out float halfH);
125
126 // Top vertex of the diamond
127 var top = GetAnchorPixelAtSceneLayerCoordinates(tile.SceneLayer, tile.SceneLayerCoordinates);

129 // Diamond fits exactly in W×H box whose top-left is (top.X - W/2, top.Y)
130 var rect = new Rectangle(top.X - (int)halfW, top.Y, W, H);
131 return TileBounds.ApplyOverhang(rect, tile.OverhangPixels, includeOverhang);
