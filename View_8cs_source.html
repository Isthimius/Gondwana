Gondwana Game Engine v2.2.4
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

87 public void ZoomAroundScreenPoint(SceneLayer layer, PointF screenPoint, float targetZoom, float durationSeconds)
88 {
89 if (layer is null)
90 throw new ArgumentNullException(nameof(layer));
91
92 targetZoom = Math.Clamp(targetZoom, MinZoom, MaxZoom);
93
94 // World under cursor BEFORE zoom changes
95 var worldUnderCursor = ScreenPxToWorldPx(layer, screenPoint);
96
97 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
98 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
99
100 float parallax = layer.Parallax;
101 if (Math.Abs(parallax) < 1e-6f)
102 parallax = 1f;
103
104 float localX = screenPoint.X - offsetX;
105 float localY = screenPoint.Y - offsetY;
106
107 // screen = offset + (world - camera*p) / zoom
108 // camera = (world - local*zoom) / p
109 float camTargetX = (worldUnderCursor.X - localX * targetZoom) / parallax;
110 float camTargetY = (worldUnderCursor.Y - localY * targetZoom) / parallax;
111
112 var cameraTargetUL = new PointF(camTargetX, camTargetY);
113
114 if (durationSeconds <= 0f)
115 {
116 Viewport.SnapZoom(targetZoom);
117 Camera.SnapTo(cameraTargetUL);
118 }
119 else
120 {
121 Viewport.ZoomToOverDuration(targetZoom, durationSeconds);
122 Camera.PanToOverDuration(cameraTargetUL, durationSeconds);
123 }
124 }

147 public PointF ScreenPxToWorldPx(SceneLayer layer, PointF screenPx)
148 {
149 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
150 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
151 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;

220 public RectangleF WorldRectToScreenRect(SceneLayer layer, RectangleF worldRect)
221 {
222 if (layer is null)
223 throw new ArgumentNullException(nameof(layer));
224
225 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
226 float inverseZoom = 1f / zoom;
227
228 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
229 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
230
231 float parallax = layer.Parallax;
232
233 // screen = offset + (world - camera * p) / zoom
234 float localLeft = worldRect.Left - Camera.PositionPx.X * parallax;
235 float localTop = worldRect.Top - Camera.PositionPx.Y * parallax;
236
237 float scaledLeft = localLeft * inverseZoom;
238 float scaledTop = localTop * inverseZoom;
239 float scaledWidth = worldRect.Width * inverseZoom;
240 float scaledHeight = worldRect.Height * inverseZoom;
241
242 float screenLeft = offsetX + scaledLeft;
243 float screenTop = offsetY + scaledTop;
244
245 return new RectangleF(screenLeft, screenTop, scaledWidth, scaledHeight);
246 }

256 public RectangleF ScreenRectToWorldRect(SceneLayer layer, RectangleF screenRect)
257 {
258 if (layer is null)
259 throw new ArgumentNullException(nameof(layer));
260
261 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
262
263 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
264 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
265
266 float parallax = layer.Parallax;
267
268 // (screen - offset)
269 float localLeft = screenRect.Left - offsetX;
270 float localTop = screenRect.Top - offsetY;
271
272 // world = camera*parallax + local * zoom
273 float worldLeft = Camera.PositionPx.X * parallax + localLeft * zoom;
274 float worldTop = Camera.PositionPx.Y * parallax + localTop * zoom;
275
276 float worldWidth = screenRect.Width * zoom;
277 float worldHeight = screenRect.Height * zoom;
278
279 return new RectangleF(worldLeft, worldTop, worldWidth, worldHeight);
