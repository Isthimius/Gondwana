Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

93 public void ZoomAroundScreenPoint(SceneLayer layer, PointF screenPoint, float targetZoom, float durationSeconds)
94 {
95 ArgumentNullException.ThrowIfNull(layer);
96
97 targetZoom = Math.Clamp(targetZoom, MinZoom, MaxZoom);
98
99 PointF worldUnderCursor = ScreenPxToWorldPx(layer, screenPoint);
100
101 // Anchored zoom owns camera positioning until the zoom completes. Cancel an
102 // unrelated explicit pan, but leave follow configuration intact so it can
103 // resume on the next update after the anchored zoom finishes.
104 Camera.CancelPan();
105
106 if (durationSeconds <= 0f)
107 {
108 ClearZoomAnchor();
109 Viewport.SnapZoom(targetZoom);
110 SnapCameraToZoomAnchor(layer, screenPoint, worldUnderCursor);
111 return;
112 }
113
114 _zoomAnchorLayer = layer;
115 _zoomAnchorScreenPoint = screenPoint;
116 _zoomAnchorWorldPoint = worldUnderCursor;
117
118 Viewport.ZoomToOverDuration(targetZoom, durationSeconds);
119 }

146 private void SnapCameraToZoomAnchor(
147 SceneLayer layer,
148 PointF screenPoint,
149 PointF worldPoint)
150 {
151 float zoom = Viewport.Zoom > 0f
152 ? Viewport.Zoom
153 : 1f;
154
155 float offsetX =
156 Viewport.TargetRectPx.Left
157 + Viewport.ScreenOffsetPx.X;
158
159 float offsetY =
160 Viewport.TargetRectPx.Top
161 + Viewport.ScreenOffsetPx.Y;
162
163 float parallax = layer.Parallax;
164 if (Math.Abs(parallax) < 1e-6f)
165 parallax = 1f;
166
167 float localX = screenPoint.X - offsetX;
168 float localY = screenPoint.Y - offsetY;
169
170 // screen = offset + (world - camera*p) * zoom
171 // camera = (world - local/zoom) / p
172 float cameraX =
173 (worldPoint.X - localX / zoom)
174 / parallax;
175
176 float cameraY =
177 (worldPoint.Y - localY / zoom)
178 / parallax;
179
180 Camera.SnapTo(new PointF(cameraX, cameraY));
181 }

211 public PointF ScreenPxToWorldPx(SceneLayer layer, PointF screenPx)
212 {
213 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
214 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
215 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
216 float parallax = layer.Parallax;
217
218 float worldX = Camera.PositionPx.X * parallax
219 + (screenPx.X - offsetX) / zoom;
220
221 float worldY = Camera.PositionPx.Y * parallax
222 + (screenPx.Y - offsetY) / zoom;
223
224 return new PointF(worldX, worldY);
225 }

284 public RectangleF WorldRectToScreenRect(SceneLayer layer, RectangleF worldRect)
285 {
286 if (layer is null)
287 throw new ArgumentNullException(nameof(layer));
288
289 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
290
291 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
292 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
293
294 float parallax = layer.Parallax;
295
296 // screen = offset + (world - camera * p) * zoom
297 float localLeft = worldRect.Left - Camera.PositionPx.X * parallax;
298 float localTop = worldRect.Top - Camera.PositionPx.Y * parallax;
299

301 float scaledTop = localTop * zoom;
302 float scaledWidth = worldRect.Width * zoom;
303 float scaledHeight = worldRect.Height * zoom;
304
305 float screenLeft = offsetX + scaledLeft;
306 float screenTop = offsetY + scaledTop;
307
308 return new RectangleF(screenLeft, screenTop, scaledWidth, scaledHeight);

319 public RectangleF ScreenRectToWorldRect(SceneLayer layer, RectangleF screenRect)
320 {
321 if (layer is null)
322 throw new ArgumentNullException(nameof(layer));
323
324 float zoom = Viewport.Zoom <= 0f ? 1f : Viewport.Zoom;
325
326 float offsetX = Viewport.TargetRectPx.Left + Viewport.ScreenOffsetPx.X;
327 float offsetY = Viewport.TargetRectPx.Top + Viewport.ScreenOffsetPx.Y;
328
329 float parallax = layer.Parallax;
330
331 // (screen - offset)
332 float localLeft = screenRect.Left - offsetX;
333 float localTop = screenRect.Top - offsetY;
334
335 // world = camera*parallax + local / zoom
336 float worldLeft = Camera.PositionPx.X * parallax + localLeft / zoom;
337 float worldTop = Camera.PositionPx.Y * parallax + localTop / zoom;
338
339 float worldWidth = screenRect.Width / zoom;
340 float worldHeight = screenRect.Height / zoom;
341
342 return new RectangleF(worldLeft, worldTop, worldWidth, worldHeight);
