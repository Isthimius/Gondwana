Gondwana Game Engine v2.2.4
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

19 public Point GetAnchorPixelAtSceneLayerCoordinates(SceneLayer sceneLayer, PointF gp)
20 {
21 int W = sceneLayer.TileWidth;
22 int H = sceneLayer.TileHeight;
23
24 int originX = sceneLayer.OriginPx.X;
25 int originY = sceneLayer.OriginPx.Y;
26
27 // gp is in grid-space (x,y) already.
28 float dx = gp.X;
29 float dy = gp.Y;
30
31 float px = (dx - dy) * (W / 2f) - originX;
32 float py = (dx + dy) * (H / 2f) - originY;
33
34 return new Point((int)Math.Floor(px), (int)Math.Floor(py));
35 }

43 public PointF GetSceneLayerCoordinatesAtPixel(SceneLayer sceneLayer, PointF pixelPt)
44 {
45 int W = sceneLayer.TileWidth;
46 int H = sceneLayer.TileHeight;
47
48 int originX = sceneLayer.OriginPx.X;
49 int originY = sceneLayer.OriginPx.Y;
50
51 float a = (pixelPt.X + originX) / (W / 2f);
52 float b = (pixelPt.Y + originY) / (H / 2f);
53
54 float dx = (a + b) / 2f;
55 float dy = (b - a) / 2f;
56
57 return new PointF(dx, dy);
58 }

67 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(SceneLayer sceneLayer, Rectangle worldPixelRange, bool includeOverhang)
68 {
69 var result = new List<SceneLayerTile>();
70 var ul = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Left, worldPixelRange.Top));

72 var ll = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(worldPixelRange.Left, worldPixelRange.Bottom));

76 int maxX = (int)Math.Ceiling(new[] { ul.X, ur.X, ll.X, lr.X }.Max()) + 1;
77 int minY = (int)Math.Floor(new[] { ul.Y, ur.Y, ll.Y, lr.Y }.Min()) - 1;
78 int maxY = (int)Math.Ceiling(new[] { ul.Y, ur.Y, ll.Y, lr.Y }.Max()) + 1;
79
80 for (int y = minY; y <= maxY; y++)
81 {
82 for (int x = minX; x <= maxX; x++)

85 if (gp == null) continue;
86 var r = GetPixelRangeForTile(gp, includeOverhang);
