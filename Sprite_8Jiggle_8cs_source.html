Gondwana Game Engine v2.2.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

138 internal void AdvanceJiggle(float deltaSeconds)
139 {
140 if (!_isJiggling)
141 return;
142
143 // Determine target update rate
144 int targetFps = Engine.Instance.Configuration.TargetFPS;
145 float step = (targetFps > 0) ? (1f / targetFps) : deltaSeconds; // if uncapped, just run
146
147 _jiggleAccumulatedSeconds += deltaSeconds;
148
149 // Only advance at most once per "frame step"
150 if (_jiggleAccumulatedSeconds < step)
151 return;
152
153 // Consume one step (don’t loop; keep it cheap and stable)
154 float dt = step;
155 _jiggleAccumulatedSeconds -= step;
156
157 _prevJiggleOffsetX = _jiggleOffsetX;
158 _prevJiggleOffsetY = _jiggleOffsetY;
159
160 _jiggleElapsedSeconds += dt;
161
162 if (!_jiggleLoop && _jiggleDurationSeconds > 0f && _jiggleElapsedSeconds >= _jiggleDurationSeconds)
163 {
164 StopJiggle();
165 return;
166 }
167
168 float intensityX = _jiggleIntensityX;
169 float intensityY = _jiggleIntensityY;
170 float scaleIntensity = _jiggleScaleIntensity;
171
172 if (_jiggleDecay && !_jiggleLoop && _jiggleDurationSeconds > 0f)
173 {
174 float lifeT = _jiggleElapsedSeconds / _jiggleDurationSeconds;
175 lifeT = Math.Clamp(lifeT, 0f, 1f);
176
177 float decayFactor = 1f - lifeT;
178
179 intensityX = _jiggleStartIntensityX * decayFactor;
180 intensityY = _jiggleStartIntensityY * decayFactor;
181 scaleIntensity = _jiggleStartScaleIntensity * decayFactor;
182 }
183
184 float t = _jiggleElapsedSeconds * _jiggleSpeed;
185
186 float xWave =
187 MathF.Sin(t + _jigglePhaseX1) +
188 (0.5f * MathF.Sin((t * 2.37f) + _jigglePhaseX2));
189
190 float yWave =
191 MathF.Cos((t * 1.13f) + _jigglePhaseY1) +
192 (0.5f * MathF.Cos((t * 2.91f) + _jigglePhaseY2));
193
194 _jiggleOffsetX = xWave * 0.6667f * intensityX;
195 _jiggleOffsetY = yWave * 0.6667f * intensityY;
196
197 if (_jiggleAffectsScale && scaleIntensity > 0f)
198 {
199 float scaleWave =
200 MathF.Sin((t * 1.41f) + _jigglePhaseScale) +
201 (0.35f * MathF.Sin((t * 3.17f) + (_jigglePhaseScale * 0.5f)));
202
203 _jiggleScale = 1f + (scaleWave * 0.7407f * scaleIntensity);
204 if (_jiggleScale < 0.01f)
205 _jiggleScale = 0.01f;
206 }
207 else
208 {
209 _jiggleScale = 1f;
210 }
211
212 bool changed =
213 Math.Abs(_jiggleOffsetX - _prevJiggleOffsetX) > 0.01f ||
214 Math.Abs(_jiggleOffsetY - _prevJiggleOffsetY) > 0.01f;
215
216 if (changed && _sceneLayer != null)
217 {
218 var rect = DrawLocationWorld;
219
220 int inflateX = (int)MathF.Ceiling(_jiggleIntensityX);
221 int inflateY = (int)MathF.Ceiling(_jiggleIntensityY);
222
223 rect.Inflate(inflateX, inflateY);
224
225 _sceneLayer.RefreshQueue.AddWorldRect(rect);
226 }
227 }

235 internal RectangleF ApplyJiggleToDestRect(RectangleF destRect)
236 {
237 if (!_isJiggling)
238 return destRect;
239
240 RectangleF result = destRect;
241 result.Offset(_jiggleOffsetX, _jiggleOffsetY);
242
243 if (Math.Abs(_jiggleScale - 1f) > 0.0001f)
244 {
245 float centerX = result.X + (result.Width / 2f);
246 float centerY = result.Y + (result.Height / 2f);
247
248 result.Width *= _jiggleScale;
249 result.Height *= _jiggleScale;
250 result.X = centerX - (result.Width / 2f);
251 result.Y = centerY - (result.Height / 2f);
252 }
253
254 return result;
255 }
