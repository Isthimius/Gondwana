Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

18 public Point GetAnchorPixelAtSceneLayerCoordinates(SceneLayer sceneLayer, PointF gp)
19 {
20 int width = sceneLayer.TileWidth;
21 int height = sceneLayer.TileHeight;
22
23 // The layout uses even-q offset coordinates. Interpolate the
24 // staggered Y offset between the surrounding integer columns.
25 int baseColumn = (int)MathF.Floor(gp.X);
26 float columnProgress = gp.X - baseColumn;
27
28 // Preserve the existing integer-division behavior for odd tile heights.
29 float halfHeight = height / 2;
30 float currentColumnOffsetY = (baseColumn & 1) == 0 ? 0f : halfHeight;
31 float nextColumnOffsetY = ((baseColumn + 1) & 1) == 0 ? 0f : halfHeight;
32 float offsetY = currentColumnOffsetY + ((nextColumnOffsetY - currentColumnOffsetY) * columnProgress);
33
34 var origin = sceneLayer.OriginPx;
35 float x = -origin.X + (gp.X * width * 0.75f);
36 float y = -origin.Y + (gp.Y * height) + offsetY;
37
38 return new Point((int)MathF.Floor(x), (int)MathF.Floor(y));
39 }

48 public PointF GetSceneLayerCoordinatesAtPixel(SceneLayer sceneLayer, PointF pixelPt)
49 {
50 int W = sceneLayer.TileWidth;
51 int H = sceneLayer.TileHeight;
52
53 float originX = sceneLayer.OriginPx.X;
54 float originY = sceneLayer.OriginPx.Y;
55
56 // --- 1) Coarse column estimate using centers, not box top-left ---
57
58 // Center of column 0 is at originX + W/2
59 float dxFromCol0Center = pixelPt.X - (-originX + W * 0.5f);
60
61 // Columns advance by 0.75 * W horizontally
62 float colF = dxFromCol0Center / (W * 0.75f);
63 int approxCol = (int)Math.Round(colF);
64
65 // --- 2) Coarse row estimate using centers and parity ---
66
67 // Center Y of row 0 in this column:
68 // even col: originY + H/2
69 // odd col: originY + H/2 + H/2 = originY + H
70 float baseCenterY =
71 -originY + H * 0.5f
72 + ((approxCol & 1) == 0 ? 0f : H * 0.5f);
73
74 float dyFromRow0Center = pixelPt.Y - baseCenterY;
75 float rowF = dyFromRow0Center / H;
76 int approxRow = (int)Math.Round(rowF);
77
78 var best = new Point(approxCol, approxRow);
79 float bestDist = float.MaxValue;
80
81 // --- 3) Refine via polygon hit-test + nearest center ---
82
83 foreach (var cand in NeighborsFlatTop(approxCol, approxRow, includeSelf: true))
84 {
85 var poly = HexPolygonFlatTop(sceneLayer, cand.X, cand.Y, includeOverhang: false);
86
87 var pInt = new Point((int)Math.Round(pixelPt.X), (int)Math.Round(pixelPt.Y));
88 if (PointInPolygon(poly, pInt))
89 return new PointF(cand.X, cand.Y);
90
91 // Center of candidate hex
92 float cx = -originX + cand.X * (W * 0.75f) + W / 2f;
93 float cy = -originY + cand.Y * (float)H
94 + ((cand.X & 1) == 0 ? 0 : H / 2) + H / 2f;
95
96 float dx = cx - pixelPt.X;
97 float dy = cy - pixelPt.Y;
98 float d = dx * dx + dy * dy;
99
100 if (d < bestDist)
101 {
102 bestDist = d;
103 best = cand;
104 }
105 }
106
107 return new PointF(best.X, best.Y);
108 }

117 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(SceneLayer sceneLayer, Rectangle worldPixelRange, bool includeOverhang)
118 {
119 var result = new List<SceneLayerTile>();
120 int W = sceneLayer.TileWidth;
121 int H = sceneLayer.TileHeight;
122
123 int originX = sceneLayer.OriginPx.X;
124 int originY = sceneLayer.OriginPx.Y;
125
126 int minCol = (int)Math.Floor((worldPixelRange.Left + originX) / (W * 0.75f)) - 2;
127 int maxCol = (int)Math.Ceiling((worldPixelRange.Right + originX) / (W * 0.75f)) + 2;
128
129 for (int col = minCol; col <= maxCol; col++)
130 {
131 int yOffset = ((col & 1) == 0 ? 0 : H / 2);
132 int minRow = (int)Math.Floor((worldPixelRange.Top + originY - yOffset) / (float)H) - 2;
133 int maxRow = (int)Math.Ceiling((worldPixelRange.Bottom + originY - yOffset) / (float)H) + 2;
134
135 for (int row = minRow; row <= maxRow; row++)
136 {
137 var gp = sceneLayer[col, row];
138 if (gp == null) continue;
139 var r = GetPixelRangeForTile(gp, includeOverhang);
140 if (r.IntersectsWith(worldPixelRange))
141 result.Add(gp);
142 }
143 }
144
145 return result;
146 }

186 public SceneLayerTile GetAdjacentSceneLayerTile(SceneLayerTile gp, CardinalDirections dir)
187 {
188 int x = gp.GridCoordinatesAbs.X; int y = gp.GridCoordinatesAbs.Y;
189 bool even = (x & 1) == 0;
190 var m = gp.SceneLayer;
191 return dir switch
192 {
193 CardinalDirections.E => m[x + 1, y],
194 CardinalDirections.W => m[x - 1, y],
195 CardinalDirections.NE => m[x + 1, y - (even ? 1 : 0)],
196 CardinalDirections.SE => m[x + 1, y + (even ? 0 : 1)],
197 CardinalDirections.NW => m[x - 1, y - (even ? 1 : 0)],

199 CardinalDirections.N => null, // not a direct neighbor in hex grid
200 CardinalDirections.S => null,

225 // wrap like square; hex maps are usually finite with wrap disabled, but keep parity sane

235 private static IEnumerable<Point> NeighborsFlatTop(int col, int row, bool includeSelf)
236 {
237 if (includeSelf) yield return new Point(col, row);

243 yield return new Point(col - 1, row + (even ? 0 : 1));
244 yield return new Point(col - 1, row - (even ? 1 : 0));

247 private static Point[] HexPolygonFlatTop(SceneLayer sceneLayer, int col, int row, bool includeOverhang)
248 {
249 int W = sceneLayer.TileWidth;
250 int H = sceneLayer.TileHeight;
251
252 int originX = sceneLayer.OriginPx.X;
253 int originY = sceneLayer.OriginPx.Y;
254
255 var p = new Point(
256 -originX + (int)Math.Floor(col * (W * 0.75f)),
257 -originY + row * H + ((col & 1) == 0 ? 0 : H / 2));
258
259 var rect = new Rectangle(p.X, p.Y, W, H);
260 var ohRect = TileBounds.ApplyOverhang(
261 rect,
262 includeOverhang ? new Spacing(0, 0, 0, 0) : Spacing.None,
263 includeOverhang); // polygon uses base box; overhang impacts range, not shape
264
265 int x = ohRect.X;
266 int y = ohRect.Y;
267
268 // Flat-top vertices from bounding box
269 return new[]
270 {
271 new Point(x + W/4, y),
272 new Point(x + 3*W/4, y),
273 new Point(x + W, y + H/2),
274 new Point(x + 3*W/4, y + H),
275 new Point(x + W/4, y + H),
276 new Point(x, y + H/2)
277 };
278 }

280 private static bool PointInPolygon(Point[] poly, Point p)
281 {
282 bool inside = false; int j = poly.Length - 1;
283 for (int i = 0; i < poly.Length; i++)
284 {
285 bool intersect = ((poly[i].Y > p.Y) != (poly[j].Y > p.Y)) &&
286 (p.X < (poly[j].X - poly[i].X) * (p.Y - poly[i].Y) / (float)(poly[j].Y - poly[i].Y) + poly[i].X);
287 if (intersect) inside = !inside; j = i;
288 }
289 return inside;
290 }
