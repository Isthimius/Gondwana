Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

132 private static TilesheetDefinition FromJson(
133 string json,
134 string? sourceDescription)
135 {
136 if (string.IsNullOrWhiteSpace(json))
137 throw new ArgumentException("GTS JSON content must be a non-empty string.", nameof(json));
138
139 try
140 {
141 var definition = JsonConvert.DeserializeObject<TilesheetDefinition>(json, Settings);
142 if (definition is null)
143 {
144 var source = string.IsNullOrWhiteSpace(sourceDescription)
145 ? "GTS JSON content"
146 : sourceDescription;
147
148 throw new InvalidDataException(
149 $"Failed to deserialize {source}. Result was null.");
150 }
151
152 // Maintain compatibility with files written before frame-level metadata existed,
153 // and tolerate explicit null collection values from hand-authored JSON.
154 definition.Regions ??= [];
155 foreach (var region in definition.Regions)
156 region.Frames ??= [];
157
158 return definition;
159 }
160 catch (JsonException ex)
161 {
162 var source = string.IsNullOrWhiteSpace(sourceDescription)
163 ? "GTS JSON content"
164 : sourceDescription;
165
166 throw new InvalidDataException($"Failed to deserialize {source}.", ex);
167 }

170 private static TilesheetImageDefinition CreateImageDefinition(
171 Tilesheet tilesheet,
172 string? baseDirectory,
173 bool makePathsRelative)
174 {
175 if (!string.IsNullOrWhiteSpace(tilesheet.ImageFilePath))
176 {
177 return new TilesheetImageDefinition
178 {
179 FilePath = NormalizePath(
180 tilesheet.ImageFilePath,
181 baseDirectory,
182 makePathsRelative)
183 };
184 }
185
186 if (tilesheet.AssetIdentifier is not null)
187 {
188 var assetIdentifier = tilesheet.AssetIdentifier;
189 if (assetIdentifier.AssetType != AssetTypes.Image)
190 {
191 throw new InvalidOperationException(
192 $"Tilesheet '{tilesheet.Name}' references an asset of type '{assetIdentifier.AssetType}', but expected '{AssetTypes.Image}'.");
193 }
194
195 if (assetIdentifier.AssetsFile is null ||
196 string.IsNullOrWhiteSpace(assetIdentifier.AssetsFile.FilePath))
197 {
198 throw new InvalidOperationException(
199 $"Tilesheet '{tilesheet.Name}' has an asset identifier, but the assets file path is missing.");
200 }
201
202 return new TilesheetImageDefinition
203 {
204 AssetsFilePath = NormalizePath(
205 assetIdentifier.AssetsFile.FilePath,
206 baseDirectory,
207 makePathsRelative),
208 AssetEntryName = assetIdentifier.AssetName
209 };
210 }
211
212 throw new InvalidOperationException(
213 $"Tilesheet '{tilesheet.Name}' cannot be converted to a TilesheetDefinition because it has no ImageFilePath or AssetIdentifier.");
214 }

216 private static TilesheetRegionDefinition CreateRegionDefinition(
217 TilesheetRegion region)
218 {
219 var definition = new TilesheetRegionDefinition
220 {
221 Name = region.Name,
222 Area = region.Area,
223 TileSize = region.TileSize,
224 TilePadding = region.TilePadding,
225 RegionMargin = region.RegionMargin,
226 Overhang = region.Overhang,
227 CollisionAdjust = region.CollisionAdjust
228 };
229
230 // Persist every effective value. This keeps the format deterministic and means
231 // deserialization never has to infer whether a frame used an inherited value.
232 for (int y = 0; y < region.Rows; y++)
233 {
234 for (int x = 0; x < region.Columns; x++)
235 {
236 definition.Frames.Add(new TilesheetFrameDefinition
237 {
238 XTile = x,
239 YTile = y,
240 CollisionAdjust = region.GetFrameCollisionAdjust(x, y)
241 });
242 }
243 }
244
245 return definition;
246 }

284 private static string NormalizePath(
285 string path,
286 string? baseDirectory,
287 bool makeRelative)
288 {
289 if (string.IsNullOrWhiteSpace(path))
290 return path;
291
292 if (!makeRelative || string.IsNullOrWhiteSpace(baseDirectory))
293 return path.Replace('\\', '/');
294
295 if (!Path.IsPathRooted(path))
296 return path.Replace('\\', '/');
297
298 var fullPath = Path.GetFullPath(path);
299 var fullBaseDirectory = Path.GetFullPath(baseDirectory);
300
301 return Path.GetRelativePath(fullBaseDirectory, fullPath).Replace('\\', '/');
