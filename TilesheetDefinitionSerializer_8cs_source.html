Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

245 private static TilesheetImageDefinition CreateImageDefinition(
246 Tilesheet tilesheet,
247 string? baseDirectory,
248 bool makePathsRelative)
249 {
250 if (!string.IsNullOrWhiteSpace(tilesheet.ImageFilePath))
251 {
252 return new TilesheetImageDefinition
253 {
254 FilePath = NormalizePath(
255 tilesheet.ImageFilePath,
256 baseDirectory,
257 makePathsRelative)
258 };
259 }
260
261 if (tilesheet.AssetIdentifier is not null)
262 {
263 var assetIdentifier = tilesheet.AssetIdentifier;
264
265 if (assetIdentifier.AssetType != AssetTypes.Image)
266 {
267 throw new InvalidOperationException(
268 $"Tilesheet '{tilesheet.Name}' references an asset of type '{assetIdentifier.AssetType}', but expected '{AssetTypes.Image}'.");
269 }
270
271 if (assetIdentifier.AssetsFile is null ||
272 string.IsNullOrWhiteSpace(assetIdentifier.AssetsFile.FilePath))
273 {
274 throw new InvalidOperationException(
275 $"Tilesheet '{tilesheet.Name}' has an asset identifier, but the assets file path is missing.");
276 }
277
278 return new TilesheetImageDefinition
279 {
280 AssetsFilePath = NormalizePath(
281 assetIdentifier.AssetsFile.FilePath,
282 baseDirectory,
283 makePathsRelative),
284
285 AssetEntryName = assetIdentifier.AssetName
286 };
287 }
288
289 throw new InvalidOperationException(
290 $"Tilesheet '{tilesheet.Name}' cannot be converted to a TilesheetDefinition because it has no ImageFilePath or AssetIdentifier.");
291 }

325 private static string NormalizePath(
326 string path,
327 string? baseDirectory,
328 bool makeRelative)
329 {
330 if (string.IsNullOrWhiteSpace(path))
331 return path;
332
333 if (!makeRelative || string.IsNullOrWhiteSpace(baseDirectory))
334 return path.Replace('\\', '/');
335
336 // Preserve already-relative paths as-authored to avoid depending on the current working directory.
337 if (!Path.IsPathRooted(path))
338 return path.Replace('\\', '/');
339
340 var fullPath = Path.GetFullPath(path);
341 var fullBaseDirectory = Path.GetFullPath(baseDirectory);
342
343 return Path.GetRelativePath(fullBaseDirectory, fullPath).Replace('\\', '/');
344 }
