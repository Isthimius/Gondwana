Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

198 public virtual bool HitTest(View view,
199 Point screenPositionPx)
200 {
201 ArgumentNullException.ThrowIfNull(view);
202
203 if (!IsInputEnabled ||
204 !IsPointerInputEnabled ||
205 !Visible)
206 {
207 return false;
208 }
209
210 if (Mode == DirectDrawingMode.View
211 && !ReferenceEquals(View, view))
212 {
213 return false;
214 }
215
216 RectangleF screenBounds = GetDrawLocationScreen(view);
217
218 return !screenBounds.IsEmpty
219 && screenBounds.Contains(screenPositionPx.X, screenPositionPx.Y);
220 }
221
229 public virtual bool HitTest(Point screenPositionPx)
230 {
231 var views = RenderSurfaceHost.ViewManager.Views;
232
233 for (int index = views.Count - 1; index >= 0; index--)
234 {
235 View view = views[index];
236
237 if (!view.Viewport.TargetRectPx.Contains(screenPositionPx))
238 {
239 continue;
240 }
241
242 return HitTest(view, screenPositionPx);
243 }
244
245 return false;
246 }
