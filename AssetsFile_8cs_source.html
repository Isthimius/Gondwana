Gondwana Game Engine v2.2.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

101 private void LoadZip()
102 {
103 if (_isLoaded)
104 return;
105
106 try
107 {
108 Engine.Logger.LogInformation("Loading assets file: {FilePath}", FilePath);
109
110 _zipFile?.Close();
111 _zipFile = null;
112 _zipEntries.Clear();
113
114 if (!File.Exists(FilePath))
115 {
116 _isLoaded = true;
117 return;
118 }
119
120 _zipFile = new ZipFile(File.OpenRead(FilePath));
121
122 if (!string.IsNullOrEmpty(Password))
123 _zipFile.Password = Password;
124
125 // Force decryption on the first entry so a bad password fails early.
126 var testEntry = _zipFile.Cast<ZipEntry>().FirstOrDefault(e => e.IsFile);
127 if (testEntry != null)
128 {
129 using var testStream = _zipFile.GetInputStream(testEntry);
130 testStream.ReadByte();
131 }
132
133 foreach (ZipEntry entry in _zipFile)
134 {
135 if (!entry.IsFile)
136 continue;
137
138 var key = AssetsFileEntry.FromString(entry.Name);
139 if (key == null)
140 {
141 Engine.Logger.LogWarning("Invalid entry in assets file: {EntryName}", entry.Name);
142 continue;
143 }
144
145 using var entryStream = _zipFile.GetInputStream(entry);
146 using var ms = new MemoryStream();
147 entryStream.CopyTo(ms);
148
149 _zipEntries[key] = ms.ToArray();
150 }
151
152 _isLoaded = true;
153 }
154 catch (ZipException zex)
155 {
156 Engine.Logger.LogError(zex, "Decryption failed — check password or file integrity.");
157 throw;
158 }
159 catch (Exception ex)
160 {
161 Engine.Logger.LogError(ex, "Failed to load asset file.");
162 throw;
163 }
164 finally
165 {
166 // Once contents are buffered, no need to keep the file handle open.
167 _zipFile?.Close();
168 _zipFile = null;
169 }
170 }

288 public Stream? Get(AssetTypes type, string name)
289 {
290 if (string.IsNullOrWhiteSpace(name))
291 return null;
292
293 EnsureLoaded();
294
295 var exactKey = new AssetsFileEntry
296 {
297 AssetType = type,
298 AssetName = name
299 };
300
301 if (_zipEntries.TryGetValue(exactKey, out var exactData))
302 return new MemoryStream(exactData, writable: false);
303
304 var requestedBaseName = Path.GetFileNameWithoutExtension(name);
305
306 var match = _zipEntries.Keys.FirstOrDefault(k =>
307 k.AssetType == type &&
308 string.Equals(
309 Path.GetFileNameWithoutExtension(k.AssetName),
310 requestedBaseName,
311 StringComparison.OrdinalIgnoreCase));
312
313 return match != null && _zipEntries.TryGetValue(match, out var fallbackData)
314 ? new MemoryStream(fallbackData, writable: false)
315 : null;
316 }

335 _zipFile?.Close();

339 using (var fs = File.Create(FilePath))
340 using (var zipStream = new ZipOutputStream(fs) { IsStreamOwner = true })
341 {
342 if (!string.IsNullOrEmpty(Password))
343 zipStream.Password = Password;
344
345 foreach (var (key, data) in _zipEntries)
346 {
347 var entry = new ZipEntry(key.ToString())
348 {
349 DateTime = DateTime.Now
350 };
351
352 if (UseEncryption && !string.IsNullOrEmpty(Password))
353 entry.AESKeySize = 256;
354
355 zipStream.PutNextEntry(entry);
356
357 using var inputStream = new MemoryStream(data, writable: false);
358 inputStream.CopyTo(zipStream);
359 zipStream.CloseEntry();
360 }
361
362 zipStream.Finish();
363 }
364
365 Engine.Logger.LogInformation(
366 "Assets file saved: {FilePath} (Encrypted: {Encrypted})",
367 FilePath,
368 UseEncryption);
369
370 // Keep the in-memory copy as the source of truth.
371 _isLoaded = true;
372 }
