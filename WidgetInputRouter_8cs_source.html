Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

288 private void OnMouseEvent(MouseEventArgs args)
289 {
290 if (!_isStarted || _disposed)
291 return;
292
293 Point currentPosition = args.CurrentPosition;
294 WidgetHit? initialHit = HitTest(currentPosition);
295
296 UpdateMouseHover(initialHit, currentPosition, args.Tick);
297
298 foreach (MouseButton mouseButton in _mouseButtons)
299 {
300 if (args.IsButtonJustPressed(mouseButton))
301 {
302 ProcessMouseDown(initialHit, currentPosition, mouseButton, args.Tick);
303 }
304 }
305
306 if (args.PreviousPosition != currentPosition)
307 {
308 ProcessMouseMove(args.PreviousPosition, currentPosition, args.Tick);
309 }
310
311 foreach (MouseButton mouseButton in _mouseButtons)
312 {
313 if (args.IsButtonJustReleased(mouseButton))
314 {
315 ProcessMouseUp(currentPosition, mouseButton, args.Tick);
316 }
317 }
318
319 UpdateMouseHover(HitTest(currentPosition), currentPosition, args.Tick);
320 }

322 private void ProcessMouseDown(WidgetHit? hit,
323 Point position,
324 MouseButton mouseButton,
325 long tick)
326 {
327 if (_mouseCapture is not null || hit is null)
328 return;
329
330 FocusFromPointer(hit.Widget);
331
332 WidgetPointerButtonEnum widgetButton = MapMouseButton(mouseButton);

334 hit.Widget.DispatchPointerDown(CreatePointerArgs(hit.Widget,
335 hit.View,

338 tick,
339 MousePointerId));

342 hit.View,
343 widgetButton,
344 position);

347 private void ProcessMouseMove(Point previousPosition,
348 Point currentPosition,
349 long tick)
350 {
351 PointerCapture? capture = GetValidMouseCapture(tick);
352 WidgetHit? hit = capture is null ? HitTest(currentPosition) : null;
353 WidgetBase? recipient = capture?.Widget ?? hit?.Widget;
354 View? view = capture?.View ?? hit?.View;
355
356 if (recipient is null || view is null)
357 return;
358
359 var delta = new Vector2(currentPosition.X - previousPosition.X,
360 currentPosition.Y - previousPosition.Y);
361
362 WidgetPointerButtonEnum button = capture?.Button ?? WidgetPointerButtonEnum.None;
363
364 recipient.DispatchPointerMove(CreatePointerArgs(recipient,
365 view,
366 currentPosition,
367 button,
368 tick,
369 MousePointerId,
370 deltaPx: delta));
371
372 if (capture is not null)
373 capture.LastPosition = currentPosition;
374 }

376 private void ProcessMouseUp(Point position,
377 MouseButton mouseButton,
378 long tick)
379 {
380 WidgetPointerButtonEnum releasedButton = MapMouseButton(mouseButton);
381 PointerCapture? capture = GetValidMouseCapture(tick);
382
383 if (capture is null || capture.Button != releasedButton)
384 return;
385
386 WidgetBase recipient = capture.Widget;
387
388 recipient.DispatchPointerUp(CreatePointerArgs(recipient,
389 capture.View,
390 position,
391 releasedButton,
392 tick,
393 MousePointerId));
394
395 WidgetHit? releaseHit = HitTest(position);
396
397 if (IsSameHit(releaseHit, recipient, capture.View))
398 {
399 recipient.DispatchPointerClick(CreatePointerArgs(recipient,
400 capture.View,
401 position,
402 releasedButton,
403 tick,
404 MousePointerId,
405 clickCount: 1));
406 }
407
408 _mouseCapture = null;
409 }

491 private void OnTouchMoved(object? sender, TouchEventArgs args)
492 {
493 if (!_isStarted || _disposed)
494 return;
495
496 TouchPoint touch = args.Touch;
497
498 if (!_touchCaptures.TryGetValue(touch.Id, out PointerCapture? capture))
499 return;
500
501 if (!IsPointerRoutable(capture.Widget, capture.View))
502 {
503 capture.Widget.DispatchPointerUp(CreatePointerArgs(capture.Widget,
504 capture.View,
505 capture.LastPosition,
506 WidgetPointerButtonEnum.Touch,
507 args.Tick,
508 touch.Id));
509
510 _touchCaptures.Remove(touch.Id);
511 return;
512 }
513
514 var delta = new Vector2(touch.Position.X - capture.LastPosition.X,
515 touch.Position.Y - capture.LastPosition.Y);
516
517 capture.Widget.DispatchPointerMove(CreatePointerArgs(capture.Widget,
518 capture.View,
519 touch.Position,
520 WidgetPointerButtonEnum.Touch,
521 args.Tick,
522 touch.Id,
523 deltaPx: delta));
524
525 capture.LastPosition = touch.Position;
526 }

528 private void OnTouchEnded(object? sender, TouchEventArgs args)
529 {
530 if (!_isStarted || _disposed)
531 return;
532
533 TouchPoint touch = args.Touch;
534
535 if (!_touchCaptures.Remove(touch.Id, out PointerCapture? capture))
536 return;
537
538 WidgetBase recipient = capture.Widget;
539
540 recipient.DispatchPointerUp(CreatePointerArgs(recipient,
541 capture.View,
542 touch.Position,
543 WidgetPointerButtonEnum.Touch,
544 args.Tick,
545 touch.Id));
546
547 WidgetHit? releaseHit = HitTest(touch.Position);
548
549 if (touch.Phase != TouchPhase.Cancelled && IsSameHit(releaseHit, recipient, capture.View))
550 {
551 recipient.DispatchPointerClick(CreatePointerArgs(recipient,
552 capture.View,
553 touch.Position,
554 WidgetPointerButtonEnum.Touch,
555 args.Tick,
556 touch.Id,
557 clickCount: 1));
558 }

561 private WidgetHit? HitTest(Point screenPositionPx)
562 {
563 View? view = GetTopmostViewAt(screenPositionPx);
564
565 if (view is null)
566 return null;
567
568 WidgetBase[] snapshot;
569
570 lock (_syncRoot)
571 snapshot = [.. _widgets];
572
573 for (int index = snapshot.Length - 1; index >= 0; index--)
574 {
575 WidgetBase widget = snapshot[index];
576
577 if (widget.HitTest(view, screenPositionPx))
578 return new WidgetHit(widget, view);
579 }
580
581 return null;
582 }

584 private View? GetTopmostViewAt(Point screenPositionPx)
585 {
586 var views = _renderSurfaceHost.ViewManager.Views;
587
588 for (int index = views.Count - 1; index >= 0; index--)
589 {
590 View view = views[index];
591
592 if (view.Viewport.TargetRectPx.Contains(screenPositionPx))
593 return view;
594 }

596 return null;

666 private void ReleasePointerState(WidgetBase widget,
667 bool dispatchPointerUp)
668 {
669 if (ReferenceEquals(_hoveredHit?.Widget, widget))
670 _hoveredHit = null;
671
672 if (ReferenceEquals(_mouseCapture?.Widget, widget))
673 {
674 PointerCapture capture = _mouseCapture;
675
676 if (dispatchPointerUp)
677 {
678 widget.DispatchPointerUp(CreatePointerArgs(widget,
679 capture.View,
680 capture.LastPosition,
681 capture.Button,
682 tick: 0,
683 pointerId: MousePointerId));
684 }
685
686 _mouseCapture = null;
687 }
688
689 int[] touchIds = _touchCaptures.Where(x => ReferenceEquals(x.Value.Widget, widget))
690 .Select(x => x.Key)
691 .ToArray();
692
693 foreach (int touchId in touchIds)
694 {
695 PointerCapture capture = _touchCaptures[touchId];
696
697 if (dispatchPointerUp)
698 {
699 widget.DispatchPointerUp(CreatePointerArgs(widget,
700 capture.View,
701 capture.LastPosition,
702 WidgetPointerButtonEnum.Touch,
703 tick: 0,
704 pointerId: touchId));
705 }
706
707 _touchCaptures.Remove(touchId);
708 }
