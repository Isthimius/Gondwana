Gondwana Game Engine v2.4.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

76 internal static Tilesheet FromDefinition(
77 TilesheetDefinition definition,
78 string? baseDirectory = null,
79 AssetsFile? defaultAssetsFile = null)
80 {
81 ArgumentNullException.ThrowIfNull(definition);
82
83 var tilesheet = CreateTilesheet(
84 definition,
85 addDefaultRegion: false,
86 baseDirectory,
87 defaultAssetsFile);
88
89 foreach (var region in definition.Regions)
90 {
91 tilesheet.AddRegion(
92 region.Name,
93 region.Area,
94 region.TileSize,
95 region.TilePadding,
96 region.RegionMargin,
97 region.Overhang);
98 }
99
100 if (definition.Mask is not null)
101 {
102 tilesheet.ApplyMask(
103 new SKColor(
104 definition.Mask.Red,
105 definition.Mask.Green,
106 definition.Mask.Blue,
107 definition.Mask.Alpha),
108 definition.Mask.Tolerance);
109 }
110 else if (definition.PremultiplyAlpha)
111 {
112 tilesheet.ApplyPremultiplyAlpha();
113 }

170 private static Tilesheet CreateTilesheet(
171 TilesheetDefinition definition,
172 bool addDefaultRegion,
173 string? baseDirectory = null,
174 AssetsFile? defaultAssetsFile = null)
175 {
176 var image = definition.Image
177 ?? throw new InvalidOperationException(
178 "TilesheetDefinition must specify an image source.");
179
180 if (!string.IsNullOrWhiteSpace(image.FilePath))
181 {
182 var imagePath = ResolvePath(image.FilePath, baseDirectory);
183
184 return new Tilesheet(
185 definition.Name,
186 imagePath,
187 addDefaultRegion);
188 }
189
190 if (!string.IsNullOrWhiteSpace(image.AssetEntryName))
191 {
192 AssetsFile assetsFile;
193
194 if (!string.IsNullOrWhiteSpace(image.AssetsFilePath))
195 {
196 var assetsPath = ResolvePath(image.AssetsFilePath, baseDirectory);
197 assetsFile = LoadExistingAssetsFile(assetsPath);
198 }
199 else if (defaultAssetsFile is not null)
200 {
201 assetsFile = defaultAssetsFile;
202 }
203 else
204 {
205 throw new InvalidOperationException(
206 "TilesheetDefinition image specifies an AssetEntryName but does not specify an AssetsFilePath, and no default AssetsFile was provided.");
207 }
208
209 return new Tilesheet(
210 assetsFile,
211 image.AssetEntryName,
212 addDefaultRegion);
213 }
214
215 throw new InvalidOperationException(
216 "TilesheetDefinition must specify either Image.FilePath or Image.AssetEntryName.");
217 }

247 private static string ResolvePath(string path, string? baseDirectory)
248 {
249 if (Path.IsPathRooted(path))

253 return path;
