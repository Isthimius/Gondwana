Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

179 public void TranslateWorldPx(int dx, int dy)
180 {
181 if (dx == 0 && dy == 0)
182 return;
183
184 // Start from current world rect
185 var rect = CollisionArea;
186 rect.X += dx;
187 rect.Y += dy;
188
189 // Convert back to whatever coordinate system Sprite uses internally
190 var sceneCoord = GetSceneLayerCoordsFromSpriteWorldRect(rect);
191
192 SetPosition(new Vector2(sceneCoord.X, sceneCoord.Y));

321 [JsonIgnore]
322 public override Rectangle DrawLocationWorld
323 {
324 get
325 {
326 // get the "top left" of the Sprite gridCoordinates value
327 Point pxlPt = _sceneLayer.CoordinateSystem.GetAnchorPixelAtSceneLayerCoordinates(_sceneLayer, SceneLayerCoordinates);
328
329 // adjust X coord
330 switch (HorizAlign)
331 {
332 case HorizontalAlignment.Left:
333 // no adjustment necessary
334 break;
335
336 case HorizontalAlignment.Center:
337 // shift right by half the difference between Tile Width values
338 // if Sprite Width > GridPt Width, Sprite will shift left
339 pxlPt.X += (_sceneLayer.TileWidth - _renderSize.Width) / 2;
340 break;
341
342 case HorizontalAlignment.Right:
343 // shift right by the entire difference between Tile Width values
344 // if Sprite Width > GridPt Width, Sprite will shift left
345 pxlPt.X += (_sceneLayer.TileWidth - _renderSize.Width);
346 break;
347
348 default:
349 // shouldn't get here...
350 break;
351 }
352
353 // adjust Y coord
354 switch (VertAlign)
355 {
356 case VerticalAlignment.Top:
357 // no adjustment necessary
358 break;
359
360 case VerticalAlignment.Middle:
361 // shift down by half the difference between Tile Height values
362 // if Sprite Height > GridPt Height, Sprite will shift up
363 pxlPt.Y += (_sceneLayer.TileHeight - _renderSize.Height) / 2;
364 break;
365
366 case VerticalAlignment.Bottom:
367 // shift down by the entire difference between Tile Height values
368 // if Sprite Height > GridPt Height, Sprite will shift up
369 pxlPt.Y += (_sceneLayer.TileHeight - _renderSize.Height);
370 break;
371
372 default:
373 // shouldn't get here...
374 break;
375 }
376
377 pxlPt.X += NudgeX;
378 pxlPt.Y += NudgeY;
379
380 return new Rectangle(pxlPt, _renderSize);
381 }

432 internal PointF GetSceneLayerCoordsFromSpriteWorldRect(Rectangle worldRectPx)
433 {
434 // work the Sprites.DrawLocation method backwards...
435 worldRectPx.X -= NudgeX;
436 worldRectPx.Y -= NudgeY;
437
438 // adjust X coord
439 switch (HorizAlign)
440 {
441 case HorizontalAlignment.Left:
442 // no adjustment necessary
443 break;
444
445 case HorizontalAlignment.Center:
446 // shift left by half the difference between Tile Width values
447 // if Sprite Width > GridPt Width, Sprite will shift right
448 worldRectPx.X -= (_sceneLayer.TileWidth - worldRectPx.Width) / 2;
449 break;
450
451 case HorizontalAlignment.Right:
452 // shift left by the entire difference between Tile Width values
453 // if Sprite Width > GridPt Width, Sprite will shift right
454 worldRectPx.X -= (_sceneLayer.TileWidth - worldRectPx.Width);
455 break;
456
457 default:
458 // shouldn't get here...
459 break;
460 }
461
462 // adjust Y coord
463 switch (VertAlign)
464 {
465 case VerticalAlignment.Top:
466 // no adjustment necessary
467 break;
468
469 case VerticalAlignment.Middle:
470 // shift up by half the difference between Tile Height values
471 // if Sprite Height > GridPt Height, Sprite will shift down
472 worldRectPx.Y -= (_sceneLayer.TileHeight - worldRectPx.Height) / 2;
473 break;
474
475 case VerticalAlignment.Bottom:
476 // shift up by the entire difference between Tile Height values
477 // if Sprite Height > GridPt Height, Sprite will shift down
478 worldRectPx.Y -= (_sceneLayer.TileHeight - worldRectPx.Height);
479 break;
480
481 default:
482 // shouldn't get here...
483 break;
484 }
485
486 // find and return the grid coordinates after the Sprite adjustments have been considered
487 return _sceneLayer.CoordinateSystem.GetSceneLayerCoordinatesAtPixel(_sceneLayer, worldRectPx.Location);
