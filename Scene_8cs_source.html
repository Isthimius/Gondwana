Gondwana Game Engine v2.2.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

453 public RectangleF GetWorldBoundsPx()
454 {
455 if (_sceneLayers.Count == 0)
456 return RectangleF.Empty;
457
458 RectangleF result = RectangleF.Empty;
459 bool hasBounds = false;
460
461 foreach (var layer in _sceneLayers)
462 {
463 var lb = layer.GetLayerBoundsPx();
464 if (lb.IsEmpty)
465 continue;
466
467 if (!hasBounds)
468 {
469 result = lb;
470 hasBounds = true;
471 }
472 else
473 {
474 result = RectangleF.Union(result, lb);
475 }
476 }
477
478 return result;
