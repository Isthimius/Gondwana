Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

319 internal void BindRenderSurfaceHost(RenderSurfaceHostBase host)
320 {
321 ArgumentNullException.ThrowIfNull(host);
322
323 // Scene.Empty is a shared null-object placeholder and is never treated as an
324 // actively rendered scene for binding ownership purposes.
325 if (ReferenceEquals(this, Empty))
326 return;
327
328 lock (_renderSurfaceHostSync)
329 {
330 if (_boundRenderSurfaceHost is not null &&
331 !ReferenceEquals(_boundRenderSurfaceHost, host))
332 {
333 throw new InvalidOperationException(
334 $"Scene '{ID}' is already bound to another RenderSurfaceHost.");
335 }
336
337 _boundRenderSurfaceHost = host;
338 }
339
340 // A GPU host redraws the full frame and never consumes RefreshQueue. Remove
341 // anything accumulated before binding; future additions are rejected by the
342 // queue's scene-policy callback.
343 if (!UsesDirtyRegionRendering)
344 {
345 foreach (var layer in _sceneLayers)
346 layer.RefreshQueue.ClearRefreshQueue();
347 }
348 }
349
354 internal void UnbindRenderSurfaceHost(RenderSurfaceHostBase host)
355 {
356 ArgumentNullException.ThrowIfNull(host);
357
358 if (ReferenceEquals(this, Empty))
359 return;
360
361 lock (_renderSurfaceHostSync)
362 {
363 if (ReferenceEquals(_boundRenderSurfaceHost, host))
364 _boundRenderSurfaceHost = null;
365 }
366 }

556 public RectangleF GetWorldBoundsPx()
557 {
558 if (_sceneLayers.Count == 0)
559 return RectangleF.Empty;
560
561 RectangleF result = RectangleF.Empty;
562 bool hasBounds = false;
563
564 foreach (var layer in _sceneLayers)
565 {
566 var lb = layer.GetLayerBoundsPx();
567 if (lb.IsEmpty)
568 continue;
569
570 if (!hasBounds)
571 {
572 result = lb;
573 hasBounds = true;
574 }
575 else
576 {
577 result = RectangleF.Union(result, lb);
578 }
579 }
580
581 return result;
