Gondwana Game Engine v2.5.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

61 private void OnPointerPressed(object? sender, PointerPressedEventArgs e)
62 {
63 // For mouse pointers, only emulate touch for the primary (left) button.
64 // Right- and middle-clicks should not generate touch events.
65 if (e.Pointer.Type == PointerType.Mouse &&
66 !e.GetCurrentPoint(sender as Control).Properties.IsLeftButtonPressed)
67 return;
68
69 var id = GetTouchId(e.Pointer);
70 var pos = GetPosition(e, sender as Control);
71 var point = new TouchPoint(id, pos, TouchPhase.Began);
72
73 _activeTouches[id] = point;
74 RebuildSnapshot();
75 }
76
77 private void OnPointerMoved(object? sender, PointerEventArgs e)
78 {
79 var id = GetTouchId(e.Pointer);
80
81 // Only track movement for contacts that are already active (i.e., button/finger is held).
82 // On desktop, PointerMoved fires even without a button down; skip those.
83 if (!_activeTouches.ContainsKey(id))
84 return;
85
86 var pos = GetPosition(e, sender as Control);
87 var point = new TouchPoint(id, pos, TouchPhase.Moved);
88
89 _activeTouches[id] = point;
90 RebuildSnapshot();
91 }
92
93 private void OnPointerReleased(object? sender, PointerReleasedEventArgs e)
94 {
95 var id = GetTouchId(e.Pointer);
96
97 if (!_activeTouches.ContainsKey(id))
98 return;
99
100 var pos = GetPosition(e, sender as Control);
101 var point = new TouchPoint(id, pos, TouchPhase.Ended);
102
103 _activeTouches.Remove(id);
104 RebuildSnapshot();
105 _pendingEnds.Enqueue(point);
106 }
107
108 private void OnPointerCaptureLost(object? sender, PointerCaptureLostEventArgs e)
109 {
110 // The system cancelled the pointer (e.g. incoming call on Android).
111 var id = GetTouchId(e.Pointer);
112
113 if (!_activeTouches.TryGetValue(id, out var existing))
114 return;
115
116 var point = new TouchPoint(existing.Id, existing.Position, TouchPhase.Cancelled);
117 _activeTouches.Remove(id);
118 RebuildSnapshot();
119 _pendingEnds.Enqueue(point);
120 }

123 public IReadOnlyList<TouchPoint> ConsumeEndedTouches()
124 {
125 if (_pendingEnds.IsEmpty)
126 return Array.Empty<TouchPoint>();
127
128 var snapshot = new List<TouchPoint>(_pendingEnds.Count);
129 while (_pendingEnds.TryDequeue(out var point))
130 snapshot.Add(point);
131 return snapshot;
132 }
133
134 private void RebuildSnapshot()
135 {
136 _activeTouchesSnapshot = _activeTouches.Values.ToArray();
137 }
