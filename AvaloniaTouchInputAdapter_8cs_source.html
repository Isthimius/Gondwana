Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

69 private void OnPointerPressed(object? sender, PointerPressedEventArgs e)
70 {
71 if (e.Pointer.Type == PointerType.Mouse && !_emulateMouse)
72 return;
73
74 // For mouse pointers, only emulate touch for the primary (left) button.
75 // Right- and middle-clicks should not generate touch events.
76 if (e.Pointer.Type == PointerType.Mouse &&
77 !e.GetCurrentPoint(sender as Control).Properties.IsLeftButtonPressed)
78 return;
79
80 var id = GetTouchId(e.Pointer);
81 var pos = GetPosition(e, sender as Control);
82 var point = new TouchPoint(id, pos, TouchPhase.Began);
83
84 _activeTouches[id] = point;
85 RebuildSnapshot();
86 _pendingBegins.Enqueue(point);
87 }
88
89 private void OnPointerMoved(object? sender, PointerEventArgs e)
90 {
91 if (e.Pointer.Type == PointerType.Mouse && !_emulateMouse)
92 return;
93
94 var id = GetTouchId(e.Pointer);
95
96 // Only track movement for contacts that are already active (i.e., button/finger is held).
97 // On desktop, PointerMoved fires even without a button down; skip those.
98 if (!_activeTouches.ContainsKey(id))
99 return;
100
101 var pos = GetPosition(e, sender as Control);
102 var point = new TouchPoint(id, pos, TouchPhase.Moved);
103
104 _activeTouches[id] = point;
105 RebuildSnapshot();
106 }
107
108 private void OnPointerReleased(object? sender, PointerReleasedEventArgs e)
109 {
110 if (e.Pointer.Type == PointerType.Mouse && !_emulateMouse)
111 return;
112
113 var id = GetTouchId(e.Pointer);
114
115 if (!_activeTouches.ContainsKey(id))
116 return;
117
118 var pos = GetPosition(e, sender as Control);
119 var point = new TouchPoint(id, pos, TouchPhase.Ended);
120
121 _activeTouches.Remove(id);
122 RebuildSnapshot();
123 _pendingEnds.Enqueue(point);
124 }
125
126 private void OnPointerCaptureLost(object? sender, PointerCaptureLostEventArgs e)
127 {
128 if (e.Pointer.Type == PointerType.Mouse && !_emulateMouse)
129 return;
130
131 // The system cancelled the pointer (e.g. incoming call on Android).
132 var id = GetTouchId(e.Pointer);
133
134 if (!_activeTouches.TryGetValue(id, out var existing))
135 return;
136
137 var point = new TouchPoint(existing.Id, existing.Position, TouchPhase.Cancelled);
138 _activeTouches.Remove(id);
139 RebuildSnapshot();
140 _pendingEnds.Enqueue(point);
141 }

144 public IReadOnlyList<TouchPoint> ConsumeBeganTouches()
145 {
146 if (_pendingBegins.IsEmpty)
147 return Array.Empty<TouchPoint>();
148
149 var snapshot = new List<TouchPoint>(_pendingBegins.Count);
150 while (_pendingBegins.TryDequeue(out var point))
151 snapshot.Add(point);
152 return snapshot;
153 }
154
156 public IReadOnlyList<TouchPoint> ConsumeEndedTouches()
157 {
158 if (_pendingEnds.IsEmpty)
159 return Array.Empty<TouchPoint>();
160
161 var snapshot = new List<TouchPoint>(_pendingEnds.Count);
162 while (_pendingEnds.TryDequeue(out var point))
163 snapshot.Add(point);
164 return snapshot;
165 }
