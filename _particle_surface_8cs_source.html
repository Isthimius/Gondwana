Gondwana Game Engine @PROJECT_VERSION@
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

293 public override void Update(long tick)
294 {
295 if (tick <= _lastTick)
296 return;
297
298 // Compute dt from ticks (seconds)
299 float dt = 0f;
300
301 // no previous tick assume first frame, so skip dt-based updates
302 if (_particlesLastTick > 0)
303 dt = HighResTimer.GetDuration(_particlesLastTick, tick);
304
305 _particlesLastTick = tick;
306
307 // (A) per-emitter motion
308 for (int i = 0; i < Emitters.Count; i++)
309 Emitters[i].OnUpdate?.Invoke(Emitters[i], dt);
310
311 // (B) emission
312 for (int i = 0; i < Emitters.Count; i++)
313 {
314 var em = Emitters[i];
315 float rate = MathF.Max(0f, em.EmitRate * GlobalEmitScale);
316 em._accumulator += rate * dt;
317
318 int toEmit = (int)em._accumulator;
319 if (toEmit > 0) { em._accumulator -= toEmit; EmitFrom(em, toEmit); }
320 }
321
322 // (C) integrate & compact
323 int write = 0;
324 for (int i = 0; i < _alive; i++)
325 {
326 ref var p = ref _particles[i];
327
328 // integrate acceleration (including gravity)
329 p.VX += p.AX * dt;
330 p.VY += p.AY * dt;
331
332 // clamp max velocity
333 if (p.MaxVelocity > 0f)
334 {
335 // thank you, Pythagoras
336 float vx = p.VX, vy = p.VY;
337 float velocity2 = (vx * vx) + (vy * vy);
338 float max2 = p.MaxVelocity * p.MaxVelocity;
339
340 if (velocity2 > max2)
341 {
342 float inv = p.MaxVelocity / MathF.Sqrt(velocity2);
343 p.VX = vx * inv;
344 p.VY = vy * inv;
345 }
346 }
347
348 // integrate velocity
349 p.X += p.VX * dt;
350 p.Y += p.VY * dt;
351
352 // to every season...
353 p.Rotation += p.AngularVel * dt;
354 p.Life -= dt;
355
356 // cull if out of bounds (with margin)
357 Rectangle cullBounds = (Mode == DirectDrawingMode.SceneLayer)
358 ? WorldBounds
359 : ScreenBounds;
360
361 bool isInView = p.X >= cullBounds.Left - CullingMarginX
362 && p.X <= cullBounds.Right + CullingMarginX
363 && p.Y >= cullBounds.Top - CullingMarginY
364 && p.Y <= cullBounds.Bottom + CullingMarginY;
365
366 if (p.Life > 0 && isInView)
367 _particles[write++] = p;
368 }
369 _alive = write;
370
371 ForceRefresh();
372
373 base.Update(tick);
374 }

403 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
404 {
405 var canvas = backbuffer.Canvas;
406
407 // Source bounds in the particle coordinate space
408 Rectangle srcBounds = (Mode == DirectDrawingMode.SceneLayer)
409 ? WorldBounds
410 : ScreenBounds;
411
412 // Guard against divide-by-zero (shouldn't happen, but cheap insurance)
413 float sx = (srcBounds.Width > 0) ? (destRectScreen.Width / srcBounds.Width) : 1f;
414 float sy = (srcBounds.Height > 0) ? (destRectScreen.Height / srcBounds.Height) : 1f;
415
416 // Usually uniform; if not, pick one (or average) for size scaling
417 float sSize = (sx + sy) * 0.5f;
418
419 for (int i = 0; i < _alive; i++)
420 {
421 ref var p = ref _particles[i];
422
423 // blend mode
424 _paint.BlendMode = p.BlendMode;
425
426 // life-based fade
427 float t = 1f - (p.Life / p.MaxLife);
428 byte a = (byte)(255 * (1f - t));
429
430 // choose tint: emitter override if set, otherwise current global
431 var tint = p.Tint ?? GlobalColorTint;
432
433 // apply global tint
434 _paint.Color = ApplyTint(p.Color, a, tint);
435
436 // Map particle position from srcBounds space -> destRectScreen space
437 float x = destRectScreen.Left + (p.X - srcBounds.Left) * sx;
438 float y = destRectScreen.Top + (p.Y - srcBounds.Top) * sy;
439
440 if (p.ParticleSprite is null)
441 {
442 float size = (p.Size * sSize) * (1f + 0.5f * t);
443 canvas.DrawCircle(x, y, size, _paint);
444 }
445 else
446 {
447 float s = p.Size * sSize;
448 var dst = new SKRect(x - s * 0.5f, y - s * 0.5f,
449 x + s * 0.5f, y + s * 0.5f);
450
451 canvas.Save();
452 canvas.RotateDegrees(p.Rotation, x, y);
453 canvas.DrawBitmap(p.ParticleSprite, dst, _paint);
454 canvas.Restore();
455 }
456 }

459 private void EmitFrom(ParticleEmitter em, int count)
460 {
461 for (int i = 0; i < count && _alive < _particles.Length; i++)
462 {
463 ref var p = ref _particles[_alive++];
464
465 // Position
466 if (em.JitterX == 0f && em.JitterY == 0f)
467 {
468 p.X = em.Position.X;
469 p.Y = em.Position.Y;
470 }
471 else
472 {
473 (float dx, float dy) = SampleSpawnOffset(em);
474 p.X = em.Position.X + dx;
475 p.Y = em.Position.Y + dy;
476 }
477
478 // Velocity
479 p.VX = NextRange(em.VelocityRangeX.Min, em.VelocityRangeX.Max);
480 p.VY = NextRange(em.VelocityRangeY.Min, em.VelocityRangeY.Max);
481
482 // Acceleration (emitter override or surface default)
483 p.AX = em.GravityX ?? this.GravityX;
484 p.AY = em.GravityY ?? this.GravityY;
485
486 // Life/Size
487 p.MaxLife = p.Life = NextRange(em.LifeRange.Min, em.LifeRange.Max);
488 p.Size = NextRange(em.SizeRange.Min, em.SizeRange.Max);
489
490 // Color & spin
491 p.Color = em.Color;
492 p.Rotation = NextRange(0f, 360f);
493 p.AngularVel = NextRange(-180f, 180f);
494
495 // choose sprite once; no emitter lookups in hot loop
496 p.ParticleSprite = em.ParticleSprite ?? this._particleSprite;
497
498 // tint override
499 p.Tint = em.Tint; // null means "use global at render"
500
501 // max speed clamp
502 p.MaxVelocity = em.MaxVelocity ?? 0f; // 0 = no clamp
503
504 // blend mode
505 p.BlendMode = em.BlendMode;
506
507 // User hook for last-mile per-particle tweaks
508 em.OnSpawn?.Invoke(ref p);
509 }
510 }

514 private (float dx, float dy) SampleSpawnOffset(ParticleEmitter em)
515 {
516 float jx = MathF.Abs(em.JitterX);
517 float jy = MathF.Abs(em.JitterY);
518
519 // Handle "line" cases cleanly (one axis zero)
520 if (jx == 0f)
521 return (0f, NextRange(-jy, jy));
522
523 if (jy == 0f)
524 return (NextRange(-jx, jx), 0f);
525
526 switch (em.SpawnDistribution)
527 {
528 default:
529 case ParticleSpawnDistribution.Rectangle:
530 return (NextRange(-jx, jx), NextRange(-jy, jy));
531
532 case ParticleSpawnDistribution.Ellipse:
533 {
534 // Uniform over area of ellipse
535 float a = NextRange(0f, MathF.Tau);
536 float r = MathF.Sqrt(NextRange(0f, 1f)); // sqrt => uniform area
537 return (MathF.Cos(a) * r * jx, MathF.Sin(a) * r * jy);
538 }
539
540 case ParticleSpawnDistribution.Ring:
541 {
542 // Uniform over area of annulus (ellipse-scaled)
543 float inner = Math.Clamp(em.RingInnerRadius01, 0f, 1f);
544 float a = NextRange(0f, MathF.Tau);
545
546 // area-uniform annulus: r = sqrt(inner^2 + u*(1-inner^2))
547 float u = NextRange(0f, 1f);
548 float r = MathF.Sqrt(inner * inner + u * (1f - inner * inner));
549
550 return (MathF.Cos(a) * r * jx, MathF.Sin(a) * r * jy);
551 }
552
553 case ParticleSpawnDistribution.Gaussian:
554 {
555 // Center-weighted, then clamped to ellipse boundary
556 float sd = Math.Clamp(em.GaussianStdDev01, 1e-4f, 10f);
557
558 float dx = NextGaussian() * (jx * sd);
559 float dy = NextGaussian() * (jy * sd);
560
561 // Clamp to ellipse: (dx/jx)^2 + (dy/jy)^2 <= 1
562 float nx = dx / jx;
563 float ny = dy / jy;
564 float q = nx * nx + ny * ny;
565
566 if (q > 1f)
567 {
568 float s = 1f / MathF.Sqrt(q);
569 dx *= s;
570 dy *= s;
571 }
572
573 return (dx, dy);
574 }
575 }
576 }

578 private float NextGaussian()
579 {
580 // Box–Muller transform (standard normal)
581 // Guard u1 from 0 to avoid log(0)
582 double u1 = 1.0 - _rng.NextDouble();
583 double u2 = 1.0 - _rng.NextDouble();
584 double mag = Math.Sqrt(-2.0 * Math.Log(u1));
585 double z0 = mag * Math.Cos(Math.Tau * u2);
586 return (float)z0;
587 }

589 // Fast, branch-free tint (multiplies RGB by tint)
590 // alpha = lifeAlpha * globalAlpha). Assumes particle base alpha = 255.
591 [MethodImpl(MethodImplOptions.AggressiveInlining)]
592 private SKColor ApplyTint(SKColor c, byte lifeAlpha, SKColor tint)
593 {
594 int r = (c.Red * tint.Red) / 255;
595 int g = (c.Green * tint.Green) / 255;
596 int b = (c.Blue * tint.Blue) / 255;
597 int a = (lifeAlpha * tint.Alpha) / 255;
598
599 return new SKColor((byte)r, (byte)g, (byte)b, (byte)a);
600 }
