Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

256 private void CreateGpuSurface(GRContext grContext, int width, int height)
257 {
258 // Rgba8888 / Premul is the natural format for an OpenGL render target.
259 var info = new SKImageInfo(width, height, SKColorType.Rgba8888, SKAlphaType.Premul);
260 _surface = SKSurface.Create(grContext, budgeted: true, info, _msaaSampleCount);
261
262 // SKSurface.Create returns null when the requested MSAA sample count is not supported
263 // by the hardware or driver. Fall back to no MSAA (sample count 1) so the surface
264 // is always valid after Initialize() completes.
265 if (_surface is null && _msaaSampleCount > 1)
266 _surface = SKSurface.Create(grContext, budgeted: true, info, sampleCount: 1);
267
268 if (_surface is null)
269 throw new InvalidOperationException(
270 $"Failed to create a {nameof(GpuBackbuffer)} GPU surface ({width}x{height}). " +
271 "The GRContext may be invalid or the pixel format is not renderable.");
272 }

274 private void CreateCpuSurface(int width, int height)
275 {
276 // Bgra8888 matches the native Windows GDI pixel format; used only before
277 // GRContext becomes available so rendering is always valid from frame one.
278 var info = new SKImageInfo(width, height, SKColorType.Bgra8888, SKAlphaType.Premul);
279 _cpuBitmap = new SKBitmap(info);
280 _surface = SKSurface.Create(info, _cpuBitmap.GetPixels(), _cpuBitmap.Info.RowBytes);
281 }
