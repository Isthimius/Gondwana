Gondwana Game Engine v2.5.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

141 public static void StopAsyncLogging(bool flush = true, TimeSpan? flushTimeout = null)
142 {
143 Channel<LogEvent>? ch;
144 Task? worker;
145 CancellationTokenSource? cts;
146
147 lock (_asyncGate)
148 {
149 ch = _channel;
150 worker = _worker;
151 cts = _cts;
152
153 if (ch == null || worker == null)
154 return;
155
156 // Completing the writer lets the worker drain then exit.
157 try { ch.Writer.TryComplete(); } catch { /* ignore */ }
158
159 // Clear shared references so other threads can't enqueue on this instance.
160 _channel = null;
161 _worker = null;
162 _cts = null;
163 }
164
165 try { cts?.Cancel(); } catch { /* never crash */ }
166
167 if (flush)
168 {
169 var timeout = flushTimeout ?? TimeSpan.FromSeconds(2);
170 try { worker!.Wait(timeout); } catch { /* never crash */ }
171 }
172
173 try { cts?.Dispose(); } catch { /* ignore */ }
174 }

283 // Must be called under _asyncGate.
284 private static void EnsureAsyncStarted_NoLock(bool forceRestart)
285 {
286 if (_mode != EngineLoggingMode.Asynchronous)
287 return;
288
289 if (!forceRestart && _worker != null)
290 return;
291
292 if (forceRestart && _channel != null)
293 {
294 // Best-effort stop without flush; caller controls flush via StopAsyncLogging.
295 try { _channel.Writer.TryComplete(); } catch { /* ignore */ }
296 }
297
298 int capacityLocal = _capacity;
299
300 var channel = Channel.CreateBounded<LogEvent>(new BoundedChannelOptions(capacityLocal)
301 {
302 SingleReader = true,
303 SingleWriter = false,
304 FullMode = BoundedChannelFullMode.DropWrite // fire-and-forget; drop if full
305 });
306
307 _channel = channel;
308
309 _cts?.Dispose();
310 _cts = new CancellationTokenSource();
311
312 // Pass reader so WorkerLoop never dereferences static _channel (avoids races with Stop).
313 var reader = channel.Reader;
314 var token = _cts.Token;
315 _worker = Task.Run(() => WorkerLoop(reader, token));
316 }

336 private static async Task WorkerLoop(ChannelReader<LogEvent> reader, CancellationToken ct)
337 {
338 try
339 {
340 while (await reader.WaitToReadAsync(ct).ConfigureAwait(false))
341 {
342 while (reader.TryRead(out var ev))
343 {
344 try
345 {
346 // Executes formatting + sinks (console) off the engine thread.
347 var logger = GetOrCreateLoggerFactory().CreateLogger(ev.CategoryName);
348 logger.Log(ev.LogLevel, ev.EventId, ev.State, ev.Exception, ev.Formatter);
349 }
350 catch (Exception ex)
351 {
352 // Fallback: write to debug output to aid diagnostics without crashing.
353 Debug.WriteLine($"[EngineLogger] Logging failed: {ex.GetType().Name}: {ex.Message}");
354
355 // Raise event for applications that want to monitor logging failures.
356 try
357 {
358 LoggingError?.Invoke(null, new LoggingErrorEventArgs(ex, ev.CategoryName, ev.LogLevel));
359 }
360 catch
361 {
362 // Never let event handlers crash the logging infrastructure.
363 }
364 }
365 }
366 }
367 }
368 catch (System.OperationCanceledException) when (ct.IsCancellationRequested)
369 {
370 // Normal cancellation: no action required.
371 }
372 catch
373 {
374 // Swallow any channel or unexpected exceptions.
375 }
376 }
