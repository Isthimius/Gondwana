Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

152 internal void OpenAnimated(MenuDropDownAnimation animation,
153 float durationSec)
154 {
155 CancelPendingClose();
156
157 if (!Visible)
158 Show();
159
160 IsPointerInputEnabled = true;
161 IsOpen = true;
162 SetEntryInputEnabled(true);
163
164 PrepareVisualsForAnimation();
165
166 if (animation == MenuDropDownAnimation.None || durationSec <= 0f)
167 {
168 SetOpacity(1f);
169 SetRevealProgress(1f);
170 return;
171 }
172
173 SetOpacity(0f);
174 SetRevealProgress(animation == MenuDropDownAnimation.FadeAndReveal ? 0f : 1f);
175 FadeIn(durationSec);
176
177 if (animation == MenuDropDownAnimation.FadeAndReveal)
178 {
179 foreach (DirectDrawingBase visual in EnumerateVisuals())
180 {
181 visual.SetRevealDirection(DirectDrawingBase.RevealDirection.TopToBottom)
182 .RevealTo(1f, durationSec, EaseOutCubic);
183 }
184 }
185 }

187 internal void CloseAnimated(MenuDropDownAnimation animation,
188 float durationSec,
189 bool immediate = false)
190 {
191 IsOpen = false;
192 IsPointerInputEnabled = false;
193 SetEntryInputEnabled(false);
194 SetSelectedIndex(-1);
195 CancelPendingClose();
196
197 if (immediate || animation == MenuDropDownAnimation.None || durationSec <= 0f)
198 {
199 CancelVisualFades();
200 SetOpacity(1f);
201 SetRevealProgress(1f);
202
203 if (Visible)
204 Hide();
205
206 return;
207 }
208
209 _closeCompletionHandler = (_, _) =>
210 {
211 CancelPendingClose();
212
213 if (Visible)
214 Hide();
215 };
216
217 Panel.FadeToCompleted += _closeCompletionHandler;
218
219 // Closing is a clean fade. Any in-progress opening reveal is snapped
220 // fully open so it cannot continue unfolding while disappearing.
221 SetRevealProgress(1f);
222 FadeOut(durationSec);
223 }

280 private void RecalculateLayout()
281 {
282 int y = _theme.DropDownVerticalPadding;
283
284 foreach (Entry entry in _entries)
285 {
286 if (entry.Item is not null)
287 {
288 var offset = new Vector2(_theme.DropDownHorizontalPadding, y);
289 SetLocalOffset(entry.Item, offset);
290
291 Point position = Point.Round(new PointF(
292 GetPosition().X + offset.X,
293 GetPosition().Y + offset.Y));
294
295 entry.Item.SetBounds(new Rectangle(
296 position.X,
297 position.Y,
298 Width - (_theme.DropDownHorizontalPadding * 2),
299 _theme.ItemHeight));
300
301 y += _theme.ItemHeight;
302 continue;
303 }
304
305 if (entry.Separator is not null)
306 {
307 int lineX = _theme.DropDownHorizontalPadding + 5;
308 int lineY = y + (_theme.SeparatorHeight / 2);
309 var offset = new Vector2(lineX, lineY);
310
311 SetLocalOffset(entry.Separator, offset);
312
313 Point position = Point.Round(new PointF(
314 GetPosition().X + offset.X,
315 GetPosition().Y + offset.Y));
316
317 entry.Separator.ScreenBounds = new Rectangle(
318 position.X,
319 position.Y,
320 Math.Max(1, Width - (lineX * 2)),
321 1);
322
323 y += _theme.SeparatorHeight;
324 }
325 }
326
327 Height = Math.Max(1, y + _theme.DropDownVerticalPadding);
328
329 Point panelPosition = Point.Round(new PointF(GetPosition().X, GetPosition().Y));
330 Panel.ScreenBounds = new Rectangle(panelPosition.X, panelPosition.Y, Width, Height);
331 }

356 private int FindNextEnabledIndex(int startIndex, int direction)
357 {
358 if (_items.Count == 0)
359 return -1;
360
361 int index = startIndex;
362
363 for (int attempt = 0; attempt < _items.Count; attempt++)
364 {
365 index = (index + direction + _items.Count) % _items.Count;
366
367 if (_items[index].IsEnabled)
368 return index;
369 }
370
371 return -1;

387 private int EstimateRequiredWidth(string text, string? shortcutText)
388 {
389 float labelWidth = text.Length * _theme.EstimatedGlyphWidth;
390 float shortcutWidth = string.IsNullOrWhiteSpace(shortcutText)
391 ? 0f
392 : shortcutText.Length * _theme.EstimatedGlyphWidth + _theme.ShortcutGap;
393
394 int padding = (_theme.DropDownHorizontalPadding * 2) +
395 (_theme.ItemHorizontalPadding * 2);
396
397 return Math.Max(
398 _theme.MinimumDropDownWidth,
399 (int)Math.Ceiling(labelWidth + shortcutWidth + padding));
400 }
