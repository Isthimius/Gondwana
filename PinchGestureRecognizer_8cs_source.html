Gondwana Game Engine v2.4.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

54 private void OnTouchBegan(object? sender, TouchEventArgs e)
55 {
56 var prevCount = _activePoints.Count;
57 _activePoints[e.Touch.Id] = e.Touch.Position;
58 var newCount = _activePoints.Count;
59
60 // Transition into "exactly 2" — baseline the distance so the first move has a reference.
61 // Any other count change (1→3, 3→4, etc.) clears the baseline so updates pause.
62 if (prevCount != 2 && newCount == 2)
63 _lastDistance = GetCurrentDistance();
64 else if (newCount != 2)
65 _lastDistance = 0;
66 }
67
68 private void OnTouchMoved(object? sender, TouchEventArgs e)
69 {
70 if (!_activePoints.ContainsKey(e.Touch.Id))
71 return;
72
73 _activePoints[e.Touch.Id] = e.Touch.Position;
74
75 // Only emit updates when exactly two contacts are tracked and a baseline exists.
76 if (_activePoints.Count == 2 && _lastDistance > 0)
77 {
78 var currentDistance = GetCurrentDistance();
79
80 if (currentDistance > 0)
81 {
82 var scaleDelta = currentDistance / _lastDistance;
83 PinchUpdated?.Invoke(this, new PinchedEventArgs(scaleDelta, currentDistance));
84 }
85
86 _lastDistance = currentDistance;
87 }
88 }
89
90 private void OnTouchEnded(object? sender, TouchEventArgs e)
91 {
92 var prevCount = _activePoints.Count;
93 _activePoints.Remove(e.Touch.Id);
94 var newCount = _activePoints.Count;
95
96 // Transition from 3+ down to exactly 2 — re-baseline so there is no position jump.
97 if (prevCount != 2 && newCount == 2)
98 _lastDistance = GetCurrentDistance();
99 else if (newCount < 2)
100 _lastDistance = 0;
101 }

103 private double GetCurrentDistance()
104 {
105 using var enumerator = _activePoints.Values.GetEnumerator();
106 if (!enumerator.MoveNext()) return 0;
107 var a = enumerator.Current;
108 if (!enumerator.MoveNext()) return 0;
109 var b = enumerator.Current;
110
111 var dx = a.X - b.X;
112 var dy = a.Y - b.Y;
113 return Math.Sqrt(dx * dx + dy * dy);
114 }
