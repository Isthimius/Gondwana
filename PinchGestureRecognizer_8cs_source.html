Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

61 private void OnTouchBegan(object? sender, TouchEventArgs e)
62 {
63 var prevCount = _activePoints.Count;
64
65 if (prevCount == 2)
66 Emit(PinchPhase.Ended, PinchEnded);
67
68 _activePoints[e.Touch.Id] = e.Touch.Position;
69 var newCount = _activePoints.Count;
70
71 if (prevCount != 2 && newCount == 2)
72 BeginPinch();
73 else if (newCount != 2)
74 ClearBaseline();
75 }
76
77 private void OnTouchMoved(object? sender, TouchEventArgs e)
78 {
79 if (!_activePoints.ContainsKey(e.Touch.Id))
80 return;
81
82 _activePoints[e.Touch.Id] = e.Touch.Position;
83
84 // Only emit updates when exactly two contacts are tracked and a baseline exists.
85 if (_activePoints.Count == 2 && _lastDistance > 0)
86 {
87 var currentDistance = GetCurrentDistance();
88
89 if (currentDistance > 0)
90 {
91 Emit(PinchPhase.Updated, PinchUpdated, currentDistance);
92 }
93
94 _lastDistance = currentDistance;
95 }
96 }
97
98 private void OnTouchEnded(object? sender, TouchEventArgs e)
99 {
100 var prevCount = _activePoints.Count;
101
102 if (_activePoints.ContainsKey(e.Touch.Id))
103 _activePoints[e.Touch.Id] = e.Touch.Position;
104
105 if (prevCount == 2)
106 Emit(PinchPhase.Ended, PinchEnded);
107
108 _activePoints.Remove(e.Touch.Id);
109 var newCount = _activePoints.Count;
110
111 if (prevCount != 2 && newCount == 2)
112 BeginPinch();
113 else if (newCount < 2)
114 ClearBaseline();
115 }

130 private void Emit(PinchPhase phase,
131 EventHandler<PinchedEventArgs>? handler,
132 double? currentDistance = null)
133 {
134 if (_activePoints.Count != 2)
135 return;
136
137 var pair = _activePoints.OrderBy(kvp => kvp.Key).Take(2).ToArray();
138 var a = pair[0].Value;
139 var b = pair[1].Value;
140 var distance = currentDistance ?? GetDistance(a, b);
141 var center = new PointF((a.X + b.X) / 2f, (a.Y + b.Y) / 2f);
142
143 handler?.Invoke(this, new PinchedEventArgs(phase,
144 new[] { pair[0].Key, pair[1].Key },
145 center,
146 _startingDistance,
147 _lastDistance,
148 distance));
149 }

151 private double GetCurrentDistance()
152 {
153 using var enumerator = _activePoints.Values.GetEnumerator();
154 if (!enumerator.MoveNext())
155 return 0;
156
157 var a = enumerator.Current;
158
159 if (!enumerator.MoveNext())
160 return 0;
161
162 var b = enumerator.Current;
163
164 return GetDistance(a, b);
165 }

167 private static double GetDistance(Point a, Point b)
168 {
169 var dx = a.X - b.X;
170 var dy = a.Y - b.Y;
171 return Math.Sqrt(dx * dx + dy * dy);
172 }
