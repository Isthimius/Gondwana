Gondwana Game Engine Version 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

198 public virtual bool HitTest(View view,
199 Point screenPositionPx)
200 {
201 ArgumentNullException.ThrowIfNull(view);
202
203 if (!IsInputEnabled ||
204 !IsPointerInputEnabled ||
205 !Visible)
206 {
207 return false;
208 }
209
210 if (Mode == DirectDrawingMode.View
211 && !ReferenceEquals(View, view))
212 {
213 return false;
214 }
215
216 if (Mode == DirectDrawingMode.SceneLayer && SceneLayer is { } layer &&
217 (layer.WrapHorizontally || layer.WrapVertically))
218 return GetWrappedHitOffset(view, screenPositionPx).HasValue;
219
220 RectangleF screenBounds = GetDrawLocationScreen(view);
221
222 return !screenBounds.IsEmpty
223 && screenBounds.Contains(screenPositionPx.X, screenPositionPx.Y);
224 }

226 internal PointF? GetWrappedHitOffset(View view, Point screenPositionPx)
227 {
228 if (Mode != DirectDrawingMode.SceneLayer || SceneLayer is not { } layer)
229 return PointF.Empty;
230 var world = view.ScreenPxToWorldPx(layer, screenPositionPx);
231 foreach (var offset in layer.GetWrappedOffsets(WorldBounds, new RectangleF(world.X, world.Y, 0.01f, 0.01f)))
232 {
233 RectangleF bounds = WorldBounds;
234 bounds.Offset(offset);
235 if (bounds.Contains(world)) return offset;
236 }
237 return null;
238 }

247 public virtual bool HitTest(Point screenPositionPx)
248 {
249 var views = RenderSurfaceHost.ViewManager.Views;
250
251 for (int index = views.Count - 1; index >= 0; index--)
252 {
253 View view = views[index];
254
255 if (!view.Viewport.TargetRectPx.Contains(screenPositionPx))
256 {
257 continue;
258 }
259
260 return HitTest(view, screenPositionPx);
261 }
262
263 return false;
