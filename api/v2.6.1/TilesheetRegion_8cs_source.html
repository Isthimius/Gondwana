Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

492 internal void BuildTileCache()
493 {
494 ThrowIfDisposed();
495 ClearTileCache();
496
497 if (Tilesheet == null ||
498 Tilesheet.SkBitmap == null ||
499 Tilesheet.SkBitmap.IsEmpty ||
500 _tileSize.Width <= 0 ||
501 _tileSize.Height <= 0 ||
502 _area.Width <= 0 ||
503 _area.Height <= 0)
504 {
505 return;
506 }
507
508 if (_tilePadding.Left < 0 || _tilePadding.Top < 0 ||
509 _tilePadding.Right < 0 || _tilePadding.Bottom < 0)
510 {
511 throw new InvalidOperationException("Tilesheet region tile padding cannot be negative.");
512 }
513
514 if (_regionMargin.Left < 0 || _regionMargin.Top < 0 ||
515 _regionMargin.Right < -_tilePadding.Right || _regionMargin.Bottom < -_tilePadding.Bottom)
516 {
517 throw new InvalidOperationException("Leading region margins cannot be negative; trailing margins may only cancel trailing tile padding.");
518 }
519
520 if (TileWidthIncludingPadding <= 0 || TileHeightIncludingPadding <= 0)
521 return;
522
523 int xTiles = (_area.Width - _regionMargin.Left - _regionMargin.Right) /
524 TileWidthIncludingPadding;
525 int yTiles = (_area.Height - _regionMargin.Top - _regionMargin.Bottom) /
526 TileHeightIncludingPadding;
527
528 if (xTiles <= 0 || yTiles <= 0)
529 return;
530
531 PruneFrameCollisionAdjustments(xTiles, yTiles);
532 PruneFrameCollisionTypes(xTiles, yTiles);
533 _tileCache = new TilesheetRegionSlice?[xTiles, yTiles];
534
535 var regionArea = Area;
536 var bitmapBounds = Tilesheet.SkBitmap.Info.Rect;
537
538 for (int y = 0; y < yTiles; y++)
539 {
540 for (int x = 0; x < xTiles; x++)
541 {
542 var srcRect = GetTileBounds(x, y);
543
544 // Prevent this region from bleeding into another region or outside the image.
545 if (!regionArea.Contains(srcRect) ||
546 !bitmapBounds.Contains(srcRect.ToSKRectI()))
547 {
548 continue;
549 }
550
551 var slice = CreateSlice(
552 srcRect,
553 GetStoredFrameCollisionAdjust(x, y));
554
555 if (slice.HasValue)
556 _tileCache[x, y] = slice.Value;
557 }
558 }
559 }

603 private TilesheetRegionSlice? CreateSlice(
604 Rectangle srcRect,
605 CollisionAdjust collisionAdjust)
606 {
607 var srcInfo = Tilesheet.SkBitmap.Info;
608 var sliceInfo = new SKImageInfo(
609 srcRect.Width,
610 srcRect.Height,
611 srcInfo.ColorType,
612 srcInfo.AlphaType);
613
614 var bmp = new SKBitmap(sliceInfo);
615 bmp.Erase(SKColors.Transparent);
616
617 if (!Tilesheet.SkBitmap.ExtractSubset(bmp, srcRect.ToSKRectI()))
618 {
619 bmp.Dispose();
620 return null;
621 }
622
623 var img = SKImage.FromBitmap(bmp);
624 return new TilesheetRegionSlice(bmp, img, collisionAdjust);
