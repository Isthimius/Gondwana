Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

352 {

354 return;

389 }

649 public void Tick()
650 {
651 if (!IsRunning)
652 return;
653
654 if (!_isTimerDriven)
655 {
656 Cycle();
657 return;
658 }
659
660 lock (RenderStateSynchronization.SyncRoot)
661 {
662 EngineDispatcher.Drain();
663
664 long driverTick = HighResTimer.GetCurrentTick();
665 var batch = _timerDrivenSteps.Advance(
666 driverTick,
667 Configuration.TimerDrivenSimulationRate,
668 Configuration.MaxTimerDrivenSimulationSteps);
669
670 bool render = IsTimerDrivenForegroundDue(driverTick);
671 double frameDelta = HighResTimer.GetDuration(_lastForegroundTick, driverTick);
672
673 if (batch.StepCount == 0)
674 {
675 if (render)
676 RenderFrame(driverTick, frameDelta);
677
678 if (Configuration.LegacyCpsSamplingTime > 0)
679 CalculateCPS(driverTick);
680
681 return;
682 }
683
684 double fixedDelta = batch.StepTicks / (double)HighResTimer.TicksPerSecond;
685
686 for (int step = 0; step < batch.StepCount && IsRunning; step++)
687 {
688 long simulationTick = batch.GetStepTick(step);
689 EngineSimulationClock.SetTimerDrivenTick(simulationTick);
690
691 bool isLastStep = step == batch.StepCount - 1;
692 RunSimulationCycle(
693 simulationTick,
694 fixedDelta,
695 render && isLastStep,
696 driverTick,
697 frameDelta,
698 sampleCps: isLastStep);
699 }
700 }

994 private void Cycle()
995 {
996 // Give a GPU paint callback that is already waiting for live render state
997 // a chance to acquire the shared gate before this unconstrained loop takes
998 // another simulation cycle.
999 RenderStateSynchronization.WaitForPendingGpuRenderers();
1000
1001 lock (RenderStateSynchronization.SyncRoot)
1002 {
1003 EngineDispatcher.Drain();
1004
1005 long tick = HighResTimer.GetCurrentTick();
1006 var deltaSeconds = HighResTimer.GetDuration(_lastCycleTick, tick);
1007 _lastCycleTick = tick;
1008
1009 RunSimulationCycle(
1010 tick,
1011 deltaSeconds,
1012 IsForegroundDue(tick),
1013 tick,
1014 deltaSeconds,
1015 sampleCps: true);
1016 }

1019 private void RunSimulationCycle(
1020 long simulationTick,
1021 double simulationDelta,
1022 bool render,
1023 long renderTick,
1024 double frameDelta,
1025 bool sampleCps)
1026 {
1027 // Refresh native gamepad state before any simulation callbacks consume it.
1028 // Use the real driver/render tick rather than the fixed-step simulation clock so
1029 // timer-driven catch-up steps cannot compress or delay hardware polling cadence.
1030 Input.RefreshGamepads(renderTick, Configuration);
1031
1032 long telemetryGeneration = _engineTelemetry.BeginSample();
1033 long telemetryStarted = telemetryGeneration == 0 ? 0 : HighResTimer.GetCurrentTick();
1034 EnginePluginRegistry.InvokePreCycle(this, simulationDelta);
1035
1036 DoBackgroundTasks(simulationTick);
1037
1038 if (render)
1039 RenderFrame(renderTick, frameDelta);
1040
1041 _grossCyclesThisMeasure++;
1042
1043 if (sampleCps && Configuration.LegacyCpsSamplingTime > 0)
1044 CalculateCPS(renderTick);
1045
1046 EnginePluginRegistry.InvokePostCycle(this, simulationDelta);
1047 if (telemetryGeneration != 0)
1048 _engineTelemetry.Record(telemetryGeneration, "cycle.cpu.ms",
1049 HighResTimer.GetDuration(telemetryStarted, HighResTimer.GetCurrentTick()) * 1000d);
1050 }

1052 private bool IsForegroundDue(long tick) =>
1053 Configuration.TargetFPS <= 0
1054 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS;
1055
1056 private bool IsTimerDrivenForegroundDue(long tick)
1057 {
1058 int targetFps = Configuration.TargetFPS;
1059 long elapsedTicks = Math.Max(0, tick - _lastTimerDrivenForegroundDriverTick);
1060 _lastTimerDrivenForegroundDriverTick = tick;
1061
1062 if (targetFps <= 0)
1063 {
1064 _timerDrivenForegroundAccumulatedTicks = 0;
1065 _timerDrivenForegroundTargetFps = targetFps;
1066 return true;
1067 }
1068
1069 if (_timerDrivenForegroundTargetFps != targetFps)
1070 {
1071 _timerDrivenForegroundTargetFps = targetFps;
1072 _timerDrivenForegroundAccumulatedTicks = 0;
1073 }
1074
1075 _timerDrivenForegroundAccumulatedTicks += elapsedTicks;
1076
1077 long frameIntervalTicks = Math.Max(1, HighResTimer.TicksPerSecond / targetFps);
1078 if (_timerDrivenForegroundAccumulatedTicks < frameIntervalTicks)
1079 return false;
1080
1081 // Render at most once per externally driven Tick(), but preserve the fractional
1082 // remainder so small requestAnimationFrame timing jitter cannot permanently lower
1083 // the effective render rate. Whole missed intervals are discarded rather than
1084 // causing an unbounded render catch-up sequence after a long browser stall.
1085 _timerDrivenForegroundAccumulatedTicks %= frameIntervalTicks;
1086 return true;
1087 }

1106 private void DoBackgroundTasks(long tick)
1107 {
1108 long telemetryGeneration = _engineTelemetry.BeginSample();
1109 long telemetryStarted = telemetryGeneration == 0 ? 0 : HighResTimer.GetCurrentTick();
1110 // find total real seconds passed since last background loop
1111 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
1112
1113 BeforeBackgroundTasksExecute?.Invoke();
1114
1115 // raise pre-cycle timer events
1116 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
1117
1118 // check for keyboard events
1119 KeyboardEventPoller.Instance?.PollForEvents(tick);
1120
1121 // check for mouse events
1122 MouseEventPoller.Instance?.PollForEvents(tick);
1123
1124 // check for touch events
1125 TouchEventPoller.Instance?.PollForEvents(tick);
1126
1127 // check for gamepad events
1128 GamepadEventPoller.Instance?.PollForEvents(tick);
1129
1130 // cycle Animator frames
1131 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
1132 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
1133
1134 // advance Sprite Movement paths
1135 SpriteManager.Instance.MoveSprites(tick);
1136
1137 // resolve collisions after movement
1138 foreach (var scene in Scenes.Scene.GetAllScenes())
1139 {
1140 foreach (var layer in scene.SceneLayers)
1141 layer.CollisionResolver.Resolve();
1142 }
1143
1144 // update cameras so any movement can mark RefreshNeeded = All.
1145 foreach (var surface in RenderSurfaceHostRegistry.All)
1146 surface.ViewManager.UpdateCameras(deltaSeconds);
1147
1148 AfterBackgroundTasksExecute?.Invoke();
1149
1150 _lastBackgroundTick = tick;
1151 if (telemetryGeneration != 0)
1152 _engineTelemetry.Record(telemetryGeneration, "background.cpu.ms",
1153 HighResTimer.GetDuration(telemetryStarted, HighResTimer.GetCurrentTick()) * 1000d);
1154 }

1156 private void DoForegroundTasks(long tick)
1157 {
1158 // raise event
1159 BeforeFrameRender?.Invoke();
1160
1161 // Effects are presentation state, so advance them on the foreground/render cadence.
1162 foreach (var surface in RenderSurfaceHostRegistry.All)
1163 surface.Effects.Update(tick);
1164
1165 // update the DirectDrawing instances' states
1166 DirectDrawingManager.Instance.UpdateAll(tick);
1167
1168 // Resolve desktop GPU visual state at foreground cadence before requesting repaint.
1169 foreach (var surface in RenderSurfaceHostRegistry.All)
1170 surface.ProduceRenderFrameSnapshot(tick);
1171
1172 // Bitmap rendering remains on the engine thread.
1173 foreach (var surface in RenderSurfaceHostRegistry.All)
1174 if (!surface.Backbuffer.IsGlThreadRendered)
1175 surface.RenderToBackbuffer(tick);
1176
1177 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
1178 foreach (var surface in RenderSurfaceHostRegistry.All)
1179 if (!surface.Backbuffer.IsGlThreadRendered)
1180 surface.PresentBackbufferToAdapter();
1181
1182 // raise event
1183 // GL thread rendering is done as part of this invocation
1184 AfterFrameRender?.Invoke();
1185
1186 // raise post-cycle timer events
1187 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
1188 }

1190 private void CalculateCPS(long tick)
1191 {
1192 // Has the sampling interval elapsed?
1193 long elapsedTicks = tick - _lastCPSSamplingTick;
1194 if (elapsedTicks < Configuration.LegacyCpsSamplingTicks)
1195 return;
1196
1197 // SNAPSHOT the counters BEFORE resetting or posting
1198 long grossCycles = _grossCyclesThisMeasure;
1199 long netCycles = _netCyclesThisMeasure;
1200
1201 // Compute using the snapshot
1202 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
1203 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1204 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1205
1206 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
1207 long totalGpuFrames = 0;
1208 int gpuSurfaceCount = 0;
1209 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
1210 {
1211 if (surface.Backbuffer is GpuBackbuffer gpuBb)
1212 {
1213 totalGpuFrames += gpuBb.ConsumeFrameCount();
1214 gpuSurfaceCount++;
1215 }
1216 }
1217 double? gpuFps = gpuSurfaceCount > 0
1218 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
1219 : null;
1220
1221 // Build immutable args NOW (so lambda doesn't read changing fields later)
1222 var args = new CyclesPerSecondCalculatedEventArgs(
1223 grossCycles,
1224 netCycles,
1225 grossCps,
1226 netCps,
1227 elapsedSec,
1228 gpuFps
1229 );
1230
1231 // Post the snapshot
1232 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
1233
1234 _grossCPS = grossCps;
1235 _netFPS = netCps;
1236
1237 // Reset for next window
1238 _lastCPSSamplingTick = tick;
1239 _grossCyclesThisMeasure = 0;
1240 _netCyclesThisMeasure = 0;
1241 }

1263 private void Dispose(bool disposing)
1264 {
1265 if (!IsDisposed)
1266 {
1267 if (disposing)
1268 {
1269 if (IsDisposing)
1270 return;
1271
1272 IsDisposing = true;
1273
1274 // stop the loop first so handlers don't race the cycle thread
1275 try
1276 {
1277 Stop();
1278 }
1279 catch (Exception ex)
1280 {
1281 Logger.LogError(ex, "Unhandled exception calling Stop()");
1282 }
1283
1284 // wait for the background loop to actually exit unless we're already on it
1285 try
1286 {
1287 var cycleTask = _cycleTask;
1288 if (cycleTask is not null && EngineDispatcher.IsOnEngineThread && !cycleTask.IsCompleted)
1289 {
1290 if (!_deferredDisposeScheduled)
1291 {
1292 _deferredDisposeScheduled = true;
1293 cycleTask.ContinueWith(
1294 _ => CompleteManagedDisposal(),
1295 System.Threading.CancellationToken.None,
1296 System.Threading.Tasks.TaskContinuationOptions.ExecuteSynchronously,
1297 System.Threading.Tasks.TaskScheduler.Default);
1298 }
1299
1300 return;
1301 }
1302
1303 if (cycleTask is not null && !EngineDispatcher.IsOnEngineThread)
1304 cycleTask.Wait();
1305 }
1306 catch (Exception ex)
1307 {
1308 Logger.LogError(ex, "Error waiting for engine loop to exit.");
1309 }
1310
1311 CompleteManagedDisposal();
1312 }
1313 else
1314 {
1315 // unmanaged cleanup...
1316 IsDisposed = true;
1317 }
1318 }
1319 }

1321 private void CompleteManagedDisposal()
1322 {

1324 return;
1325
1326 // raise Disposing on UI thread if possible; otherwise inline
1327 if (UiDispatcher is not null)
1328 UiDispatcher.Post(() => SafeInvoke(Disposing));
1329 else
1330 SafeInvoke(Disposing);
1331
1332 Profiler.Dispose();
1333
