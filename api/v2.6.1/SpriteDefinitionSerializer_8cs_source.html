Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

117 public static SpriteDefinition Rebase(SpriteDefinition definition, string directory)
118 {
119 var clone = FromJson(ToJson(definition));
120 var origin = definition.Source.GsprFilePath ?? definition.Source.AssetsFilePath;
121 if (string.IsNullOrWhiteSpace(origin)) return clone;
122 var oldDirectory = Path.GetDirectoryName(Path.GetFullPath(origin))!;
123 string? RebasePath(string? path) => string.IsNullOrWhiteSpace(path) ? path :
124 Path.GetRelativePath(Path.GetFullPath(directory), Path.GetFullPath(path, oldDirectory));
125 foreach (var source in clone.TilesheetSources)
126 {
127 source.GtsPath = RebasePath(source.GtsPath);
128 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
129 }
130 foreach (var source in clone.SceneSources)
131 {
132 source.GscnPath = RebasePath(source.GscnPath);
133 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
134 }
135 return clone;
136 }

223 // EngineState retains its historical last-incoming-nickname-wins merge contract.
224 internal static List<Sprite> ToSprites(SpriteDefinition definition, bool allowDuplicateNicknames)
225 {
226 var errors = SpriteDefinitionValidator.Validate(definition, allowDuplicateNicknames);
227 if (errors.Count != 0) throw new InvalidDataException("Invalid GSPR definition:\n" + string.Join("\n", errors));
228 var created = new List<Sprite>();
229 try
230 {
231 foreach (var entry in definition.Sprites)
232 {
233 var scenes = Scene._allScenes.Where(scene => scene.ID == entry.SceneId).Take(2).ToArray();
234 if (scenes.Length != 1)
235 throw new InvalidDataException($"Sprite '{entry.Nickname}': Scene '{entry.SceneId}' is missing or ambiguous.");
236 var scene = scenes[0];
237 var layers = scene.SceneLayers.Where(layer => layer.ID == entry.SceneLayerId).Take(2).ToArray();
238 if (layers.Length != 1)
239 throw new InvalidDataException($"Sprite '{entry.Nickname}': SceneLayer '{entry.SceneLayerId}' is missing or ambiguous.");
240 var layer = layers[0];
241 var frame = entry.Frame is null ? default : ResolveFrame(entry.Frame);
242 if (!string.IsNullOrWhiteSpace(entry.CollisionProfileName) && !scene.CollisionProfiles.GetProfileNames().Contains(entry.CollisionProfileName))
243 throw new InvalidDataException($"Sprite '{entry.Nickname}': collision profile '{entry.CollisionProfileName}' does not exist.");
244 var sprite = SpriteManager.Instance.CreateSprite(layer, frame, entry.Nickname, entry.CollisionProfileName);
245 created.Add(sprite);
246 if (entry.Id != Guid.Empty) sprite.Id = entry.Id;
247 sprite.SetPosition(new Vector2(entry.Position.X, entry.Position.Y));
248 sprite.Visible = entry.Visible;
249 sprite.ZOrder = entry.ZOrder;
250 sprite.Rotation = entry.Rotation;
251 sprite.HorizAlign = entry.HorizAlign;
252 sprite.VertAlign = entry.VertAlign;
253 sprite.NudgeX = entry.NudgeX;
254 sprite.NudgeY = entry.NudgeY;
255 if (!entry.RenderSize.IsEmpty) sprite.RenderSize = entry.RenderSize;
256 sprite.EnableFog = entry.EnableFog;
257 sprite.AdjustCollisionArea = entry.AdjustCollisionArea;
258 sprite.CollisionType = entry.CollisionType;
259 sprite.AdjustCollisionAreaByFrame = entry.AdjustCollisionAreaByFrame;
260 sprite.CollisionTypeByFrame = entry.CollisionTypeByFrame;
261 sprite.CollisionsEnabled = entry.CollisionsEnabled;
262 }
263 SpriteManager.Instance.DefinitionProvenance = definition.Sprites.All(sprite => sprite.Id != Guid.Empty &&
264 !string.IsNullOrWhiteSpace(sprite.Nickname))
265 ? definition.LoadStamp?.Materialized(definition, () => FromSprites(created))
266 : null;
267 return created;
268 }
269 catch
270 {
271 foreach (var sprite in created)
272 {
273 SpriteManager.Instance._spriteList.Remove(sprite);
274 sprite.DisposeImmediate();
275 }
276 throw;
277 }
278 }

280 private static Frame ResolveFrame(SpriteFrameDefinition definition)
281 {
282 var sheet = TilesheetRegistry.Instance.GetOrNull(definition.Tilesheet)
283 ?? throw new InvalidDataException($"GSPR tilesheet '{definition.Tilesheet}' is not registered.");
284 var region = sheet.GetRegion(definition.RegionName)
285 ?? throw new InvalidDataException($"GSPR region '{definition.RegionName}' is missing on '{definition.Tilesheet}'.");
286 if (definition.XTile >= region.Columns || definition.YTile >= region.Rows)
287 throw new InvalidDataException($"GSPR frame ({definition.XTile}, {definition.YTile}) is outside region '{definition.RegionName}'.");
288 return sheet.GetFrame(definition.RegionName, definition.XTile, definition.YTile);
289 }
