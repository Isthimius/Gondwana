Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

176 {
177 get => frame;
178 set
179 {
180 // Animation may change tile size, so bitmap rendering must invalidate
181 // the old and new bounds. GPU rendering redraws the full frame and does
182 // not consume RefreshQueue, so avoid calculating those bounds entirely.
183 bool usesDirtyRegionRendering =
184 SceneLayer.Scene?.UsesDirtyRegionRendering ?? true;
185
186 if (usesDirtyRegionRendering)
187 SceneLayer.RefreshQueue.AddWorldRect(DrawLocationWorld);
188

190
191 bool hasFrame = value.Tilesheet is not null;
192 if (hasFrame &&
193 (_adjustCollisionAreaByFrame ||
194 (!_hasAssignedFrame && !_collisionAdjustExplicitlySet)))

202 {
203 SetCollisionTypeFromFrame(value);
204 }
205
206 _hasAssignedFrame = hasFrame;
207

536 public int CompareTo(Tile? tile)
537 {
538 if (tile is null)
539 return -1;
540
541 var thisKey = GetSortKeyForCompare();
542 var tileKey = tile.GetSortKeyForCompare();
543
544 if (thisKey.IsPositionFixed && !tileKey.IsPositionFixed)
545 return -1;
546
547 if (!thisKey.IsPositionFixed && tileKey.IsPositionFixed)
548 return 1;
549
550 return (thisKey.Location, thisKey.ZOrder, thisKey.SceneLayerX)
551 .CompareTo((tileKey.Location, tileKey.ZOrder, tileKey.SceneLayerX));
552 }
553
554 private TileSortKey GetSortKeyForCompare()
555 {
556 var renderContext = RenderContext.Current;
557
558 // Outside an active render pass preserve the historical live comparison
559 // semantics. During rendering, capture the expensive virtual/property values
560 // once per tile and reuse them for every comparison in this sort pass.
561 if (renderContext is null)
562 return CaptureSortKey();
563
564 long renderPassId = renderContext.PassId;
565 if (_sortKeyRenderPassId != renderPassId)
566 {
567 _sortKey = CaptureSortKey();
568 _sortKeyRenderPassId = renderPassId;
569 }
570
571 return _sortKey;
572 }
573
574 private TileSortKey CaptureSortKey()
575 {
576 bool isPositionFixed = IsPositionFixed;
577 Rectangle drawLocationWorld = DrawLocationWorld;
578 Spacing overhang = Overhang;
579 float location = isPositionFixed
580 ? drawLocationWorld.Top + overhang.Top
581 : drawLocationWorld.Bottom - overhang.Bottom - 1;
582
583 return new TileSortKey(
584 isPositionFixed,
585 location,
586 zOrder,
587 SceneLayerCoordinates.X);
588 }
589

591 bool IsPositionFixed,
592 float Location,
593 int ZOrder,
594 float SceneLayerX);
