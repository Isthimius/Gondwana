Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 public static void Validate(string path, string? password = null, bool testData = true)
51 {
52 using var archive = new ZipFile(File.OpenRead(path));
53 archive.Password = password;
54 var keys = new HashSet<AssetsFileEntry>();
55 foreach (ZipEntry entry in archive)
56 {
57 if (!entry.IsFile) continue;
58 var key = AssetsFileEntry.FromString(entry.Name);

60 throw new InvalidDataException($"Invalid bundle entry key: {entry.Name}");
61 if (!keys.Add(key)) throw new InvalidDataException($"Duplicate bundle entry key: {entry.Name}");

213 private void LoadZip(Stream stream)
214 {
215 if (_isLoaded)
216 return;
217
218 try
219 {
220 Engine.Logger.LogInformation("Loading assets file.");
221
222 _zipFile?.Close();
223 _zipFile = null;
224 _zipEntries.Clear();
225
226 // SharpZipLib's ZipFile expects seekable input. Buffering here also gives
227 // stream-loaded GAFs the same lifetime semantics as path-loaded GAFs:
228 // the source stream is needed only for the duration of this call.
229 MemoryStream? bufferedStream = stream.CanSeek ? null : new MemoryStream();
230 Stream archiveStream = stream;
231 if (bufferedStream is not null)
232 {
233 stream.CopyTo(bufferedStream);
234 bufferedStream.Position = 0;
235 archiveStream = bufferedStream;
236 }
237
238 _zipFile = new ZipFile(archiveStream) { IsStreamOwner = false };
239
240 if (!string.IsNullOrEmpty(Password))
241 _zipFile.Password = Password;
242
243 // Force decryption on the first entry so a bad password fails early.
244 var testEntry = _zipFile.Cast<ZipEntry>().FirstOrDefault(e => e.IsFile);
245 if (testEntry != null)
246 {
247 using var testStream = _zipFile.GetInputStream(testEntry);
248 testStream.ReadByte();
249 }
250
251 foreach (ZipEntry entry in _zipFile)
252 {
253 if (!entry.IsFile)
254 continue;
255
256 var key = AssetsFileEntry.FromString(entry.Name);
257 if (key == null)
258 {
259 Engine.Logger.LogWarning("Invalid entry in assets file: {EntryName}", entry.Name);
260 continue;
261 }
262
263 using var entryStream = _zipFile.GetInputStream(entry);
264 using var ms = new MemoryStream();
265 entryStream.CopyTo(ms);
266
267 _zipEntries[key] = ms.ToArray();
268 }
269
270 _isLoaded = true;
271 }
272 catch (ZipException zex)
273 {
274 Engine.Logger.LogError(zex, "Decryption failed — check password or file integrity.");
275 throw;
276 }
277 catch (Exception ex)
278 {
279 Engine.Logger.LogError(ex, "Failed to load asset file.");
280 throw;
281 }
282 finally
283 {
284 // Once contents are buffered, no need to keep the archive or source stream open.
285 _zipFile?.Close();
286 _zipFile = null;
287 }
288 }

406 public Stream? Get(AssetTypes type, string name)
407 {
408 if (string.IsNullOrWhiteSpace(name))
409 return null;
410
411 EnsureLoaded();
412
413 var exactKey = new AssetsFileEntry
414 {
415 AssetType = type,
416 AssetName = name
417 };
418
419 if (_zipEntries.TryGetValue(exactKey, out var exactData))
420 return new MemoryStream(exactData, writable: false);
421
422 var requestedBaseName = Path.GetFileNameWithoutExtension(name);
423
424 var match = _zipEntries.Keys.FirstOrDefault(k =>
425 k.AssetType == type &&
426 string.Equals(
427 Path.GetFileNameWithoutExtension(k.AssetName),
428 requestedBaseName,
429 StringComparison.OrdinalIgnoreCase));
430
431 return match != null && _zipEntries.TryGetValue(match, out var fallbackData)
432 ? new MemoryStream(fallbackData, writable: false)
433 : null;
434 }

449 public void Save()
450 {
451 EnsureLoaded();
452
453 _zipFile?.Close();
454 _zipFile = null;
455 _isLoaded = false;
456
457 using (var fs = File.Create(FilePath))
458 using (var zipStream = new ZipOutputStream(fs) { IsStreamOwner = true })
459 {
460 if (!string.IsNullOrEmpty(Password))
461 zipStream.Password = Password;
462
463 foreach (var (key, data) in _zipEntries)
464 {
465 var entry = new ZipEntry(key.ToString())
466 {
467 DateTime = DateTime.Now
468 };
469
470 if (UseEncryption && !string.IsNullOrEmpty(Password))
471 entry.AESKeySize = 256;
472
473 zipStream.PutNextEntry(entry);
474
475 using var inputStream = new MemoryStream(data, writable: false);
476 inputStream.CopyTo(zipStream);
477 zipStream.CloseEntry();
478 }
479
480 zipStream.Finish();
481 }
482
483 Engine.Logger.LogInformation(
484 "Assets file saved (Encrypted: {Encrypted}).",
485 UseEncryption);
486
487 // Keep the in-memory copy as the source of truth.
488 _isLoaded = true;
489 }
