Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

27 internal static TelemetrySource? Register(bool gpu)
28 {
29 var source = Engine.Instance.Profiler.TryRegisterSource("Render surface",
30 gpu ? (OperatingSystem.IsBrowser() ? "WebGL" : "Desktop GPU") : "Bitmap");
31 if (source is null) return null;
32 foreach (var name in Metrics)
33 {
34 var availability = TelemetryAvailability.NotYetSampled;
35 if (!gpu) availability = TelemetryAvailability.Unsupported;
36 else if (name.StartsWith("gate.", StringComparison.Ordinal))
37 availability = OperatingSystem.IsBrowser() ? TelemetryAvailability.Unsupported : TelemetryAvailability.NotApplicable;
38 else if (OperatingSystem.IsBrowser() && Array.IndexOf(Metrics, name) is >= 9 and <= 17)
39 availability = TelemetryAvailability.NotApplicable;
40 else if (OperatingSystem.IsBrowser() && name.StartsWith("atlas.", StringComparison.Ordinal))
41 availability = TelemetryAvailability.Unsupported;
42 else if (Array.IndexOf(Metrics, name) >= 20)
43 availability = TelemetryAvailability.Unsupported;
44 source.Define(name, availability, GetMetricKind(name));
45 }
46 source.Define("presentation.count", gpu ? TelemetryAvailability.NotYetSampled : TelemetryAvailability.Unsupported);
47 source.Define("layers.omitted", gpu ? TelemetryAvailability.NotYetSampled : TelemetryAvailability.Unsupported);
48 foreach (var layer in Layers)
49 {
50 for (int index = 0; index < layer.Length; index++)
51 {
52 source.Define(
53 layer[index],
54 gpu ? TelemetryAvailability.NotYetSampled : TelemetryAvailability.Unsupported,
55 index <= 3 ? TelemetryMetricKind.Sample : TelemetryMetricKind.Gauge);
56 }
57 }
58
59 return source;
60 }

62 private static TelemetryMetricKind GetMetricKind(string name)
63 {
64 if (name.EndsWith(".lifetime", StringComparison.Ordinal))

66

69 name.StartsWith("visible.", StringComparison.Ordinal) ||
70 name.StartsWith("atlas.", StringComparison.Ordinal))
71 {
72 return TelemetryMetricKind.Sample;
73 }
74
75 return TelemetryMetricKind.Gauge;
76 }
