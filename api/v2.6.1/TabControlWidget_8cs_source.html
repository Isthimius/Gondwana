402 public TabControlWidget MoveTab(int fromIndex,
403 int toIndex)
404 {
405 ObjectDisposedException.ThrowIf(_disposed, this);
406
407 if (fromIndex < 0 || fromIndex >= _tabs.Count)
408 throw new ArgumentOutOfRangeException(nameof(fromIndex));
409
410 if (toIndex < 0 || toIndex >= _tabs.Count)
411 throw new ArgumentOutOfRangeException(nameof(toIndex));
412

414 return this;
415
416 TabPageWidget page = _tabs[fromIndex];
417 TabPageWidget? selectedPage = SelectedTab;
418
419 _tabs.RemoveAt(fromIndex);

421 _selectedIndex = selectedPage is null ? -1 : _tabs.IndexOf(selectedPage);

424 TabReordered?.Invoke(page, fromIndex, toIndex);
425
426 return this;

451 public TabControlWidget Relayout()
452 {
453 ObjectDisposedException.ThrowIf(_disposed, this);
454
455 Rectangle bounds = Bounds;
456 (_displayedRowCount, _displayedColumnCount) = ResolveGrid(bounds);
457 Rectangle contentBounds = CreateContentBounds(bounds, _displayedRowCount);
458
459 for (int index = 0; index < _tabs.Count; index++)
460 {
461 TabPageWidget page = _tabs[index];
462 TabHeaderWidget header = _headers[page];
463 Rectangle headerBounds = CreateHeaderBounds(bounds, index, _displayedColumnCount);
464
465 header.SetBounds(headerBounds);
466 page.HeaderBounds = headerBounds;
467 page.SetContentBounds(contentBounds);
468
469 SetLocalOffset(header, new Vector2(headerBounds.Left - bounds.Left, headerBounds.Top - bounds.Top));
470 SetLocalOffset(page, new Vector2(contentBounds.Left - bounds.Left, contentBounds.Top - bounds.Top));
471 }
472
473 ApplyZOrder();
474 return this;

598 private bool RemoveTabCore(TabPageWidget page,
599 bool dispose,
600 bool raiseSelectionChanged)
601 {
602 int removedIndex = _tabs.IndexOf(page);
603 if (removedIndex < 0)
604 return false;
605
606 TabPageWidget? previouslySelected = SelectedTab;
607 TabHeaderWidget header = _headers[page];
608
609 page.TitleChanged -= OnPageTitleChanged;
610 page.Disposing -= OnPageDisposing;
611 header.Invoked -= OnHeaderInvoked;
612 header.ReorderDragged -= OnHeaderDragged;
613 header.ReorderDragEnded -= OnHeaderDragEnded;
614 header.Disposing -= OnHeaderDisposing;
615
616 _tabs.RemoveAt(removedIndex);
617 _headers.Remove(page);
618
619 Remove(header);
620 Remove(page);
621
622 header.Dispose();
623
624 if (dispose)
625 page.Dispose();
626
627 if (_tabs.Count == 0)
628 {
629 _selectedIndex = -1;
630 }
631 else if (ReferenceEquals(previouslySelected, page))
632 {
633 _selectedIndex = Math.Min(removedIndex, _tabs.Count - 1);
634 }
635 else
636 {
637 _selectedIndex = previouslySelected is null ? 0 : _tabs.IndexOf(previouslySelected);
638 }
639
640 Relayout();
641 UpdateSelectedTabVisuals();
642 UpdatePageVisibility();
643
644 if (raiseSelectionChanged && SelectedTab is not null && !ReferenceEquals(previouslySelected, SelectedTab))
645 SelectedTabChanged?.Invoke(SelectedTab);
646
647 return true;
648 }

780 private (int Rows, int Columns) ResolveGrid(Rectangle bounds)
781 {
782 if (!AutoExpandRows)
783 {
784 int columns = Math.Max(1, (int)Math.Ceiling(_tabs.Count / (double)TabRowCount));
785 return (TabRowCount, columns);
786 }
787
788 int autoColumns = _tabs.Count == 0
789 ? 1
790 : Math.Min(_tabs.Count, Math.Max(1, bounds.Width / PreferredTabWidth));
791 int autoRows = Math.Max(1, (int)Math.Ceiling(_tabs.Count / (double)autoColumns));
792
793 return (autoRows, autoColumns);
794 }

796 private Rectangle CreateHeaderBounds(Rectangle bounds,
797 int index,
798 int columns)
799 {
800 int row = index / columns;
801 int column = index % columns;

803 int right = bounds.Left + (column + 1) * bounds.Width / columns;
804
805 return Rectangle.FromLTRB(
806 left,
807 bounds.Top + row * TabHeight,
808 right,
809 bounds.Top + (row + 1) * TabHeight);
