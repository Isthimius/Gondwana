Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

14public sealed class RadioButtonGroup

16 private readonly HashSet<RadioButtonWidget> _buttons = [];

21 public RadioButtonWidget? SelectedButton => _buttons.FirstOrDefault(static button => button.IsSelected);
22
23 internal void Register(RadioButtonWidget button)
24 {
25 ArgumentNullException.ThrowIfNull(button);
26 _buttons.Add(button);
27
28 if (button.IsSelected)
29 Select(button);
30 }
31
32 internal void Unregister(RadioButtonWidget button)
33 {
34 ArgumentNullException.ThrowIfNull(button);
35 _buttons.Remove(button);
36 }
37
38 internal void Select(RadioButtonWidget selected)
39 {
40 ArgumentNullException.ThrowIfNull(selected);
41
42 foreach (RadioButtonWidget button in _buttons.ToArray())
43 button.SetSelectedFromGroup(ReferenceEquals(button, selected));
44 }
45}
