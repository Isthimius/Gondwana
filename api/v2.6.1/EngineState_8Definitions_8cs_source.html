Gondwana Game Engine Version 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

124 private static T LoadPacked<T>(DefinitionStateEntry entry, string? baseDirectory, AssetTypes expectedType, Func<AssetsFile, string, T> load)
125 {
126 var path = ResolvePath(entry.AssetsFilePath!, baseDirectory);
127 try
128 {
129 var archive = FindAssetsFile(path);
130 if (archive is null)
131 {
132 if (!File.Exists(path)) throw new FileNotFoundException("Archive not found.", path);
133 archive = AssetsFile.LoadOrCreate(path);
134 }
135 // EngineState references are exact identities, not the convenience basename
136 // aliases accepted by AssetsFile.Get for interactive runtime lookup.
137 if (!archive.GetAllEntries().Any(candidate => candidate.AssetType == expectedType &&
138 string.Equals(candidate.AssetName, entry.AssetEntryName, StringComparison.Ordinal)))
139 throw new InvalidDataException($"Entry does not exist with expected type '{expectedType}'.");
140 return load(archive, entry.AssetEntryName!);
141 }
142 catch (Exception ex) when (ex is not OutOfMemoryException)
143 {
144 throw new InvalidDataException($"Cannot load packed definition '{entry.AssetEntryName}' from GAF '{path}'.", ex);
145 }
