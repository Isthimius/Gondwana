Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

272 public TextBoxWidget SetText(string? text)
273 {
274 string normalized = NormalizeText(text);
275
276 if (MaxLength.HasValue && normalized.Length > MaxLength.Value)
277 normalized = normalized[..MaxLength.Value];
278
279 if (_text == normalized)
280 {
281 _caretIndex = Math.Min(_caretIndex, _text.Length);
282 RefreshDisplayedText();
283 return this;
284 }
285
286 _text = normalized;
287 _caretIndex = _text.Length;
288 RefreshDisplayedText();
289 TextChanged?.Invoke(_text);
290 return this;
291 }

298 public TextBoxWidget InsertText(string? text)
299 {
300 if (IsReadOnly || string.IsNullOrEmpty(text))
301 return this;
302
303 string insertion = NormalizeText(text);
304 if (insertion.Length == 0)
305 return this;
306
307 if (MaxLength.HasValue)
308 {
309 int available = MaxLength.Value - _text.Length;
310 if (available <= 0)
311 return this;
312
313 if (insertion.Length > available)
314 insertion = insertion[..available];
315 }
316
317 _text = _text.Insert(_caretIndex, insertion);
318 _caretIndex += insertion.Length;
319 RefreshDisplayedText();
320 TextChanged?.Invoke(_text);
321 return this;
322 }

446 protected override void OnKeyboardInput(WidgetKeyboardEventArgs args)
447 {
448 base.OnKeyboardInput(args);
449
450 if (args.KeyAction == KeyAction.Released)
451 {
452 if (_repeatingKey == args.Key)
453 ResetKeyRepeatState();
454
455 return;
456 }
457
458 if (args.KeyAction is not KeyAction.Pressed and not KeyAction.Repeated)
459 return;
460
461 long currentTick = HighResTimer.GetCurrentTick();
462
463 if (args.KeyAction == KeyAction.Pressed)
464 {
465 _repeatingKey = args.Key;
466 _keyPressedTick = currentTick;
467 _lastRepeatTick = currentTick;
468 }
469 else
470 {
471 if (_repeatingKey != args.Key ||
472 _keyPressedTick == 0 ||
473 HighResTimer.GetDuration(_keyPressedTick, currentTick) < _initialKeyRepeatDelaySeconds ||
474 (_lastRepeatTick != 0 &&
475 HighResTimer.GetDuration(_lastRepeatTick, currentTick) < _repeatedKeyIntervalSeconds))
476 {
477 args.Handled = true;
478 return;
479 }
480
481 _lastRepeatTick = currentTick;
482 }
483
484 if (args.Key == SubmitKey)
485 {
486 if (args.KeyAction != KeyAction.Pressed)
487 return;
488
489 args.Handled = true;
490 Submitted?.Invoke(_text);
491 return;
492 }
493
494 if (args.Key == LeftKey)
495 {
496 args.Handled = true;
497 MoveCaret(-1);
498 return;
499 }
500
501 if (args.Key == RightKey)
502 {
503 args.Handled = true;
504 MoveCaret(1);
505 return;
506 }
507
508 if (args.Key == HomeKey)
509 {
510 args.Handled = true;
511 CaretIndex = 0;
512 return;
513 }
514
515 if (args.Key == EndKey)
516 {
517 args.Handled = true;
518 CaretIndex = _text.Length;
519 return;
520 }
521
522 if (args.Key == BackspaceKey)
523 {
524 args.Handled = true;
525 Backspace();
526 return;
527 }
528
529 if (args.Key == DeleteKey)
530 {
531 args.Handled = true;
532 Delete();
533 return;
534 }
535
536 if (IsReadOnly)
537 return;
538
539 char? character = CharacterResolver(args);
540 if (!character.HasValue)
541 return;
542
543 args.Handled = true;
544 InsertText(character.Value.ToString());
545 }

634 private static char? ResolveWindowsVirtualKeyCharacter(WidgetKeyboardEventArgs args)
635 {
636 if ((args.Modifiers & (KeyboardModifierState.Ctrl | KeyboardModifierState.Alt)) != 0)
637 return null;
638
639 bool shift = (args.Modifiers & KeyboardModifierState.Shift) != 0;
640 int key = args.Key;
641
642 if (key is >= 65 and <= 90)
643 {
644 char letter = (char)key;
645 return shift ? letter : char.ToLowerInvariant(letter);
646 }
647
648 if (key is >= 48 and <= 57)
649 {
650 if (!shift)
651 return (char)key;
652
653 const string shiftedDigits = ")!@#$%^&*(";
654 return shiftedDigits[key - 48];
655 }
656
657 if (key is >= 96 and <= 105)
658 return (char)('0' + key - 96);
659
660 return key switch
661 {
662 32 => ' ',
663 106 => '*',
664 107 => '+',
665 109 => '-',
666 110 => '.',
667 111 => '/',
668 186 => shift ? ':' : ';',
669 187 => shift ? '+' : '=',
670 188 => shift ? '<' : ',',
671 189 => shift ? '_' : '-',
672 190 => shift ? '>' : '.',
673 191 => shift ? '?' : '/',
674 192 => shift ? '~' : '`',
675 219 => shift ? '{' : '[',
676 220 => shift ? '|' : '\\',
677 221 => shift ? '}' : ']',
678 222 => shift ? '"' : '\'',
679 _ => null
680 };
681 }
