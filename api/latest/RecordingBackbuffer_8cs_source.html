Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

196 SceneLayer layer,
197 Rectangle worldRect,
198 out FixedGridRenderPlan plan)
199 {
200 plan = default;
201
202 if (layer.WrapHorizontally ||
203 layer.WrapVertically ||
204 layer.TransformedTiles.Count != 0 ||
205 layer.ShowGridLines ||
206 layer.ShowCollisionBoxes ||
207 !layer.IsFixedGridSnapshotFastPathEligible ||
208 !RenderTileQuery.IsRenderOrdered(layer))

210 return false;
211 }

214 queryRect.Inflate(layer.TileWidth, layer.TileHeight);
215 queryRect.Inflate(1, 1);

227 internal int DrawFixedGridLayer(
228 View view,
229 FixedGridRenderPlan plan,
230 Rectangle clipRect)
231 {
232 var layer = plan.Layer;
233 LayerScreenMap screen = LayerScreenMap.Create(view, layer);
234 bool snapOrthogonalEdges = layer.CoordinateSystem is OrthogonalCoordinates;
235 int tileCount = 0;
236
237 Canvas.Save();
238 Canvas.ClipRect(clipRect.ToSKRect());
239
240 if (!snapOrthogonalEdges)
241 {
242 var worldToScreen = screen.CreateWorldToScreenMatrix();
243 Canvas.Concat(ref worldToScreen);
244 }
245
246 try
247 {
248 RenderTileQuery.VisitCandidates(
249 layer.CoordinateSystem,
250 layer,
251 plan.WorldRect,
252 includeOverhang: true,
253 tile =>
254 {
255 if (!tile.Visible || tile.CurrentFrame.Tilesheet is null)
256 return true;
257
258 tileCount++;
259
260 if (snapOrthogonalEdges)
261 DrawOrthogonalFixedGridTileSnapped(layer, screen, tile);
262 else
263 DrawFixedGridTileInWorldSpace(layer, tile);
264
265 return true;
266 });
267
268 if (snapOrthogonalEdges)
269 FlushSnappedVertexBatch();
270 else
271 FlushTileBatch();
272
273 return tileCount;
274 }
275 finally
276 {
277 Canvas.Restore();
278 }

281 internal bool TryDrawFixedGridDrawables(
282 View view,
283 IReadOnlyList<IDrawable> drawables,
284 Rectangle clipRect)
285 {
286 if (drawables.Count == 0)
287 return true;
288
289 SceneLayer? layer = null;
290
291 // This path deliberately handles only the engine's fixed tile type and no
292 // post-tile debug/fog overlays. Anything richer retains the generic renderer.
293 for (int i = 0; i < drawables.Count; i++)
294 {
295 if (drawables[i].GetType() != typeof(SceneLayerTile) ||
296 drawables[i] is not SceneLayerTile tile ||
297 tile.EnableFog)
298 {
299 return false;
300 }
301
302 layer ??= tile.SceneLayer;
303 if (!ReferenceEquals(layer, tile.SceneLayer))
304 return false;
305 }
306
307 if (layer is null ||
308 layer.ShowGridLines ||
309 layer.ShowCollisionBoxes)
310 {
311 return false;
312 }
313
314 LayerScreenMap screen = LayerScreenMap.Create(view, layer);
315
316 Canvas.Save();
317 Canvas.ClipRect(clipRect.ToSKRect());
318
319 if (layer.CoordinateSystem is OrthogonalCoordinates)
320 {
321 try
322 {
323 for (int i = 0; i < drawables.Count; i++)
324 DrawOrthogonalFixedGridTileSnapped(
325 layer,
326 screen,
327 (SceneLayerTile)drawables[i]);
328
329 FlushSnappedVertexBatch();
330 return true;
331 }
332 finally
333 {
334 Canvas.Restore();
335 }
336 }
337
338 var worldToScreen = screen.CreateWorldToScreenMatrix();
339 bool worldSpaceActive = false;
340
341 try
342 {
343 for (int i = 0; i < drawables.Count; i++)
344 {
345 var tile = (SceneLayerTile)drawables[i];
346
347 if (tile.Transform == TileTransform.Identity &&
348 TryGetFrameAtlasInfo(tile.CurrentFrame, out _))
349 {
350 if (!worldSpaceActive)
351 {
352 Canvas.Save();
353 Canvas.Concat(ref worldToScreen);
354 worldSpaceActive = true;
355 }
356
357 DrawFixedGridTileInWorldSpace(layer, tile);
358 continue;
359 }
360
361 if (worldSpaceActive)
362 {
363 FlushTileBatch();
364 Canvas.Restore();
365 worldSpaceActive = false;
366 }
367
368 DrawFixedGridTile(view, layer, screen, tile);
369 }
370
371 if (worldSpaceActive)
372 {
373 FlushTileBatch();
374 Canvas.Restore();
375 worldSpaceActive = false;
376 }
377
378 return true;
379 }
380 finally
381 {
382 if (worldSpaceActive)
383 {
384 FlushTileBatch();
385 Canvas.Restore();
386 }
387
388 Canvas.Restore();
389 }

392 private void DrawOrthogonalFixedGridTileSnapped(
393 SceneLayer layer,
394 LayerScreenMap screen,
395 SceneLayerTile tile)
396 {
397 if (tile.Transform == TileTransform.Identity &&
398 TryGetFrameAtlasInfo(tile.CurrentFrame, out var frame))
399 {
400 Rectangle world = RenderTileQuery.GetFixedTileCellBounds(layer, tile);
401 if (!frame.Overhang.IsEmpty)
402 {
403 world = Rectangle.FromLTRB(
404 world.Left - frame.Overhang.Left,
405 world.Top - frame.Overhang.Top,
406 world.Right + frame.Overhang.Right,
407 world.Bottom + frame.Overhang.Bottom);
408 }
409
410 RectangleF destination = screen.MapPixelSnapped(world);
411 if (destination.Width <= 0f || destination.Height <= 0f)
412 return;
413
414 QueueSnappedVertexAtlas(
415 frame.Atlas,
416 frame.Source,
417 destination);
418 return;
419 }
420
421 FlushSnappedVertexBatch();
422
423 RectangleF transformedDestination =
424 screen.MapPixelSnapped(tile.DrawLocationWorld);
425 if (transformedDestination.Width > 0f &&
426 transformedDestination.Height > 0f)
427 {
428 tile.Draw(this, transformedDestination);
429 }

432 private void DrawFixedGridTileInWorldSpace(
433 SceneLayer layer,
434 SceneLayerTile tile)
435 {
436 if (!TryGetFrameAtlasInfo(tile.CurrentFrame, out var frame))
437 return;
438
439 Rectangle world = RenderTileQuery.GetFixedTileCellBounds(layer, tile);
440 if (!frame.Overhang.IsEmpty)
441 {
442 world = Rectangle.FromLTRB(
443 world.Left - frame.Overhang.Left,
444 world.Top - frame.Overhang.Top,
445 world.Right + frame.Overhang.Right,
446 world.Bottom + frame.Overhang.Bottom);
447 }
448
449 // Record atlas entries in integer world coordinates. A single canvas transform
450 // scales/translates the complete fixed-grid layer to screen space, so adjacent
451 // tile edges are transformed from the same world boundary instead of from
452 // independently rounded fractional destination rectangles.
453 if (TryQueueAtlas(frame.Atlas, frame.Source, world))
454 return;
455
456 FlushTileBatch();
457 Canvas.DrawImage(
458 frame.Atlas,
459 frame.Source,
460 world.ToSKRect(),
461 SKSamplingOptions.Default);
462 }

503 private void QueueSnappedVertexAtlas(
504 SKImage atlas,
505 SKRect source,
506 RectangleF destination)
507 {
508 if (_pendingSnappedVertexAtlas is not null &&
509 !ReferenceEquals(_pendingSnappedVertexAtlas, atlas))
510 {
511 FlushSnappedVertexBatch();
512 }
513
514 _pendingSnappedVertexAtlas = atlas;
515
516 int tileIndex = _pendingSnappedVertexCount++;
517 int vertexIndex = tileIndex * 4;
518
519 float left = destination.Left;
520 float top = destination.Top;
521 float right = destination.Right;
522 float bottom = destination.Bottom;
523
524 _pendingSnappedPositions[vertexIndex] = new(left, top);
525 _pendingSnappedPositions[vertexIndex + 1] = new(right, top);
526 _pendingSnappedPositions[vertexIndex + 2] = new(right, bottom);
527 _pendingSnappedPositions[vertexIndex + 3] = new(left, bottom);
528
529 _pendingSnappedTextureCoordinates[vertexIndex] =
530 new(source.Left, source.Top);
531 _pendingSnappedTextureCoordinates[vertexIndex + 1] =
532 new(source.Right, source.Top);
533 _pendingSnappedTextureCoordinates[vertexIndex + 2] =
534 new(source.Right, source.Bottom);
535 _pendingSnappedTextureCoordinates[vertexIndex + 3] =
536 new(source.Left, source.Bottom);
537
538 if (_pendingSnappedVertexCount == MaxSnappedVertexBatchSize)
539 FlushFullSnappedVertexBatch();
540 }

542 private void FlushSnappedVertexBatch()
543 {
544 if (_pendingSnappedVertexCount == 0)
545 {
546 _pendingSnappedVertexAtlas = null;
547 return;
548 }
549
550 if (_pendingSnappedVertexAtlas is null)
551 {
552 throw new InvalidOperationException(
553 "Snapped vertex batch has queued tiles without an atlas image.");
554 }
555
556 int offset = 0;
557 int remaining = _pendingSnappedVertexCount;
558
559 foreach (int size in PartialSnappedVertexBatchSizes)
560 {
561 while (remaining >= size)
562 {
563 var buffers = _partialSnappedVertexBuffers[size];
564 Array.Copy(
565 _pendingSnappedPositions,
566 offset * 4,
567 buffers.Positions,
568 0,
569 size * 4);
570 Array.Copy(
571 _pendingSnappedTextureCoordinates,
572 offset * 4,
573 buffers.TextureCoordinates,
574 0,
575 size * 4);
576
577 DrawSnappedVertexBatch(
578 _pendingSnappedVertexAtlas,
579 buffers.Positions,
580 buffers.TextureCoordinates,
581 buffers.Indices,
582 size);
583
584 offset += size;
585 remaining -= size;
586 }
587 }
588
589 ResetSnappedVertexBatch();
590 }

611 SKImage atlas,
612 SKPoint[] positions,

645 private static ushort[] CreateQuadIndices(int tileCount)
646 {
647 var indices = new ushort[tileCount * 6];
648
649 for (int tile = 0; tile < tileCount; tile++)
650 {
651 int vertex = tile * 4;
652 int index = tile * 6;
653
654 indices[index] = (ushort)vertex;
655 indices[index + 1] = (ushort)(vertex + 1);
656 indices[index + 2] = (ushort)(vertex + 2);
657 indices[index + 3] = (ushort)vertex;
658 indices[index + 4] = (ushort)(vertex + 2);
659 indices[index + 5] = (ushort)(vertex + 3);
660 }
661
662 return indices;

725 private bool TryQueueAtlas(
726 SKImage atlas,
727 SKRect source,
728 RectangleF destination)
729 {
730 if (source.Width <= 0 ||
731 source.Height <= 0 ||
732 destination.Width <= 0 ||
733 destination.Height <= 0)
734 {
735 return false;
736 }
737
738 float scaleX = destination.Width / source.Width;
739 float scaleY = destination.Height / source.Height;
740 float tolerance = MathF.Max(0.0001f, MathF.Abs(scaleX) * 0.0001f);
741
742 if (!float.IsFinite(scaleX) ||
743 !float.IsFinite(scaleY) ||
744 MathF.Abs(scaleX - scaleY) > tolerance)
745 {
746 return false;
747 }
748
749 if (_pendingAtlas is not null && !ReferenceEquals(_pendingAtlas, atlas))
750 FlushTileBatch();
751
752 _pendingAtlas = atlas;
753 int index = _pendingCount++;
754 _pendingSprites[index] = source;
755 _pendingTransforms[index] = new SKRotationScaleMatrix(
756 scaleX,
757 0f,
758 destination.Left,
759 destination.Top);
760
761 if (_pendingCount == MaxAtlasBatchSize)
762 FlushFullTileBatch();
763
764 return true;
765 }

767 internal void FlushTileBatch()
768 {
769 if (_pendingCount == 0)
770 {
771 _pendingAtlas = null;
772 return;
773 }
774
775 if (_pendingAtlas is null)
776 throw new InvalidOperationException("Atlas batch has queued tiles without an atlas image.");
777
778 int offset = 0;
779 int remaining = _pendingCount;
780
781 foreach (int size in PartialBatchSizes)
782 {
783 while (remaining >= size)
784 {
785 var buffers = _partialBatchBuffers[size];
786 Array.Copy(_pendingSprites, offset, buffers.Sprites, 0, size);
787 Array.Copy(_pendingTransforms, offset, buffers.Transforms, 0, size);
788
789 Canvas.DrawAtlas(
790 _pendingAtlas,
791 buffers.Sprites,
792 buffers.Transforms,
793 SKSamplingOptions.Default);
794
795 AtlasBatchCount++;
796 AtlasBatchedTileCount += size;
797 offset += size;
798 remaining -= size;
799 }
800 }
801
802 ResetPendingBatch();
803 }
