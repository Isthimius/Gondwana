Gondwana Game Engine Development (master) - 2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

78 public override void Present(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)
79 {
80 // No UI target — dispose immediately to avoid leak
81 if (_control.IsDisposed || !_control.IsHandleCreated)
82 {
83 bufferImage.Dispose();
84 return;
85 }
86
87 var sourceRect = SKRectI.Intersect(
88 bufferRect,
89 new SKRectI(0, 0, bufferImage.Width, bufferImage.Height));
90
91 var invalidateRect = global::System.Drawing.Rectangle.Intersect(
92 global::System.Drawing.Rectangle.FromLTRB(
93 sourceRect.Left,
94 sourceRect.Top,
95 sourceRect.Right,
96 sourceRect.Bottom),
97 _control.ClientRectangle);
98
99 if (sourceRect.IsEmpty || invalidateRect.IsEmpty)
100 {
101 bufferImage.Dispose();
102 return;
103 }
104
105 lock (_presentSync)
106 {
107 if (_disposed)
108 {
109 bufferImage.Dispose();
110 return;
111 }
112
113 // Swap into current; old one is disposed after the paint that consumes the newest
114 // complete snapshot. Coalesce dirty regions in case Windows combines invalidations.
115 var old = _currentImage;
116 _currentImage = bufferImage;
117 if (!ReferenceEquals(old, _currentImage) && old is not null)
118 _toDispose.Enqueue(old);
119
120 _pendingSourceRect = Union(_pendingSourceRect, sourceRect);
121 }
122
123 _control.Invalidate(invalidateRect);
124 }

126 private void OnPaintSurface(object? sender, SKPaintSurfaceEventArgs e)
127 {
128 var canvas = e.Surface.Canvas;
129
130 SKImage? img;
131 SKRectI sourceRect;
132 lock (_presentSync)
133 {
134 img = _currentImage;
135 sourceRect = _pendingSourceRect;
136 _pendingSourceRect = SKRectI.Empty;
137 }
138
139 try
140 {
141 if (img == null) return;
142
143 var srcI = SKRectI.Intersect(sourceRect, new SKRectI(0, 0, img.Width, img.Height));
144 if (srcI.IsEmpty) return;
145
146 var destI = new SKRectI(srcI.Left, srcI.Top, srcI.Right, srcI.Bottom);
147
148 var boundsI = new SKRectI(0, 0, e.Info.Width, e.Info.Height);
149 var clippedDestI = SKRectI.Intersect(destI, boundsI);
150 if (clippedDestI.IsEmpty) return;
151
152 var dx = clippedDestI.Left - destI.Left;
153 var dy = clippedDestI.Top - destI.Top;
154 var clippedSrcI = new SKRectI(
155 srcI.Left + dx, srcI.Top + dy,
156 srcI.Left + dx + clippedDestI.Width,
157 srcI.Top + dy + clippedDestI.Height);
158
159 // clear the destination patch (overwrite)
160 canvas.DrawRect(clippedDestI, _clearPaint);
161
162 // blit the updated patch (overwrite)
163 canvas.DrawImage(img, clippedSrcI, clippedDestI, _presentPaint);
164 }
165 finally
166 {
167 DisposeStaleImages();
168 }
169 }
170
171 private static SKRectI Union(SKRectI left, SKRectI right)
172 {
173 if (left.IsEmpty) return right;
174 if (right.IsEmpty) return left;
175
176 return new SKRectI(
177 Math.Min(left.Left, right.Left),
178 Math.Min(left.Top, right.Top),
179 Math.Max(left.Right, right.Right),
180 Math.Max(left.Bottom, right.Bottom));
181 }

192 public void Dispose()
193 {
194 SKImage? currentImage;
195 lock (_presentSync)
196 {
197 if (_disposed) return;
198 _disposed = true;
199
200 currentImage = _currentImage;
201 _currentImage = null;
202 _pendingSourceRect = SKRectI.Empty;
203 }
204

206 {
207 _control.PaintSurface -= OnPaintSurface;
208 _control.HandleCreated -= OnHandleCreated;
209 _control.ClientSizeChanged -= OnSizeChanged;
210 }
211

213
214 currentImage?.Dispose();
215 }
