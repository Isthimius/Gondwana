629 private void OnAfterBackgroundTasksExecute()
630 {
631 if (_collectionRequest is null)
632 return;
633
634 long now = Stopwatch.GetTimestamp();
635 if (_lastRefreshTimestamp != 0 &&
636 Stopwatch.GetElapsedTime(_lastRefreshTimestamp, now) < RefreshInterval)
637 {
638 return;
639 }
640
641 _lastRefreshTimestamp = now;
642 Refresh();
643 }

646 {
647 Engine.Instance.AfterBackgroundTasksExecute -= OnAfterBackgroundTasksExecute;

653 private string FormatSnapshot(TelemetrySnapshot? snapshot)
654 {
655 var text = new StringBuilder();
656
657 if (ShowHeader)
658 text.AppendLine(HeaderText);
659
660 View? view = View;
661 if (view is not null)
662 {
663 var extensionContext = new ProfilerWidgetExtensionContext(
664 snapshot,
665 RenderSurfaceHost,
666 view);
667
668 AppendAdditionalLines(text, extensionContext);
669 AppendRuntimeContext(text, extensionContext);
670 }
671
672 if (snapshot is null)
673 {
674 text.Append("Waiting for first completed telemetry window...");
675 return text.ToString().TrimEnd();
676 }
677
678 if (ShowSnapshotMetadata)
679 {
680 text.Append("Generation ")
681 .Append(snapshot.Generation)
682 .Append(" | window ")
683 .Append((snapshot.ElapsedSeconds * 1000d).ToString("0.0"))
684 .Append(" ms");
685
686 if (snapshot.Truncated)
687 text.Append(" | detail truncated");
688
689 text.AppendLine();
690 }
691
692 bool wroteSource = false;
693 foreach (TelemetrySourceSnapshot source in snapshot.Sources)
694 {
695 if (!IsSourceVisible(source.Name))
696 continue;
697
698 var metrics = source.Metrics
699 .Where(pair => ResolveMeasurementVisibility(
700 source.Name,
701 pair.Key,
702 pair.Value.Availability))
703 .OrderBy(pair => pair.Key, StringComparer.Ordinal)
704 .ToArray();
705
706 if (metrics.Length == 0)
707 continue;
708
709 if (wroteSource)
710 text.AppendLine();
711
712 if (ShowSourceHeaders)
713 {
714 text.Append(source.Name)
715 .Append(" [")
716 .Append(source.Backend)
717 .AppendLine("]");
718 }
719
720 foreach (KeyValuePair<string, TelemetrySummary> pair in metrics)
721 {
722 text.Append(" ")
723 .Append(pair.Key)
724 .Append(": ")
725 .AppendLine(FormatSummary(pair.Value, snapshot.ElapsedSeconds));
726 }
727
728 wroteSource = true;
729 }
730
731 if (!wroteSource)
732 text.Append("No measurements selected.");
733
734 return text.ToString().TrimEnd();
735 }

750 StringBuilder text,
751 ProfilerWidgetExtensionContext context)
752 {
753 if ((ContextInfo & ProfilerContextInfo.Scene) != 0)

761 text.Append("Layers / grid cells: ")

763 .Append(" / ")
764 .AppendLine(gridCells.ToString("N0"));

789 .Append(" / ")

791 .Append('x')

794 else if (includeViewport)
795 {
796 text.Append("Viewport: ")
797 .Append(viewport.Width)
798 .Append('x')
799 .AppendLine(viewport.Height.ToString());
800 }
801 else if (includeBackbuffer)
802 {
803 var backbuffer = context.RenderSurfaceHost.Backbuffer;
804 text.Append("Backbuffer: ")
805 .Append(backbuffer.Width)
806 .Append('x')
807 .AppendLine(backbuffer.Height.ToString());

809

817

965 private string BuildMetricSelectorSignature(TelemetrySnapshot snapshot)
966 {
967 var signature = new StringBuilder()
968 .Append(Bounds.Width)
969 .Append('x')
970 .Append(Bounds.Height);
971
972 foreach (var sourceGroup in snapshot.Sources
973 .GroupBy(source => source.Name, StringComparer.OrdinalIgnoreCase)
974 .OrderBy(group => group.Key, StringComparer.OrdinalIgnoreCase))
975 {
976 signature.Append('|').Append(sourceGroup.Key);
977
978 foreach (string backend in sourceGroup
979 .Select(source => source.Backend)
980 .Distinct(StringComparer.Ordinal)
981 .OrderBy(value => value, StringComparer.Ordinal))
982 {
983 signature.Append('[').Append(backend).Append(']');
984 }
985
986 foreach (string metricKey in sourceGroup
987 .SelectMany(source => source.Metrics.Keys)
988 .Distinct(StringComparer.Ordinal)
989 .OrderBy(value => value, StringComparer.Ordinal))
990 {
991 signature.Append(';').Append(metricKey);
992 }
993 }
994
995 return signature.ToString();
996 }

998 private void RefreshMetricSelectorRows(TelemetrySnapshot snapshot)
999 {
1000 if (_metricSelectorList is null)
1001 return;
1002
1003 _metricSelectorRows.Clear();
1004
1005 AddSelectorRow(
1006 MeasurementTooltipsEnabled,
1007 "Hover definitions",
1008 () => MeasurementTooltipsEnabled = !MeasurementTooltipsEnabled);
1009 AddSelectorRow(
1010 ShowUnavailableMeasurements,
1011 "Show unavailable",
1012 () => ShowUnavailableMeasurements = !ShowUnavailableMeasurements);
1013 _metricSelectorRows.Add(new MetricSelectorRow("Show all measurements", () =>
1014 {
1015 MeasurementVisibilityMode = ProfilerMeasurementVisibilityMode.All;
1016 ClearVisibilityOverrides();
1017 }));
1018 _metricSelectorRows.Add(new MetricSelectorRow("Hide all measurements", () =>
1019 {
1020 MeasurementVisibilityMode = ProfilerMeasurementVisibilityMode.Selected;
1021 ClearVisibilityOverrides();
1022 }));
1023 _metricSelectorRows.Add(new MetricSelectorRow("------------------------------", null));
1024
1025 foreach (var sourceGroup in snapshot.Sources
1026 .GroupBy(source => source.Name, StringComparer.OrdinalIgnoreCase)
1027 .OrderBy(group => group.Key, StringComparer.OrdinalIgnoreCase))
1028 {
1029 string sourceName = sourceGroup.Key;
1030 string backend = string.Join(
1031 "/",
1032 sourceGroup
1033 .Select(source => source.Backend)
1034 .Distinct(StringComparer.Ordinal)
1035 .OrderBy(value => value, StringComparer.Ordinal));
1036
1037 bool sourceVisible = IsSourceVisible(sourceName);
1038 AddSelectorRow(
1039 sourceVisible,
1040 $"{sourceName} [{backend}]",
1041 () => SetSourceVisible(sourceName, !IsSourceVisible(sourceName)));
1042
1043 foreach (string metricKey in sourceGroup
1044 .SelectMany(source => source.Metrics.Keys)
1045 .Distinct(StringComparer.Ordinal)
1046 .OrderBy(value => value, StringComparer.Ordinal))
1047 {
1048 TelemetryAvailability availability = sourceGroup
1049 .Where(source => source.Metrics.ContainsKey(metricKey))
1050 .Select(source => source.Metrics[metricKey].Availability)
1051 .DefaultIfEmpty(TelemetryAvailability.Unsupported)
1052 .Max();
1053 bool metricVisible = ResolveMeasurementVisibility(
1054 sourceName,
1055 metricKey,
1056 availability);
1057 string capturedMetricKey = metricKey;
1058 string capturedSourceName = sourceName;
1059 TelemetryAvailability capturedAvailability = availability;
1060 string availabilitySuffix = availability == TelemetryAvailability.Available
