Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

639 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
640 {
641 var canvas = backbuffer.Canvas;
642
643 // get SKImage to draw
644 var img = _image ?? (_bitmap != null ? SKImage.FromBitmap(_bitmap) : null);
645
646 if (img is null)
647 return;
648
649 if (backbuffer is RecordingBackbuffer && img.IsTextureBacked)
650 throw new NotSupportedException(
651 "Desktop frame recording requires a CPU image, not a context-bound texture.");
652
653 // source rect (pixels in image space)
654 SKRect src = _src ?? new SKRect(
655 0,
656 0,
657 img.Width,
658 img.Height);
659
660 // destination rect (screen space), computed by scale mode
661 SKRect dst = ComputeDestRect(destRectScreen.ToPixelAlignedRect(), src, _scale);
662
663 // apply tint/opacity via color filter
664 // combine tint with opacity (multiply in linear-ish sRGB)
665 var tint = _tint ?? SKColors.White;
666
667 var withAlpha = new SKColor(
668 tint.Red,
669 tint.Green,
670 tint.Blue,
671 _opacity);
672
673 using var cf = SKColorFilter.CreateBlendMode(withAlpha, SKBlendMode.Modulate);
674 _paint.ColorFilter = cf;
675
676 // rotation about anchor inside dst
677 if (_rotationDeg != 0f)
678 {
679 float ax = dst.Left + _anchorX * dst.Width;
680 float ay = dst.Top + _anchorY * dst.Height;
681 canvas.Save();
682 canvas.RotateDegrees(_rotationDeg, ax, ay);
683 }
684
685 // draw
686 canvas.DrawImage(
687 img,
688 src,
689 dst,
690 _paint);
691
692 if (_rotationDeg != 0f)
693 canvas.Restore();
694
695 // if we created a transient SKImage from a bitmap, dispose it now
696 if (_image is null && img is not null)
697 img.Dispose();
698 }

700 private static SKRect ComputeDestRect(Rectangle bounds, SKRect src, ScaleMode mode)
701 {
702 var dst = bounds.ToSKRect();
703
704 if (mode == ScaleMode.Stretch)
705 return dst;
706
707 if (mode == ScaleMode.Center) // center, no scale
708 {
709 float w = MathF.Min(dst.Width, src.Width);
710 float h = MathF.Min(dst.Height, src.Height);
711 float dx = dst.MidX - w * 0.5f, dy = dst.MidY - h * 0.5f;
712
713 return new SKRect(
714 dx,
715 dy,
716 dx + w,
717 dy + h);
718 }
719
720 float srcW = src.Width, srcH = src.Height;
721 float dstW = dst.Width, dstH = dst.Height;
722
723 float scaleX = dstW / srcW;
724 float scaleY = dstH / srcH;
725
726 float scale = mode switch
727 {
728 ScaleMode.Fit => MathF.Min(scaleX, scaleY),
729 ScaleMode.Fill => MathF.Max(scaleX, scaleY),
730 ScaleMode.PixelPerfect => MathF.Max(1f, MathF.Floor(MathF.Min(scaleX, scaleY))),
731 _ => 1f
732 };
733
734 float w2 = srcW * scale, h2 = srcH * scale;
735 float x = dst.MidX - w2 * 0.5f;
736 float y = dst.MidY - h2 * 0.5f;
737
738 return new SKRect(
739 x,
740 y,
741 x + w2,
742 y + h2);
743 }
