Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

154 public void Open(Uri source)
155 {
156 ArgumentNullException.ThrowIfNull(source);
157
158 if (!source.IsAbsoluteUri)
159 source = new Uri(Path.GetFullPath(source.OriginalString));
160
161 lock (_control)
162 {
163 ThrowIfDisposed();
164
165 if (!ReplaceMedia(() => new Media(_vlc, source), out long generation))
166 return;
167
168 var media = _media!;
169 _parseCancellation = new CancellationTokenSource();
170
171 _parseTask = media.Parse(
172 source.IsFile ? MediaParseOptions.ParseLocal : MediaParseOptions.ParseNetwork,
173 cancellationToken: _parseCancellation.Token);
174
175 // The continuation does not own the media. Generation validation under _control
176 // prevents any access after replacement/disposal. Open never waits for network parse.
177 _ = ObserveParseAsync(_parseTask, media, generation);
178 }
179 }

215 private bool ReplaceMedia(Func<Media> create, out long generation)
216 {
217 lock (_frames)
218 _acceptFrames = false;
219
220 ++_generation;
221 long replacementGeneration = _generation;
222 ReleaseMedia();
223
224 lock (_frames)
225 _naturalSize = (0, 0);
226
227 _metadata.Reset();
228 Volatile.Write(ref _lastError, null);
229 StateChanged?.Invoke(this, new("MediaOpening"));
230
231 if (_disposed || replacementGeneration != _generation)
232 {
233 generation = 0;
234 return false;
235 }
236
237 _media = create();
238 _player.Media = _media;
239 generation = _generation = _metadata.Begin();
240 StateChanged?.Invoke(this, new("MediaOpened"));
241
242 if (_disposed || generation != _generation)
243 return false;
244
245 lock (_frames)
246 _acceptFrames = true;
247
248 return true;
249 }

251 private async Task ObserveParseAsync(Task<MediaParsedStatus> parse, Media media, long generation)
252 {
253 MediaParsedStatus status;
254
255 try
256 {
257 status = await parse.ConfigureAwait(false);
258 }

260 {
261 return;
262 }
263 catch (Exception ex)
264 {
265 Volatile.Write(ref _lastError, ex);
266 status = MediaParsedStatus.Failed;

269 // Always queue: even a synchronous Parse completion must not raise user events inside Open.

360 private uint SetupFormat(
361 ref IntPtr opaque,
362 IntPtr chroma,
363 ref uint width,
364 ref uint height,
365 ref uint pitches,
366 ref uint lines)
367 {
368 try
369 {
370 var buffer = new VideoDecodeBuffer(checked((int)width), checked((int)height));
371
372 lock (_frames)
373 {
374 _decodeBuffer?.Dispose();
375 _decodeBuffer = buffer;
376 }
377
378 // LibVLC 3 vmem masks: R=0xff0000 G=0xff00 B=0xff. On little endian
379 // RV32 is B,G,R,X (not alpha). Skia uses Bgra8888 + Opaque without swizzling.
380 Marshal.Copy(
381 new byte[] { (byte)'R', (byte)'V', (byte)'3', (byte)'2' },
382 0,
383 chroma,
384 4);
385
386 pitches = (uint)buffer.Stride;
387 lines = (uint)buffer.Lines;
388
389 lock (_frames)
390 _naturalSize = (buffer.Width, buffer.Height);
391
392 return 1;
393 }
394 catch (Exception ex)
395 {
396 Volatile.Write(ref _lastError, ex);
397 return 0;
398 }
