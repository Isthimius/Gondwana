Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

101 public void Open(Uri source)
102 {
103 ArgumentNullException.ThrowIfNull(source);
104 if (!source.IsAbsoluteUri) source = new Uri(Path.GetFullPath(source.OriginalString));
105 lock (_control)
106 {
107 ThrowIfDisposed();
108 if (!ReplaceMedia(() => new Media(_vlc, source), out long generation)) return;
109 var media = _media!;
110 _parseCancellation = new CancellationTokenSource();
111 _parseTask = media.Parse(source.IsFile ? MediaParseOptions.ParseLocal : MediaParseOptions.ParseNetwork,
112 cancellationToken: _parseCancellation.Token);
113 // The continuation does not own the media. Generation validation under _control
114 // prevents any access after replacement/disposal. Open never waits for network parse.
115 _ = ObserveParseAsync(_parseTask, media, generation);
116 }
117 }

120 public void Open(Stream source, bool leaveOpen = false)
121 {
122 ArgumentNullException.ThrowIfNull(source);
123 if (!source.CanRead) throw new ArgumentException("Video stream must be readable.", nameof(source));
124 lock (_control)
125 {
126 ThrowIfDisposed();
127 var input = new StreamMediaInput(source);
128 try
129 {
130 if (!ReplaceMedia(() => new Media(_vlc, input), out _))
131 {
132 input.Dispose();
133 return;
134 }
135 _input = input;
136 _ownedStream = leaveOpen ? null : source;
137 }
138 catch { input.Dispose(); throw; }
139 // Do not preparse and play against the same stream cursor concurrently.
140 // LibVLC discovers stream tracks during playback; Playing publishes metadata.
141 }

144 private bool ReplaceMedia(Func<Media> create, out long generation)
145 {
146 lock (_frames) _acceptFrames = false;
147 ++_generation;
148 long replacementGeneration = _generation;
149 ReleaseMedia();
150 lock (_frames) _naturalSize = (0, 0);
151 _metadata.Reset();
152 Volatile.Write(ref _lastError, null);
153 StateChanged?.Invoke(this, new("MediaOpening"));
154 if (_disposed || replacementGeneration != _generation)
155 {
156 generation = 0;
157 return false;
158 }
159 _media = create();
160 _player.Media = _media;
161 generation = _generation = _metadata.Begin();
162 StateChanged?.Invoke(this, new("MediaOpened"));
163 if (_disposed || generation != _generation) return false;
164 lock (_frames) _acceptFrames = true;
165 return true;
166 }

168 private async Task ObserveParseAsync(Task<MediaParsedStatus> parse, Media media, long generation)
169 {
170 MediaParsedStatus status;
171 try { status = await parse.ConfigureAwait(false); }
172 catch (OperationCanceledException) { return; }
173 catch (Exception ex) { Volatile.Write(ref _lastError, ex); status = MediaParsedStatus.Failed; }
174 // Always queue: even a synchronous Parse completion must not raise user events inside Open.
175 QueueForSource(generation, () => PublishMetadata(media, status == MediaParsedStatus.Done));
176 }

214 private void OnEnded(object? sender, EventArgs e)
215 {
216 long revision = Volatile.Read(ref _playbackRevision);
217 QueueForSource(Volatile.Read(ref _generation), () =>
218 {
219 long generation = _generation;
220 Ended?.Invoke(this, EventArgs.Empty);
221 if (!_disposed && _playRequested && generation == _generation && revision == _playbackRevision && Loop)
222 {
223 // Explicit controls, including Stop from an Ended handler, cancel a queued loop.
224 // Never reenter libvlc from its EndReached callback (documented deadlock risk).
225 _player.Stop();
226 _player.Play();
227 }
228 });

231 private uint SetupFormat(ref IntPtr opaque, IntPtr chroma, ref uint width, ref uint height, ref uint pitches, ref uint lines)
232 {
233 try
234 {
235 var buffer = new VideoDecodeBuffer(checked((int)width), checked((int)height));
236 lock (_frames)
237 {
238 _decodeBuffer?.Dispose();
239 _decodeBuffer = buffer;
240 }
241 // LibVLC 3 vmem masks: R=0xff0000 G=0xff00 B=0xff. On little endian
242 // RV32 is B,G,R,X (not alpha). Skia uses Bgra8888 + Opaque without swizzling.
243 Marshal.Copy(new byte[] { (byte)'R', (byte)'V', (byte)'3', (byte)'2' }, 0, chroma, 4);
244 pitches = (uint)buffer.Stride;
245 lines = (uint)buffer.Lines;
246 lock (_frames) _naturalSize = (buffer.Width, buffer.Height);
247 return 1;
248 }
249 catch (Exception ex) { Volatile.Write(ref _lastError, ex); return 0; }
250 }

314 _parseCancellation = null;
