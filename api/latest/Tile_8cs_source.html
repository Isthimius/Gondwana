Gondwana Game Engine Development (master) - 2.6.0-unreleased
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

425 public int CompareTo(Tile? tile)
426 {
427 if (tile is null)
428 return -1;
429
430 var thisKey = GetSortKeyForCompare();
431 var tileKey = tile.GetSortKeyForCompare();
432
433 if (thisKey.IsPositionFixed && !tileKey.IsPositionFixed)
434 return -1;
435
436 if (!thisKey.IsPositionFixed && tileKey.IsPositionFixed)
437 return 1;
438
439 return (thisKey.Location, thisKey.ZOrder, thisKey.SceneLayerX)
440 .CompareTo((tileKey.Location, tileKey.ZOrder, tileKey.SceneLayerX));
441 }
442
443 private TileSortKey GetSortKeyForCompare()
444 {
445 var renderContext = RenderContext.Current;
446
447 // Outside an active render pass preserve the historical live comparison
448 // semantics. During rendering, capture the expensive virtual/property values
449 // once per tile and reuse them for every comparison in this sort pass.
450 if (renderContext is null)
451 return CaptureSortKey();
452
453 long renderPassId = renderContext.PassId;
454 if (_sortKeyRenderPassId != renderPassId)
455 {
456 _sortKey = CaptureSortKey();
457 _sortKeyRenderPassId = renderPassId;
458 }
459
460 return _sortKey;
461 }
462
463 private TileSortKey CaptureSortKey()
464 {
465 bool isPositionFixed = IsPositionFixed;
466 Rectangle drawLocationWorld = DrawLocationWorld;
467 Spacing overhang = Overhang;
468 float location = isPositionFixed
469 ? drawLocationWorld.Top + overhang.Top
470 : drawLocationWorld.Bottom - overhang.Bottom - 1;
471
472 return new TileSortKey(
473 isPositionFixed,
474 location,
475 zOrder,
476 SceneLayerCoordinates.X);
477 }
478

480 bool IsPositionFixed,
481 float Location,
482 int ZOrder,
483 float SceneLayerX);
