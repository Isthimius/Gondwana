Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

188 {

191 // not consume RefreshQueue, so avoid calculating those bounds entirely.
192 bool usesDirtyRegionRendering =
193 SceneLayer.Scene?.UsesDirtyRegionRendering ?? true;

548 public int CompareTo(Tile? tile)
549 {
550 if (tile is null)
551 return -1;
552
553 var thisKey = GetSortKeyForCompare();
554 var tileKey = tile.GetSortKeyForCompare();
555
556 if (thisKey.IsPositionFixed && !tileKey.IsPositionFixed)
557 return -1;
558
559 if (!thisKey.IsPositionFixed && tileKey.IsPositionFixed)
560 return 1;
561
562 return (thisKey.Location, thisKey.ZOrder, thisKey.SceneLayerX)
563 .CompareTo((tileKey.Location, tileKey.ZOrder, tileKey.SceneLayerX));
564 }
565
566 private TileSortKey GetSortKeyForCompare()
567 {
568 var renderContext = RenderContext.Current;
569
570 // Outside an active render pass preserve the historical live comparison
571 // semantics. During rendering, capture the expensive virtual/property values
572 // once per tile and reuse them for every comparison in this sort pass.
573 if (renderContext is null)
574 return CaptureSortKey();
575
576 long renderPassId = renderContext.PassId;
577
578 if (_sortKeyRenderPassId != renderPassId)
579 {
580 _sortKey = CaptureSortKey();
581 _sortKeyRenderPassId = renderPassId;
582 }
583
584 return _sortKey;
585 }
586
587 private TileSortKey CaptureSortKey()
588 {
589 bool isPositionFixed = IsPositionFixed;
590 Rectangle drawLocationWorld = DrawLocationWorld;
591 Spacing overhang = Overhang;
592
593 float location = isPositionFixed
594 ? drawLocationWorld.Top + overhang.Top
595 : drawLocationWorld.Bottom - overhang.Bottom - 1;
596
597 return new TileSortKey(
598 isPositionFixed,
599 location,
600 zOrder,
601 SceneLayerCoordinates.X);
602 }

605 bool IsPositionFixed,
606 float Location,
607 int ZOrder,
608 float SceneLayerX);
