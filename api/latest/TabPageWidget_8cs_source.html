Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

382 private static void SetWidgetZOrder(
383 WidgetBase widget,
384 int zOrder)
385 {
386 DirectDrawingBase[] visuals = EnumerateVisuals(widget).ToArray();
387
388 if (visuals.Length == 0)
389 return;
390
391 int currentMinimum = visuals.Min(static visual => visual.ZOrder);
392 int delta = zOrder - currentMinimum;
393
394 foreach (DirectDrawingBase visual in visuals)
395 visual.ZOrder += delta;
396 }
397
398 private static IEnumerable<DirectDrawingBase> EnumerateVisuals(IDirectCompositeContainer container)
399 {
400 foreach (IDirectCompositeChild child in container.Children)
401 {
402 if (child is DirectDrawingBase drawing)
403 yield return drawing;
404
405 if (child is IDirectCompositeContainer nestedContainer)
406 {
407 foreach (DirectDrawingBase nestedDrawing in EnumerateVisuals(nestedContainer))
408 yield return nestedDrawing;
409 }
410 }
