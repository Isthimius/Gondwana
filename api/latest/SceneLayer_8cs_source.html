Gondwana Game Engine Development (master) - 2.6.0-unreleased
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

770 public virtual RectangleF GetLayerBoundsPx()
771 {
772 if (GridColumnCount == 0 || GridRowCount == 0)
773 return RectangleF.Empty;
774
775 int maxCol = GridColumnCount - 1;
776 int maxRow = GridRowCount - 1;
777
778 // Corner tiles
779 var corners = new[]
780 {
781 this[0, 0],
782 this[maxCol, 0],
783 this[0, maxRow],
784 this[maxCol, maxRow]
785 };
786
787 RectangleF result = RectangleF.Empty;
788 bool hasBounds = false;
789
790 foreach (var tile in corners)
791 {
792 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
793 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
794
795 if (!hasBounds)
796 {
797 result = rf;
798 hasBounds = true;
799 }
800 else
801 {
802 result = RectangleF.Union(result, rf);
803 }
804 }
805
806 return result;
807 }

865 internal virtual List<IDrawable> GetDrawablesInWorldRect(Rectangle worldRect, bool includeOverhang = true)
866 {
867 // Make selection rect covering so we never miss the edge tile.
868 // Drawing is still clipped later, so over-selecting is safe.
869 var queryRect = worldRect;
870 queryRect.Inflate(TileWidth, TileHeight); // <- KEY (tile-sized)
871 queryRect.Inflate(1, 1); // optional boundary insurance
872
873 // Gather into a list so we can sort it.
874 var list = new List<IDrawable>(64);
875
876 // 1) Grid tiles
877 var sceneLayerTiles = CoordinateSystem.GetSceneLayerTilesInPixelRange(
878 this,
879 worldRect,
880 includeOverhang: includeOverhang);
881
882 if (sceneLayerTiles != null)
883 {
884 for (int i = 0; i < sceneLayerTiles.Count; i++)
885 {
886 var tile = sceneLayerTiles[i];
887
888 if (tile is null)
889 continue;
890
891 if (!tile.Visible)
892 continue;
893
894 // Defensive overlap check (same idea as sprites)
895 if (!tile.DrawLocationWorld.IntersectsWith(queryRect))
896 continue;
897
898 list.Add(tile);
899 }
900 }
901
902 // 2) Sprites
903 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(queryRect, this, fullEnclosures: false);
904
905 for (int i = 0; i < sprites.Count; i++)
906 {
907 var sprite = sprites[i];
908
909 if (sprite is null)
910 continue;
911
912 // Defensive overlap check (cheap)
913 if (!sprite.VisualBoundsWorld.IntersectsWith(queryRect))
914 continue;
915
916 list.Add(sprite);
917 }
918
919 // 3) DirectDrawing instances
920 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
921
922 for (int i = 0; i < drawings.Count; i++)
923 {
924 var drawing = drawings[i];
925
926 if (!drawing.Visible)
927 continue;
928
929 // Must be SceneLayer-mode by definition if it's "for layer", but be defensive:
930 if (drawing.Mode != DirectDrawingMode.SceneLayer)
931 continue;
932
933 // Only include if it intersects this dirty rect
934 if (!drawing.WorldBounds.IntersectsWith(worldRect))
935 continue;
936
937 list.Add(drawing);
938 }
939
940 // 4) Sort using Tile.CompareTo
941 list.Sort(CompareDrawables); // ← this calls Tile.CompareTo internally
942 return list;
943 }
944
945 private static int CompareDrawables(IDrawable a, IDrawable b)
946 {
947 int z = a.ZOrder.CompareTo(b.ZOrder);
948 if (z != 0)
949 return z;
950
951 // Preserve legacy ordering for tiles/sprites when Z ties
952 if (a is Tile ta && b is Tile tb)
953 return ta.CompareTo(tb);
954
955 // Stable tie-breaker (avoid flicker)
956 return a.Id.CompareTo(b.Id);
957 }
