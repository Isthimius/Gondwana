Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

821 public virtual RectangleF GetLayerBoundsPx()
822 {
823 if (GridColumnCount == 0 || GridRowCount == 0)
824 return RectangleF.Empty;
825
826 int maxCol = GridColumnCount - 1;
827 int maxRow = GridRowCount - 1;
828
829 // Corner tiles
830 var corners = new[]
831 {
832 this[0, 0],
833 this[maxCol, 0],
834 this[0, maxRow],
835 this[maxCol, maxRow]
836 };
837
838 RectangleF result = RectangleF.Empty;
839 bool hasBounds = false;
840
841 foreach (var tile in corners)
842 {
843 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
844 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
845
846 if (!hasBounds)
847 {
848 result = rf;
849 hasBounds = true;
850 }
851 else
852 {
853 result = RectangleF.Union(result, rf);
854 }
855 }
856
857 return result;
858 }

916 internal virtual List<IDrawable> GetDrawablesInWorldRect(Rectangle worldRect, bool includeOverhang = true)
917 {
918 if (WrapHorizontally || WrapVertically)
919 return GetWrappedDrawables(worldRect);
920
921 // Make selection rect covering so we never miss the edge tile.
922 // Drawing is still clipped later, so over-selecting is safe.
923 var queryRect = worldRect;
924 queryRect.Inflate(TileWidth, TileHeight); // <- KEY (tile-sized)
925 queryRect.Inflate(1, 1); // optional boundary insurance
926
927 // Gather into a list so we can sort it.
928 var list = new List<IDrawable>(64);
929
930 // 1) Grid tiles
931 var sceneLayerTiles = CoordinateSystem.GetSceneLayerTilesInPixelRange(
932 this,
933 worldRect,
934 includeOverhang: includeOverhang);
935
936 if (sceneLayerTiles != null)
937 {
938 for (int i = 0; i < sceneLayerTiles.Count; i++)
939 {
940 var tile = sceneLayerTiles[i];
941
942 if (tile is null)
943 continue;
944
945 if (!tile.Visible)
946 continue;
947
948 // Defensive overlap check (same idea as sprites)
949 if (!tile.DrawLocationWorld.IntersectsWith(queryRect))
950 continue;
951
952 list.Add(tile);
953 }
954 }
955
956 // 2) Sprites
957 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(queryRect, this, fullEnclosures: false);
958
959 for (int i = 0; i < sprites.Count; i++)
960 {
961 var sprite = sprites[i];
962
963 if (sprite is null)
964 continue;
965
966 // Defensive overlap check (cheap)
967 if (!sprite.VisualBoundsWorld.IntersectsWith(queryRect))
968 continue;
969
970 list.Add(sprite);
971 }
972
973 // 3) DirectDrawing instances
974 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
975
976 for (int i = 0; i < drawings.Count; i++)
977 {
978 var drawing = drawings[i];
979
980 if (!drawing.Visible)
981 continue;
982
983 // Must be SceneLayer-mode by definition if it's "for layer", but be defensive:
984 if (drawing.Mode != DirectDrawingMode.SceneLayer)
985 continue;
986
987 // Only include if it intersects this dirty rect
988 if (!drawing.WorldBounds.IntersectsWith(worldRect))
989 continue;
990
991 list.Add(drawing);
992 }
993
994 // 4) Sort using Tile.CompareTo
995 list.Sort(CompareDrawables); // ← this calls Tile.CompareTo internally
996 return list;
997 }

999 private List<IDrawable> GetWrappedDrawables(Rectangle worldRect)
1000 {
1001 var period = GetPeriod();
1002 var instances = new List<IDrawable>();
1003 void Add(IDrawable drawable, RectangleF bounds)
1004 {
1005 if (!drawable.Visible) return;
1006 foreach (var offset in period.Offsets(bounds, worldRect))
1007 {
1008 if (instances.Count >= 1_000_000)
1009 throw new InvalidOperationException("Too many visible wrapped instances.");
1010 instances.Add(new WrappedDrawable(drawable, this, offset));
1011 }
1012 }
1013 // Inspect actual artwork bounds, including overhang, rather than assuming
1014 // a tile-sized margin can enclose all content.
1015 foreach (var tile in _sceneLayerTileArray)
1016 if (tile is not null) Add(tile, tile.DrawLocationWorld);
1017 foreach (var sprite in SpriteManager.Instance.AllSprites)
1018 if (ReferenceEquals(sprite.SceneLayer, this)) Add(sprite, sprite.VisualBoundsWorld);
1019 foreach (var drawing in DirectDrawingManager.Instance.GetDrawingsForLayer(this))
1020 if (drawing.Mode == DirectDrawingMode.SceneLayer) Add(drawing, drawing.WorldBounds);
1021 instances.Sort((a, b) =>
1022 {
1023 var x = (WrappedDrawable)a;
1024 var y = (WrappedDrawable)b;
1025 int z = a.ZOrder.CompareTo(b.ZOrder);
1026 if (z != 0) return z;
1027 if (x.Owner is Tile ta && y.Owner is Tile tb)
1028 {
1029 if (ta.IsPositionFixed != tb.IsPositionFixed) return ta.IsPositionFixed ? -1 : 1;
1030 float ay = (ta.IsPositionFixed ? ta.DrawLocationWorld.Top + ta.Overhang.Top : ta.DrawLocationWorld.Bottom - ta.Overhang.Bottom - 1) + x.Offset.Y;
1031 float by = (tb.IsPositionFixed ? tb.DrawLocationWorld.Top + tb.Overhang.Top : tb.DrawLocationWorld.Bottom - tb.Overhang.Bottom - 1) + y.Offset.Y;
1032 int depth = ay.CompareTo(by);
1033 if (depth != 0) return depth;
1034 int horizontal = (ta.DrawLocationWorld.Left + x.Offset.X).CompareTo(tb.DrawLocationWorld.Left + y.Offset.X);
1035 if (horizontal != 0) return horizontal;
1036 }
1037 int identity = a.Id.CompareTo(b.Id);
1038 return identity != 0 ? identity : (x.Offset.Y, x.Offset.X).CompareTo((y.Offset.Y, y.Offset.X));
1039 });
1040 return instances;
1041 }

1043 private static int CompareDrawables(IDrawable a, IDrawable b)
1044 {
1045 int z = a.ZOrder.CompareTo(b.ZOrder);
1046 if (z != 0)
1047 return z;
1048
1049 // Preserve legacy ordering for tiles/sprites when Z ties
1050 if (a is Tile ta && b is Tile tb)
1051 return ta.CompareTo(tb);
1052
1053 // Stable tie-breaker (avoid flicker)
1054 return a.Id.CompareTo(b.Id);
1055 }
