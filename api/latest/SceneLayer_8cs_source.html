Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

875 public virtual RectangleF GetLayerBoundsPx()
876 {
877 if (GridColumnCount == 0 || GridRowCount == 0)
878 return RectangleF.Empty;
879
880 int maxCol = GridColumnCount - 1;
881 int maxRow = GridRowCount - 1;
882
883 // Corner tiles
884 var corners = new[]
885 {
886 this[0, 0],
887 this[maxCol, 0],
888 this[0, maxRow],
889 this[maxCol, maxRow]
890 };
891
892 RectangleF result = RectangleF.Empty;
893 bool hasBounds = false;
894
895 foreach (var tile in corners)
896 {
897 var px = CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: true);
898 var rf = RectangleF.FromLTRB(px.Left, px.Top, px.Right, px.Bottom);
899
900 if (!hasBounds)
901 {
902 result = rf;
903 hasBounds = true;
904 }
905 else
906 {
907 result = RectangleF.Union(result, rf);
908 }
909 }
910
911 // Interior placements can extend beyond the corner cells after rotation.
912 foreach (var tile in TransformedTiles)
913 result = RectangleF.Union(result, tile.DrawLocationWorld);
914
915 return result;
916 }

1040 private List<IDrawable> GetDrawablesInWorldRectCore(
1041 Rectangle worldRect,
1042 bool includeOverhang,
1043 bool collectSortDiagnostics,
1044 out long sortTicks)
1045 {
1046 if (WrapHorizontally || WrapVertically)
1047 return GetWrappedDrawables(worldRect, collectSortDiagnostics, out sortTicks);
1048
1049 // Make selection rect covering so we never miss the edge tile.
1050 // Drawing is still clipped later, so over-selecting is safe.
1051 var queryRect = worldRect;
1052 queryRect.Inflate(TileWidth, TileHeight);
1053 queryRect.Inflate(1, 1);
1054
1055 // 1) Grid tiles. Every projection supplies a conservative render query whose
1056 // traversal follows fixed-tile depth order for that coordinate system.
1057 var renderTiles = CoordinateSystem.GetSceneLayerTilesForRendering(
1058 this,
1059 worldRect,
1060 includeOverhang);
1061 var sceneLayerTiles = renderTiles.Tiles;
1062
1063 var list = new List<IDrawable>(sceneLayerTiles.Count);
1064 bool uniformGridZOrder = true;
1065 int? gridZOrder = null;
1066
1067 for (int i = 0; i < sceneLayerTiles.Count; i++)
1068 {
1069 var tile = sceneLayerTiles[i];
1070
1071 if (!tile.Visible || !HasRenderableTileContent(tile))
1072 continue;
1073
1074 int tileZ = tile.ZOrder;
1075 if (gridZOrder.HasValue && gridZOrder.Value != tileZ)
1076 uniformGridZOrder = false;
1077 else
1078 gridZOrder ??= tileZ;
1079
1080 list.Add(tile);
1081 }
1082
1083 bool hasNonGridDrawables = false;
1084
1085 // 2) Sprites
1086 var sprites = SpriteManager.Instance.GetSpritesInWorldRectRange(
1087 queryRect,
1088 this,
1089 fullEnclosures: false);
1090
1091 for (int i = 0; i < sprites.Count; i++)
1092 {
1093 var sprite = sprites[i];
1094
1095 if (sprite is null)
1096 continue;
1097
1098 if (!sprite.VisualBoundsWorld.IntersectsWith(queryRect))
1099 continue;
1100
1101 list.Add(sprite);
1102 hasNonGridDrawables = true;
1103 }
1104
1105 // 3) DirectDrawing instances
1106 var drawings = DirectDrawingManager.Instance.GetDrawingsForLayer(this);
1107
1108 for (int i = 0; i < drawings.Count; i++)
1109 {
1110 var drawing = drawings[i];
1111
1112 if (!drawing.Visible)
1113 continue;
1114
1115 if (drawing.Mode != DirectDrawingMode.SceneLayer)
1116 continue;
1117
1118 if (!drawing.WorldBounds.IntersectsWith(worldRect))
1119 continue;
1120
1121 list.Add(drawing);
1122 hasNonGridDrawables = true;
1123 }
1124
1125 // Projection-specific render queries are emitted in fixed-tile depth order.
1126 // Per-tile Z differences or geometry-changing transforms still fall back to
1127 // verification/sort so the optimization cannot change rendering semantics.
1128 bool gridOrderGuaranteed =
1129 renderTiles.IsRenderOrdered &&
1130 uniformGridZOrder;
1131
1132 long sortStartTick = collectSortDiagnostics ? HighResTimer.GetCurrentTick() : 0;
1133 if (hasNonGridDrawables)
1134 {
1135 list.Sort(CompareDrawables);
1136 }
1137 else if (!gridOrderGuaranteed && !IsSorted(list, CompareDrawables))
1138 {
1139 list.Sort(CompareDrawables);
1140 }
1141
1142 sortTicks = collectSortDiagnostics
1143 ? Math.Max(0, HighResTimer.GetCurrentTick() - sortStartTick)
1144 : 0;
1145
1146 return list;
1147 }

1149 private List<IDrawable> GetWrappedDrawables(
1150 Rectangle worldRect,
1151 bool collectSortDiagnostics,
1152 out long sortTicks)
1153 {
1154 var period = GetPeriod();
1155 var instances = new List<IDrawable>();
1156 void Add(IDrawable drawable, RectangleF bounds)
1157 {
1158 if (!drawable.Visible) return;
1159 foreach (var offset in period.Offsets(bounds, worldRect))
1160 {
1161 if (instances.Count >= 1_000_000)
1162 throw new InvalidOperationException("Too many visible wrapped instances.");
1163 instances.Add(new WrappedDrawable(drawable, this, offset));
1164 }
1165 }
1166
1167 // Inspect actual artwork bounds, including overhang, rather than assuming
1168 // a tile-sized margin can enclose all content.
1169 foreach (var tile in _sceneLayerTileArray)
1170 if (tile is not null) Add(tile, tile.DrawLocationWorld);
1171 foreach (var sprite in SpriteManager.Instance.AllSprites)
1172 if (ReferenceEquals(sprite.SceneLayer, this)) Add(sprite, sprite.VisualBoundsWorld);
1173 foreach (var drawing in DirectDrawingManager.Instance.GetDrawingsForLayer(this))
1174 if (drawing.Mode == DirectDrawingMode.SceneLayer) Add(drawing, drawing.WorldBounds);
1175
1176 long sortStartTick = collectSortDiagnostics ? HighResTimer.GetCurrentTick() : 0;
1177 instances.Sort(CompareWrappedDrawables);
1178 sortTicks = collectSortDiagnostics
1179 ? Math.Max(0, HighResTimer.GetCurrentTick() - sortStartTick)
1180 : 0;
1181
1182 return instances;
1183 }

1214 private static int CompareWrappedDrawables(IDrawable a, IDrawable b)
1215 {
1216 var x = (WrappedDrawable)a;
1217 var y = (WrappedDrawable)b;
1218 int z = a.ZOrder.CompareTo(b.ZOrder);
1219 if (z != 0) return z;
1220
1221 if (x.Owner is Tile ta && y.Owner is Tile tb)
1222 {
1223 if (ta.IsPositionFixed != tb.IsPositionFixed)
1224 return ta.IsPositionFixed ? -1 : 1;
1225
1226 float ay =
1227 (ta.IsPositionFixed
1228 ? ta.DrawLocationWorld.Top + ta.Overhang.Top
1229 : ta.DrawLocationWorld.Bottom - ta.Overhang.Bottom - 1) +
1230 x.Offset.Y;
1231 float by =
1232 (tb.IsPositionFixed
1233 ? tb.DrawLocationWorld.Top + tb.Overhang.Top
1234 : tb.DrawLocationWorld.Bottom - tb.Overhang.Bottom - 1) +
1235 y.Offset.Y;
1236 int depth = ay.CompareTo(by);
1237 if (depth != 0) return depth;
1238
