Gondwana Game Engine Development (master) - 2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

112 public void ZoomAroundScreenPoint(SceneLayer layer, PointF screenPoint, float targetZoom, float durationSeconds)
113 {
114 ArgumentNullException.ThrowIfNull(layer);
115
116 targetZoom = Math.Clamp(targetZoom, MinZoom, MaxZoom);
117
118 PointF worldUnderCursor = ScreenPxToWorldPx(layer, screenPoint);
119
120 // Anchored zoom owns camera positioning until the zoom completes. Cancel an
121 // unrelated explicit pan, but leave follow configuration intact so it can
122 // resume on the next update after the anchored zoom finishes.
123 Camera.CancelPan();
124
125 if (durationSeconds <= 0f)
126 {
127 ClearZoomAnchor();
128 Viewport.SnapZoom(targetZoom);
129 SnapCameraToZoomAnchor(layer, screenPoint, worldUnderCursor);
130 return;
131 }
132
133 _zoomAnchorLayer = layer;
134 _zoomAnchorScreenPoint = screenPoint;
135 _zoomAnchorWorldPoint = worldUnderCursor;
136
137 Viewport.ZoomToOverDuration(targetZoom, durationSeconds);
138 }

143 internal void Update(float dtSeconds)
144 {
145 if (_zoomAnchorLayer is { } anchorLayer)
146 {
147 // Update zoom first, then derive the one camera position that keeps

149 Viewport.UpdateZoom(dtSeconds);

153 _zoomAnchorWorldPoint);
154

156 ClearZoomAnchor();

165 private void SnapCameraToZoomAnchor(
166 SceneLayer layer,
167 PointF screenPoint,
168 PointF worldPoint)
169 {
170 float zoom = Viewport.Zoom > 0f
171 ? Viewport.Zoom
172 : 1f;
173
174 PointF effectOffset = GetEffectOffsetPx(layer);
175
176 float offsetX =
177 Viewport.TargetRectPx.Left
178 + Viewport.ScreenOffsetPx.X
179 + effectOffset.X;
180
181 float offsetY =
182 Viewport.TargetRectPx.Top
183 + Viewport.ScreenOffsetPx.Y
184 + effectOffset.Y;
185
186 float parallax = layer.Parallax;
187 if (Math.Abs(parallax) < 1e-6f)
188 parallax = 1f;
189
190 float localX = screenPoint.X - offsetX;
191 float localY = screenPoint.Y - offsetY;
192
193 // screen = offset + (world - camera*p) * zoom
194 // camera = (world - local/zoom) / p
195 float cameraX =
196 (worldPoint.X - localX / zoom)
197 / parallax;
198
199 float cameraY =
200 (worldPoint.Y - localY / zoom)
201 / parallax;
202
203 Camera.SnapTo(new PointF(cameraX, cameraY));
204 }

234 public PointF ScreenPxToWorldPx(SceneLayer layer, PointF screenPx)
235 {
236 RenderContext? context = GetCurrentRenderContext();
237 float zoom = context?.ViewportZoom
238 ?? (Viewport.Zoom <= 0f ? 1f : Viewport.Zoom);
239 Rectangle targetRect = context?.ViewportTargetRectPx
240 ?? Viewport.TargetRectPx;
241 PointF screenOffset = context?.ViewportScreenOffsetPx
242 ?? Viewport.ScreenOffsetPx;
243 PointF cameraPosition = context?.CameraPositionPx
244 ?? Camera.PositionPx;
245 PointF effectOffset = GetEffectOffsetPx(layer);
246 float offsetX = targetRect.Left + screenOffset.X + effectOffset.X;
247 float offsetY = targetRect.Top + screenOffset.Y + effectOffset.Y;
248 float parallax = layer.Parallax;
249
250 float worldX = cameraPosition.X * parallax
251 + (screenPx.X - offsetX) / zoom;
252
253 float worldY = cameraPosition.Y * parallax
254 + (screenPx.Y - offsetY) / zoom;
255
256 return new PointF(worldX, worldY);
257 }

404 internal PointF GetEffectOffsetPx(SceneLayer? layer)
405 {
406 RenderContext? context = GetCurrentRenderContext();
407 PointF factor = context?.ViewEffectOffsetFactor

412 ?? Viewport.TargetRectPx;

416 factor = new PointF(

426 pixels.Y + factor.Y * targetRect.Height);
