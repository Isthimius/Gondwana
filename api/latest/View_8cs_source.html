Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

110 public void ZoomAroundScreenPoint(SceneLayer layer, PointF screenPoint, float targetZoom, float durationSeconds)
111 {
112 ArgumentNullException.ThrowIfNull(layer);
113
114 targetZoom = Math.Clamp(targetZoom, MinZoom, MaxZoom);
115
116 PointF worldUnderCursor = ScreenPxToWorldPx(layer, screenPoint);
117
118 // Anchored zoom owns camera positioning until the zoom completes. Cancel an
119 // unrelated explicit pan, but leave follow configuration intact so it can
120 // resume on the next update after the anchored zoom finishes.
121 Camera.CancelPan();
122
123 if (durationSeconds <= 0f)
124 {
125 ClearZoomAnchor();
126 Viewport.SnapZoom(targetZoom);
127 SnapCameraToZoomAnchor(layer, screenPoint, worldUnderCursor);
128 return;
129 }
130
131 _zoomAnchorLayer = layer;
132 _zoomAnchorScreenPoint = screenPoint;
133 _zoomAnchorWorldPoint = worldUnderCursor;
134
135 Viewport.ZoomToOverDuration(targetZoom, durationSeconds);
136 }

163 private void SnapCameraToZoomAnchor(
164 SceneLayer layer,
165 PointF screenPoint,
166 PointF worldPoint)
167 {
168 float zoom = Viewport.Zoom > 0f
169 ? Viewport.Zoom
170 : 1f;
171
172 PointF effectOffset = GetEffectOffsetPx(layer);
173
174 float offsetX =
175 Viewport.TargetRectPx.Left
176 + Viewport.ScreenOffsetPx.X
177 + effectOffset.X;
178
179 float offsetY =
180 Viewport.TargetRectPx.Top
181 + Viewport.ScreenOffsetPx.Y
182 + effectOffset.Y;
183
184 float parallax = layer.Parallax;
185 if (Math.Abs(parallax) < 1e-6f)
186 parallax = 1f;
187
188 float localX = screenPoint.X - offsetX;
189 float localY = screenPoint.Y - offsetY;
190
191 // screen = offset + (world - camera*p) * zoom
192 // camera = (world - local/zoom) / p
193 float cameraX =
194 (worldPoint.X - localX / zoom)
195 / parallax;
196
197 float cameraY =
198 (worldPoint.Y - localY / zoom)
199 / parallax;
200
201 Camera.SnapTo(new PointF(cameraX, cameraY));
202 }

232 public PointF ScreenPxToWorldPx(SceneLayer layer, PointF screenPx)
233 {
234 RenderContext? context = GetCurrentRenderContext();
235 float zoom = context?.ViewportZoom
236 ?? (Viewport.Zoom <= 0f ? 1f : Viewport.Zoom);
237 Rectangle targetRect = context?.ViewportTargetRectPx
238 ?? Viewport.TargetRectPx;
239 PointF screenOffset = context?.ViewportScreenOffsetPx
240 ?? Viewport.ScreenOffsetPx;
241 PointF cameraPosition = context?.CameraPositionPx
242 ?? Camera.PositionPx;
243 PointF effectOffset = GetEffectOffsetPx(layer);
244 float offsetX = targetRect.Left + screenOffset.X + effectOffset.X;
245 float offsetY = targetRect.Top + screenOffset.Y + effectOffset.Y;
246 float parallax = layer.Parallax;
247
248 float worldX = cameraPosition.X * parallax
249 + (screenPx.X - offsetX) / zoom;

252 + (screenPx.Y - offsetY) / zoom;

402 internal PointF GetEffectOffsetPx(SceneLayer? layer)

404 RenderContext? context = GetCurrentRenderContext();

406 ?? EffectOffsetFactor;

408 ?? EffectOffsetPx;
409 Rectangle targetRect = context?.ViewportTargetRectPx
410 ?? Viewport.TargetRectPx;

414 factor = new PointF(
415 factor.X + layer.EffectOffsetFactor.X,

417 pixels = new PointF(
418 pixels.X + layer.EffectOffsetPx.X,
