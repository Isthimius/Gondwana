Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

258 internal void ReloadIntoManager(bool forceReload = false)
259 {
260 if (string.IsNullOrWhiteSpace(Key))
261 throw new InvalidOperationException("AudioResource has no Key and cannot be reloaded.");
262
263 var manager = AudioResourceManager.Instance;
264
265 if (!forceReload && manager.TryGet(Key, out var existing) && existing is not null)
266 {
267 existing.Volume = Volume;
268 existing.Pan = Pan;
269 existing.PlaybackSpeed = PlaybackSpeed;
270 existing.IsLooping = IsLooping;
271 return;
272 }
273
274 if (forceReload && manager.Contains(Key))
275 manager.Unload(Key);
276
277 AudioResource loaded;
278 if (AssetIdentifier is not null)
279 {
280 using var stream = AssetIdentifier.Data;
281 if (stream is null)
282 throw new InvalidOperationException($"Missing asset data for {Key}.");
283
284 loaded = manager.LoadFromStream(Key, stream, SourceExtension ?? ".wav", Volume, Pan, PlaybackSpeed);
285 loaded.SetAssetIdentifier(AssetIdentifier);
286 }
287 else if (!string.IsNullOrWhiteSpace(SourceFilePath))
288 {
289 loaded = manager.LoadFromFile(Key, SourceFilePath, Volume, Pan, PlaybackSpeed);
290 }
291 else if (!string.IsNullOrWhiteSpace(SourceUri))
292 {
293 loaded = manager.LoadFromUri(Key, SourceUri, Volume, Pan, PlaybackSpeed);
294 }
295 else
296 {
297 throw new InvalidOperationException($"AudioResource '{Key}' has no persisted source.");
298 }
299
300 loaded.IsLooping = IsLooping;
301 }
