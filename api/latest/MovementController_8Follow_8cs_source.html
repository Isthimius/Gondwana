Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

356 private bool AdvanceFollow()
357 {
358 // --- Pixel-follow ---
359 if (_followPixel is not null)
360 {
361 var goalPx = _followPixel() + _followOffsetPx;
362
363 if (_mover.PositionSpace == MovementSpace.Grid)
364 {
365 // grid-space mover following pixel goal...
366 return Advance_TargetPixel_MoverGrid(goalPx);
367 }
368 else
369 {
370 // pixel mover following pixel goal...
371 return Advance_TargetPixel_MoverPixel(goalPx);
372 }
373 }
374
375 // --- Grid-follow (target provides GRID coords each frame) ---
376 if (_followGridTarget is not null)
377 {
378 var layer = _followGridTarget.SceneLayer;
379
380 // target grid coordinate (+ any grid offset)
381 var goalCoordinates = _followGridTarget.GetPosition() + _followGridOffset;
382
383 if (_mover.PositionSpace == MovementSpace.Grid)
384 {
385 // GRID follower -> set/use GRID directly (no conversion)
386 return Advance_TargetGrid_MoverGrid(goalCoordinates);
387 }
388 else
389 {
390 // PIXEL follower -> convert GRID -> PIXEL, then apply optional pixel offset
391 return Advance_TargetGrid_MoverPixel(goalCoordinates, layer);
392 }
393 }
394
395 return false; // not following this frame
396 }

398 private bool Advance_TargetPixel_MoverGrid(Vector2 goalPx)
399 {
400 // follower is grid-space: convert pixel -> grid
401 if (_sceneLayer is null || _coords is null)
402 throw new InvalidOperationException("Pixel->Grid follow requires scene layer/coords.");
403
404 var gridPt = _coords.GetSceneLayerCoordinatesAtPixel(_sceneLayer, new PointF(goalPx.X, goalPx.Y));
405 var goalGrid = new Vector2(gridPt.X, gridPt.Y);
406
407 if (_followHard)
408 {
409 _mover.SetPosition(goalGrid);
410 return true;
411 }
412
413 if (_followEasing is not null && _followDurationSec > 0f)
414 {
415 MoveTo(
416 goalGrid,
417 _followDurationSec,
418 _followEasing,
419 _followSnapTiles);
420
421 return false;
422 }
423
424 MoveToward(goalGrid, _followSpeedTilesPerSec, _followSnapTiles);
425 return false;
426 }

432 _mover.SetPosition(goalPx);

478 private bool Advance_TargetGrid_MoverPixel(Vector2 goalCoordinates, SceneLayer sceneLayer)
479 {
480 var targetPx = sceneLayer.CoordinateSystem.GetAnchorPixelAtSceneLayerCoordinates(
481 sceneLayer,
482 new PointF(goalCoordinates.X, goalCoordinates.Y));
483
484 var goalPx = new Vector2(targetPx.X, targetPx.Y) + _followOffsetPx;
485
486 if (_followHard)
487 {
488 _mover.SetPosition(goalPx);
489 return true;
490 }
491
492 if (_followEasing is not null && _followDurationSec > 0f)
493 {
494 MoveTo(
495 goalPx,
496 _followDurationSec,
497 _followEasing,
498 _followSnapPx);
499
500 return false;
501 }
502
503 // tiles/sec → px/sec (sample +X to estimate px-per-tile length)
504 var pxRight = sceneLayer.CoordinateSystem.GetAnchorPixelAtSceneLayerCoordinates(
505 sceneLayer,
506 new PointF(goalCoordinates.X + 1f, goalCoordinates.Y));
507
508 float pxPerTile = MathF.Max(1f, new Vector2(pxRight.X - targetPx.X, pxRight.Y - targetPx.Y).Length());
509 float speedPxPerSec = _followSpeedTilesPerSec * pxPerTile;
510 float snap = _followSnapTiles * pxPerTile;
511
512 MoveToward(goalPx, speedPxPerSec, snap);
513 return false;
514 }
