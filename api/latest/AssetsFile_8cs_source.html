Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 public static void Validate(string path, string? password = null, bool testData = true)
51 {
52 using var archive = new ZipFile(File.OpenRead(path));
53 archive.Password = password;
54 var keys = new HashSet<AssetsFileEntry>();
55

58 if (!entry.IsFile)

64 throw new InvalidDataException($"Invalid bundle entry key: {entry.Name}");

66 if (!keys.Add(key))

68 }

245 private void LoadZip(Stream stream)
246 {
247 if (_isLoaded)
248 return;
249
250 try
251 {
252 Engine.Logger.LogInformation("Loading assets file.");
253
254 _zipFile?.Close();
255 _zipFile = null;
256 _zipEntries.Clear();
257
258 // SharpZipLib's ZipFile expects seekable input. Buffering here also gives
259 // stream-loaded GAFs the same lifetime semantics as path-loaded GAFs:
260 // the source stream is needed only for the duration of this call.
261 MemoryStream? bufferedStream = stream.CanSeek ? null : new MemoryStream();
262 Stream archiveStream = stream;
263
264 if (bufferedStream is not null)
265 {
266 stream.CopyTo(bufferedStream);
267 bufferedStream.Position = 0;
268 archiveStream = bufferedStream;
269 }
270
271 _zipFile = new ZipFile(archiveStream) { IsStreamOwner = false };
272
273 if (!string.IsNullOrEmpty(Password))
274 _zipFile.Password = Password;
275
276 // Force decryption on the first entry so a bad password fails early.
277 var testEntry = _zipFile.Cast<ZipEntry>().FirstOrDefault(e => e.IsFile);
278
279 if (testEntry != null)
280 {
281 using var testStream = _zipFile.GetInputStream(testEntry);
282 testStream.ReadByte();
283 }
284
285 foreach (ZipEntry entry in _zipFile)
286 {
287 if (!entry.IsFile)
288 continue;
289
290 // Preserve the loaded archive's encryption policy when tooling stages a replacement.
291 if (string.IsNullOrEmpty(FilePath))
292 UseEncryption |= entry.IsCrypted;
293
294 var key = AssetsFileEntry.FromString(entry.Name);
295
296 if (key == null)
297 {
298 Engine.Logger.LogWarning("Invalid entry in assets file: {EntryName}", entry.Name);
299 continue;
300 }
301
302 using var entryStream = _zipFile.GetInputStream(entry);
303 using var ms = new MemoryStream();
304 entryStream.CopyTo(ms);
305
306 _zipEntries[key] = ms.ToArray();
307 }
308
309 _isLoaded = true;
310 }
311 catch (ZipException zex)
312 {
313 Engine.Logger.LogError(zex, "Decryption failed — check password or file integrity.");
314 throw;
315 }
316 catch (Exception ex)
317 {
318 Engine.Logger.LogError(ex, "Failed to load asset file.");
319 throw;
320 }
321 finally
322 {
323 // Once contents are buffered, no need to keep the archive or source stream open.
324 _zipFile?.Close();
325 _zipFile = null;
326 }
327 }

445 public Stream? Get(AssetTypes type, string name)
446 {
447 if (string.IsNullOrWhiteSpace(name))
448 return null;
449
450 EnsureLoaded();
451
452 var exactKey = new AssetsFileEntry
453 {
454 AssetType = type,
455 AssetName = name
456 };
457
458 if (_zipEntries.TryGetValue(exactKey, out var exactData))
459 return new MemoryStream(exactData, writable: false);
460
461 var requestedBaseName = Path.GetFileNameWithoutExtension(name);
462
463 var match = _zipEntries.Keys.FirstOrDefault(k =>
464 k.AssetType == type &&
465 string.Equals(
466 Path.GetFileNameWithoutExtension(k.AssetName),
467 requestedBaseName,
468 StringComparison.OrdinalIgnoreCase));
469
470 return match != null && _zipEntries.TryGetValue(match, out var fallbackData)
471 ? new MemoryStream(fallbackData, writable: false)
472 : null;
473 }

488 public void Save()
489 {
490 EnsureLoaded();
491
492 _zipFile?.Close();
493 _zipFile = null;
494 _isLoaded = false;
495
496 using (var fs = File.Create(FilePath))
497 using (var zipStream = new ZipOutputStream(fs) { IsStreamOwner = true })
498 {
499 if (!string.IsNullOrEmpty(Password))
500 zipStream.Password = Password;
501
502 foreach (var (key, data) in _zipEntries)
503 {
504 var entry = new ZipEntry(key.ToString())
505 {
506 DateTime = DateTime.Now
507 };
508
509 if (UseEncryption && !string.IsNullOrEmpty(Password))
510 entry.AESKeySize = 256;
511
512 zipStream.PutNextEntry(entry);
513
514 using var inputStream = new MemoryStream(data, writable: false);
515 inputStream.CopyTo(zipStream);
516 zipStream.CloseEntry();
517 }
518
519 zipStream.Finish();
520 }
521
522 Engine.Logger.LogInformation(
523 "Assets file saved (Encrypted: {Encrypted}).",
524 UseEncryption);
525
526 // Keep the in-memory copy as the source of truth.
527 _isLoaded = true;
528 }
