Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

50 public static void Validate(string path, string? password = null, bool testData = true)
51 {
52 using var archive = new ZipFile(File.OpenRead(path));
53 archive.Password = password;
54 var keys = new HashSet<AssetsFileEntry>();
55 foreach (ZipEntry entry in archive)
56 {
57 if (!entry.IsFile) continue;

60 throw new InvalidDataException($"Invalid bundle entry key: {entry.Name}");
61 if (!keys.Add(key)) throw new InvalidDataException($"Duplicate bundle entry key: {entry.Name}");
62 }

124 private void LoadZip()
125 {
126 if (_isLoaded)
127 return;
128
129 try
130 {
131 Engine.Logger.LogInformation("Loading assets file.");
132
133 _zipFile?.Close();
134 _zipFile = null;
135 _zipEntries.Clear();
136
137 if (!File.Exists(FilePath))
138 {
139 _isLoaded = true;
140 return;
141 }
142
143 _zipFile = new ZipFile(File.OpenRead(FilePath));
144
145 if (!string.IsNullOrEmpty(Password))
146 _zipFile.Password = Password;
147
148 // Force decryption on the first entry so a bad password fails early.
149 var testEntry = _zipFile.Cast<ZipEntry>().FirstOrDefault(e => e.IsFile);
150 if (testEntry != null)
151 {
152 using var testStream = _zipFile.GetInputStream(testEntry);
153 testStream.ReadByte();
154 }
155
156 foreach (ZipEntry entry in _zipFile)
157 {
158 if (!entry.IsFile)
159 continue;
160
161 var key = AssetsFileEntry.FromString(entry.Name);
162 if (key == null)
163 {
164 Engine.Logger.LogWarning("Invalid entry in assets file: {EntryName}", entry.Name);
165 continue;
166 }
167
168 using var entryStream = _zipFile.GetInputStream(entry);
169 using var ms = new MemoryStream();
170 entryStream.CopyTo(ms);
171
172 _zipEntries[key] = ms.ToArray();
173 }
174
175 _isLoaded = true;
176 }
177 catch (ZipException zex)
178 {
179 Engine.Logger.LogError(zex, "Decryption failed — check password or file integrity.");
180 throw;
181 }
182 catch (Exception ex)
183 {
184 Engine.Logger.LogError(ex, "Failed to load asset file.");
185 throw;
186 }
187 finally
188 {
189 // Once contents are buffered, no need to keep the file handle open.
190 _zipFile?.Close();
191 _zipFile = null;
192 }
193 }

311 public Stream? Get(AssetTypes type, string name)
312 {
313 if (string.IsNullOrWhiteSpace(name))
314 return null;
315
316 EnsureLoaded();
317
318 var exactKey = new AssetsFileEntry
319 {
320 AssetType = type,
321 AssetName = name
322 };
323
324 if (_zipEntries.TryGetValue(exactKey, out var exactData))
325 return new MemoryStream(exactData, writable: false);
326
327 var requestedBaseName = Path.GetFileNameWithoutExtension(name);
328
329 var match = _zipEntries.Keys.FirstOrDefault(k =>
330 k.AssetType == type &&
331 string.Equals(
332 Path.GetFileNameWithoutExtension(k.AssetName),
333 requestedBaseName,
334 StringComparison.OrdinalIgnoreCase));
335
336 return match != null && _zipEntries.TryGetValue(match, out var fallbackData)
337 ? new MemoryStream(fallbackData, writable: false)
338 : null;
339 }

354 public void Save()
355 {
356 EnsureLoaded();
357
358 _zipFile?.Close();
359 _zipFile = null;
360 _isLoaded = false;
361
362 using (var fs = File.Create(FilePath))
363 using (var zipStream = new ZipOutputStream(fs) { IsStreamOwner = true })
364 {
365 if (!string.IsNullOrEmpty(Password))
366 zipStream.Password = Password;
367
368 foreach (var (key, data) in _zipEntries)
369 {
370 var entry = new ZipEntry(key.ToString())
371 {
372 DateTime = DateTime.Now
373 };
374
375 if (UseEncryption && !string.IsNullOrEmpty(Password))
376 entry.AESKeySize = 256;
377
378 zipStream.PutNextEntry(entry);
379
380 using var inputStream = new MemoryStream(data, writable: false);
381 inputStream.CopyTo(zipStream);
382 zipStream.CloseEntry();
383 }
384
385 zipStream.Finish();
386 }
387
388 Engine.Logger.LogInformation(
389 "Assets file saved (Encrypted: {Encrypted}).",
390 UseEncryption);
391
392 // Keep the in-memory copy as the source of truth.
393 _isLoaded = true;
