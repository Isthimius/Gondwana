Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 public static void Validate(string path, string? password = null, bool testData = true)
51 {
52 using var archive = new ZipFile(File.OpenRead(path));
53 archive.Password = password;
54 var keys = new HashSet<AssetsFileEntry>();
55 foreach (ZipEntry entry in archive)
56 {
57 if (!entry.IsFile) continue;

61 if (!keys.Add(key)) throw new InvalidDataException($"Duplicate bundle entry key: {entry.Name}");

147 private void LoadZip()
148 {
149 if (_isLoaded)
150 return;
151
152 try
153 {
154 Engine.Logger.LogInformation("Loading assets file.");
155
156 _zipFile?.Close();
157 _zipFile = null;
158 _zipEntries.Clear();
159
160 if (!File.Exists(FilePath))
161 {
162 _isLoaded = true;
163 return;
164 }
165
166 _zipFile = new ZipFile(File.OpenRead(FilePath));
167
168 if (!string.IsNullOrEmpty(Password))
169 _zipFile.Password = Password;
170
171 // Force decryption on the first entry so a bad password fails early.
172 var testEntry = _zipFile.Cast<ZipEntry>().FirstOrDefault(e => e.IsFile);
173 if (testEntry != null)
174 {
175 using var testStream = _zipFile.GetInputStream(testEntry);
176 testStream.ReadByte();
177 }
178
179 foreach (ZipEntry entry in _zipFile)
180 {
181 if (!entry.IsFile)
182 continue;
183
184 var key = AssetsFileEntry.FromString(entry.Name);
185 if (key == null)
186 {
187 Engine.Logger.LogWarning("Invalid entry in assets file: {EntryName}", entry.Name);
188 continue;
189 }
190
191 using var entryStream = _zipFile.GetInputStream(entry);
192 using var ms = new MemoryStream();
193 entryStream.CopyTo(ms);
194
195 _zipEntries[key] = ms.ToArray();
196 }
197
198 _isLoaded = true;
199 }
200 catch (ZipException zex)
201 {
202 Engine.Logger.LogError(zex, "Decryption failed — check password or file integrity.");
203 throw;
204 }
205 catch (Exception ex)
206 {
207 Engine.Logger.LogError(ex, "Failed to load asset file.");
208 throw;
209 }
210 finally
211 {
212 // Once contents are buffered, no need to keep the file handle open.
213 _zipFile?.Close();
214 _zipFile = null;
215 }
216 }

334 public Stream? Get(AssetTypes type, string name)
335 {
336 if (string.IsNullOrWhiteSpace(name))
337 return null;
338
339 EnsureLoaded();
340
341 var exactKey = new AssetsFileEntry
342 {
343 AssetType = type,
344 AssetName = name
345 };
346
347 if (_zipEntries.TryGetValue(exactKey, out var exactData))
348 return new MemoryStream(exactData, writable: false);
349
350 var requestedBaseName = Path.GetFileNameWithoutExtension(name);
351
352 var match = _zipEntries.Keys.FirstOrDefault(k =>
353 k.AssetType == type &&
354 string.Equals(
355 Path.GetFileNameWithoutExtension(k.AssetName),
356 requestedBaseName,
357 StringComparison.OrdinalIgnoreCase));
358
359 return match != null && _zipEntries.TryGetValue(match, out var fallbackData)
360 ? new MemoryStream(fallbackData, writable: false)
361 : null;
362 }

385 using (var fs = File.Create(FilePath))
386 using (var zipStream = new ZipOutputStream(fs) { IsStreamOwner = true })
387 {
388 if (!string.IsNullOrEmpty(Password))
389 zipStream.Password = Password;
390
391 foreach (var (key, data) in _zipEntries)
392 {
393 var entry = new ZipEntry(key.ToString())
394 {
395 DateTime = DateTime.Now
396 };
397
398 if (UseEncryption && !string.IsNullOrEmpty(Password))
399 entry.AESKeySize = 256;
400
401 zipStream.PutNextEntry(entry);
402
403 using var inputStream = new MemoryStream(data, writable: false);
404 inputStream.CopyTo(zipStream);
405 zipStream.CloseEntry();
406 }

408 zipStream.Finish();
409 }
