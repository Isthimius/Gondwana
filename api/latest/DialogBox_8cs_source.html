Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

215 private void BringVisualTreeToFront()
216 {
217 DirectDrawingBase[] visuals = GetVisualDescendants(this).ToArray();
218 if (visuals.Length == 0)
219 return;
220
221 int minimumZOrder = visuals.Min(static visual => visual.ZOrder);
222 int maximumZOrder = visuals.Max(static visual => visual.ZOrder);
223 int span = maximumZOrder - minimumZOrder;
224
225 int targetMinimum;
226 lock (_zOrderSyncRoot)
227 {
228 targetMinimum = _nextDialogZOrder;
229 _nextDialogZOrder = checked(targetMinimum + span + DialogZOrderGap);
230 }
231
232 int delta = targetMinimum - minimumZOrder;
233 foreach (DirectDrawingBase visual in visuals)
234 visual.ZOrder += delta;
235 }
236
237 private static IEnumerable<DirectDrawingBase> GetVisualDescendants(IDirectCompositeContainer container)
238 {
239 foreach (IDirectCompositeChild child in container.Children)
240 {
241 if (child is DirectDrawingBase visual)
242 yield return visual;
243
244 if (child is not IDirectCompositeContainer childContainer)
245 continue;
246
247 foreach (DirectDrawingBase descendant in GetVisualDescendants(childContainer))
248 yield return descendant;
249 }
250 }
