Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

279 private void BringVisualTreeToFront()
280 {
281 DirectDrawingBase[] visuals = GetVisualDescendants(this).ToArray();
282
283 if (visuals.Length == 0)
284 return;
285
286 int minimumZOrder = visuals.Min(static visual => visual.ZOrder);
287 int maximumZOrder = visuals.Max(static visual => visual.ZOrder);
288 int span = maximumZOrder - minimumZOrder;
289
290 int targetMinimum;
291
292 lock (_zOrderSyncRoot)
293 {
294 targetMinimum = _nextDialogZOrder;
295 _nextDialogZOrder = checked(targetMinimum + span + DialogZOrderGap);
296 }
297
298 int delta = targetMinimum - minimumZOrder;
299
300 foreach (DirectDrawingBase visual in visuals)
301 visual.ZOrder += delta;
302 }
303
304 private static IEnumerable<DirectDrawingBase> GetVisualDescendants(IDirectCompositeContainer container)
305 {
306 foreach (IDirectCompositeChild child in container.Children)
307 {
308 if (child is DirectDrawingBase visual)
309 yield return visual;
310
311 if (child is not IDirectCompositeContainer childContainer)
312 continue;
313
314 foreach (DirectDrawingBase descendant in GetVisualDescendants(childContainer))
315 yield return descendant;
316 }
317 }
