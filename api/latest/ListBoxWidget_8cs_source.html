Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

395 protected override void OnPointerClick(WidgetPointerEventArgs args)
396 {
397 base.OnPointerClick(args);
398
399 if (!args.IsPrimaryButton || _items.Count == 0)
400 return;
401
402 if (_suppressScrollBarClick || IsPointerOverScrollBar(args))
403 {
404 _suppressScrollBarClick = false;
405 args.Handled = true;
406 return;
407 }
408
409 RectangleF screenBounds = Background.GetDrawLocationScreen(args.View);
410 screenBounds.Offset(args.WrappedOffsetWorldPx.X * args.View.Viewport.Zoom, args.WrappedOffsetWorldPx.Y * args.View.Viewport.Zoom);
411 if (screenBounds.Height <= 0f || Bounds.Height <= 0)
412 return;
413
414 float scaleY = screenBounds.Height / Bounds.Height;
415 float localY = args.ScreenPositionPx.Y - screenBounds.Top - ContentPadding * scaleY;
416 float rowHeightScreen = ItemHeight * scaleY;
417
418 if (localY < 0f || rowHeightScreen <= 0f)
419 return;
420
421 int row = (int)(localY / rowHeightScreen);
422 int index = TopIndex + row;
423
424 if (row < 0 || row >= VisibleItemCount || index >= _items.Count)
425 return;
426
427 args.Handled = true;
428 SetSelectedIndex(index);
429 SelectionCommitted?.Invoke(index);
430 }

433 protected override void OnPointerDown(WidgetPointerEventArgs args)
434 {
435 base.OnPointerDown(args);
436
437 if (_scrollBarPointerId.HasValue)
438 return;
439
440 _suppressScrollBarClick = false;
441
442 if (!args.IsPrimaryButton || !IsVerticalScrollBarVisible)
443 return;
444
445 PointF local = GetLocalPointerPosition(args);
446 Rectangle thumbBounds = GetScrollBarThumbBounds(Bounds);
447 Rectangle trackBounds = GetScrollBarTrackBounds(Bounds);
448
449 if (!trackBounds.Contains(Point.Round(local)))
450 return;
451
452 args.Handled = true;
453 _scrollBarPointerId = args.PointerId;
454 _suppressScrollBarClick = true;
455 if (thumbBounds.Contains(Point.Round(local)))
456 {
457 _isDraggingScrollBarThumb = true;
458 _scrollBarDragOffset = local.Y - thumbBounds.Top;
459 return;
460 }
461
462 TopIndex += local.Y < thumbBounds.Top ? -VisibleItemCount : VisibleItemCount;
463 }

466 protected override void OnPointerMove(WidgetPointerEventArgs args)
467 {
468 base.OnPointerMove(args);
469
470 if (!_isDraggingScrollBarThumb || _scrollBarPointerId != args.PointerId || !IsVerticalScrollBarVisible)
471 return;
472
473 Rectangle trackBounds = GetScrollBarTrackBounds(Bounds);
474 Rectangle thumbBounds = GetScrollBarThumbBounds(Bounds);
475 float travel = trackBounds.Height - thumbBounds.Height;
476
477 if (travel <= 0f || GetMaximumTopIndex() == 0)
478 return;

480 float desiredTop = GetLocalPointerPosition(args).Y - _scrollBarDragOffset;
481 float fraction = Math.Clamp((desiredTop - trackBounds.Top) / travel, 0f, 1f);
482 TopIndex = (int)MathF.Round(fraction * GetMaximumTopIndex());
483 args.Handled = true;
484 }

500 protected override void OnMouseWheel(WidgetMouseWheelEventArgs args)
501 {
502 base.OnMouseWheel(args);
503

505 return;

514 protected override void OnKeyboardInput(WidgetKeyboardEventArgs args)
515 {
516 base.OnKeyboardInput(args);
517
518 if (_items.Count == 0)
519 return;
520
521 if (args.Key == 13)
522 {
523 if (args.KeyAction != KeyAction.Pressed || _selectedIndex < 0)
524 return;
525
526 args.Handled = true;
527 SelectionCommitted?.Invoke(_selectedIndex);
528 return;
529 }
530
531 if (args.KeyAction is not KeyAction.Pressed and not KeyAction.Repeated)
532 return;
533
534 switch (args.Key)
535 {
536 case 38: // Up
537 SelectPrevious();
538 break;
539 case 40: // Down
540 SelectNext();
541 break;
542 case 36: // Home
543 SetSelectedIndex(0);
544 break;
545 case 35: // End
546 SetSelectedIndex(_items.Count - 1);
547 break;
548 case 33: // Page Up
549 SetSelectedIndex(Math.Max(0, (_selectedIndex < 0 ? 0 : _selectedIndex) - VisibleItemCount));
550 break;
551 case 34: // Page Down
552 SetSelectedIndex(Math.Min(_items.Count - 1, (_selectedIndex < 0 ? 0 : _selectedIndex) + VisibleItemCount));
553 break;
554 default:
555 return;
556 }
557
558 args.Handled = true;
559 }

606 private void RefreshRows()
607 {
608 bool wasVisible = Visible;
609
610 RefreshScrollBar();
611
612 foreach (TextBlock row in _rowTextBlocks)
613 {
614 Remove(row);
615 row.Dispose();
616 }
617
618 _rowTextBlocks.Clear();
619
620 Rectangle bounds = Bounds;
621 int count = Math.Min(VisibleItemCount, Math.Max(0, _items.Count - TopIndex));
622
623 for (int rowIndex = 0; rowIndex < count; rowIndex++)
624 {
625 int itemIndex = TopIndex + rowIndex;
626 Rectangle rowBounds = GetRowBounds(bounds, rowIndex);
627 TextBlock row = CreateRowText(rowBounds, _items[itemIndex]);
628 row.ZOrder = _baseZOrder + 2;
629 Add(row);
630 _rowTextBlocks.Add(row);
631 }
632
633 RefreshSelectionHighlight();
634
635 if (!wasVisible)
636 SetIsVisible(false);
637 }
