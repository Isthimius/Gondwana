Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

132 public static SceneDefinition FromScene(Scene scene)
133 {
134 ArgumentNullException.ThrowIfNull(scene);
135
136 var definition = new SceneDefinition
137 {
138 ID = scene.ID,
139 CollisionGroups = scene.CollisionGroups.GetGroupNames().ToList(),
140 CollisionProfiles = scene.CollisionProfiles.GetProfileNames()
141 .Select(name =>
142 {
143 var profile = scene.CollisionProfiles.Get(name);
144 return new SceneCollisionProfileDefinition
145 {
146 Name = profile.Name,
147 CollisionGroup = profile.CollisionGroup,
148 CollidesWith = profile.CollidesWith.ToList(),
149 CollidesWithAll = profile.CollidesWithAll
150 };
151 })
152 .ToList(),
153 Source = SceneDefinitionSource.Generated()
154 };
155
156 foreach (var layer in scene.SceneLayers)
157 definition.Layers.Add(CreateLayerDefinition(layer));
158
159 return definition;
160 }

172 public static Scene ToScene(SceneDefinition definition)
173 {
174 ArgumentNullException.ThrowIfNull(definition);
175
176 var errors = SceneDefinitionValidator.Validate(definition);
177 if (errors.Count != 0)
178 {
179 throw new InvalidDataException(
180 "GSCN definition is invalid:" + Environment.NewLine +
181 string.Join(Environment.NewLine, errors.Select(error => "- " + error)));
182 }
183
184 var scene = new Scene();
185
186 try
187 {
188 if (!string.IsNullOrWhiteSpace(definition.ID))
189 scene.ID = definition.ID;
190
191 foreach (var group in definition.CollisionGroups ?? [])
192 scene.CollisionGroups.Define(group);
193
194 foreach (var profile in definition.CollisionProfiles ?? [])
195 {
196 scene.CollisionProfiles.Define(
197 profile.Name,
198 profile.CollisionGroup,
199 profile.CollidesWith,
200 profile.CollidesWithAll);
201 }
202
203 foreach (var layerDefinition in definition.Layers ?? [])
204 MaterializeLayer(scene, layerDefinition);
205
206 return scene;
207 }
208 catch
209 {
210 scene.Dispose();
211 throw;
212 }
213 }

226 private static SceneDefinition FromJson(string json, string? sourceDescription)
227 {
228 if (string.IsNullOrWhiteSpace(json))
229 throw new ArgumentException("GSCN JSON content must be a non-empty string.", nameof(json));
230
231 try
232 {
233 var definition = JsonConvert.DeserializeObject<SceneDefinition>(json, Settings);
234 if (definition is null)
235 {
236 var source = string.IsNullOrWhiteSpace(sourceDescription)
237 ? "GSCN JSON content"
238 : sourceDescription;
239
240 throw new InvalidDataException($"Failed to deserialize {source}. Result was null.");
241 }
242
243 definition.CollisionGroups ??= [];
244 definition.CollisionProfiles ??= [];
245 definition.Layers ??= [];
246
247 foreach (var profile in definition.CollisionProfiles)
248 {
249 if (profile is null)
250 throw new InvalidDataException("GSCN CollisionProfiles cannot contain null entries.");
251 profile.CollidesWith ??= [];
252 }
253
254 foreach (var layer in definition.Layers)
255 {
256 if (layer is null)
257 throw new InvalidDataException("GSCN Layers cannot contain null entries.");
258 layer.Tiles ??= [];
259 }
260
261 return definition;
262 }
263 catch (JsonException ex)
264 {
265 var source = string.IsNullOrWhiteSpace(sourceDescription)
266 ? "GSCN JSON content"
267 : sourceDescription;
268
269 throw new InvalidDataException($"Failed to deserialize {source}.", ex);
270 }
271 }

424 private static Frame ResolveFrame(SceneFrameDefinition definition)
425 {
426 var tilesheet = TilesheetRegistry.Instance.GetOrNull(definition.Tilesheet)
427 ?? throw new InvalidDataException(
428 $"GSCN references tilesheet '{definition.Tilesheet}', but it is not registered.");
429
430 var region = tilesheet.GetRegion(definition.RegionName)
431 ?? throw new InvalidDataException(
432 $"GSCN references region '{definition.RegionName}' on tilesheet '{definition.Tilesheet}', but it does not exist.");
433
434 if (definition.XTile < 0 ||
435 definition.YTile < 0 ||
436 definition.XTile >= region.Columns ||
437 definition.YTile >= region.Rows)
438 {
439 throw new InvalidDataException(
440 $"GSCN frame ({definition.XTile}, {definition.YTile}) is outside region '{definition.RegionName}' on tilesheet '{definition.Tilesheet}'.");
441 }
442
443 return tilesheet.GetFrame(
444 definition.RegionName,
445 definition.XTile,
446 definition.YTile);
447 }
