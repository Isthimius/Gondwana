Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

240 public void PanToOverDuration(PointF worldTopLeftPx, float durationSeconds)
241 {
242 if (durationSeconds <= 0f)
243 {
244 // Degenerate case — behave like SnapTo().
245 _panTargetUpperLeftPx = null;
246 SnapTo(worldTopLeftPx);
247 return;
248 }
249
250 // In an exponential smoothing system:
251 // pos(t) = target + (pos0 - target) * exp(-k * t)
252 //
253 // To reach ~99% of target position in T seconds:
254 // exp(-k*T) = 0.01 → k = -ln(0.01) / T
255 //
256 // This produces a visually nice, predictable time-to-arrive.
257 float k = -(float)Math.Log(0.01f) / durationSeconds;
258
259 // Cancel following and start a timed pan using your existing machinery.
260 _followAxis = 0;
261 _followLayer = null;
262 _followWorldPx = null;
263 _hardFollow = false;
264
265 _panLerpPerSecond = k;
266 _panTargetUpperLeftPx = ClampToWorldBounds(worldTopLeftPx);
267 }

504 internal void Update(float dtSeconds)
505 {
506 // 1) Explicit pan-to-UL (PanTo) takes priority over center-follow.
507 if (_panTargetUpperLeftPx is { } panTarget)
508 {
509 var clamped = ClampToWorldBounds(panTarget);
510
511 if (_hardFollow || _panLerpPerSecond <= 0f)
512 {
513 PositionPx = clamped;
514 _panTargetUpperLeftPx = null;
515 }
516 else
517 {
518 float t = 1f - (float)Math.Exp(-_panLerpPerSecond * Math.Max(0f, dtSeconds));
519 var newPos = new PointF(
520 PositionPx.X + (clamped.X - PositionPx.X) * t,
521 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
522
523 PositionPx = newPos;
524
525 // Close enough → snap and finish.
526 if (Math.Abs(newPos.X - clamped.X) < 0.5f &&
527 Math.Abs(newPos.Y - clamped.Y) < 0.5f)
528 {
529 PositionPx = clamped;
530 _panTargetUpperLeftPx = null;
531 }
532 }
533
534 // If we're actively panning, don't also run center-follow this frame.
535 if (_panTargetUpperLeftPx is not null)
536 return;
537 }
538
539 // 2) Center-follow / dead-zone logic.
540 if (_followWorldPx is null)
541 return;
542
543 var target = _followWorldPx();
544 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
545 {
546 var visible = GetVisibleWorldSizePx();
547 var center = new PointF(PositionPx.X + visible.Width / 2, PositionPx.Y + visible.Height / 2);
548 var period = layer.GetPeriod();
549 if (_followAxis == 1)
550 {
551 target = NearestWrappedByAxis(period, target, center, followX: true);
552 target.Y = center.Y;
553 }
554 else if (_followAxis == 2)
555 {
556 target = NearestWrappedByAxis(period, target, center, followX: false);
557 target.X = center.X;
558 }
559 else
560 {
561 target = period.Nearest(target, center);
562 }
563 }
564 var desiredUL = DesiredUpperLeftToContainTarget(target);
565 if (_hardFollow || FollowLerpPerSecond <= 0f)
566 {
567 PositionPx = ClampToWorldBounds(desiredUL);
568 }
569 else
570 {
571 float t = 1f - (float)Math.Exp(-FollowLerpPerSecond * Math.Max(0f, dtSeconds));
572 var clamped = ClampToWorldBounds(desiredUL);
573 PositionPx = new PointF(
574 PositionPx.X + (clamped.X - PositionPx.X) * t,
575 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
576 }
577 }

581 private PointF DesiredUpperLeftToContainTarget(PointF targetWorldPx)
582 {
583 var vis = GetVisibleWorldSizePx();
584 if (DeadZonePx == Rectangle.Empty)
585 return new PointF(targetWorldPx.X - vis.Width * 0.5f,
586 targetWorldPx.Y - vis.Height * 0.5f);
587
588 var viewWorld = new RectangleF(PositionPx.X, PositionPx.Y, vis.Width, vis.Height);
589 var dzWorld = new RectangleF(viewWorld.X + DeadZonePx.X,
590 viewWorld.Y + DeadZonePx.Y,
591 DeadZonePx.Width, DeadZonePx.Height);
592
593 if (dzWorld.Contains(targetWorldPx))
594 return PositionPx;
595
596 float newX = PositionPx.X;
597 float newY = PositionPx.Y;
598
599 if (targetWorldPx.X < dzWorld.Left)
600 newX -= dzWorld.Left - targetWorldPx.X;
601
602 if (targetWorldPx.X > dzWorld.Right)
603 newX += targetWorldPx.X - dzWorld.Right;
604
605 if (targetWorldPx.Y < dzWorld.Top)
606 newY -= dzWorld.Top - targetWorldPx.Y;
607
608 if (targetWorldPx.Y > dzWorld.Bottom)
609 newY += targetWorldPx.Y - dzWorld.Bottom;
610
611 return new PointF(newX, newY);
612 }

614 private static PointF NearestWrappedByAxis(LayerPeriod period, PointF point, PointF reference, bool followX)
615 {
616 var delta = new PointF(reference.X - point.X, reference.Y - point.Y);
617 var coefficients = period.Coefficients(delta);
618 int baseColumn = (int)Math.Round(coefficients.X);
619 int baseRow = (int)Math.Round(coefficients.Y);
620
621 int minColumn = period.WrapColumns ? baseColumn - 4 : 0;
622 int maxColumn = period.WrapColumns ? baseColumn + 4 : 0;
623 int minRow = period.WrapRows ? baseRow - 4 : 0;
624 int maxRow = period.WrapRows ? baseRow + 4 : 0;
625
626 var best = period.Nearest(point, reference);
627 float bestAxisDistance = followX ? Math.Abs(best.X - reference.X) : Math.Abs(best.Y - reference.Y);
628 float bestDistanceSquared = (best.X - reference.X) * (best.X - reference.X) + (best.Y - reference.Y) * (best.Y - reference.Y);
629
630 for (int c = minColumn; c <= maxColumn; c++)
631 for (int r = minRow; r <= maxRow; r++)
632 {
633 var offset = period.Offset(c, r);
634 var candidate = new PointF(point.X + offset.X, point.Y + offset.Y);
635 float axisDistance = followX ? Math.Abs(candidate.X - reference.X) : Math.Abs(candidate.Y - reference.Y);
636 float distanceSquared = (candidate.X - reference.X) * (candidate.X - reference.X) +
637 (candidate.Y - reference.Y) * (candidate.Y - reference.Y);
638 if (axisDistance < bestAxisDistance ||
639 (axisDistance == bestAxisDistance && distanceSquared < bestDistanceSquared))
640 {
641 best = candidate;
642 bestAxisDistance = axisDistance;
643 bestDistanceSquared = distanceSquared;
644 }
645 }
646
647 return best;
648 }

650 private PointF ClampToWorldBounds(PointF ul)
651 {
652 if (WorldBoundsPx == RectangleF.Empty)
653 return ul;
654
655 var vis = GetVisibleWorldSizePx();
656 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
657 {
658 var period = layer.GetPeriod();
659 var position = period.Coefficients(ul);
660 PointF[] Corners(RectangleF rect) => new[] {
661 period.Coefficients(new(rect.Left, rect.Top)), period.Coefficients(new(rect.Right, rect.Top)),
662 period.Coefficients(new(rect.Left, rect.Bottom)), period.Coefficients(new(rect.Right, rect.Bottom)) };
663 var bounds = Corners(WorldBoundsPx);
664 var viewport = Corners(new(0, 0, vis.Width, vis.Height));
665 float lowX = bounds.Min(p => p.X) - viewport.Min(p => p.X);
666 float lowY = bounds.Min(p => p.Y) - viewport.Min(p => p.Y);
667 float highX = Math.Max(lowX, bounds.Max(p => p.X) - viewport.Max(p => p.X));
668 float highY = Math.Max(lowY, bounds.Max(p => p.Y) - viewport.Max(p => p.Y));
669 return period.Offset(layer.WrapHorizontally ? position.X : Math.Clamp(position.X, lowX, highX),
670 layer.WrapVertically ? position.Y : Math.Clamp(position.Y, lowY, highY));
671 }
672 float minX = WorldBoundsPx.Left;
673 float minY = WorldBoundsPx.Top;
674 float maxX = WorldBoundsPx.Right - vis.Width;
675 float maxY = WorldBoundsPx.Bottom - vis.Height;
676
677 if (maxX < minX)
678 maxX = minX;
679
680 if (maxY < minY)
681 maxY = minY;
682
683 return new PointF(
684 Math.Clamp(ul.X, minX, maxX),
685 Math.Clamp(ul.Y, minY, maxY));
686 }
