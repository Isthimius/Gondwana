Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

247 public void PanToOverDuration(PointF worldTopLeftPx, float durationSeconds)
248 {
249 if (durationSeconds <= 0f)
250 {
251 // Degenerate case — behave like SnapTo().
252 _panTargetUpperLeftPx = null;
253 SnapTo(worldTopLeftPx);
254 return;
255 }
256
257 // In an exponential smoothing system:
258 // pos(t) = target + (pos0 - target) * exp(-k * t)
259 //
260 // To reach ~99% of target position in T seconds:
261 // exp(-k*T) = 0.01 → k = -ln(0.01) / T
262 //
263 // This produces a visually nice, predictable time-to-arrive.
264 float k = -(float)Math.Log(0.01f) / durationSeconds;
265
266 // Cancel following and start a timed pan using your existing machinery.
267 _followAxis = 0;
268 _followLayer = null;
269 _followWorldPx = null;
270 _hardFollow = false;
271
272 _panLerpPerSecond = k;
273 _panTargetUpperLeftPx = ClampToWorldBounds(worldTopLeftPx);
274 }

612 internal void Update(float dtSeconds)
613 {
614 // 1) Explicit pan-to-UL (PanTo) takes priority over center-follow.
615 if (_panTargetUpperLeftPx is { } panTarget)
616 {
617 var clamped = ClampToWorldBounds(panTarget);
618
619 if (_hardFollow || _panLerpPerSecond <= 0f)
620 {
621 PositionPx = clamped;
622 _panTargetUpperLeftPx = null;
623 }
624 else
625 {
626 float t = 1f - (float)Math.Exp(-_panLerpPerSecond * Math.Max(0f, dtSeconds));
627 var newPos = new PointF(
628 PositionPx.X + (clamped.X - PositionPx.X) * t,
629 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
630
631 PositionPx = newPos;
632
633 // Close enough → snap and finish.
634 if (Math.Abs(newPos.X - clamped.X) < 0.5f &&
635 Math.Abs(newPos.Y - clamped.Y) < 0.5f)
636 {
637 PositionPx = clamped;
638 _panTargetUpperLeftPx = null;
639 }
640 }
641
642 // If we're actively panning, don't also run center-follow this frame.
643 if (_panTargetUpperLeftPx is not null)
644 return;
645 }
646
647 // 2) Center-follow / dead-zone logic.
648 if (_followWorldPx is null)
649 return;
650
651 var target = _followWorldPx();
652 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
653 {
654 var visible = GetVisibleWorldSizePx();
655 var followReference = new PointF(
656 PositionPx.X + visible.Width * _followAnchor.X,
657 PositionPx.Y + visible.Height * _followAnchor.Y);
658 var period = layer.GetPeriod();
659 if (_followAxis == 1)
660 {
661 target = NearestWrappedByAxis(period, target, followReference, followX: true);
662 target.Y = followReference.Y;
663 }
664 else if (_followAxis == 2)
665 {
666 target = NearestWrappedByAxis(period, target, followReference, followX: false);
667 target.X = followReference.X;
668 }
669 else
670 {
671 target = period.Nearest(target, followReference);
672 }
673 }
674 var desiredUL = DesiredUpperLeftToContainTarget(target);
675 if (_hardFollow || FollowLerpPerSecond <= 0f)
676 {
677 PositionPx = ClampToWorldBounds(desiredUL);
678 }
679 else
680 {
681 float t = 1f - (float)Math.Exp(-FollowLerpPerSecond * Math.Max(0f, dtSeconds));
682 var clamped = ClampToWorldBounds(desiredUL);
683 PositionPx = new PointF(
684 PositionPx.X + (clamped.X - PositionPx.X) * t,
685 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
686 }
687 }

691 private PointF DesiredUpperLeftToContainTarget(PointF targetWorldPx)
692 {
693 var vis = GetVisibleWorldSizePx();
694 if (DeadZonePx == Rectangle.Empty)
695 return new PointF(
696 targetWorldPx.X - vis.Width * _followAnchor.X,
697 targetWorldPx.Y - vis.Height * _followAnchor.Y);
698
699 var viewWorld = new RectangleF(PositionPx.X, PositionPx.Y, vis.Width, vis.Height);
700 var dzWorld = new RectangleF(viewWorld.X + DeadZonePx.X,
701 viewWorld.Y + DeadZonePx.Y,
702 DeadZonePx.Width, DeadZonePx.Height);
703
704 if (dzWorld.Contains(targetWorldPx))
705 return PositionPx;
706
707 float newX = PositionPx.X;
708 float newY = PositionPx.Y;
709
710 if (targetWorldPx.X < dzWorld.Left)
711 newX -= dzWorld.Left - targetWorldPx.X;
712
713 if (targetWorldPx.X > dzWorld.Right)
714 newX += targetWorldPx.X - dzWorld.Right;
715
716 if (targetWorldPx.Y < dzWorld.Top)
717 newY -= dzWorld.Top - targetWorldPx.Y;
718
719 if (targetWorldPx.Y > dzWorld.Bottom)
720 newY += targetWorldPx.Y - dzWorld.Bottom;
721
722 return new PointF(newX, newY);
723 }

725 private static PointF NearestWrappedByAxis(LayerPeriod period, PointF point, PointF reference, bool followX)
726 {
727 var delta = new PointF(reference.X - point.X, reference.Y - point.Y);
728 var coefficients = period.Coefficients(delta);
729 int baseColumn = (int)Math.Round(coefficients.X);
730 int baseRow = (int)Math.Round(coefficients.Y);
731
732 int minColumn = period.WrapColumns ? baseColumn - 4 : 0;
733 int maxColumn = period.WrapColumns ? baseColumn + 4 : 0;
734 int minRow = period.WrapRows ? baseRow - 4 : 0;
735 int maxRow = period.WrapRows ? baseRow + 4 : 0;
736
737 var best = period.Nearest(point, reference);
738 float bestAxisDistance = followX ? Math.Abs(best.X - reference.X) : Math.Abs(best.Y - reference.Y);
739 float bestDistanceSquared = (best.X - reference.X) * (best.X - reference.X) + (best.Y - reference.Y) * (best.Y - reference.Y);
740
741 for (int c = minColumn; c <= maxColumn; c++)
742 for (int r = minRow; r <= maxRow; r++)
743 {
744 var offset = period.Offset(c, r);
745 var candidate = new PointF(point.X + offset.X, point.Y + offset.Y);
746 float axisDistance = followX ? Math.Abs(candidate.X - reference.X) : Math.Abs(candidate.Y - reference.Y);
747 float distanceSquared = (candidate.X - reference.X) * (candidate.X - reference.X) +
748 (candidate.Y - reference.Y) * (candidate.Y - reference.Y);
749 if (axisDistance < bestAxisDistance ||
750 (axisDistance == bestAxisDistance && distanceSquared < bestDistanceSquared))
751 {
752 best = candidate;
753 bestAxisDistance = axisDistance;
754 bestDistanceSquared = distanceSquared;
755 }
756 }
757
758 return best;
759 }

761 private PointF ClampToWorldBounds(PointF ul)
762 {
763 if (WorldBoundsPx == RectangleF.Empty)
764 return ul;
765
766 var vis = GetVisibleWorldSizePx();
767 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
768 {
769 var period = layer.GetPeriod();
770 var position = period.Coefficients(ul);
771 PointF[] Corners(RectangleF rect) => new[] {
772 period.Coefficients(new(rect.Left, rect.Top)), period.Coefficients(new(rect.Right, rect.Top)),
773 period.Coefficients(new(rect.Left, rect.Bottom)), period.Coefficients(new(rect.Right, rect.Bottom)) };
774 var bounds = Corners(WorldBoundsPx);
775 var viewport = Corners(new(0, 0, vis.Width, vis.Height));
776 float lowX = bounds.Min(p => p.X) - viewport.Min(p => p.X);
777 float lowY = bounds.Min(p => p.Y) - viewport.Min(p => p.Y);
778 float highX = Math.Max(lowX, bounds.Max(p => p.X) - viewport.Max(p => p.X));
779 float highY = Math.Max(lowY, bounds.Max(p => p.Y) - viewport.Max(p => p.Y));
780 return period.Offset(layer.WrapHorizontally ? position.X : Math.Clamp(position.X, lowX, highX),
781 layer.WrapVertically ? position.Y : Math.Clamp(position.Y, lowY, highY));
782 }
783 float minX = WorldBoundsPx.Left;
784 float minY = WorldBoundsPx.Top;
785 float maxX = WorldBoundsPx.Right - vis.Width;
786 float maxY = WorldBoundsPx.Bottom - vis.Height;
787
788 if (maxX < minX)
789 maxX = minX;
790
791 if (maxY < minY)
792 maxY = minY;
793
794 return new PointF(
795 Math.Clamp(ul.X, minX, maxX),
796 Math.Clamp(ul.Y, minY, maxY));
797 }
