Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

254 public void PanToOverDuration(PointF worldTopLeftPx, float durationSeconds)
255 {
256 if (durationSeconds <= 0f)
257 {
258 // Degenerate case — behave like SnapTo().
259 _panTargetUpperLeftPx = null;
260 SnapTo(worldTopLeftPx);
261 return;
262 }
263
264 // In an exponential smoothing system:
265 // pos(t) = target + (pos0 - target) * exp(-k * t)
266 //
267 // To reach ~99% of target position in T seconds:
268 // exp(-k*T) = 0.01 → k = -ln(0.01) / T
269 //
270 // This produces a visually nice, predictable time-to-arrive.
271 float k = -(float)Math.Log(0.01f) / durationSeconds;
272
273 // Cancel following and start a timed pan using your existing machinery.
274 _followAxis = 0;
275 _followLayer = null;
276 _followWorldPx = null;
277 _hardFollow = false;
278
279 _panLerpPerSecond = k;
280 _panTargetUpperLeftPx = ClampToWorldBounds(worldTopLeftPx);
281 }

660 internal void Update(float dtSeconds)
661 {
662 // 1) Explicit pan-to-UL (PanTo) takes priority over center-follow.
663 if (_panTargetUpperLeftPx is { } panTarget)
664 {
665 var clamped = ClampToWorldBounds(panTarget);
666
667 if (_hardFollow || _panLerpPerSecond <= 0f)
668 {
669 PositionPx = clamped;
670 _panTargetUpperLeftPx = null;
671 }
672 else
673 {
674 float t = 1f - (float)Math.Exp(-_panLerpPerSecond * Math.Max(0f, dtSeconds));
675
676 var newPos = new PointF(
677 PositionPx.X + (clamped.X - PositionPx.X) * t,
678 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
679
680 PositionPx = newPos;
681
682 // Close enough → snap and finish.
683 if (Math.Abs(newPos.X - clamped.X) < 0.5f &&
684 Math.Abs(newPos.Y - clamped.Y) < 0.5f)
685 {
686 PositionPx = clamped;
687 _panTargetUpperLeftPx = null;
688 }
689 }
690
691 // If we're actively panning, don't also run center-follow this frame.
692 if (_panTargetUpperLeftPx is not null)
693 return;
694 }
695
696 // 2) Center-follow / dead-zone logic.
697 if (_followWorldPx is null)
698 return;
699
700 var target = _followWorldPx();
701
702 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
703 {
704 var visible = GetVisibleWorldSizePx();
705
706 var followReference = new PointF(
707 PositionPx.X + visible.Width * _followAnchor.X,
708 PositionPx.Y + visible.Height * _followAnchor.Y);
709
710 var period = layer.GetPeriod();
711
712 if (_followAxis == 1)
713 {
714 target = NearestWrappedByAxis(
715 period,
716 target,
717 followReference,
718 followX: true);
719
720 target.Y = followReference.Y;
721 }
722 else if (_followAxis == 2)
723 {
724 target = NearestWrappedByAxis(
725 period,
726 target,
727 followReference,
728 followX: false);
729
730 target.X = followReference.X;
731 }
732 else
733 {
734 target = period.Nearest(target, followReference);
735 }
736 }
737
738 var desiredUL = DesiredUpperLeftToContainTarget(target);
739
740 if (_hardFollow || FollowLerpPerSecond <= 0f)
741 {
742 PositionPx = ClampToWorldBounds(desiredUL);
743 }
744 else
745 {
746 float t = 1f - (float)Math.Exp(-FollowLerpPerSecond * Math.Max(0f, dtSeconds));
747 var clamped = ClampToWorldBounds(desiredUL);
748
749 PositionPx = new PointF(
750 PositionPx.X + (clamped.X - PositionPx.X) * t,
751 PositionPx.Y + (clamped.Y - PositionPx.Y) * t);
752 }
753 }

757 private PointF DesiredUpperLeftToContainTarget(PointF targetWorldPx)
758 {
759 var vis = GetVisibleWorldSizePx();
760
761 if (DeadZonePx == Rectangle.Empty)
762 return new PointF(
763 targetWorldPx.X - vis.Width * _followAnchor.X,
764 targetWorldPx.Y - vis.Height * _followAnchor.Y);
765
766 var viewWorld = new RectangleF(
767 PositionPx.X,
768 PositionPx.Y,
769 vis.Width,
770 vis.Height);
771
772 var dzWorld = new RectangleF(
773 viewWorld.X + DeadZonePx.X,
774 viewWorld.Y + DeadZonePx.Y,
775 DeadZonePx.Width,
776 DeadZonePx.Height);
777
778 if (dzWorld.Contains(targetWorldPx))
779 return PositionPx;
780
781 float newX = PositionPx.X;
782 float newY = PositionPx.Y;
783
784 if (targetWorldPx.X < dzWorld.Left)
785 newX -= dzWorld.Left - targetWorldPx.X;
786
787 if (targetWorldPx.X > dzWorld.Right)
788 newX += targetWorldPx.X - dzWorld.Right;
789
790 if (targetWorldPx.Y < dzWorld.Top)
791 newY -= dzWorld.Top - targetWorldPx.Y;
792
793 if (targetWorldPx.Y > dzWorld.Bottom)
794 newY += targetWorldPx.Y - dzWorld.Bottom;
795
796 return new PointF(newX, newY);
797 }

799 private static PointF NearestWrappedByAxis(
800 LayerPeriod period,
801 PointF point,
802 PointF reference,
803 bool followX)
804 {
805 var delta = new PointF(reference.X - point.X, reference.Y - point.Y);
806 var coefficients = period.Coefficients(delta);
807 int baseColumn = (int)Math.Round(coefficients.X);
808 int baseRow = (int)Math.Round(coefficients.Y);
809
810 int minColumn = period.WrapColumns ? baseColumn - 4 : 0;
811 int maxColumn = period.WrapColumns ? baseColumn + 4 : 0;
812 int minRow = period.WrapRows ? baseRow - 4 : 0;
813 int maxRow = period.WrapRows ? baseRow + 4 : 0;
814
815 var best = period.Nearest(point, reference);
816 float bestAxisDistance = followX ? Math.Abs(best.X - reference.X) : Math.Abs(best.Y - reference.Y);
817
818 float bestDistanceSquared = (best.X - reference.X) * (best.X - reference.X) +
819 (best.Y - reference.Y) * (best.Y - reference.Y);
820
821 for (int c = minColumn; c <= maxColumn; c++)
822 for (int r = minRow; r <= maxRow; r++)
823 {
824 var offset = period.Offset(c, r);
825 var candidate = new PointF(point.X + offset.X, point.Y + offset.Y);
826 float axisDistance = followX ? Math.Abs(candidate.X - reference.X) : Math.Abs(candidate.Y - reference.Y);
827
828 float distanceSquared = (candidate.X - reference.X) * (candidate.X - reference.X) +
829 (candidate.Y - reference.Y) * (candidate.Y - reference.Y);
830
831 if (axisDistance < bestAxisDistance ||
832 (axisDistance == bestAxisDistance && distanceSquared < bestDistanceSquared))
833 {
834 best = candidate;
835 bestAxisDistance = axisDistance;
836 bestDistanceSquared = distanceSquared;
837 }
838 }
839
840 return best;
841 }

843 private PointF ClampToWorldBounds(PointF ul)
844 {
845 if (WorldBoundsPx == RectangleF.Empty)
846 return ul;
847
848 var vis = GetVisibleWorldSizePx();
849
850 if (TopologyLayer is { } layer && (layer.WrapHorizontally || layer.WrapVertically))
851 {
852 var period = layer.GetPeriod();
853 var position = period.Coefficients(ul);
854
855 PointF[] Corners(RectangleF rect) => new[]
856 {
857 period.Coefficients(new(rect.Left, rect.Top)), period.Coefficients(new(rect.Right, rect.Top)),
858 period.Coefficients(new(rect.Left, rect.Bottom)), period.Coefficients(new(rect.Right, rect.Bottom))
859 };
860
861 var bounds = Corners(WorldBoundsPx);
862
863 var viewport = Corners(
864 new(
865 0,
866 0,
867 vis.Width,
868 vis.Height));
869
870 float lowX = bounds.Min(p => p.X) - viewport.Min(p => p.X);
871 float lowY = bounds.Min(p => p.Y) - viewport.Min(p => p.Y);
872 float highX = Math.Max(lowX, bounds.Max(p => p.X) - viewport.Max(p => p.X));
873 float highY = Math.Max(lowY, bounds.Max(p => p.Y) - viewport.Max(p => p.Y));
874
875 return period.Offset(
876 layer.WrapHorizontally ? position.X : Math.Clamp(position.X, lowX, highX),
877 layer.WrapVertically ? position.Y : Math.Clamp(position.Y, lowY, highY));
878 }
879
880 float minX = WorldBoundsPx.Left;
881 float minY = WorldBoundsPx.Top;
882 float maxX = WorldBoundsPx.Right - vis.Width;
883 float maxY = WorldBoundsPx.Bottom - vis.Height;
884
885 if (maxX < minX)
886 maxX = minX;
887
888 if (maxY < minY)
889 maxY = minY;
890
891 return new PointF(
892 Math.Clamp(ul.X, minX, maxX),
893 Math.Clamp(ul.Y, minY, maxY));
894 }
