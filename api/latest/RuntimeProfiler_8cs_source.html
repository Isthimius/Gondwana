Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

152 internal bool Define(TelemetrySource source, string name, TelemetryAvailability availability, TelemetryMetricKind kind)
153 {

155 lock (_sync)
156 {
157 if (!_sources.ContainsKey(source.Id)) return false;
158 if (source.Metrics.TryGetValue(name, out var existing))

175 internal void Record(TelemetrySource source, long generation, string metric, double value)
176 {
177 if (generation == 0 || !IsCollecting || !double.IsFinite(value)) return;
178 lock (_sync)
179 {
180 if (_requests == 0 || generation != _generation || !_sources.ContainsKey(source.Id)) return;
181 Advance();
182 if (!source.Metrics.TryGetValue(metric, out var accumulator)) return;
183 if (accumulator.Availability is TelemetryAvailability.Unsupported or TelemetryAvailability.NotApplicable) return;

221 private void Advance()
222 {
223 if (_requests == 0) return;
224 double now = _clock();
225 if (now - _started >= _options.Interval.TotalSeconds) Publish(now);
226 }
227
228 private void Publish(double now)
229 {
230 var sources = _sources.Values.Select(source => new TelemetrySourceSnapshot(source.Id, source.Name, source.Backend,
231 new ReadOnlyDictionary<string, TelemetrySummary>(source.Metrics.ToDictionary(p => p.Key, p => p.Value.Snapshot())))).ToArray();
232 if (_history.Count == _options.HistoryCapacity) _history.Dequeue();
233 _latestSnapshot = new(_generation, _started, now, Array.AsReadOnly(sources), _truncated);
234 _history.Enqueue(_latestSnapshot);
235 ClearAccumulators();
236 _started = now;
237 }

240 {
241 lock (_sync)
242 {

244 if (_requests == 1)

246 double now = _clock();
247 if (now > _started) Publish(now);

249 }

274 {
275 _min = _count == 0 ? value : Math.Min(_min, value);
276 _max = _count == 0 ? value : Math.Max(_max, value);
277 _count++;
278 _sum += value;

291 Kind = Kind,
