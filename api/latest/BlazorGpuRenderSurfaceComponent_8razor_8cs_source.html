Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

105 private void HandlePaintSurface(SKPaintGLSurfaceEventArgs e)
106 {
107 if (_disposed)
108 return;
109
110 var width = e.Info.Width;
111 var height = e.Info.Height;
112 Adapter.BeginPaint(width, height);
113
114 if (width <= 0 || height <= 0 || e.Surface.Context is not GRContext grContext)
115 {
116 e.Surface.Canvas.Clear(SKColors.Black);
117 return;
118 }
119
120 var backbuffer = (GpuBackbuffer)Host.Backbuffer;
121 if (!ReferenceEquals(_grContext, grContext)
122 || backbuffer.Width != width
123 || backbuffer.Height != height)
124 {
125 _grContext = grContext;
126 backbuffer.Initialize(grContext, width, height);
127 _backbufferNeedsRender = true;
128 }
129
130 var engine = Engine.Instance;
131 if (!engine.IsRunning)
132 {
133 e.Surface.Canvas.Clear(backbuffer.ClearColor);
134 return;
135 }
136
137 // SKGLView owns the browser rAF for the GPU path. Advance simulation first so a
138 // foreground frame request, when due, can be rendered in this same paint callback.
139 engine.Tick();
140
141 bool engineFrameRequested = Adapter.ConsumeFrameRequest();
142 bool renderScene = _backbufferNeedsRender || engineFrameRequested;
143
144 bool presented = renderScene
145 ? Host.GlRenderToCanvas(e.Surface.Canvas)
146 : Host.GlDrawCurrentFrameToCanvas(e.Surface.Canvas);
147
148 if (!presented)
149 {
150 e.Surface.Canvas.Clear(backbuffer.ClearColor);
151 return;
152 }
153
154 if (renderScene)
155 _backbufferNeedsRender = false;
156
157 // Count successful WebGL paint/presentation callbacks. This intentionally measures the
158 // browser GPU presentation cadence, which may differ from the engine's TargetFPS.
159 backbuffer.RecordFrame();
160 }
