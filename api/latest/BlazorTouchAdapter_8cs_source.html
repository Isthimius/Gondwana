Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

69 private void OnTouchStart(BrowserTouchEventArgs e)
70 {
71 if (_isDisposed)
72 return;
73
74 foreach (var t in e.ChangedTouches)
75 {
76 var point = new GondwanaTouchPoint((int)t.Identifier, GetPosition(t), TouchPhase.Began);
77 _activeTouches[t.Identifier] = point;
78 _pendingBegins.Enqueue(point);
79 }
80

82 }

84 private void OnTouchMove(BrowserTouchEventArgs e)
85 {
86 if (_isDisposed)
87 return;
88
89 foreach (var t in e.ChangedTouches)
90 {
91 if (!_activeTouches.ContainsKey(t.Identifier))
92 continue;
93
94 var point = new GondwanaTouchPoint((int)t.Identifier, GetPosition(t), TouchPhase.Moved);
95 _activeTouches[t.Identifier] = point;
96 }
97
98 RebuildSnapshot();
99 }

101 private void OnTouchEnd(BrowserTouchEventArgs e)
102 {
103 if (_isDisposed)
104 return;
105
106 foreach (var t in e.ChangedTouches)
107 {
108 if (!_activeTouches.ContainsKey(t.Identifier))
109 continue;
110
111 var point = new GondwanaTouchPoint((int)t.Identifier, GetPosition(t), TouchPhase.Ended);
112 _activeTouches.Remove(t.Identifier);
113 _pendingEnds.Enqueue(point);
114 }

117 }

119 private void OnTouchCancel(BrowserTouchEventArgs e)
120 {
121 if (_isDisposed)
122 return;
123
124 foreach (var t in e.ChangedTouches)
125 {
126 if (!_activeTouches.TryGetValue(t.Identifier, out var existing))
127 continue;
128
129 var point = new GondwanaTouchPoint(existing.Id, existing.Position, TouchPhase.Cancelled);
130 _activeTouches.Remove(t.Identifier);
131 _pendingEnds.Enqueue(point);
132 }
133
134 RebuildSnapshot();
135 }

138 public IReadOnlyList<GondwanaTouchPoint> ConsumeBeganTouches()
139 {
140 if (_pendingBegins.IsEmpty)
141 return Array.Empty<GondwanaTouchPoint>();
142
143 var snapshot = new List<GondwanaTouchPoint>(_pendingBegins.Count);
144
145 while (_pendingBegins.TryDequeue(out var point))
146 snapshot.Add(point);
147
148 return snapshot;
149 }
