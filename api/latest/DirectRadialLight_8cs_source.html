Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

336 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
337 {
338 if (destRectScreen.Width <= 0f || destRectScreen.Height <= 0f)
339 return;
340
341 var canvas = backbuffer.Canvas;
342 var worldBounds = WorldBounds;
343
344 if (worldBounds.Width <= 0 || worldBounds.Height <= 0)
345 return;
346
347 var rect = new SKRect(destRectScreen.Left, destRectScreen.Top, destRectScreen.Right, destRectScreen.Bottom);
348 float scaleX = destRectScreen.Width / worldBounds.Width;
349 float scaleY = destRectScreen.Height / worldBounds.Height;
350
351 // Use the actual world-space center/radius rather than the integer-rounded
352 // WorldBounds midpoint. The bounds are intentionally pixel-aligned for dirty
353 // rectangles, but the shader geometry should preserve sub-pixel light movement
354 // so a moving torch does not pop by ~0.5px as its bounds floor/ceil change.
355 var center = new SKPoint(
356 destRectScreen.Left + (_centerWorldPx.X - worldBounds.Left) * scaleX,
357 destRectScreen.Top + (_centerWorldPx.Y - worldBounds.Top) * scaleY);
358
359 var radius = Math.Max(Math.Abs(_radiusWorldPx * scaleX), Math.Abs(_radiusWorldPx * scaleY));
360
361 if (radius <= 0f)
362 return;
363
364 var hotspot = Math.Clamp(_hotspotRadiusRatio, 0f, 0.95f);
365 var midpoint = Math.Clamp(_midpointRadiusRatio, hotspot + 0.001f, 0.999f);
366 var effectiveIntensity = EffectiveIntensity;
367
368 var inner = ToSkColor(_lightColor, effectiveIntensity);
369 var middle = ToSkColor(_lightColor, effectiveIntensity * _midpointIntensityRatio);
370 var outer = ToSkColor(_lightColor, 0f);
371
372 using var shader = SKShader.CreateRadialGradient(
373 center,
374 radius,
375 new[] { inner, inner, middle, outer },
376 new[] { 0f, hotspot, midpoint, 1f },
377 SKShaderTileMode.Clamp);
378
379 using var paint = new SKPaint
380 {
381 Shader = shader,
382 Style = SKPaintStyle.Fill,
383 BlendMode = _blendMode,
384 IsAntialias = _isAntialias
385 };
386
387 canvas.DrawRect(rect, paint);
388 }

390 private void UpdateFlicker(long tick)
391 {
392 if (!_flickerEnabled)
393 return;
394
395 if (_lastFlickerRefreshTick == 0)
396 {
397 _lastFlickerRefreshTick = tick;
398 return;
399 }
400
401 if (tick <= _lastFlickerRefreshTick)
402 return;
403
404 var dt = HighResTimer.GetDuration(_lastFlickerRefreshTick, tick);
405 var refreshIntervalSec = 1f / _flickerRefreshHz;
406
407 if (dt < refreshIntervalSec)
408 return;
409
410 _flickerPhaseSec += dt;
411
412 var primary = 0.5f + 0.5f * MathF.Sin(_flickerPhaseSec * MathF.Tau * _flickerRefreshHz);
413 var secondary = 0.5f + 0.5f * MathF.Sin(_flickerPhaseSec * MathF.Tau * (_flickerRefreshHz * 0.37f + 0.73f));
414 var mixed = primary * 0.65f + secondary * 0.35f;
415
416 _flickerMultiplier = 1f - _flickerAmount + mixed * _flickerAmount * 2f;
417 _lastFlickerRefreshTick = tick;
418
419 ForceRefresh();
420 OnChanged();
421 }

434 private static Rectangle BoundsFromCenter(PointF centerWorldPx, float radiusWorldPx)
435 {
436 var radius = Math.Max(1f, radiusWorldPx);
437 var left = (int)MathF.Floor(centerWorldPx.X - radius);
438 var top = (int)MathF.Floor(centerWorldPx.Y - radius);
439 var right = (int)MathF.Ceiling(centerWorldPx.X + radius);
440 var bottom = (int)MathF.Ceiling(centerWorldPx.Y + radius);
441
442 return Rectangle.FromLTRB(left, top, right, bottom);
443 }
