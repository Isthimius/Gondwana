Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

176 private void DrawCurrentSurface(SKCanvas canvas)
177 {
178 var presentation = RenderSurfaceAdapter?.Presentation ?? default;
179 canvas.Clear(Backbuffer.ClearColor);
180 if (presentation.Scale <= 0) return;
181 canvas.Save();
182 try
183 {
184 canvas.Translate(presentation.DestinationRect.Left, presentation.DestinationRect.Top);
185 canvas.Scale(presentation.Scale);
186 bool needsLinearSampling = Engine.Instance.Configuration.RenderScalingFilter == RenderScalingFilter.Linear &&
187 (presentation.Scale != 1f ||
188 presentation.DestinationRect.Left != MathF.Floor(presentation.DestinationRect.Left) ||
189 presentation.DestinationRect.Top != MathF.Floor(presentation.DestinationRect.Top));
190 if (needsLinearSampling)
191 {
192 // Skia's DrawSurface does not expose sampling and ignores SKPaint.FilterQuality.
193 // A scoped snapshot is a GPU texture view, not a readback or an intermediate copy.
194 using var image = Backbuffer.Snapshot();
195 canvas.DrawImage(image, 0, 0, RenderSurfaceAdapterBase.PresentationSampling);
196 }
197 else
198 {
199 canvas.DrawSurface(Backbuffer.Canvas.Surface, 0, 0);
200 }
201 }
202 finally { canvas.Restore(); }
203 }
