Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

108 public SKImage? GlRenderAndSnapshot()
109 {
110 if (!Backbuffer.IsGlThreadRendered)
111 return null;
112
113 bool collectSynchronizationDiagnostics =
114 GpuRenderSynchronizationDiagnosticsCalculated is not null;
115 long waitStarted = collectSynchronizationDiagnostics
116 ? HighResTimer.GetCurrentTick()
117 : 0;
118 long lockAcquired = 0;
119 long lockReleased = 0;
120 SKImage? image;
121
122 RenderStateSynchronization.EnterGpuRender();
123 try
124 {
125 if (collectSynchronizationDiagnostics)
126 lockAcquired = HighResTimer.GetCurrentTick();
127
128 var tick = HighResTimer.GetCurrentTick();
129
130 RenderToBackbuffer(tick);
131 Backbuffer.EndFrame();
132
133 image = Backbuffer.Snapshot();
134
135 Backbuffer.BeginFrame();
136
137 if (collectSynchronizationDiagnostics)
138 lockReleased = HighResTimer.GetCurrentTick();
139 }
140 finally
141 {
142 RenderStateSynchronization.ExitGpuRender();
143 }
144
145 if (collectSynchronizationDiagnostics)
146 {
147 GpuRenderSynchronizationDiagnosticsCalculated?.Invoke(
148 new GpuRenderSynchronizationDiagnostics(
149 HighResTimer.GetDuration(waitStarted, lockAcquired) * 1000d,
150 HighResTimer.GetDuration(lockAcquired, lockReleased) * 1000d));
151 }
152
153 return image;
154 }

221 private void DrawCurrentSurface(SKCanvas canvas)
222 {
223 var presentation = RenderSurfaceAdapter?.Presentation ?? default;
224 canvas.Clear(Backbuffer.ClearColor);
225 if (presentation.Scale <= 0) return;
226 canvas.Save();
227 try
228 {
229 canvas.Translate(presentation.DestinationRect.Left, presentation.DestinationRect.Top);
230 canvas.Scale(presentation.Scale);
231 bool needsLinearSampling = Engine.Instance.Configuration.RenderScalingFilter == RenderScalingFilter.Linear &&
232 (presentation.Scale != 1f ||
233 presentation.DestinationRect.Left != MathF.Floor(presentation.DestinationRect.Left) ||
234 presentation.DestinationRect.Top != MathF.Floor(presentation.DestinationRect.Top));
235 if (needsLinearSampling)
236 {
237 // Skia's DrawSurface does not expose sampling and ignores SKPaint.FilterQuality.
238 // A scoped snapshot is a GPU texture view, not a readback or an intermediate copy.
239 using var image = Backbuffer.Snapshot();
240 canvas.DrawImage(image, 0, 0, RenderSurfaceAdapterBase.PresentationSampling);
241 }
242 else
243 {
244 canvas.DrawSurface(Backbuffer.Canvas.Surface, 0, 0);
245 }
246 }
247 finally { canvas.Restore(); }
248 }
