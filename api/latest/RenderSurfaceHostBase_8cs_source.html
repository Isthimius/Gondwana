Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

114 public SKImage? GlRenderAndSnapshot()
115 {
116 if (!Backbuffer.IsGlThreadRendered)
117 return null;
118
119 if (UsesRenderFrameSnapshots)
120 return ReplayRenderFrameSnapshot();
121
122 bool collectSynchronizationDiagnostics =
123 GpuRenderSynchronizationDiagnosticsCalculated is not null;
124 long waitStarted = collectSynchronizationDiagnostics
125 ? HighResTimer.GetCurrentTick()
126 : 0;
127 long lockAcquired = 0;
128 long lockReleased = 0;
129 SKImage? image;
130
131 RenderStateSynchronization.EnterGpuRender();
132 try
133 {
134 if (collectSynchronizationDiagnostics)
135 lockAcquired = HighResTimer.GetCurrentTick();
136
137 var tick = HighResTimer.GetCurrentTick();
138
139 RenderToBackbuffer(tick);
140 Backbuffer.EndFrame();
141
142 image = Backbuffer.Snapshot();
143
144 Backbuffer.BeginFrame();
145
146 if (collectSynchronizationDiagnostics)
147 lockReleased = HighResTimer.GetCurrentTick();
148 }
149 finally
150 {
151 RenderStateSynchronization.ExitGpuRender();
152 }
153
154 if (collectSynchronizationDiagnostics)
155 {
156 GpuRenderSynchronizationDiagnosticsCalculated?.Invoke(
157 new GpuRenderSynchronizationDiagnostics(
158 HighResTimer.GetDuration(waitStarted, lockAcquired) * 1000d,
159 HighResTimer.GetDuration(lockAcquired, lockReleased) * 1000d));
160 }
161
162 return image;
163 }

165 private SKImage? ReplayRenderFrameSnapshot()
166 {
167 var diagnostics = GpuRenderSynchronizationDiagnosticsCalculated;
168 long telemetryGeneration = Telemetry?.BeginSample() ?? 0;
169 bool collectDiagnostics = diagnostics is not null || telemetryGeneration != 0;
170 var slot = FrameMailbox.TryAcquire();
171 if (slot is null) return Backbuffer.Snapshot();
172
173 long acquired = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
174 var counters = !collectDiagnostics ? default : FrameMailbox.Counters;
175 long replayEnd = acquired;
176 bool replayCompleted = false;
177 long pictureReplayTicks = 0;
178 long backbufferFlushTicks = 0;
179 long snapshotTicks = 0;
180 int commands = !collectDiagnostics ? 0 : slot.Frame!.CommandCount;
181 double age = !collectDiagnostics ? 0 :
182 HighResTimer.GetDuration(slot.Frame!.ProducedTick, acquired) * 1000d;
183
184 try
185 {
186 var frame = slot.Frame!;
187
188 // A logical resize invalidates geometry, but context/MSAA recreation does
189 // not: recordings contain CPU resources and are uploaded by the current GL context.
190 if (frame.Width != Backbuffer.Width || frame.Height != Backbuffer.Height)
191 {
192 long snapshotStart = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
193 var resizedSnapshot = Backbuffer.Snapshot();
194 if (collectDiagnostics)
195 snapshotTicks = HighResTimer.GetCurrentTick() - snapshotStart;
196 return resizedSnapshot;
197 }
198
199 Backbuffer.BeginFrame();
200
201 long pictureStart = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
202 frame.Replay(Backbuffer.Canvas);
203 long pictureEnd = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
204
205 Backbuffer.EndFrame();
206 long flushEnd = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
207 replayCompleted = true;
208
209 if (collectDiagnostics)
210 {
211 pictureReplayTicks = pictureEnd - pictureStart;
212 backbufferFlushTicks = flushEnd - pictureEnd;
213 replayEnd = flushEnd;
214 }
215
216 long snapshotStartTick = !collectDiagnostics ? 0 : HighResTimer.GetCurrentTick();
217 var snapshot = Backbuffer.Snapshot();
218 if (collectDiagnostics)
219 snapshotTicks = HighResTimer.GetCurrentTick() - snapshotStartTick;
220
221 return snapshot;
222 }
223 finally
224 {
225 try { Backbuffer.BeginFrame(); }
226 finally { FrameMailbox.Release(slot); }
227
228 if (collectDiagnostics)
229 {
230 double replayMs = HighResTimer.GetDuration(acquired, replayEnd) * 1000d;
231 double pictureMs = HighResTimer.GetDuration(0, pictureReplayTicks) * 1000d;
232 double flushMs = HighResTimer.GetDuration(0, backbufferFlushTicks) * 1000d;
233 double snapshotMs = HighResTimer.GetDuration(0, snapshotTicks) * 1000d;
234 if (telemetryGeneration != 0)
235 {
236 if (replayCompleted)
237 {
238 Telemetry!.Record(telemetryGeneration, "replay.cpu.ms", replayMs);
239 Telemetry.Record(telemetryGeneration, "picture.cpu.ms", pictureMs);
240 Telemetry.Record(telemetryGeneration, "backbuffer.flush.cpu.ms", flushMs);
241 }
242
243 Telemetry!.Record(telemetryGeneration, "snapshot.cpu.ms", snapshotMs);
244 Telemetry.Record(telemetryGeneration, "snapshot.age.ms", age);
245 Telemetry.Record(telemetryGeneration, "mailbox.published.lifetime", counters.Published);
246 Telemetry.Record(telemetryGeneration, "mailbox.dropped.lifetime", counters.Dropped);
247 Telemetry.Record(telemetryGeneration, "mailbox.slots", counters.InUse);
248 Telemetry.Record(telemetryGeneration, "snapshot.commands.approximate", commands);
249 }
250 diagnostics?.Invoke(new(0, 0)
251 {
252 ReplayMilliseconds = replayMs,
253 PictureReplayMilliseconds = pictureMs,
254 BackbufferFlushMilliseconds = flushMs,
255 SnapshotMilliseconds = snapshotMs,
256 SnapshotAgeMilliseconds = age,
257 PublishedSnapshots = counters.Published,
258 DroppedSnapshots = counters.Dropped,
259 SnapshotSlotsInUse = counters.InUse,
260 SnapshotCommandCount = commands
261 });
262 }
263 }
264 }

338 private void DrawCurrentSurface(SKCanvas canvas)
339 {
340 var presentation = RenderSurfaceAdapter?.Presentation ?? default;
341 canvas.Clear(Backbuffer.ClearColor);
342 if (presentation.Scale <= 0) return;
343 canvas.Save();
344 try
345 {
346 canvas.Translate(presentation.DestinationRect.Left, presentation.DestinationRect.Top);
347 canvas.Scale(presentation.Scale);
348 bool needsLinearSampling = Engine.Instance.Configuration.RenderScalingFilter == RenderScalingFilter.Linear &&
349 (presentation.Scale != 1f ||
350 presentation.DestinationRect.Left != MathF.Floor(presentation.DestinationRect.Left) ||
351 presentation.DestinationRect.Top != MathF.Floor(presentation.DestinationRect.Top));
352 if (needsLinearSampling)
353 {
354 // Skia's DrawSurface does not expose sampling and ignores SKPaint.FilterQuality.
355 // A scoped snapshot is a GPU texture view, not a readback or an intermediate copy.
356 using var image = Backbuffer.Snapshot();
357 canvas.DrawImage(image, 0, 0, RenderSurfaceAdapterBase.PresentationSampling);
358 }
359 else
360 {
361 canvas.DrawSurface(Backbuffer.Canvas.Surface, 0, 0);
362 }
363 }
364 finally { canvas.Restore(); }
365 }
