Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

182 private void DrawCurrentSurface(SKCanvas canvas)
183 {
184 var presentation = RenderSurfaceAdapter?.Presentation ?? default;
185 canvas.Clear(Backbuffer.ClearColor);
186 if (presentation.Scale <= 0) return;
187 canvas.Save();
188 try
189 {
190 canvas.Translate(presentation.DestinationRect.Left, presentation.DestinationRect.Top);
191 canvas.Scale(presentation.Scale);
192 bool needsLinearSampling = Engine.Instance.Configuration.RenderScalingFilter == RenderScalingFilter.Linear &&
193 (presentation.Scale != 1f ||
194 presentation.DestinationRect.Left != MathF.Floor(presentation.DestinationRect.Left) ||
195 presentation.DestinationRect.Top != MathF.Floor(presentation.DestinationRect.Top));
196 if (needsLinearSampling)
197 {
198 // Skia's DrawSurface does not expose sampling and ignores SKPaint.FilterQuality.
199 // A scoped snapshot is a GPU texture view, not a readback or an intermediate copy.
200 using var image = Backbuffer.Snapshot();
201 canvas.DrawImage(image, 0, 0, RenderSurfaceAdapterBase.PresentationSampling);
202 }
203 else
204 {
205 canvas.DrawSurface(Backbuffer.Canvas.Surface, 0, 0);
206 }
207 }
208 finally { canvas.Restore(); }
209 }
