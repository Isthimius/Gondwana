463 public TabControlWidget MoveTab(
464 int fromIndex,
465 int toIndex)
466 {
467 ObjectDisposedException.ThrowIf(_disposed, this);
468
469 if (fromIndex < 0 || fromIndex >= _tabs.Count)
470 throw new ArgumentOutOfRangeException(nameof(fromIndex));
471
472 if (toIndex < 0 || toIndex >= _tabs.Count)
473 throw new ArgumentOutOfRangeException(nameof(toIndex));
474
475 if (fromIndex == toIndex)
476 return this;
477
478 TabPageWidget page = _tabs[fromIndex];
479 TabPageWidget? selectedPage = SelectedTab;
480
481 _tabs.RemoveAt(fromIndex);
482 _tabs.Insert(toIndex, page);
483 _selectedIndex = selectedPage is null ? -1 : _tabs.IndexOf(selectedPage);
484
485 Relayout();
486 TabReordered?.Invoke(page, fromIndex, toIndex);
487
488 return this;
489 }

515 public TabControlWidget Relayout()
516 {
517 ObjectDisposedException.ThrowIf(_disposed, this);
518
519 Rectangle bounds = Bounds;
520 (_displayedRowCount, _displayedColumnCount) = ResolveGrid(bounds);
521 Rectangle contentBounds = CreateContentBounds(bounds, _displayedRowCount);
522
523 for (int index = 0; index < _tabs.Count; index++)
524 {
525 TabPageWidget page = _tabs[index];
526 TabHeaderWidget header = _headers[page];
527 Rectangle headerBounds = CreateHeaderBounds(bounds, index, _displayedColumnCount);
528
529 header.SetBounds(headerBounds);
530 page.HeaderBounds = headerBounds;
531 page.SetContentBounds(contentBounds);
532
533 SetLocalOffset(header, new Vector2(headerBounds.Left - bounds.Left, headerBounds.Top - bounds.Top));
534 SetLocalOffset(page, new Vector2(contentBounds.Left - bounds.Left, contentBounds.Top - bounds.Top));
535 }
536
537 ApplyZOrder();
538 return this;
539 }

663 private bool RemoveTabCore(
664 TabPageWidget page,
665 bool dispose,
666 bool raiseSelectionChanged)
667 {
668 int removedIndex = _tabs.IndexOf(page);
669
670 if (removedIndex < 0)
671 return false;
672
673 TabPageWidget? previouslySelected = SelectedTab;
674 TabHeaderWidget header = _headers[page];
675
676 page.TitleChanged -= OnPageTitleChanged;
677 page.Disposing -= OnPageDisposing;
678 header.Invoked -= OnHeaderInvoked;
679 header.ReorderDragged -= OnHeaderDragged;
680 header.ReorderDragEnded -= OnHeaderDragEnded;
681 header.Disposing -= OnHeaderDisposing;
682
683 _tabs.RemoveAt(removedIndex);
684 _headers.Remove(page);
685
686 Remove(header);
687 Remove(page);
688
689 header.Dispose();
690
691 if (dispose)
692 page.Dispose();
693
694 if (_tabs.Count == 0)
695 {
696 _selectedIndex = -1;
697 }
698 else if (ReferenceEquals(previouslySelected, page))
699 {
700 _selectedIndex = Math.Min(removedIndex, _tabs.Count - 1);
701 }
702 else
703 {
704 _selectedIndex = previouslySelected is null ? 0 : _tabs.IndexOf(previouslySelected);
705 }
706
707 Relayout();
708 UpdateSelectedTabVisuals();
709 UpdatePageVisibility();
710
711 if (raiseSelectionChanged && SelectedTab is not null && !ReferenceEquals(previouslySelected, SelectedTab))
712 SelectedTabChanged?.Invoke(SelectedTab);
713
714 return true;
715 }

856 private (int Rows, int Columns) ResolveGrid(Rectangle bounds)
857 {
858 if (!AutoExpandRows)
859 {
860 int columns = Math.Max(1, (int)Math.Ceiling(_tabs.Count / (double)TabRowCount));
861 return (TabRowCount, columns);
862 }
863
864 int autoColumns = _tabs.Count == 0
865 ? 1
866 : Math.Min(_tabs.Count, Math.Max(1, bounds.Width / PreferredTabWidth));
867
868 int autoRows = Math.Max(1, (int)Math.Ceiling(_tabs.Count / (double)autoColumns));
869
870 return (autoRows, autoColumns);
871 }

873 private Rectangle CreateHeaderBounds(
874 Rectangle bounds,
875 int index,
876 int columns)
877 {
878 int row = index / columns;
879 int column = index % columns;
880 int left = bounds.Left + column * bounds.Width / columns;
881 int right = bounds.Left + (column + 1) * bounds.Width / columns;
882
883 return Rectangle.FromLTRB(
884 left,
885 bounds.Top + row * TabHeight,
886 right,
887 bounds.Top + (row + 1) * TabHeight);
