Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

92 public IReadOnlyList<TouchPoint> ConsumeBeganTouches()
93 {
94 lock (_sync)
95 {
96 var beginnings = new List<TouchPoint>();
97 var endings = new List<TouchPoint>();
98 bool reachedEnding = false;
99
100 // The original poller emits all beginnings before all endings.
101 // Stop before the next beginning after any ending, so separate
102 // taps never look like simultaneous fingers. This also prevents
103 // a reused native ID being deduplicated within a single tick.
104 while (_pending.Count > 0)
105 {
106 TouchPoint next = _pending.Peek();
107
108 if (reachedEnding && next.Phase == TouchPhase.Began)
109 break;
110
111 _pending.Dequeue();
112
113 switch (next.Phase)
114 {
115 case TouchPhase.Began:
116 beginnings.Add(next);
117 _reportedActive[next.Id] = next;
118 break;
119 case TouchPhase.Ended:
120 case TouchPhase.Cancelled:
121 endings.Add(next);
122 _reportedActive.Remove(next.Id);
123 reachedEnding = true;
124 break;
125 case TouchPhase.Moved:
126 if (_reportedActive.ContainsKey(next.Id))
127 _reportedActive[next.Id] = next;
128
129 break;
130 }
131 }
132
133 _batchBeginnings = beginnings.ToArray();
134 _batchEndings = endings.ToArray();
135 _snapshot = _reportedActive.Values.ToArray();
136 return _batchBeginnings;
137 }
138 }
