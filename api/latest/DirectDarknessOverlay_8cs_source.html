234 public RevealSource TrackLight(DirectRadialLight light,
235 float radiusScale = 1f,
236 float intensityScale = 1f,
237 bool trackIntensity = true,
238 string? nickname = null)
239 {
240 if (light is null)
241 throw new ArgumentNullException(nameof(light));
242
243 if (_trackedLightReveals.Any(t => ReferenceEquals(t.Light, light)))
244 return _trackedLightReveals.First(t => ReferenceEquals(t.Light, light)).Source;
245
246 var source = new RevealSource(this, light.CenterWorldPx, light.RadiusWorldPx * Math.Max(0f, radiusScale), nickname ?? light.Nickname);
247 _revealSources.Add(source);
248
249 var tracked = new TrackedLightReveal(light, source, Math.Max(0f, radiusScale), Math.Max(0f, intensityScale), trackIntensity);
250 _trackedLightReveals.Add(tracked);
251
252 light.Changed += OnTrackedLightChanged;
253 light.Disposing += OnTrackedLightDisposing;
254
255 SyncTrackedLight(tracked);
256 ForceRefresh();
257 return source;
258 }

273 public void TrackLightLayer(DirectLightLayer lightLayer,
274 float radiusScale = 1f,
275 float intensityScale = 1f,
276 bool trackIntensity = true)

411 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
412 {
413 var canvas = backbuffer.Canvas;
414 var destRect = destRectScreen.ToSKRect();
415
416 // Work in an isolated layer so "holes" reveal the already-rendered scene beneath,
417 // rather than clearing the backbuffer itself.
418 canvas.SaveLayer(destRect, null);
419
420 canvas.DrawRect(destRect, _darknessPaint);
421
422 for (int i = 0; i < _revealSources.Count; i++)
423 {
424 var source = _revealSources[i];
425 if (!source.Enabled || source.RadiusWorldPx <= 0.01f || source.Intensity <= 0.001f)
426 continue;
427
428 var centerScreen = View!.WorldPxToScreenPx(ProjectionLayer, source.CenterWorldPx);
429
430 var radiusProbeWorld = new PointF(source.CenterWorldPx.X + source.RadiusWorldPx, source.CenterWorldPx.Y);
431 var radiusProbeScreen = View.WorldPxToScreenPx(ProjectionLayer, radiusProbeWorld);
432 float screenRadius = Math.Abs(radiusProbeScreen.X - centerScreen.X);
433
434 if (screenRadius <= 0.01f)
435 continue;
436
437 using var revealShader = BuildRevealShader(centerScreen, screenRadius, source.Intensity);
438 using var revealPaint = new SKPaint
439 {
440 IsAntialias = true,
441 BlendMode = SKBlendMode.DstOut,
442 Shader = revealShader,
443 Style = SKPaintStyle.Fill
444 };
445
446 canvas.DrawRect(destRect, revealPaint);
447 }
448
449 canvas.Restore();
450 }

510 private SKShader BuildRevealShader(PointF centerScreenPx, float radiusScreenPx, float intensity)
511 {
512 byte centerAlpha = (byte)Math.Clamp((int)Math.Round(255f * Math.Clamp(intensity, 0f, 1f)), 0, 255);
513 byte midAlpha = (byte)Math.Clamp((int)Math.Round(255f * Math.Clamp(intensity, 0f, 1f) * _midpointStrength), 0, 255);
514
515 return SKShader.CreateRadialGradient(
516 new SKPoint(centerScreenPx.X, centerScreenPx.Y),
517 radiusScreenPx,
518 colors:
519 [
520 new SKColor(255, 255, 255, centerAlpha),
521 new SKColor(255, 255, 255, centerAlpha),
522 new SKColor(255, 255, 255, midAlpha),
523 new SKColor(255, 255, 255, 0)
524 ],
525 colorPos:
526 [
527 0f,
528 _innerClearRadiusRatio,
529 _midpointRadiusRatio,
530 1f
531 ],
532 mode: SKShaderTileMode.Clamp);
533 }
