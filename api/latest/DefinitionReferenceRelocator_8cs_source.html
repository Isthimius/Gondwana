Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

239 private readonly AssetLocation _source;
240 private readonly AssetLocation _destination;
241 private readonly ReferenceRelocation[] _mappings;
242 private readonly bool[] _used;

249 internal RelocationContext(AssetTypes type, AssetLocation source, AssetLocation destination, IEnumerable<ReferenceRelocation> mappings)
250 {
251 ArgumentNullException.ThrowIfNull(source);
252 ArgumentNullException.ThrowIfNull(destination);
253 ArgumentNullException.ThrowIfNull(mappings);
254 if (source.AssetType != type || destination.AssetType != type)
255 throw new ArgumentException($"Containing locations must both have type {type}.");
256 _source = source;
257 _destination = destination;
258 _mappings = mappings.ToArray();
259 _used = new bool[_mappings.Length];
260 if (_mappings.Any(mapping => mapping is null)) throw new ArgumentException("Relocations must not contain null.", nameof(mappings));
261 for (int i = 0; i < _mappings.Length; i++)
262 for (int j = 0; j < _mappings.Length; j++)
263 {
264 if (i == j) continue;
265 if (_mappings[i].Source.Matches(_mappings[j].Source))
266 throw new InvalidDataException($"Ambiguous duplicate relocation source: {_mappings[i].Source}.");
267 if (_mappings[i].Destination.Matches(_mappings[j].Source) &&
268 !_mappings[j].Source.Matches(_mappings[j].Destination))
269 throw new InvalidDataException($"Relocation chains/cycles are ambiguous on repeated application: {_mappings[j].Source}.");
270 }
271 }

281 internal void Rewrite(AssetTypes type, string? file, string? archive, string? entry, int? kind,
282 string label, Action<string?, string?, string?, bool> apply)
283 {
284 // GTS may implicitly reference an image in its containing GAF.
285 if (kind is null && string.IsNullOrWhiteSpace(file) && string.IsNullOrWhiteSpace(archive) &&
286 !string.IsNullOrWhiteSpace(entry) && _source.AssetEntryName is not null)
287 archive = _source.Path;
288 bool loose = !string.IsNullOrWhiteSpace(file);
289 bool packed = !string.IsNullOrWhiteSpace(archive) && !string.IsNullOrWhiteSpace(entry);
290 if (loose == packed || (loose && (!string.IsNullOrWhiteSpace(archive) || !string.IsNullOrWhiteSpace(entry))) ||
291 (kind is not null && kind != (packed ? 1 : 0)))
292 throw new InvalidDataException($"{label}: reference must identify exactly one loose file or typed GAF entry consistent with its source kind.");
293 var directory = Path.GetDirectoryName(_source.Path)!;
294 AssetLocation location;
295 try { location = new AssetLocation(type, Path.GetFullPath(loose ? file! : archive!, directory), packed ? entry : null); }
296 catch (Exception ex) when (ex is ArgumentException or NotSupportedException or PathTooLongException)
297 {
298 throw new InvalidDataException($"{label}: invalid physical reference path.", ex);
299 }
300 var target = location;
301 for (int i = 0; i < _mappings.Length; i++)
302 {
303 if (location.Matches(_mappings[i].Destination)) _used[i] = true;
304 if (!location.Matches(_mappings[i].Source)) continue;
305 _used[i] = true;
306 target = _mappings[i].Destination;
307 }
308 if (target.Matches(location) && _source.Matches(_destination)) return;
309 var relative = Path.GetRelativePath(Path.GetDirectoryName(_destination.Path)!, target.Path);
310 apply(target.AssetEntryName is null ? relative : null,
311 target.AssetEntryName is null ? null : relative, target.AssetEntryName, target.AssetEntryName is not null);
312 }
313
316 internal void Complete(bool requireAll)
317 {
318 if (!requireAll) return;
319 var missing = _mappings.Where((mapping, index) => !_used[index]).Select(mapping => mapping.Source.ToString()).ToArray();
320 if (missing.Length != 0)
321 throw new InvalidDataException("Relocation sources not found in document references: " + string.Join("; ", missing));
322 }
323 }
