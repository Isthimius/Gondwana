Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

340 private int CreateGpuSurface(
341 GRContext grContext,
342 int width,
343 int height,
344 int requestedSampleCount)
345 {
346 // Rgba8888 / Premul is the natural format for an OpenGL render target.
347 var info = new SKImageInfo(width, height, SKColorType.Rgba8888, SKAlphaType.Premul);
348 _surface = SKSurface.Create(grContext, budgeted: true, info, requestedSampleCount);
349
350 if (_surface is not null)
351 return requestedSampleCount;
352
353 // SKSurface.Create returns null when the requested MSAA sample count is not supported
354 // by the hardware or driver. Fall back to no MSAA (sample count 1) so the surface
355 // is always valid after Initialize() completes.
356 if (requestedSampleCount > 1)
357 _surface = SKSurface.Create(grContext, budgeted: true, info, sampleCount: 1);
358
359 if (_surface is null)
360 throw new InvalidOperationException(
361 $"Failed to create a {nameof(GpuBackbuffer)} GPU surface ({width}x{height}). " +
362 "The GRContext may be invalid or the pixel format is not renderable.");
363
364 return 1;
365 }

367 private void CreateCpuSurface(int width, int height)
368 {
369 // Bgra8888 matches the native Windows GDI pixel format; used only before
370 // GRContext becomes available so rendering is always valid from frame one.
371 var info = new SKImageInfo(width, height, SKColorType.Bgra8888, SKAlphaType.Premul);
372 _cpuBitmap = new SKBitmap(info);
373 _surface = SKSurface.Create(info, _cpuBitmap.GetPixels(), _cpuBitmap.Info.RowBytes);
374 }
