Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

9 public static IReadOnlyList<string> Validate(TilesheetDefinition definition, int? imageWidth = null, int? imageHeight = null)
10 {
11 ArgumentNullException.ThrowIfNull(definition);
12 var errors = new List<string>();
13 var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
14 foreach (var region in definition.Regions ?? [])
15 {
16 if (region is null) { errors.Add("Regions contains a null entry."); continue; }
17 var label = $"Region '{region.Name}'";
18 if (string.IsNullOrWhiteSpace(region.Name)) errors.Add($"{label}: name is empty.");
19 if (!names.Add(region.Name)) errors.Add($"{label}: duplicate name.");
20 if (region.Area.Width <= 0 || region.Area.Height <= 0 || region.TileSize.Width <= 0 || region.TileSize.Height <= 0)
21 {
22 errors.Add($"{label}: area and tile dimensions must be positive.");
23 continue;
24 }
25 if (region.Area.X < 0 || region.Area.Y < 0 ||
26 (imageWidth.HasValue && (long)region.Area.X + region.Area.Width > imageWidth) ||
27 (imageHeight.HasValue && (long)region.Area.Y + region.Area.Height > imageHeight)) errors.Add($"{label}: area is outside the source image.");
28 if (Negative(region.TilePadding) || Negative(region.RegionMargin)) errors.Add($"{label}: tile padding and region margins cannot be negative.");
29 if (Negative(region.Overhang)) errors.Add($"{label}: overhang cannot be negative.");
30 var (columns, rows) = GridSize(region);
31 if (columns <= 0 || rows <= 0) errors.Add($"{label}: layout contains no complete frames.");
32 CheckCollision(region.CollisionAdjust, region, label, errors);
33 if (!Enum.IsDefined(region.CollisionType)) errors.Add($"{label}: unknown collision type.");
34 var coordinates = new HashSet<(int, int)>();
35 foreach (var frame in region.Frames ?? [])
36 {
37 if (frame is null) { errors.Add($"{label}: null frame metadata."); continue; }
38 var frameLabel = $"{label}, frame ({frame.XTile}, {frame.YTile})";
39 if (!coordinates.Add((frame.XTile, frame.YTile))) errors.Add($"{frameLabel}: duplicate frame metadata.");
40 if (frame.XTile < 0 || frame.YTile < 0 || frame.XTile >= columns || frame.YTile >= rows) errors.Add($"{frameLabel}: coordinates are outside the frame grid.");
41 if (frame.CollisionAdjust is { } adjust) CheckCollision(adjust, region, frameLabel, errors);
42 if (frame.CollisionType is { } type && !Enum.IsDefined(type)) errors.Add($"{frameLabel}: unknown collision type.");
43 }
44 }
45 return errors;
46 }
