Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

142 string key,
143 string uri,
144 float volume = 1.0f,
145 float pan = 0.0f,
146 float playbackSpeed = 1.0f)
147 {
148 lock (_backendLock)
149 {
150 if (string.IsNullOrWhiteSpace(uri))

153 ValidateSettings(key, volume, pan, playbackSpeed);

191 public List<AudioResource> LoadFromEngineAssetsFile(
192 AssetsFile resourceFile,
193 float defaultVolume = 1.0f,
194 float defaultPan = 0.0f,
195 float defaultPlaybackSpeed = 1.0f)
196 {
197 List<AudioResource> loadedSounds = [];
198
199 foreach (var entry in resourceFile.GetAllEntries())
200 {
201 if (entry.AssetType != AssetTypes.Audio)
202 continue;
203
204 lock (_backendLock)
205 {
206 if (_soundResources.ContainsKey(entry.AssetName))
207 {
208 Engine.Logger.LogDebug("AudioResource '{Key}' already loaded. Skipping.", entry.AssetName);
209 continue;
210 }
211
212 var stream = resourceFile.Get(entry.AssetType, entry.AssetName);
213 if (stream is null)
214 {
215 Engine.Logger.LogWarning("Failed to retrieve stream for audio resource: {Key}", entry.AssetName);
216 continue;
217 }
218
219 try
220 {
221 using (stream)
222 using (var ms = new MemoryStream())
223 {
224 stream.CopyTo(ms);
225 var sound = LoadFromBytes(
226 entry.AssetName,
227 ms.ToArray(),
228 entry.AssetName,
229 defaultVolume,
230 defaultPan,
231 defaultPlaybackSpeed);
232 sound.SetAssetIdentifier(new AssetsFileIdentifier(resourceFile, AssetTypes.Audio, entry.AssetName));
233 loadedSounds.Add(sound);
234 }
235
236 Engine.Logger.LogInformation("Loaded sound: {Key}", entry.AssetName);
237 }
238 catch (Exception ex)
239 {
240 Engine.Logger.LogError(ex, "Error loading sound from asset file for key: {Key}", entry.AssetName);
241 throw;
242 }
243 }
244 }
245
246 return loadedSounds;
247 }

271 public AudioResource? Clone(
272 string key,
273 string? newKey = null,
274 float? volume = null,
275 float? pan = null,
276 float? playbackSpeed = null)
277 {
278 lock (_backendLock)
279 {
280 if (!_soundResources.TryGetValue(key, out var original))
281 {
282 Engine.Logger.LogWarning("Attempted to clone non-existent AudioResource with key: {Key}", key);
283 return null;
284 }
285
286 newKey ??= $"{key}_clone_{Guid.NewGuid()}";
287 if (_soundResources.ContainsKey(newKey))
288 {
289 Engine.Logger.LogWarning("AudioResource with key '{Key}' already exists. Cannot clone.", newKey);
290 return null;
291 }
292
293 AudioResource clone;
294 if (original.OriginalBytes is not null && !string.IsNullOrEmpty(original.SourceExtension))
295 {
296 using var stream = new MemoryStream(original.OriginalBytes, writable: false);
297 clone = LoadFromStream(
298 newKey,
299 stream,
300 original.SourceExtension,
301 volume ?? original.Volume,
302 pan ?? original.Pan,
303 playbackSpeed ?? original.PlaybackSpeed);
304 }
305 else if (!string.IsNullOrWhiteSpace(original.SourceUri))
306 {
307 clone = LoadFromUri(
308 newKey,
309 original.SourceUri,
310 volume ?? original.Volume,
311 pan ?? original.Pan,
312 playbackSpeed ?? original.PlaybackSpeed);
313 }
314 else
315 {
316 Engine.Logger.LogWarning("Cannot clone AudioResource '{Key}' because its source cannot be recreated.", key);
317 return null;
318 }
319
320 clone.CopySourceFrom(original);
321 clone.IsLooping = original.IsLooping;
322 return clone;
323 }
324 }
