Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

277 public List<AudioResource> LoadFromEngineAssetsFile(
278 AssetsFile resourceFile,
279 float defaultVolume = 1.0f,
280 float defaultPan = 0.0f,
281 float defaultPlaybackSpeed = 1.0f)
282 {
283 List<AudioResource> loadedSounds = [];
284
285 foreach (var entry in resourceFile.GetAllEntries())
286 {
287 if (entry.AssetType != AssetTypes.Audio)
288 continue;
289
290 lock (_backendLock)
291 {
292 if (_soundResources.ContainsKey(entry.AssetName))
293 {
294 Engine.Logger.LogDebug("AudioResource '{Key}' already loaded. Skipping.", entry.AssetName);
295 continue;
296 }
297
298 var stream = resourceFile.Get(entry.AssetType, entry.AssetName);
299
300 if (stream is null)
301 {
302 Engine.Logger.LogWarning("Failed to retrieve stream for audio resource: {Key}", entry.AssetName);
303 continue;
304 }
305
306 try
307 {
308 using (stream)
309 using (var ms = new MemoryStream())
310 {
311 stream.CopyTo(ms);
312
313 var sound = LoadFromBytes(
314 entry.AssetName,
315 ms.ToArray(),
316 entry.AssetName,
317 defaultVolume,
318 defaultPan,
319 defaultPlaybackSpeed);
320
321 sound.SetAssetIdentifier(
322 new AssetsFileIdentifier(resourceFile, AssetTypes.Audio, entry.AssetName));
323
324 loadedSounds.Add(sound);
325 }
326
327 Engine.Logger.LogInformation("Loaded sound: {Key}", entry.AssetName);
328 }
329 catch (Exception ex)
330 {
331 Engine.Logger.LogError(ex, "Error loading sound from asset file for key: {Key}", entry.AssetName);
332 throw;
333 }
334 }
335 }
336
337 return loadedSounds;
338 }

371 public AudioResource? Clone(
372 string key,
373 string? newKey = null,
374 float? volume = null,
375 float? pan = null,
376 float? playbackSpeed = null)
377 {
378 lock (_backendLock)
379 {
380 if (!_soundResources.TryGetValue(key, out var original))
381 {
382 Engine.Logger.LogWarning("Attempted to clone non-existent AudioResource with key: {Key}", key);
383 return null;
384 }
385
386 newKey ??= $"{key}_clone_{Guid.NewGuid()}";
387
388 if (_soundResources.ContainsKey(newKey))
389 {
390 Engine.Logger.LogWarning("AudioResource with key '{Key}' already exists. Cannot clone.", newKey);
391 return null;
392 }
393
394 AudioResource clone;
395
396 if (original.OriginalBytes is not null && !string.IsNullOrEmpty(original.SourceExtension))
397 {
398 using var stream = new MemoryStream(original.OriginalBytes, writable: false);
399
400 clone = LoadFromStream(
401 newKey,
402 stream,
403 original.SourceExtension,
404 volume ?? original.Volume,
405 pan ?? original.Pan,
406 playbackSpeed ?? original.PlaybackSpeed);
407 }
408 else if (!string.IsNullOrWhiteSpace(original.SourceUri))
409 {
410 clone = LoadFromUri(
411 newKey,
412 original.SourceUri,
413 volume ?? original.Volume,
414 pan ?? original.Pan,
415 playbackSpeed ?? original.PlaybackSpeed);
416 }
417 else
418 {
419 Engine.Logger.LogWarning(
420 "Cannot clone AudioResource '{Key}' because its source cannot be recreated.",
421 key);
422
423 return null;
424 }
425
426 clone.CopySourceFrom(original);
427 clone.IsLooping = original.IsLooping;
428 return clone;
429 }

523 internal T LoadDefinitionDocument<T>(Func<T> load)
524 {
525 lock (_backendLock)
526 {
527 if (_retainedDefinitionResources is not null)
528 throw new InvalidOperationException("A definition document load is already in progress.");
529
530 var previous = GetAll();
531
532 var settings = previous.Values.ToDictionary(
533 resource => resource,
534 resource => (resource.Volume, resource.Pan, resource.PlaybackSpeed, resource.IsLooping));
535
536 var provenance = DefinitionProvenance;
537 _retainedDefinitionResources = previous.Values.ToHashSet();
538 T result;
539
540 try
541 {
542 result = load();
543 }
544 catch
545 {
546 foreach (var resource in _soundResources.Values.ToArray())
547 if (!_retainedDefinitionResources.Contains(resource))
548 resource.Dispose();
549
550 _soundResources.Clear();
551
552 foreach (var pair in previous)
553 _soundResources[pair.Key] = pair.Value;
554
555 foreach (var pair in settings)
556 {
557 pair.Key.Volume = pair.Value.Volume;
558 pair.Key.Pan = pair.Value.Pan;
559 pair.Key.PlaybackSpeed = pair.Value.PlaybackSpeed;
560 pair.Key.IsLooping = pair.Value.IsLooping;
561 }
562
563 DefinitionProvenance = provenance;
564 throw;
565 }
566 finally
567 {
568 _retainedDefinitionResources = null;
569 }
570
571 foreach (var pair in previous)
572 if (!_soundResources.TryGetValue(pair.Key, out var current) || !ReferenceEquals(current, pair.Value))
573 pair.Value.Dispose();
574
575 return result;
576 }
577 }
