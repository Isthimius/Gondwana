Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

168 private static AnimationDefinition PrepareDocumentMember(AnimationDefinition definition, string path)
169 {
170 var clone = ParseDefinition(ToJson(definition), null);
171 var origin = definition.Source.GaniFilePath ?? definition.Source.AssetsFilePath;
172
173 var oldDirectory = definition.LoadStamp?.ReferenceBaseDirectory ??
174 (string.IsNullOrWhiteSpace(origin) || !Path.IsPathFullyQualified(origin)
175 ? null
176 : Path.GetDirectoryName(origin));
177
178 var newDirectory = Path.GetDirectoryName(path)!;
179
180 string? RebasePath(string? value) =>
181 string.IsNullOrWhiteSpace(value) || (oldDirectory is null && !Path.IsPathRooted(value))
182 ? value
183 : Path.GetRelativePath(
184 newDirectory,
185 oldDirectory is null ? Path.GetFullPath(value) : Path.GetFullPath(value, oldDirectory));
186
187 foreach (var source in clone.TilesheetSources)
188 {
189 source.GtsPath = RebasePath(source.GtsPath);
190 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
191 }
192
193 clone.Source = AnimationDefinitionSource.LooseDefinitionFile(path);
194 return clone;

277 public static IReadOnlyList<Cycle> ToAllCycles(AnimationDefinitionDocument document)
278 {
279 var errors = ValidateDocument(document);
280
281 if (errors.Count != 0)
282 throw new InvalidDataException(string.Join(Environment.NewLine, errors));
283
284 var created = new List<Cycle>();
285 var previous = new Dictionary<string, Cycle>(Cycle._cycles, StringComparer.OrdinalIgnoreCase);
286
287 try
288 {
289 foreach (var definition in document.Definitions)
290 created.Add(MaterializeCycle(definition));
291
292 for (int i = 0; i < created.Count; i++)
293 ApplyNextCycle(created[i], document.Definitions[i]);
294
295 return created;
296 }
297 catch
298 {
299 foreach (var item in created)
300 item.Dispose();
301
302 foreach (var pair in previous)
303 Cycle._cycles[pair.Key] = pair.Value;
304
305 throw;
306 }
307 }
