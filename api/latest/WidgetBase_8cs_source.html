Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

203 public virtual bool HitTest(View view,
204 Point screenPositionPx)
205 {
206 ArgumentNullException.ThrowIfNull(view);
207
208 if (!IsInputEnabled ||
209 !IsPointerInputEnabled ||
210 !Visible)
211 {
212 return false;
213 }
214
215 if (Mode == DirectDrawingMode.View
216 && !ReferenceEquals(View, view))
217 {
218 return false;
219 }
220
221 if (Mode == DirectDrawingMode.SceneLayer && SceneLayer is { } layer &&
222 (layer.WrapHorizontally || layer.WrapVertically))
223 return GetWrappedHitOffset(view, screenPositionPx).HasValue;
224
225 RectangleF screenBounds = GetDrawLocationScreen(view);
226
227 return !screenBounds.IsEmpty
228 && screenBounds.Contains(screenPositionPx.X, screenPositionPx.Y);
229 }

231 internal PointF? GetWrappedHitOffset(View view, Point screenPositionPx)
232 {
233 if (Mode != DirectDrawingMode.SceneLayer || SceneLayer is not { } layer)
234 return PointF.Empty;
235 var world = view.ScreenPxToWorldPx(layer, screenPositionPx);
236 foreach (var offset in layer.GetWrappedOffsets(WorldBounds, new RectangleF(world.X, world.Y, 0.01f, 0.01f)))
237 {
238 RectangleF bounds = WorldBounds;
239 bounds.Offset(offset);
240 if (bounds.Contains(world)) return offset;
241 }
242 return null;
243 }

252 public virtual bool HitTest(Point screenPositionPx)
253 {
254 var views = RenderSurfaceHost.ViewManager.Views;
255
256 for (int index = views.Count - 1; index >= 0; index--)
257 {
258 View view = views[index];
259
260 if (!view.Viewport.TargetRectPx.Contains(screenPositionPx))
261 {
262 continue;
263 }
264
265 return HitTest(view, screenPositionPx);
266 }
267
268 return false;
