Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

208 public virtual bool HitTest(
209 View view,
210 Point screenPositionPx)
211 {
212 ArgumentNullException.ThrowIfNull(view);
213
214 if (!IsInputEnabled ||
215 !IsPointerInputEnabled ||
216 !Visible)
217 {
218 return false;
219 }
220
221 if (Mode == DirectDrawingMode.View
222 && !ReferenceEquals(View, view))
223 {
224 return false;
225 }
226
227 if (Mode == DirectDrawingMode.SceneLayer && SceneLayer is { } layer &&
228 (layer.WrapHorizontally || layer.WrapVertically))
229 return GetWrappedHitOffset(view, screenPositionPx).HasValue;
230
231 RectangleF screenBounds = GetDrawLocationScreen(view);
232
233 return !screenBounds.IsEmpty
234 && screenBounds.Contains(screenPositionPx.X, screenPositionPx.Y);
235 }

237 internal PointF? GetWrappedHitOffset(View view, Point screenPositionPx)
238 {
239 if (Mode != DirectDrawingMode.SceneLayer || SceneLayer is not { } layer)
240 return PointF.Empty;
241
242 var world = view.ScreenPxToWorldPx(layer, screenPositionPx);
243
244 foreach (var offset in layer.GetWrappedOffsets(
245 WorldBounds,
246 new RectangleF(
247 world.X,
248 world.Y,
249 0.01f,
250 0.01f)))
251 {
252 RectangleF bounds = WorldBounds;
253 bounds.Offset(offset);
254
255 if (bounds.Contains(world))
256 return offset;
257 }
258
259 return null;
260 }

269 public virtual bool HitTest(Point screenPositionPx)
270 {
271 var views = RenderSurfaceHost.ViewManager.Views;
272
273 for (int index = views.Count - 1; index >= 0; index--)
274 {
275 View view = views[index];
276
277 if (!view.Viewport.TargetRectPx.Contains(screenPositionPx))
278 {
279 continue;
280 }
281
282 return HitTest(view, screenPositionPx);
283 }
284
285 return false;
