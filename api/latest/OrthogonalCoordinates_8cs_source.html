Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

59 public List<SceneLayerTile> GetSceneLayerTilesInPixelRange(
60 SceneLayer sceneLayer,
61 Rectangle worldPixelRange,
62 bool includeOverhang)
63 {
64 var retVal = new List<SceneLayerTile>();
65 var candidateRange = TileBounds.GetTransformedTileCandidateRange(sceneLayer, worldPixelRange, includeOverhang);
66
67 // 1) Find coarse grid bounds via inverse transform (unchanged)
68 PointF ptUL = GetSceneLayerCoordinatesAtPixel(sceneLayer, new PointF(candidateRange.Left, candidateRange.Top));
69
70 PointF ptBR = GetSceneLayerCoordinatesAtPixel(
71 sceneLayer,
72 new PointF(candidateRange.Right - 1, candidateRange.Bottom - 1));
73
74 int minY = (int)Math.Floor(ptUL.Y) - 1;
75 int maxY = (int)Math.Ceiling(ptBR.Y) + 1;
76 int minX = (int)Math.Floor(ptUL.X) - 1;
77 int maxX = (int)Math.Ceiling(ptBR.X) + 1;
78
79 // 2) Scan candidate grid cells and include if their overhang-aware rect intersects
80 for (int y = minY; y <= maxY; y++)
81 {
82 for (int x = minX; x <= maxX; x++)
83 {
84 var gPt = sceneLayer[x, y];
85
86 if (gPt == null)
87 continue;
88
89 // Overhang-aware pixel rect
90 var rect = GetPixelRangeForTile(gPt, includeOverhang);
91
92 if (rect.IntersectsWith(worldPixelRange))
93 retVal.Add(gPt);
94 }
95 }
96
97 return retVal;
98 }
