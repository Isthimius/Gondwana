Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

320 internal void DrawDrawables(View view, IEnumerable<IDrawable> drawables, Rectangle clipRect)
321 {
322 if (this is RecordingBackbuffer recording &&
323 drawables is IReadOnlyList<IDrawable> ordered &&
324 recording.TryDrawFixedGridDrawables(view, ordered, clipRect))
325 {
326 return;
327 }
328
329 Canvas.Save();
330 Canvas.ClipRect(clipRect.ToSKRect());
331
332 var tiles = new List<(Tile Tile, WrappedDrawable? Instance)>();
333
334 foreach (var entry in drawables)
335 {
336 var instance = entry as WrappedDrawable;
337 var drawable = instance?.Owner ?? entry;
338 using var scope = instance?.Enter(view);
339
340 if (!drawable.Visible)
341 continue;
342
343 var destRectScreen = drawable.GetDrawLocationScreen(view);
344
345 if (drawable is Sprite sprite)
346 {
347 destRectScreen = sprite.ApplyJiggleToDestRect(destRectScreen);
348 }
349
350 bool batched = this is RecordingBackbuffer recorder &&
351 instance is null &&
352 drawable.GetType() == typeof(SceneLayerTile) &&
353 drawable is SceneLayerTile layerTile &&
354 recorder.TryQueueTile(layerTile, destRectScreen);
355
356 if (!batched)
357 {
358 (this as RecordingBackbuffer)?.FlushTileBatch();
359 drawable.Draw(this, destRectScreen);
360 }
361
362 // GPU/GL surfaces always present the complete backbuffer. Avoid calculating visual
363 // bounds just to pass them to dirty-region tracking that the GPU path never consumes.
364 if (!IsGlThreadRendered)
365 {
366 RectangleF visualBoundsScreen = drawable is Sprite drawnSprite
367 ? drawnSprite.GetVisualBoundsScreen(destRectScreen)
368 : destRectScreen;
369
370 AddToBackbufferDirtyRectangle(visualBoundsScreen.ToPixelAlignedRect());
371 }
372
373 if (drawable is Tile tile)
374 tiles.Add((tile, instance));
375 }
376
377 (this as RecordingBackbuffer)?.FlushTileBatch();
378
379 PostDrawTiles(view, tiles);
380
381 Canvas.Restore();
382 }

384 private void PostDrawTiles(View view, List<(Tile Tile, WrappedDrawable? Instance)> tiles)
385 {
386 foreach (var entry in tiles)
387 {
388 var tile = entry.Tile;
389 using var scope = entry.Instance?.Enter(view);
390 var layer = tile.SceneLayer;
391 bool drawFog = tile.EnableFog;
392 bool drawGrid = layer.ShowGridLines && tile.Visible && tile.IsPositionFixed;
393 bool drawCollision = layer.ShowCollisionBoxes && tile.Visible && tile.CollisionsEnabled;
394
395 if (!drawFog && !drawGrid && !drawCollision)
396 continue;
397
398 // Orthogonal grid outlines are rectangles. Avoid building two transient point arrays
399 // and performing four point-wise world-to-screen transforms for every fixed tile.
400 // Rectangle conversion uses the same view transform while preserving the grid cell
401 // bounds (includeOverhang: false) used by the legacy polygon path.
402 if (drawGrid
403 && !drawFog
404 && layer.CoordinateSystemType == Gondwana.Drawing.Coordinates.CoordinateSystemTypes.Orthogonal)
405 {
406 var worldRect = layer.CoordinateSystem.GetPixelRangeForTile(tile, includeOverhang: false);
407 var screenRect = view.WorldRectToScreenRect(layer, worldRect);
408 Canvas.DrawRect(screenRect.ToSKRect(), GridLinePaint);
409
410 if (drawCollision)
411 {
412 var colRectScreen = tile.GetCollisionAreaScreen(view).ToSKRect();
413 Canvas.DrawRect(colRectScreen, CollisionBoxPaint);
414 }
415
416 continue;
417 }
418
419 // Polygon geometry is only needed for fog or non-orthogonal grid outlines.
420 if (drawFog || drawGrid)
421 {
422 var worldPts = tile.OutlinePointsWorld;
423 var ptsScreen = new SKPoint[worldPts.Length];
424
425 for (int i = 0; i < worldPts.Length; i++)
426 {
427 var p = worldPts[i];
428
429 var sp = view.WorldPxToScreenPx(
430 layer,
431 new PointF(p.X, p.Y)
432 );
433
434 ptsScreen[i] = new SKPoint(sp.X, sp.Y);
435 }
436
437 if (drawFog)
438 {
439 using var path = new SKPath();
440 path.AddPoly(ptsScreen, close: true);
441 Canvas.DrawPath(path, FogPaint);
442 }
443
444 if (drawGrid)
445 {
446 var enclosed = new SKPoint[ptsScreen.Length + 1];
447 Array.Copy(ptsScreen, enclosed, ptsScreen.Length);
448 enclosed[^1] = ptsScreen[0];
449 Canvas.DrawPoints(SKPointMode.Polygon, enclosed, GridLinePaint);
450 }
451 }
452
453 if (drawCollision)
454 {
455 var colRectScreen = tile.GetCollisionAreaScreen(view).ToSKRect();
456 Canvas.DrawRect(colRectScreen, CollisionBoxPaint);
457 }
458 }
459 }
