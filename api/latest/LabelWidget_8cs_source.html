Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

292 protected override void OnPointerDown(WidgetPointerEventArgs args)
293 {
294 base.OnPointerDown(args);
295
296 if (_scrollBarPointerId.HasValue || !args.IsPrimaryButton || !IsVerticalScrollBarVisible)
297 return;
298
299 PointF local = GetLocalPointerPosition(args);
300 Rectangle trackBounds = GetScrollBarTrackBounds();
301 Rectangle thumbBounds = GetScrollBarThumbBounds();
302
303 if (!trackBounds.Contains(Point.Round(local)))
304 return;
305
306 args.Handled = true;
307 _scrollBarPointerId = args.PointerId;
308
309 if (thumbBounds.Contains(Point.Round(local)))
310 {
311 _isDraggingScrollBarThumb = true;
312 _scrollBarDragOffset = local.Y - thumbBounds.Top;
313 return;
314 }
315
316 float page = Math.Max(1f, GetViewportContentHeight());
317 SetVerticalScrollOffset(_verticalScrollOffsetPx + (local.Y < thumbBounds.Top ? -page : page));
318 }
319
321 protected override void OnPointerMove(WidgetPointerEventArgs args)
322 {
323 base.OnPointerMove(args);
324
325 if (!_isDraggingScrollBarThumb ||
326 _scrollBarPointerId != args.PointerId ||
327 !IsVerticalScrollBarVisible)
328 {
329 return;
330 }
331
332 Rectangle trackBounds = GetScrollBarTrackBounds();
333 Rectangle thumbBounds = GetScrollBarThumbBounds();
334 float travel = trackBounds.Height - thumbBounds.Height;
335 float maximumScroll = _maximumVerticalScrollOffsetPx;
336
337 if (travel <= 0f || maximumScroll <= 0f)
338 return;
339
340 float desiredTop = GetLocalPointerPosition(args).Y - _scrollBarDragOffset;
341 float fraction = Math.Clamp((desiredTop - trackBounds.Top) / travel, 0f, 1f);
342 SetVerticalScrollOffset(fraction * maximumScroll);
343 args.Handled = true;
344 }
345
347 protected override void OnPointerUp(WidgetPointerEventArgs args)
348 {
349 base.OnPointerUp(args);
350
351 if (_scrollBarPointerId != args.PointerId)
352 return;
353
354 _isDraggingScrollBarThumb = false;
355 _scrollBarPointerId = null;
356 args.Handled = true;
357 }

360 protected override void OnMouseWheel(WidgetMouseWheelEventArgs args)
361 {
362 base.OnMouseWheel(args);
363
364 float maximumScroll = _maximumVerticalScrollOffsetPx;
365 if (args.Delta == 0 || maximumScroll <= 0f)
366 return;
367
368 long notches = Math.Max(1, Math.Abs((long)args.Delta) / 120);
369 float change = notches * MouseWheelScrollPixels * (args.Delta > 0 ? -1f : 1f);
370 SetVerticalScrollOffset(_verticalScrollOffsetPx + change);
371 args.Handled = true;
372 }

387 private void RefreshScrollState()
388 {
389 bool scrollingEnabled = VerticalScrollBarVisibility != ScrollBarVisibility.Never;
390
391 Rectangle bounds = Bounds;
392 SetTextBlockBounds(bounds);
393
394 if (!scrollingEnabled)
395 {
396 _verticalScrollOffsetPx = 0f;
397 TextBlock.VerticalScrollOffsetPx = 0f;
398 }
399
400 _maximumVerticalScrollOffsetPx = TextBlock.MeasureMaximumVerticalScrollOffsetPx();
401
402 bool showScrollBar = VerticalScrollBarVisibility == ScrollBarVisibility.Always ||
403 (VerticalScrollBarVisibility == ScrollBarVisibility.Auto &&
404 _maximumVerticalScrollOffsetPx > 0.5f);
405
406 if (showScrollBar)
407 {
408 Rectangle contentBounds = new(
409 bounds.Left,
410 bounds.Top,
411 Math.Max(1, bounds.Width - ScrollBarWidth - ScrollBarMargin * 2),
412 bounds.Height);
413 SetTextBlockBounds(contentBounds);
414 _maximumVerticalScrollOffsetPx = TextBlock.MeasureMaximumVerticalScrollOffsetPx();
415 }
416
417 IsInputEnabled = showScrollBar;
418 IsPointerInputEnabled = showScrollBar;
419 IsKeyboardInputEnabled = false;
420 CanReceiveFocus = false;
421
422 SetVerticalScrollOffset(_verticalScrollOffsetPx);
423
424 bool visible = Visible && showScrollBar;
425 VerticalScrollBarTrack.Visible = visible;
426 VerticalScrollBarThumb.Visible = visible;
427
428 if (visible)
429 RefreshScrollBarBounds();
430 }

466 private Rectangle GetScrollBarThumbBounds()
467 {
468 Rectangle trackBounds = GetScrollBarTrackBounds();
469 float maximumScroll = _maximumVerticalScrollOffsetPx;
470 float viewportHeight = GetViewportContentHeight();
471
472 if (maximumScroll <= 0f || viewportHeight <= 0f)
473 return trackBounds;
474
475 float contentHeight = viewportHeight + maximumScroll;
476 int thumbHeight = Math.Clamp(
477 (int)MathF.Round(trackBounds.Height * Math.Min(1f, viewportHeight / contentHeight)),
478 Math.Min(MinimumScrollBarThumbHeight, trackBounds.Height),
479 trackBounds.Height);
480 int travel = trackBounds.Height - thumbHeight;
481 int offset = maximumScroll <= 0f
482 ? 0
483 : (int)MathF.Round(travel * (_verticalScrollOffsetPx / maximumScroll));
484
485 return new Rectangle(
486 trackBounds.Left,
487 trackBounds.Top + offset,
488 trackBounds.Width,
489 thumbHeight);
490 }

492 private PointF GetLocalPointerPosition(WidgetPointerEventArgs args)
493 {
494 RectangleF screenBounds = TextBlock.GetDrawLocationScreen(args.View);
495 screenBounds.Offset(
496 args.WrappedOffsetWorldPx.X * args.View.Viewport.Zoom,
497 args.WrappedOffsetWorldPx.Y * args.View.Viewport.Zoom);
498
499 Rectangle textBounds = Mode == DirectDrawingMode.View
500 ? TextBlock.ScreenBounds
501 : TextBlock.WorldBounds;
502
503 float scaleX = screenBounds.Width / textBounds.Width;
504 float scaleY = screenBounds.Height / textBounds.Height;
505
506 return new PointF(
507 Bounds.Left + (args.ScreenPositionPx.X - screenBounds.Left) / scaleX,
508 Bounds.Top + (args.ScreenPositionPx.Y - screenBounds.Top) / scaleY);
509 }
