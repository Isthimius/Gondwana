Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

157 private static AudioDefinition PrepareDocumentMember(AudioDefinition definition, string path)
158 {
159 var clone = ParseDefinition(ToJson(definition), null);
160
161 if (string.IsNullOrWhiteSpace(clone.Key))
162 clone.Key = Path.GetFileNameWithoutExtension(path);
163
164 var origin = definition.Source.GsndFilePath ?? definition.Source.AssetsFilePath;
165
166 var oldDirectory = definition.LoadStamp?.ReferenceBaseDirectory ??
167 (string.IsNullOrWhiteSpace(origin) || !Path.IsPathFullyQualified(origin)
168 ? null
169 : Path.GetDirectoryName(origin));
170
171 var newDirectory = Path.GetDirectoryName(path)!;
172
173 string? RebasePath(string? value) =>
174 string.IsNullOrWhiteSpace(value) || (oldDirectory is null && !Path.IsPathRooted(value))
175 ? value
176 : Path.GetRelativePath(
177 newDirectory,
178 oldDirectory is null ? Path.GetFullPath(value) : Path.GetFullPath(value, oldDirectory));
179
180 foreach (var resource in clone.Resources)
181 {
182 if (resource.SourceKind == AudioResourceSourceKind.Uri)
183 continue;
184
185 resource.FilePath = RebasePath(resource.FilePath);
186 resource.AssetsFilePath = RebasePath(resource.AssetsFilePath);
187 }
188
189 clone.Source = AudioDefinitionSource.LooseDefinitionFile(path);
190 return clone;
191 }
