Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

79 public static SpriteDefinition Rebase(SpriteDefinition definition, string directory)
80 {
81 var clone = FromJson(ToJson(definition));
82 var origin = definition.Source.GsprFilePath ?? definition.Source.AssetsFilePath;
83 if (string.IsNullOrWhiteSpace(origin)) return clone;
84 var oldDirectory = Path.GetDirectoryName(Path.GetFullPath(origin))!;
85 string? RebasePath(string? path) => string.IsNullOrWhiteSpace(path) ? path :
86 Path.GetRelativePath(Path.GetFullPath(directory), Path.GetFullPath(path, oldDirectory));
87 foreach (var source in clone.TilesheetSources)
88 {
89 source.GtsPath = RebasePath(source.GtsPath);
90 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
91 }
92 foreach (var source in clone.SceneSources)
93 {
94 source.GscnPath = RebasePath(source.GscnPath);
95 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
96 }
97 return clone;
98 }

137 // EngineState retains its historical last-incoming-nickname-wins merge contract.
138 internal static List<Sprite> ToSprites(SpriteDefinition definition, bool allowDuplicateNicknames)
139 {
140 var errors = SpriteDefinitionValidator.Validate(definition, allowDuplicateNicknames);
141 if (errors.Count != 0) throw new InvalidDataException("Invalid GSPR definition:\n" + string.Join("\n", errors));
142 var created = new List<Sprite>();
143 try
144 {
145 foreach (var entry in definition.Sprites)
146 {
147 var scenes = Scene._allScenes.Where(scene => scene.ID == entry.SceneId).Take(2).ToArray();
148 if (scenes.Length != 1)
149 throw new InvalidDataException($"Sprite '{entry.Nickname}': Scene '{entry.SceneId}' is missing or ambiguous.");
150 var scene = scenes[0];
151 var layers = scene.SceneLayers.Where(layer => layer.ID == entry.SceneLayerId).Take(2).ToArray();
152 if (layers.Length != 1)
153 throw new InvalidDataException($"Sprite '{entry.Nickname}': SceneLayer '{entry.SceneLayerId}' is missing or ambiguous.");
154 var layer = layers[0];
155 var frame = entry.Frame is null ? default : ResolveFrame(entry.Frame);
156 if (!string.IsNullOrWhiteSpace(entry.CollisionProfileName) && !scene.CollisionProfiles.GetProfileNames().Contains(entry.CollisionProfileName))
157 throw new InvalidDataException($"Sprite '{entry.Nickname}': collision profile '{entry.CollisionProfileName}' does not exist.");
158 var sprite = SpriteManager.Instance.CreateSprite(layer, frame, entry.Nickname, entry.CollisionProfileName);
159 created.Add(sprite);
160 if (entry.Id != Guid.Empty) sprite.Id = entry.Id;
161 sprite.SetPosition(new Vector2(entry.Position.X, entry.Position.Y));
162 sprite.Visible = entry.Visible;
163 sprite.ZOrder = entry.ZOrder;
164 sprite.Rotation = entry.Rotation;
165 sprite.HorizAlign = entry.HorizAlign;
166 sprite.VertAlign = entry.VertAlign;
167 sprite.NudgeX = entry.NudgeX;
168 sprite.NudgeY = entry.NudgeY;
169 if (!entry.RenderSize.IsEmpty) sprite.RenderSize = entry.RenderSize;
170 sprite.EnableFog = entry.EnableFog;
171 sprite.AdjustCollisionArea = entry.AdjustCollisionArea;
172 sprite.CollisionType = entry.CollisionType;
173 sprite.AdjustCollisionAreaByFrame = entry.AdjustCollisionAreaByFrame;
174 sprite.CollisionTypeByFrame = entry.CollisionTypeByFrame;
175 sprite.CollisionsEnabled = entry.CollisionsEnabled;
176 }
177 return created;
178 }
179 catch
180 {
181 foreach (var sprite in created)
182 {
183 SpriteManager.Instance._spriteList.Remove(sprite);
184 sprite.DisposeImmediate();
185 }
186 throw;
187 }
188 }

190 private static Frame ResolveFrame(SpriteFrameDefinition definition)
191 {
192 var sheet = TilesheetRegistry.Instance.GetOrNull(definition.Tilesheet)
193 ?? throw new InvalidDataException($"GSPR tilesheet '{definition.Tilesheet}' is not registered.");
194 var region = sheet.GetRegion(definition.RegionName)
195 ?? throw new InvalidDataException($"GSPR region '{definition.RegionName}' is missing on '{definition.Tilesheet}'.");
196 if (definition.XTile >= region.Columns || definition.YTile >= region.Rows)
197 throw new InvalidDataException($"GSPR frame ({definition.XTile}, {definition.YTile}) is outside region '{definition.RegionName}'.");
198 return sheet.GetFrame(definition.RegionName, definition.XTile, definition.YTile);
199 }
