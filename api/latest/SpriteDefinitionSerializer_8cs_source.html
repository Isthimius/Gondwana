Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

105 public static SpriteDefinition Rebase(SpriteDefinition definition, string directory)
106 {
107 var clone = FromJson(ToJson(definition));
108 var origin = definition.Source.GsprFilePath ?? definition.Source.AssetsFilePath;
109 if (string.IsNullOrWhiteSpace(origin)) return clone;
110 var oldDirectory = Path.GetDirectoryName(Path.GetFullPath(origin))!;
111 string? RebasePath(string? path) => string.IsNullOrWhiteSpace(path) ? path :
112 Path.GetRelativePath(Path.GetFullPath(directory), Path.GetFullPath(path, oldDirectory));
113 foreach (var source in clone.TilesheetSources)
114 {
115 source.GtsPath = RebasePath(source.GtsPath);
116 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
117 }
118 foreach (var source in clone.SceneSources)
119 {
120 source.GscnPath = RebasePath(source.GscnPath);
121 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
122 }
123 foreach (var source in clone.AnimationSources)
124 {
125 source.GaniPath = RebasePath(source.GaniPath);
126 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
127 }
128 return clone;
129 }

149 public static SpriteDefinition FromSprites(IEnumerable<Sprite> sprites)
150 {
151 ArgumentNullException.ThrowIfNull(sprites);
152 var instances = sprites.ToList();
153 var definition = new SpriteDefinition
154 {
155 Sprites = instances.Select(FromSprite).ToList(),
156 Source = SpriteDefinitionSource.Generated()
157 };
158
159 var tilesheets = new HashSet<string>(StringComparer.Ordinal);
160 var scenes = new HashSet<string>(StringComparer.Ordinal);
161 var animations = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
162 foreach (var sprite in instances)
163 {
164 if (sprite.CurrentFrame.Tilesheet is { } sheet && tilesheets.Add(sheet.Name))
165 {
166 var source = DefinitionDependencyProvenance.ForTilesheet(sheet);
167 if (source?.FilePath is { } loose)
168 definition.TilesheetSources.Add(SpriteTilesheetSourceDefinition.Loose(sheet.Name, loose));
169 else if (source?.AssetsFilePath is { } archive && source.AssetEntryName is { } entry)
170 definition.TilesheetSources.Add(SpriteTilesheetSourceDefinition.Packed(sheet.Name, archive, entry));
171 }
172
173 if (sprite.SceneLayer.Scene is { } scene && scenes.Add(scene.ID))
174 {
175 var source = DefinitionDependencyProvenance.ForScene(scene);
176 if (source?.FilePath is { } loose)
177 definition.SceneSources.Add(SpriteSceneSourceDefinition.Loose(scene.ID, loose));
178 else if (source?.AssetsFilePath is { } archive && source.AssetEntryName is { } entry)
179 definition.SceneSources.Add(SpriteSceneSourceDefinition.Packed(scene.ID, archive, entry));
180 }
181
182 if (sprite.TileAnimator.CurrentCycle?.CycleKey is { Length: > 0 } key)
183 {
184 foreach (var (animationKey, source) in DefinitionDependencyProvenance.ForAnimationChain(key, animations))
185 {
186 if (source.FilePath is { } loose)
187 definition.AnimationSources.Add(SpriteAnimationSourceDefinition.Loose(animationKey, loose));
188 else if (source.AssetsFilePath is { } archive && source.AssetEntryName is { } entry)
189 definition.AnimationSources.Add(SpriteAnimationSourceDefinition.Packed(animationKey, archive, entry));
190 }
191 }
192 }
193
194 return definition;
195 }

260 // EngineState retains its historical last-incoming-nickname-wins merge contract.
261 internal static List<Sprite> ToSprites(SpriteDefinition definition, bool allowDuplicateNicknames)
262 {
263 var errors = SpriteDefinitionValidator.Validate(definition, allowDuplicateNicknames);
264 if (errors.Count != 0) throw new InvalidDataException("Invalid GSPR definition:\n" + string.Join("\n", errors));
265 var created = new List<Sprite>();
266 try
267 {
268 foreach (var entry in definition.Sprites)
269 {
270 var scenes = Scene._allScenes.Where(scene => scene.ID == entry.SceneId).Take(2).ToArray();
271 if (scenes.Length != 1)
272 throw new InvalidDataException($"Sprite '{entry.Nickname}': Scene '{entry.SceneId}' is missing or ambiguous.");
273 var scene = scenes[0];
274 var layers = scene.SceneLayers.Where(layer => layer.ID == entry.SceneLayerId).Take(2).ToArray();
275 if (layers.Length != 1)
276 throw new InvalidDataException($"Sprite '{entry.Nickname}': SceneLayer '{entry.SceneLayerId}' is missing or ambiguous.");
277 var layer = layers[0];
278 var frame = entry.Frame is null ? default : ResolveFrame(entry.Frame);
279 if (!string.IsNullOrWhiteSpace(entry.CollisionProfileName) && !scene.CollisionProfiles.GetProfileNames().Contains(entry.CollisionProfileName))
280 throw new InvalidDataException($"Sprite '{entry.Nickname}': collision profile '{entry.CollisionProfileName}' does not exist.");
281 var sprite = SpriteManager.Instance.CreateSprite(layer, frame, entry.Nickname, entry.CollisionProfileName);
282 created.Add(sprite);
283 if (entry.Id != Guid.Empty) sprite.Id = entry.Id;
284 sprite.SetPosition(new Vector2(entry.Position.X, entry.Position.Y));
285 sprite.Visible = entry.Visible;
286 sprite.ZOrder = entry.ZOrder;
287 sprite.Rotation = entry.Rotation;
288 sprite.HorizAlign = entry.HorizAlign;
289 sprite.VertAlign = entry.VertAlign;
290 sprite.NudgeX = entry.NudgeX;
291 sprite.NudgeY = entry.NudgeY;
292 if (!entry.RenderSize.IsEmpty) sprite.RenderSize = entry.RenderSize;
293 sprite.EnableFog = entry.EnableFog;
294 sprite.AdjustCollisionArea = entry.AdjustCollisionArea;
295 sprite.CollisionType = entry.CollisionType;
296 sprite.AdjustCollisionAreaByFrame = entry.AdjustCollisionAreaByFrame;
297 sprite.CollisionTypeByFrame = entry.CollisionTypeByFrame;
298 sprite.CollisionsEnabled = entry.CollisionsEnabled;
299 if (!string.IsNullOrWhiteSpace(entry.AnimationKey))
300 {
301 // Cycle.GetAnimationCycle returns a per-sprite clone with independent playback state.
302 var cycle = Cycle.GetAnimationCycle(entry.AnimationKey)
303 ?? throw new InvalidDataException(
304 $"Sprite '{entry.Nickname}': animation '{entry.AnimationKey}' is not registered.");
305 sprite.TileAnimator.CurrentCycle = cycle;
306 if (entry.StartAnimation) sprite.TileAnimator.StartAnimation();
307 }
308 }
309 SpriteManager.Instance.DefinitionProvenance = definition.Sprites.All(sprite => sprite.Id != Guid.Empty &&
310 !string.IsNullOrWhiteSpace(sprite.Nickname))
311 ? definition.LoadStamp?.Materialized(definition, () => FromSprites(created))
312 : null;
313 foreach (var sprite in created) sprite.DefinitionProvenance = SpriteManager.Instance.DefinitionProvenance;
314 return created;
315 }
316 catch
317 {
318 foreach (var sprite in created)
319 {
320 SpriteManager.Instance._spriteList.Remove(sprite);
321 sprite.DisposeImmediate();
322 }
323 throw;
324 }
325 }

327 private static Frame ResolveFrame(SpriteFrameDefinition definition)
328 {
329 var sheet = TilesheetRegistry.Instance.GetOrNull(definition.Tilesheet)
330 ?? throw new InvalidDataException($"GSPR tilesheet '{definition.Tilesheet}' is not registered.");
331 var region = sheet.GetRegion(definition.RegionName)
332 ?? throw new InvalidDataException($"GSPR region '{definition.RegionName}' is missing on '{definition.Tilesheet}'.");
333 if (definition.XTile >= region.Columns || definition.YTile >= region.Rows)
334 throw new InvalidDataException($"GSPR frame ({definition.XTile}, {definition.YTile}) is outside region '{definition.RegionName}'.");
335 return sheet.GetFrame(definition.RegionName, definition.XTile, definition.YTile);
