Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

101 public StackPanelWidget Relayout()
102 {
103 WidgetBase[] widgets = ChildWidgets.ToArray();
104 float cursor = 0f;
105 float cross = 0f;
106
107 for (int index = 0; index < widgets.Length; index++)
108 {
109 WidgetBase widget = widgets[index];
110 Rectangle bounds = Mode == DirectDrawingMode.View
111 ? widget.ScreenBounds
112 : widget.WorldBounds;
113
114 Vector2 offset = _orientation == WidgetOrientation.Vertical
115 ? new Vector2(0f, cursor)
116 : new Vector2(cursor, 0f);
117
118 SetLocalOffset(widget, offset);
119
120 if (_orientation == WidgetOrientation.Vertical)
121 {
122 cursor += bounds.Height;
123 cross = Math.Max(cross, bounds.Width);
124 }
125 else
126 {
127 cursor += bounds.Width;
128 cross = Math.Max(cross, bounds.Height);
129 }
130
131 if (index < widgets.Length - 1)
132 cursor += _spacing;
133 }
134
135 _contentSize = _orientation == WidgetOrientation.Vertical
136 ? new SizeF(cross, cursor)
137 : new SizeF(cursor, cross);
138
139 return this;
140 }
