Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

284 public static SKBitmap PremultiplyAlpha(this SKBitmap bitmap)
285 {
286 if (bitmap == null || bitmap.IsEmpty)
287 throw new ArgumentException("Invalid bitmap.");
288
289 if (bitmap.AlphaType == SKAlphaType.Premul)
290 return bitmap;
291
292 var info = new SKImageInfo(
293 bitmap.Width,
294 bitmap.Height,
295 bitmap.ColorType,
296 SKAlphaType.Premul);
297
298 var premulBitmap = new SKBitmap(info);
299
300 int width = bitmap.Width;
301 int height = bitmap.Height;
302
303 for (int y = 0; y < height; y++)
304 {
305 for (int x = 0; x < width; x++)
306 {
307 var c = bitmap.GetPixel(x, y);
308 byte a = c.Alpha;
309
310 byte r = (byte)((c.Red * a + 127) / 255);
311 byte g = (byte)((c.Green * a + 127) / 255);
312 byte b = (byte)((c.Blue * a + 127) / 255);
313
314 premulBitmap.SetPixel(
315 x,
316 y,
317 new SKColor(
318 r,
319 g,
320 b,
321 a));
322 }
323 }
324
325 premulBitmap.NotifyPixelsChanged();
326 return premulBitmap;
