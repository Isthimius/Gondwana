Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

216 internal static void RaiseTimerEvents(TimerType type, long engineTick)
217 {
218 var expired = new List<string>();
219
220 foreach (var (key, timer) in _timers.ToArray())
221 {
222 // checking this TimerType (i.e., PreCycle or PostCycle)
223 if (timer.Type != type) continue;
224
225 // "push" time forward for Paused Timer instances
226 if (PausedAll || timer.Paused)
227 {
228 timer._lastEventTick += (engineTick - timer._lastEventTick);
229 continue;
230 }
231
232 // check if Timer.Length time has passed
233 while (engineTick - timer._lastEventTick >= timer.Length)
234 {
235 // save the time this event was scheduled to run
236 // might be different from current system tick, but storing this value
237 // will ensure that a lag in repeating timer events does not
238 // accumulate over time
239 timer._lastEventTick += timer.Length;
240 timer.RaiseTickEvent();
241
242 // check for any expired timers
243 if (timer.Cycles == TimerCycles.Once)
244 {
245 expired.Add(key);
246 break;
247 }
248 }
249 }
250
251 foreach (var key in expired)
252 _timers.TryRemove(key, out _);
253 }
