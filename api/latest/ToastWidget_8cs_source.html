Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

425 private void AdvanceTransition(float elapsedSec, bool entering)
426 {
427 _phaseElapsedSec += elapsedSec;
428
429 float progress = TransitionDurationSec <= 0f
430 ? 1f
431 : Math.Clamp(_phaseElapsedSec / TransitionDurationSec, 0f, 1f);
432
433 Func<float, float> easing = EasingFunctions.From(
434 entering ? EntranceEasing : ExitEasing);
435
436 float eased = Math.Clamp(easing(progress), 0f, 1f);
437
438 SetPosition(Vector2.Lerp(
439 _phaseStartPosition,
440 _phaseEndPosition,
441 eased));
442
443 SetOpacity(
444 _phaseStartOpacity
445 + (_phaseEndOpacity - _phaseStartOpacity) * eased);
446
447 if (progress < 1f)
448 return;
449
450 if (entering)
451 CompleteEntrance();
452 else
453 CompleteDismissal();
454 }

456 private void CompleteEntrance()
457 {
458 SetPosition(TargetBounds.X, TargetBounds.Y);
459 SetOpacity(1f);
460 CurrentState = ToastState.Holding;
461 _phaseElapsedSec = 0f;
462
463 if (HoldDurationSec == 0f)
464 Dismiss();
465 }

514 private static Rectangle GetAnchoredBounds(
515 View view,
516 Size size,
517 WidgetAnchor anchor,
518 int marginPx)
519 {
520 ArgumentNullException.ThrowIfNull(view);
521
522 if (size.Width <= 0 || size.Height <= 0)
523 throw new ArgumentOutOfRangeException(nameof(size));
524
525 if (marginPx < 0)
526 throw new ArgumentOutOfRangeException(nameof(marginPx));
527
528 Rectangle viewport = view.Viewport.TargetRectPx;
529
530 int x = anchor switch
531 {
532 WidgetAnchor.TopLeft or WidgetAnchor.CenterLeft or WidgetAnchor.BottomLeft => viewport.Left + marginPx,
533 WidgetAnchor.TopCenter or WidgetAnchor.Center or WidgetAnchor.BottomCenter => viewport.Left + (viewport.Width - size.Width) / 2,
534 WidgetAnchor.TopRight or WidgetAnchor.CenterRight or WidgetAnchor.BottomRight => viewport.Right - size.Width - marginPx,
535 _ => throw new ArgumentOutOfRangeException(nameof(anchor), anchor, null)
536 };
537
538 int y = anchor switch
539 {
540 WidgetAnchor.TopLeft or WidgetAnchor.TopCenter or WidgetAnchor.TopRight => viewport.Top + marginPx,
541 WidgetAnchor.CenterLeft or WidgetAnchor.Center or WidgetAnchor.CenterRight => viewport.Top + (viewport.Height - size.Height) / 2,
542 WidgetAnchor.BottomLeft or WidgetAnchor.BottomCenter or WidgetAnchor.BottomRight => viewport.Bottom - size.Height - marginPx,
543 _ => throw new ArgumentOutOfRangeException(nameof(anchor), anchor, null)
544 };
545
546 return new Rectangle(x, y, size.Width, size.Height);
547 }
