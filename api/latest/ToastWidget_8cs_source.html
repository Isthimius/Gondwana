Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

444 private void AdvanceTransition(float elapsedSec, bool entering)
445 {
446 _phaseElapsedSec += elapsedSec;
447
448 float progress = TransitionDurationSec <= 0f
449 ? 1f
450 : Math.Clamp(_phaseElapsedSec / TransitionDurationSec, 0f, 1f);
451
452 Func<float, float> easing = EasingFunctions.From(
453 entering ? EntranceEasing : ExitEasing);
454
455 float eased = Math.Clamp(easing(progress), 0f, 1f);
456
457 SetPosition(Vector2.Lerp(
458 _phaseStartPosition,
459 _phaseEndPosition,
460 eased));
461
462 SetOpacity(
463 _phaseStartOpacity
464 + (_phaseEndOpacity - _phaseStartOpacity) * eased);
465
466 if (progress < 1f)
467 return;
468
469 if (entering)
470 CompleteEntrance();
471 else
472 CompleteDismissal();
473 }

475 private void CompleteEntrance()
476 {
477 SetPosition(TargetBounds.X, TargetBounds.Y);
478 SetOpacity(1f);
479 CurrentState = ToastState.Holding;
480 _phaseElapsedSec = 0f;
481
482 if (HoldDurationSec == 0f)
483 Dismiss();
484 }

533 private static Rectangle GetAnchoredBounds(
534 View view,
535 Size size,
536 WidgetAnchor anchor,
537 int marginPx)
538 {
539 ArgumentNullException.ThrowIfNull(view);
540
541 if (size.Width <= 0 || size.Height <= 0)
542 throw new ArgumentOutOfRangeException(nameof(size));
543
544 if (marginPx < 0)
545 throw new ArgumentOutOfRangeException(nameof(marginPx));
546
547 Rectangle viewport = view.Viewport.TargetRectPx;
548
549 int x = anchor switch
550 {
551 WidgetAnchor.TopLeft or WidgetAnchor.CenterLeft or WidgetAnchor.BottomLeft => viewport.Left + marginPx,
552 WidgetAnchor.TopCenter or WidgetAnchor.Center or WidgetAnchor.BottomCenter => viewport.Left + (viewport.Width - size.Width) / 2,
553 WidgetAnchor.TopRight or WidgetAnchor.CenterRight or WidgetAnchor.BottomRight => viewport.Right - size.Width - marginPx,
554 _ => throw new ArgumentOutOfRangeException(nameof(anchor), anchor, null)
555 };
556
557 int y = anchor switch
558 {
559 WidgetAnchor.TopLeft or WidgetAnchor.TopCenter or WidgetAnchor.TopRight => viewport.Top + marginPx,
560 WidgetAnchor.CenterLeft or WidgetAnchor.Center or WidgetAnchor.CenterRight => viewport.Top + (viewport.Height - size.Height) / 2,
561 WidgetAnchor.BottomLeft or WidgetAnchor.BottomCenter or WidgetAnchor.BottomRight => viewport.Bottom - size.Height - marginPx,
562 _ => throw new ArgumentOutOfRangeException(nameof(anchor), anchor, null)
563 };
564
565 return new Rectangle(x, y, size.Width, size.Height);
566 }
