Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

237 internal Frame AdvanceFrame()
238 {
239 switch (SequenceCycleType)
240 {
241 case CycleType.PingPong:
242 currentFrameIdx += curFrameIncrement;
243 if ((currentFrameIdx <= 0) || (currentFrameIdx >= frameList.Count - 1))
244 curFrameIncrement *= -1;
245
246 if (currentFrameIdx < 0)
247 currentFrameIdx = 0;
248
249 if (currentFrameIdx > frameList.Count - 1)
250 currentFrameIdx = frameList.Count - 1;
251
252 cycleFinished = false;
253 break;
254
255 case CycleType.Repeating:
256 if (++currentFrameIdx >= frameList.Count)
257 currentFrameIdx = 0;
258
259 cycleFinished = false;
260 break;
261
262 case CycleType.Simple:
263 if (++currentFrameIdx > frameList.Count - 1)
264 {
265 currentFrameIdx = frameList.Count - 1;
266 cycleFinished = true;
267 }
268 else
269 cycleFinished = false;
270 break;
271
272 default:
273 throw new InvalidOperationException("Invalid CycleType: " + SequenceCycleType.ToString());
274 }
275
276 return frameList[currentFrameIdx];
277 }
