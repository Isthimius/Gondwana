Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

233 internal Frame AdvanceFrame()
234 {
235 switch (SequenceCycleType)
236 {
237 case CycleType.PingPong:
238 currentFrameIdx += curFrameIncrement;
239 if ((currentFrameIdx <= 0) || (currentFrameIdx >= frameList.Count - 1))
240 curFrameIncrement *= -1;
241
242 if (currentFrameIdx < 0)
243 currentFrameIdx = 0;
244
245 if (currentFrameIdx > frameList.Count - 1)
246 currentFrameIdx = frameList.Count - 1;
247
248 cycleFinished = false;
249 break;
250
251 case CycleType.Repeating:
252 if (++currentFrameIdx >= frameList.Count)
253 currentFrameIdx = 0;
254
255 cycleFinished = false;
256 break;
257
258 case CycleType.Simple:
259 if (++currentFrameIdx > frameList.Count - 1)
260 {
261 currentFrameIdx = frameList.Count - 1;
262 cycleFinished = true;
263 }
264 else
265 cycleFinished = false;
266 break;
267
268 default:
269 throw new InvalidOperationException("Invalid CycleType: " + SequenceCycleType.ToString());
270 }
271
272 return frameList[currentFrameIdx];
273 }
