Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

104 Name = name;
105 SkBitmap = SKBitmap.Decode(file)

108 ImageFilePath = file;

304 public void ApplyMask(SKColor? maskColor = null, byte tolerance = 5)
305 {
306 if (SkBitmap == null || SkBitmap.IsEmpty)
307 throw new ArgumentException("Invalid bitmap.");
308
309 var targetColor = maskColor ?? SKColors.White;
310
311 MaskColor = targetColor;
312 MaskTolerance = tolerance;
313 Premultiplied = true;
314
315 InvalidateAtlasImage();
316 ClearTileCache();
317
318 if (SkBitmap.Info.AlphaType == SKAlphaType.Opaque)
319 {
320 var info = new SKImageInfo(
321 SkBitmap.Width,
322 SkBitmap.Height,
323 SkBitmap.Info.ColorType,
324 SKAlphaType.Premul);
325
326 var withAlpha = new SKBitmap(info);
327 using (var canvas = new SKCanvas(withAlpha))
328 canvas.DrawBitmap(SkBitmap, 0, 0);
329
330 SkBitmap.Dispose();
331 SkBitmap = withAlpha;
332 }
333
334 SkBitmapOriginal?.Dispose();
335 SkBitmapOriginal = SkBitmap.Copy();
336
337 SkBitmap.ApplyAlphaMask(targetColor, tolerance);
338 SkBitmap = SkBitmap.PremultiplyAlpha();
339
340 BuildTileCache();
341 }

390 public void PersistImageToFile(
391 string imageFilePath,
392 SKEncodedImageFormat format = SKEncodedImageFormat.Png,
393 int quality = 100)
394 {
395 if (_disposed)
396 throw new ObjectDisposedException(nameof(Tilesheet));

398 if (string.IsNullOrWhiteSpace(imageFilePath))
399 {
400 throw new ArgumentException(

402 nameof(imageFilePath));
403 }
404
405 if (quality is < 0 or > 100)

407 throw new ArgumentOutOfRangeException(

415 throw new InvalidOperationException(

417 }
418
419 var fullPath = Path.GetFullPath(imageFilePath);
420 var directory = Path.GetDirectoryName(fullPath);
421 if (!string.IsNullOrWhiteSpace(directory))
422 Directory.CreateDirectory(directory);
423
424 var sourceBitmap = SkBitmapOriginal ?? SkBitmap;
