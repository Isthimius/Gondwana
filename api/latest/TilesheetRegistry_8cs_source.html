Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

41 internal void Register(Tilesheet sheet, bool disposeReplaced = true)
42 {
43 if (sheet is null)
44 {
45 Engine.Logger.LogError("TilesheetRegistry: Attempt to register a null Tilesheet.");
46 return;
47 }
48
49 Tilesheet? replaced = null;
50
51 lock (_gate)
52 {
53 if (_sheets.TryGetValue(sheet.Name, out var existing) && !ReferenceEquals(existing, sheet))
54 {
55 _sheets[sheet.Name] = sheet;
56 replaced = existing;
57 }
58 else
59 {
60 _sheets[sheet.Name] = sheet;
61 }
62
63 // Avoid duplicate subscriptions if the same instance is registered again.
64 sheet.Disposed -= OnTilesheetDisposed;
65 sheet.Disposed += OnTilesheetDisposed;
66 }
67
68 if (replaced is not null)
69 {
70 replaced.Disposed -= OnTilesheetDisposed;
71
72 if (disposeReplaced)
73 replaced.Dispose();
74 }
75 }

310 internal void OnTilesheetRenamed(
311 string oldName,
312 string newName,
313 Tilesheet sheet,
314 bool disposeReplaced = true)
315 {
316 if (oldName is null)
317 throw new ArgumentNullException(nameof(oldName));
318
319 if (newName is null)
320 throw new ArgumentNullException(nameof(newName));
321
322 if (sheet is null)
323 throw new ArgumentNullException(nameof(sheet));
324
325 Tilesheet? replaced = null;
326
327 lock (_gate)
328 {
329 // If old mapping points to this sheet, remove it.
330 if (_sheets.TryGetValue(oldName, out var existingAtOld) && ReferenceEquals(existingAtOld, sheet))
331 _sheets.Remove(oldName);
332
333 // If something else is already at newName, warn and replace.
334 if (_sheets.TryGetValue(newName, out var existingAtNew) && !ReferenceEquals(existingAtNew, sheet))
335 {
336 Engine.Logger.LogWarning(
337 "TilesheetRegistry: Renaming '{Old}'→'{New}' replaced a different registered Tilesheet instance.",
338 oldName,
339 newName);
340
341 replaced = existingAtNew;
342 }
343
344 _sheets[newName] = sheet;
345 }
346
347 if (replaced is not null)
348 {
349 replaced.Disposed -= OnTilesheetDisposed;
350
351 if (disposeReplaced)
352 replaced.Dispose();
353 }
354 }

511 private IReadOnlyList<Tilesheet> LoadAllDefinitions(
512 TilesheetDefinitionDocument document,
513 string? directory,
514 AssetsFile? archive)
515 {
516 var errors = TilesheetDefinitionSerializer.ValidateDocument(document);
517
518 if (errors.Count != 0)
519 throw new InvalidDataException(string.Join(Environment.NewLine, errors));
520
521 var created = new List<Tilesheet>();
522
523 try
524 {
525 foreach (var definition in document.Definitions)
526 created.Add(TilesheetFactory.FromDefinition(definition, directory, archive));
527 }
528 catch
529 {
530 foreach (var sheet in created)
531 sheet.Dispose();
532
533 throw;
534 }
535
536 foreach (var sheet in created)
537 Register(sheet);
538
539 return created;
540 }
