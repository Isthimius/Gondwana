Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

147 public override void Draw(Gondwana.Rendering.Backbuffers.BackbufferBase backbuffer, RectangleF destRectScreen)
148 {
149 if (Transform == TileTransform.Identity)
150 {
151 base.Draw(backbuffer, destRectScreen);
152 return;
153 }
154
155 var origin = TileTransformGeometry.MapToDestination(new(0, 0), destRectScreen, Transform);
156 var x = TileTransformGeometry.MapToDestination(new(1, 0), destRectScreen, Transform);
157 var y = TileTransformGeometry.MapToDestination(new(0, 1), destRectScreen, Transform);
158
159 var matrix = new global::SkiaSharp.SKMatrix
160 {
161 ScaleX = x.X - origin.X,
162 SkewX = y.X - origin.X,
163 TransX = origin.X,
164 SkewY = x.Y - origin.Y,
165 ScaleY = y.Y - origin.Y,
166 TransY = origin.Y,
167 Persp2 = 1
168 };
169
170 var canvas = backbuffer.Canvas;
171 canvas.Save();
172
173 try
174 {
175 canvas.Concat(ref matrix);
176
177 base.Draw(
178 backbuffer,
179 new RectangleF(
180 0,
181 0,
182 1,
183 1));
184 }
185 finally
186 {
187 canvas.Restore();
188 }
