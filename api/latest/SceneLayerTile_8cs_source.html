Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

125 public override void Draw(Gondwana.Rendering.Backbuffers.BackbufferBase backbuffer, RectangleF destRectScreen)
126 {
127 if (Transform == TileTransform.Identity)
128 {
129 base.Draw(backbuffer, destRectScreen);
130 return;
131 }
132
133 var origin = TileTransformGeometry.MapToDestination(new(0, 0), destRectScreen, Transform);
134 var x = TileTransformGeometry.MapToDestination(new(1, 0), destRectScreen, Transform);
135 var y = TileTransformGeometry.MapToDestination(new(0, 1), destRectScreen, Transform);
136 var matrix = new global::SkiaSharp.SKMatrix
137 {
138 ScaleX = x.X - origin.X,
139 SkewX = y.X - origin.X,
140 TransX = origin.X,
141 SkewY = x.Y - origin.Y,
142 ScaleY = y.Y - origin.Y,
143 TransY = origin.Y,
144 Persp2 = 1
145 };
146 var canvas = backbuffer.Canvas;
147 canvas.Save();
148 try
149 {
150 canvas.Concat(ref matrix);
151 base.Draw(backbuffer, new RectangleF(0, 0, 1, 1));
152 }
153 finally { canvas.Restore(); }
154 }
