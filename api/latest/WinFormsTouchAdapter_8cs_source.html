Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

121 private void RefreshNativeChildren()
122 {
123 if (_disposed || !_surface.IsHandleCreated || _refreshingNativeChildren)
124 return;
125
126 _refreshingNativeChildren = true;
127 try
128 {
129 var descendants = new HashSet<IntPtr>();
130 EnumChildWindows(_surface.Handle, (hwnd, _) =>
131 {
132 descendants.Add(hwnd);
133 return true;
134 }, IntPtr.Zero);
135
136 var managedHandles = _hooks.Values
137 .Where(static hook => hook.Handle != IntPtr.Zero)
138 .Select(static hook => hook.Handle)
139 .ToHashSet();
140
141 foreach (IntPtr hwnd in _nativeChildren.Keys.ToArray())
142 {
143 if (descendants.Contains(hwnd) && !managedHandles.Contains(hwnd))
144 continue;
145
146 _nativeChildren.Remove(hwnd, out NativeChildHook? stale);
147 stale?.Dispose();
148 }
149
150 foreach (IntPtr hwnd in descendants)
151 {
152 if (managedHandles.Contains(hwnd) || _nativeChildren.ContainsKey(hwnd))
153 continue;
154
155 _nativeChildren.Add(hwnd, new NativeChildHook(this, hwnd));
156 }
157 }
158 finally
159 {
160 _refreshingNativeChildren = false;
161 }
162 }

172 private bool HandleTouchMessage(IntPtr wParam, IntPtr lParam)
173 {
174 int count = unchecked((ushort)(wParam.ToInt64() & 0xFFFF));
175 try
176 {
177 if (count == 0)
178 return true;
179
180 var inputs = new TouchInput[count];
181 if (!GetTouchInputInfo(lParam, (uint)count, inputs,
182 Marshal.SizeOf<TouchInput>()))
183 return true;
184
185 foreach (TouchInput input in inputs)
186 {
187 // TOUCHINPUT coordinates are physical screen pixels in hundredths.
188 // Convert through the surface's WinForms client space, and then
189 // through the existing presentation transform to logical ScreenPx.
190 var physical = new Point(
191 (int)Math.Round(input.X / 100d),
192 (int)Math.Round(input.Y / 100d));
193 Point local = _surface.PointToClient(physical);
194 Point position = _surface switch
195 {
196 WinFormGpuRenderSurfaceControl gpu when gpu.Host is not null =>
197 gpu.Adapter.AdapterPxToScreenPx(local),
198 WinFormBitmapRenderSurfaceControl bitmap when bitmap.Host is not null =>
199 bitmap.Adapter.AdapterPxToScreenPx(local),
200 _ => local
201 };
202
203 int id = unchecked((int)input.Id);
204 if ((input.Flags & TouchDown) != 0)
205 _state.Begin(id, position);
206 else if ((input.Flags & TouchUp) != 0)
207 _state.End(id, position);
208 else
209 _state.Move(id, position);
210 }
211
212 return true;
213 }
214 finally
215 {
216 // Once handled, WM_TOUCH must not be passed to DefWindowProc.
217 CloseTouchInputHandle(lParam);
218 }
219 }
