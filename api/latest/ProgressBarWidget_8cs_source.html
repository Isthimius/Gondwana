Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

240 private void UpdateFillBounds()
241 {
242 int innerWidth = Math.Max(0, _size.Width - _padding * 2);
243 int innerHeight = Math.Max(0, _size.Height - _padding * 2);
244
245 Size fillSize;
246 Vector2 offset;
247
248 if (_orientation == WidgetOrientation.Horizontal)
249 {
250 int fillWidth = (int)MathF.Round(innerWidth * _value);
251 fillSize = new Size(fillWidth, innerHeight);
252 offset = new Vector2(_padding, _padding);
253 }
254 else
255 {
256 int fillHeight = (int)MathF.Round(innerHeight * _value);
257 fillSize = new Size(innerWidth, fillHeight);
258 offset = new Vector2(_padding, _padding + innerHeight - fillHeight);
259 }
260
261 Rectangle current = FillBounds;
262 Rectangle resized = new(current.Location, fillSize);
263
264 if (Mode == DirectDrawingMode.View)
265 Fill.ScreenBounds = resized;
266 else
267 Fill.WorldBounds = resized;
268
269 SetLocalOffset(Fill, offset);
270 Fill.Visible = _value > 0f && Track.Visible;
271 }
