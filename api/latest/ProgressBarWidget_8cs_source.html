Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

236 private void CompleteInitialization(float value)
237 {
238 Add(Track, keepCurrentOffset: false, explicitLocalOffsetPx: Vector2.Zero);
239 Add(Fill, keepCurrentOffset: false, explicitLocalOffsetPx: new Vector2(_padding, _padding));
240
241 IsInputEnabled = false;

243 IsKeyboardInputEnabled = false;
244 CanReceiveFocus = false;
245
246 SetProgressZOrder(0);
247
248 _value = Math.Clamp(
249 float.IsFinite(value) ? value : throw new ArgumentOutOfRangeException(nameof(value)),
250 0f,
251 1f);
252
253 UpdateFillBounds();
254 }

256 private void UpdateFillBounds()
257 {
258 int innerWidth = Math.Max(0, _size.Width - _padding * 2);
259 int innerHeight = Math.Max(0, _size.Height - _padding * 2);
260
261 Size fillSize;
262 Vector2 offset;
263
264 if (_orientation == WidgetOrientation.Horizontal)
265 {
266 int fillWidth = (int)MathF.Round(innerWidth * _value);
267 fillSize = new Size(fillWidth, innerHeight);
268 offset = new Vector2(_padding, _padding);
269 }
270 else
271 {
272 int fillHeight = (int)MathF.Round(innerHeight * _value);
273 fillSize = new Size(innerWidth, fillHeight);
274 offset = new Vector2(_padding, _padding + innerHeight - fillHeight);
275 }
276
277 Rectangle current = FillBounds;
278 Rectangle resized = new(current.Location, fillSize);
279
280 if (Mode == DirectDrawingMode.View)
281 Fill.ScreenBounds = resized;
282 else
283 Fill.WorldBounds = resized;
284
285 SetLocalOffset(Fill, offset);
286 Fill.Visible = _value > 0f && Track.Visible;
287 }
