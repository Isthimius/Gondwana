Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

220 private void UpdateFillBounds()
221 {
222 int innerWidth = Math.Max(0, _size.Width - _padding * 2);
223 int innerHeight = Math.Max(0, _size.Height - _padding * 2);
224
225 Size fillSize;
226 Vector2 offset;
227
228 if (_orientation == WidgetOrientation.Horizontal)
229 {
230 int fillWidth = (int)MathF.Round(innerWidth * _value);
231 fillSize = new Size(fillWidth, innerHeight);
232 offset = new Vector2(_padding, _padding);
233 }
234 else
235 {
236 int fillHeight = (int)MathF.Round(innerHeight * _value);
237 fillSize = new Size(innerWidth, fillHeight);
238 offset = new Vector2(_padding, _padding + innerHeight - fillHeight);
239 }
240
241 Rectangle current = FillBounds;
242 Rectangle resized = new(current.Location, fillSize);
243
244 if (Mode == DirectDrawingMode.View)
245 Fill.ScreenBounds = resized;
246 else
247 Fill.WorldBounds = resized;
248
249 SetLocalOffset(Fill, offset);
250 Fill.Visible = _value > 0f && Track.Visible;
251 }
