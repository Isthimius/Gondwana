376 public void SaveToFile(string path, EngineStateSaveOptions options)
377 {
378 if (string.IsNullOrWhiteSpace(path))
379 throw new ArgumentException("Engine state path must be a non-empty string.", nameof(path));
380
381 var fullPath = Path.GetFullPath(path);
382 var baseDirectory = Path.GetDirectoryName(fullPath);
383
384 ArgumentNullException.ThrowIfNull(options);
385 foreach (var mode in new[] { options.Tilesheets, options.Cycles, options.Scenes, options.Audio, options.Sprites })
386 if (!Enum.IsDefined(mode)) throw new ArgumentOutOfRangeException(nameof(options));
387 var parts = options.Parts;
388 if (parts.HasFlag(EngineStateParts.Sprites) && options.Sprites == DefinitionPersistence.PreserveSource)
389 parts |= EngineStateParts.Scenes | EngineStateParts.Cycles | EngineStateParts.Tilesheets;
390 if (parts.HasFlag(EngineStateParts.Scenes) && options.Scenes == DefinitionPersistence.PreserveSource)
391 parts |= EngineStateParts.Cycles | EngineStateParts.Tilesheets;
392 var snapshot = BuildSnapshot(NormalizeParts(parts), baseDirectory, fullPath, options);
393
394 var json = JsonConvert.SerializeObject(snapshot, JsonSerializerSettings);
395
396 if (options.Compress)
397 {
398 using var file = File.Create(fullPath);
399 using var zip = new GZipStream(file, CompressionMode.Compress);
400 using var writer = new StreamWriter(zip);
401 writer.Write(json);
402 }
403 else
404 {
405 File.WriteAllText(fullPath, json);
406 }
407 }

508 private static EngineStateParts NormalizeParts(EngineStateParts parts, EngineStateSnapshot? snapshot = null)
509 {
510 // Expand content dependencies for packed definitions without changing legacy
511 // sprite-only and scene-only merges that deliberately use existing registries.
512 if (parts.HasFlag(EngineStateParts.Sprites) && snapshot?.Sprites?.AssetsFilePath is not null)
513 {
514 if (snapshot.Scenes is not null) parts |= EngineStateParts.Scenes;
515 if (snapshot.Cycles is not null) parts |= EngineStateParts.Cycles;
516 if (snapshot.Tilesheets is not null) parts |= EngineStateParts.Tilesheets;
517 }
518 if (parts.HasFlag(EngineStateParts.Scenes) && snapshot?.Scenes?.Any(entry => entry?.AssetsFilePath is not null) == true)
519 {
520 if (snapshot.Cycles is not null) parts |= EngineStateParts.Cycles;
521 if (snapshot.Tilesheets is not null) parts |= EngineStateParts.Tilesheets;
522 }
523
524 // GANI definitions resolve Frame references through the TilesheetRegistry.
525 if (parts.HasFlag(EngineStateParts.Cycles))
526 parts |= EngineStateParts.Tilesheets;
527
528 // Tilesheets and Audio may depend on AssetsFiles for AssetIdentifier.Data.
529 if (parts.HasFlag(EngineStateParts.Tilesheets) ||
530 parts.HasFlag(EngineStateParts.Audio) ||
531 (parts.HasFlag(EngineStateParts.Scenes) && snapshot?.Scenes?.Any(entry => entry?.AssetsFilePath is not null) == true) ||
532 (parts.HasFlag(EngineStateParts.Sprites) && snapshot?.Sprites?.AssetsFilePath is not null))
533 {
534 parts |= EngineStateParts.AssetsFiles;
535 }
536
537 return parts;
538 }

540 private EngineStateSnapshot BuildSnapshot(EngineStateParts parts,
541 string? baseDirectory,
542 string engineStatePath,
543 EngineStateSaveOptions options)
544 {
545 return new EngineStateSnapshot
546 {
547 AssetsFiles = parts.HasFlag(EngineStateParts.AssetsFiles)
548 ? AssetsFiles.Where(file => !string.IsNullOrWhiteSpace(file.SourcePath))
549 .Select(file => new AssetsFileStateEntry
550 {
551 FilePath = MakeRelativePath(file.SourcePath!, baseDirectory),
552 Password = file.Password,
553 UseEncryption = file.UseEncryption
554 }).ToList()
555 : null,
556
557 Tilesheets = parts.HasFlag(EngineStateParts.Tilesheets)
558 ? CaptureTilesheetEntries(
559 baseDirectory,
560 engineStatePath,
561 options.Tilesheets)
562 : null,
563
564 Cycles = parts.HasFlag(EngineStateParts.Cycles)
565 ? CaptureAnimationEntries(
566 baseDirectory,
567 engineStatePath,
568 options.Cycles)
569 : null,
570
571 Scenes = parts.HasFlag(EngineStateParts.Scenes)
572 ? CaptureSceneEntries(
573 baseDirectory,
574 engineStatePath,
575 options.Scenes)
576 : null,
577
578 Sprites = parts.HasFlag(EngineStateParts.Sprites)
579 ? CaptureSpriteEntry(baseDirectory, engineStatePath, options.Sprites)
580 : null,
581
582 Audio = parts.HasFlag(EngineStateParts.Audio)
583 ? CaptureAudioEntry(
584 baseDirectory,
585 engineStatePath,
586 options.Audio)
587 : null,
588
589 // New saves use the clean GSND definition shape. This member remains
590 // only so older EngineState files can still be read.
591 SoundResources = null,
592 };

608 return new() { GsprPath = MakeRelativePath(path, baseDirectory) };

611 private static AudioStateEntry CaptureAudioEntry(
612 string? baseDirectory,
613 string engineStatePath,
614 DefinitionPersistence persistence)
615 {
616 var entry = new AudioStateEntry();
617 var currentAudio = AudioDefinitionSerializer.FromManager(AudioResourceManager.Instance);
618 if (TryPreserve(entry, AudioResourceManager.Instance.DefinitionProvenance,
619 currentAudio, persistence, baseDirectory, out var loosePath))
620 {
621 entry.GsndPath = loosePath;
622 return entry;
623 }
624 // Anonymous stream packages cannot be named in a reloadable GSND. Export their
625 // audio payloads to loose files; do not invent or mutate an archive.
626 var anonymousAudio = currentAudio.Resources.Where(resource =>
627 resource.SourceKind == AudioResourceSourceKind.PackedAsset &&
628 string.IsNullOrWhiteSpace(resource.AssetsFilePath)).ToList();
629 if (anonymousAudio.Count != 0)
630 {
631 var directory = GetAudioStateDirectory(engineStatePath);
632 Directory.CreateDirectory(directory);
633 for (int i = 0; i < anonymousAudio.Count; i++)
634 {
635 var resource = anonymousAudio[i];
636 var runtime = AudioResourceManager.Instance.Get(resource.Key)!;
637 var extension = Path.GetExtension(runtime.SourceExtension ?? ".wav");
638 var mediaPath = Path.Combine(directory, $"{i}-{SanitizeFileName(resource.Key)}{extension}");
639 using (var source = runtime.AssetIdentifier!.Data)
640 using (var output = File.Create(mediaPath))
641 source.CopyTo(output);
642 resource.SourceKind = AudioResourceSourceKind.LooseFile;
643 resource.FilePath = mediaPath;
644 resource.AssetsFilePath = null;
645 resource.AssetEntryName = null;
646 }
647 var definitionPath = Path.Combine(directory, "audio.gsnd");
648 AudioDefinitionSerializer.Save(definitionPath, currentAudio);
649 return new AudioStateEntry { GsndPath = MakeRelativePath(definitionPath, baseDirectory) };
650 }
651 if (persistence == DefinitionPersistence.Loose)
652 {
653 var gsndDirectory = GetAudioStateDirectory(engineStatePath);
654 Directory.CreateDirectory(gsndDirectory);
655
656 var gsndFullPath = Path.Combine(gsndDirectory, "audio.gsnd");
657 AudioDefinitionSerializer.Save(
658 gsndFullPath,
659 AudioResourceManager.Instance);
660
661 return new AudioStateEntry
662 {
663 GsndPath = MakeRelativePath(gsndFullPath, baseDirectory)
664 };
665 }
666
667 return new AudioStateEntry
668 {
669 Definition = AudioDefinitionSerializer.FromManager(
670 AudioResourceManager.Instance,
671 baseDirectory,
672 makePathsRelative: !string.IsNullOrWhiteSpace(baseDirectory))
673 };
674 }

676 private static Dictionary<string, AnimationStateEntry> CaptureAnimationEntries(
677 string? baseDirectory,
678 string engineStatePath,
679 DefinitionPersistence persistence)
680 {
681 var result = new Dictionary<string, AnimationStateEntry>(StringComparer.Ordinal);
682 var usedFileNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
683
684 // Work from a stable registry snapshot in case definitions are changed while
685 // the EngineState snapshot is being assembled.
686 foreach (var (key, cycle) in Cycle._cycles.ToList())
687 {
688 if (cycle is null)
689 continue;
690
691 var preserved = new AnimationStateEntry();
692 if (cycle.DefinitionProvenance is { } provenance &&
693 TryPreserve(preserved, provenance, AnimationDefinitionSerializer.FromCycle(cycle), persistence, baseDirectory, out var loosePath))
694 {
695 preserved.GaniPath = loosePath;
696 result[key] = preserved;
697 continue;
698 }
699
700 if (persistence == DefinitionPersistence.Loose)
701 {
702 var ganiDirectory = GetAnimationStateDirectory(engineStatePath);
703 Directory.CreateDirectory(ganiDirectory);
704
705 var baseFileName = SanitizeFileName(key);
706 var ganiFileName = $"{baseFileName}.gani";
707 int suffix = 2;
708
709 while (!usedFileNames.Add(ganiFileName))
710 ganiFileName = $"{baseFileName}_{suffix++}.gani";
711
712 var ganiFullPath = Path.Combine(ganiDirectory, ganiFileName);
713
714 // Use the standalone serializer so .gani files do not acquire
715 // EngineState-specific $id/$ref metadata.
716 AnimationDefinitionSerializer.Save(ganiFullPath, cycle);
717
718 result[key] = new AnimationStateEntry
719 {
720 GaniPath = MakeRelativePath(ganiFullPath, baseDirectory)
721 };
722 }
723 else
724 {
725 result[key] = new AnimationStateEntry
726 {
727 Definition = AnimationDefinitionSerializer.FromCycle(cycle)
728 };
729 }
730 }
731
732 return result;
733 }

736 string? baseDirectory,
737 string engineStatePath,
738 DefinitionPersistence persistence)
739 {
740 var result = new List<SceneStateEntry>();
741 var usedFileNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
742
743 // Work from a stable snapshot. Scene construction/disposal may occur on
744 // other threads while a state file is being assembled.
745 foreach (var scene in Scene._allScenes.ToList())
746 {
747 if (scene is null)
748 continue;
749
750 var preserved = new SceneStateEntry();
751 if (scene.DefinitionProvenance is { } provenance &&
752 TryPreserve(preserved, provenance, SceneDefinitionSerializer.FromScene(scene), persistence, baseDirectory, out var loosePath))
753 {
754 preserved.GscnPath = loosePath;
755 result.Add(preserved);
756 continue;
757 }
758
759 if (persistence == DefinitionPersistence.Loose)
760 {
761 var gscnDirectory = GetSceneStateDirectory(engineStatePath);
762 Directory.CreateDirectory(gscnDirectory);
763
764 var baseFileName = SanitizeFileName(scene.ID);
765 var gscnFileName = $"{baseFileName}.gscn";
766 int suffix = 2;
767

769 gscnFileName = $"{baseFileName}_{suffix++}.gscn";
770
771 var gscnFullPath = Path.Combine(gscnDirectory, gscnFileName);
772
773 // Use the standalone GSCN serializer so the external file remains a
774 // clean scene definition without EngineState $id/$ref metadata.
775 SceneDefinitionSerializer.Save(gscnFullPath, scene);
776
777 result.Add(new SceneStateEntry
778 {
779 GscnPath = MakeRelativePath(
780 gscnFullPath,
781 baseDirectory)
782 });
783 }
784 else
785 {
786 result.Add(new SceneStateEntry

789 });
790 }
791 }

815 private static void ApplySnapshot(
816 EngineStateSnapshot snapshot,
817 bool clearExisting,
818 bool overwriteExisting,
819 EngineStateParts parts,
820 string? baseDirectory)
821 {
822 parts = NormalizeParts(parts, snapshot);
823
824 // Legacy raw objects may register themselves while the snapshot DTO is
825 // deserialized. Detach those incoming instances before clearing or merging so
826 // they are treated as snapshot data rather than pre-existing live state.
827 DetachLegacySnapshotScenes(snapshot.Scenes);
828 DetachSnapshotSprites(snapshot.Sprites?.LegacySprites);
829
830 ValidateSelectedSources(snapshot, parts);
831
832 // clear only what we're about to load.
833 if (clearExisting)
834 ClearSelected(parts);
835
836 if (parts.HasFlag(EngineStateParts.AssetsFiles))
837 LoadAssetsFiles(snapshot.AssetsFiles ?? Enumerable.Empty<AssetsFileStateEntry>(), overwriteExisting, baseDirectory, PackedEntries(snapshot, parts));
838
839 if (parts.HasFlag(EngineStateParts.Audio))
840 MergeAudio(
841 snapshot.AssetsFiles?.Select(raw => FindAssetsFile(ResolvePath(raw.FilePath, baseDirectory)))
842 .OfType<AssetsFile>().ToList(),
843 snapshot.Audio,
844 snapshot.SoundResources,
845 overwriteExisting,
846 baseDirectory);
847
848 if (parts.HasFlag(EngineStateParts.Tilesheets))
849 MergeTilesheets(snapshot.Tilesheets, overwriteExisting, baseDirectory);
850
851 if (parts.HasFlag(EngineStateParts.Cycles))
852 MergeAnimations(snapshot.Cycles, overwriteExisting, baseDirectory);
853
854 if (parts.HasFlag(EngineStateParts.Scenes))
855 MergeScenes(snapshot.Scenes, overwriteExisting, baseDirectory);
856
857 if (parts.HasFlag(EngineStateParts.Sprites))
858 {
859 var entry = snapshot.Sprites;
860 if (entry is not null)
861 {
862 var incoming = entry.LegacySprites;
863 if (incoming is null)
864 {
865 var definition = entry.AssetsFilePath is not null
866 ? LoadPacked(entry, baseDirectory, AssetTypes.SpriteDefinition, SpriteDefinitionSerializer.Load)
867 : !string.IsNullOrWhiteSpace(entry.GsprPath)
868 ? SpriteDefinitionSerializer.Load(ResolvePath(entry.GsprPath, baseDirectory))
869 : entry.Definition ?? throw new InvalidDataException("Sprite state entry requires GsprPath or Definition.");
870 incoming = WithDefinitionContext(entry, baseDirectory,
871 () => SpriteDefinitionSerializer.ToSprites(definition, allowDuplicateNicknames: true));
872 DetachSnapshotSprites(incoming);
873 }
874 MergeSprites(incoming, overwriteExisting);
875 }
876 }

880 {
881 if (parts.HasFlag(EngineStateParts.AssetsFiles))
882 {
883 AssetsFile.ClearAll();
884 SvgResourceManager.Instance.Clear();
885 }
886
887 if (parts.HasFlag(EngineStateParts.Tilesheets))

889
890 if (parts.HasFlag(EngineStateParts.Cycles))
891 Cycle.ClearAllAnimationCycles();
892
893 if (parts.HasFlag(EngineStateParts.Scenes))
894 Scene.ClearAllScenes();
895
896 if (parts.HasFlag(EngineStateParts.Sprites))
897 {
898 SpriteManager.Instance.DefinitionProvenance = null;

903 sprite.DisposeImmediate();
904 }
905 }
906
907 if (parts.HasFlag(EngineStateParts.Audio))
908 AudioResourceManager.Instance.Dispose();
909 }

1232 private static void MergeAnimations(
1233 Dictionary<string, AnimationStateEntry>? animations,
1234 bool overwriteExisting,
1235 string? baseDirectory)
1236 {
1237 if (animations is null || animations.Count == 0)
1238 return;
1239
1240 var definitions = new List<AnimationDefinition>();
1241 var sourceEntries = new Dictionary<AnimationDefinition, AnimationStateEntry>();
1242
1243 foreach (var (key, entry) in animations)
1244 {
1245 if (entry is null)
1246 continue;
1247
1248 if (!overwriteExisting && Cycle._cycles.ContainsKey(key))
1249 continue;
1250
1251 AnimationDefinition definition;
1252
1253 if (entry.AssetsFilePath is not null)
1254 {
1255 definition = LoadPacked(entry, baseDirectory, AssetTypes.AnimationDefinition, AnimationDefinitionSerializer.Load);
1256 }
1257 else if (!string.IsNullOrWhiteSpace(entry.GaniPath))
1258 {
1259 var ganiPath = ResolvePath(entry.GaniPath, baseDirectory);
1260 definition = AnimationDefinitionSerializer.Load(ganiPath);
1261 }
1262 else if (entry.Definition is not null)
1263 {
1264 definition = entry.Definition;
1265 }
1266 else
1267 {
1268 throw new InvalidDataException(
1269 $"Animation state entry '{key}' does not contain a GANI path or inline definition.");
1270 }
1271
1272 if (string.IsNullOrWhiteSpace(definition.Key))
1273 {
1274 definition.Key = key;
1275 }
1276 else if (!string.Equals(definition.Key, key, StringComparison.Ordinal))
1277 {
1278 throw DefinitionError(entry, baseDirectory, new InvalidDataException(
1279 $"Animation state entry '{key}' contains definition key '{definition.Key}'."));
1280 }
1281
1282 definitions.Add(definition);
1283 sourceEntries.Add(definition, entry);
1284 }
1285
1286 // Validate next-cycle identities before mutating the registry. Two-phase
1287 // materialization allows circular transitions such as idle -> blink -> idle.
1288 var availableKeys = new HashSet<string>(
1289 Cycle._cycles.Keys,
1290 StringComparer.Ordinal);
1291
1292 foreach (var definition in definitions)
1293 availableKeys.Add(definition.Key);
1294
1295 foreach (var definition in definitions)
1296 {
1297 if (!string.IsNullOrWhiteSpace(definition.NextCycleKey) &&
1298 !string.Equals(
1299 definition.NextCycleKey,
1300 definition.Key,
1301 StringComparison.Ordinal) &&
1302 !availableKeys.Contains(definition.NextCycleKey))
1303 {
1304 throw DefinitionError(sourceEntries[definition], baseDirectory, new InvalidDataException(
1305 $"GANI animation '{definition.Key}' references next cycle '{definition.NextCycleKey}', but no matching animation is available."));
1306 }
1307 }
1308
1309 var materialized = new List<(Cycle Cycle, AnimationDefinition Definition)>();
1310
1311 foreach (var definition in definitions)
1312 {
1313 var cycle = WithDefinitionContext(sourceEntries[definition], baseDirectory,
1314 () => AnimationDefinitionSerializer.MaterializeCycle(definition));
1315 materialized.Add((cycle, definition));
1316 }
1317
1318 foreach (var (cycle, definition) in materialized)
1319 WithDefinitionContext(sourceEntries[definition], baseDirectory, () =>
1320 {
1321 AnimationDefinitionSerializer.ApplyNextCycle(cycle, definition);
1322 return cycle;
1323 });
1324 }

1338 private static void MergeScenes(
1339 List<SceneStateEntry>? scenes,
1340 bool overwriteExisting,
1341 string? baseDirectory)
1342 {
1343 if (scenes is null || scenes.Count == 0)
1344 return;
1345
1346 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
1347 for (int i = 0; i < Scene._allScenes.Count; i++)
1348 {
1349 var id = Scene._allScenes[i].ID;
1350 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
1351 existingIndexById.Add(id, i);
1352 }
1353
1354 // Avoid duplicating the same incoming ID twice. A duplicate within one
1355 // snapshot is treated as an overwrite so the last incoming entry wins.
1356 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
1357
1358 foreach (var entry in scenes)
1359 {
1360 if (entry is null)
1361 continue;
1362
1363 Scene incoming;
1364 bool materializedFromDefinition = false;
1365
1366 if (entry.LegacyScene is not null)
1367 {
1368 incoming = entry.LegacyScene;
1369 }
1370 else if (entry.AssetsFilePath is not null)
1371 {
1372 incoming = LoadPacked(entry, baseDirectory, AssetTypes.SceneDefinition, SceneDefinitionSerializer.LoadScene);
1373 materializedFromDefinition = true;
1374 }
1375 else if (!string.IsNullOrWhiteSpace(entry.GscnPath))
1376 {
1377 var gscnPath = ResolvePath(entry.GscnPath, baseDirectory);
1378 incoming = SceneDefinitionSerializer.LoadScene(gscnPath);
1379 materializedFromDefinition = true;
1380 }
1381 else if (entry.Definition is not null)
1382 {
1383 incoming = SceneDefinitionSerializer.ToScene(entry.Definition);
1384 materializedFromDefinition = true;
1385 }
1386 else
1387 {
1388 throw new InvalidDataException(
1389 "Scene state entry does not contain a GSCN path or inline definition.");
1390 }
1391
1392 // Scene materialization registers runtime scenes immediately. Detach newly
1393 // materialized definitions so this method remains the single merge owner.
1394 if (materializedFromDefinition)
1395 Scene._allScenes.Remove(incoming);
1396
1397 if (string.IsNullOrWhiteSpace(incoming.ID))
1398 incoming.ID = Guid.NewGuid().ToString();
1399
1400 bool duplicateIncoming = !seenIncoming.Add(incoming.ID);
1401
1402 if (existingIndexById.TryGetValue(incoming.ID, out int existingIndex))
1403 {
1404 if (!overwriteExisting && !duplicateIncoming)
1405 {
1406 if (materializedFromDefinition)
1407 incoming.Dispose();
1408
1409 continue;
1410 }
1411
1412 Scene._allScenes[existingIndex] = incoming;
1413 }
1414 else
1415 {
1416 existingIndexById[incoming.ID] = Scene._allScenes.Count;
1417 Scene._allScenes.Add(incoming);
1418 }
1419 }

1422 private static void RebindSpriteSceneLayer(Sprite sprite)
1423 {
1424 var layerId = sprite.SerializedSceneLayerIdForBinding;
1425 if (string.IsNullOrWhiteSpace(layerId))
1426 {
1427 if (!ReferenceEquals(sprite.SceneLayer, SceneLayer.Empty))
1428 return;
1429
1430 throw new InvalidDataException(
1431 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' does not contain a SceneLayer identity.");
1432 }
1433
1434 SceneLayer? targetLayer = null;
1435 var sceneId = sprite.SerializedSceneIdForBinding;
1436
1437 if (!string.IsNullOrWhiteSpace(sceneId))
1438 {
1439 var scene = Scene._allScenes.FirstOrDefault(
1440 candidate => string.Equals(candidate.ID, sceneId, StringComparison.Ordinal));
1441
1442 targetLayer = scene?.GetSceneLayerByID(layerId);
1443 }
1444 else
1445 {
1446 // Older data may have a layer ID but no scene ID. Layer IDs are normally
1447 // GUIDs; accept a unique match and reject ambiguity.
1448 var matches = Scene._allScenes
1449 .SelectMany(scene => scene.SceneLayers)
1450 .Where(layer => string.Equals(layer.ID, layerId, StringComparison.Ordinal))
1451 .Take(2)
1452 .ToList();
1453
1454 if (matches.Count == 1)
1455 targetLayer = matches[0];
1456 else if (matches.Count > 1)
1457 throw new InvalidDataException(
1458 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' references ambiguous SceneLayer ID '{layerId}'.");
1459 }
1460
1461 if (targetLayer is null)
1462 {
1463 throw new InvalidDataException(
1464 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' could not resolve SceneLayer '{layerId}'" +
1465 (string.IsNullOrWhiteSpace(sceneId) ? "." : $" in Scene '{sceneId}'."));
1466 }
1467
1468 sprite.RebindSceneLayerAfterDeserialization(targetLayer);
