77 {
78 public override SceneStateEntry? ReadJson(
79 JsonReader reader,
80 Type objectType,
81 SceneStateEntry? existingValue,
82 bool hasExistingValue,
83 JsonSerializer serializer)
84 {
85 if (reader.TokenType == JsonToken.Null)
86 return null;
87
88 var obj = JObject.Load(reader);
89
90 if (obj.Property(nameof(SceneStateEntry.GscnPath), StringComparison.OrdinalIgnoreCase) is not null ||
91 obj.Property(nameof(SceneStateEntry.Definition), StringComparison.OrdinalIgnoreCase) is not null)
92 {
93 return new SceneStateEntry
94 {
95 GscnPath = obj.GetValue(
96 nameof(SceneStateEntry.GscnPath),
97 StringComparison.OrdinalIgnoreCase)?.ToObject<string>(serializer),
98 Definition = obj.GetValue(
99 nameof(SceneStateEntry.Definition),
100 StringComparison.OrdinalIgnoreCase)?.ToObject<SceneDefinition>(serializer)
101 };
102 }
103
104 // Backward compatibility: pre-GSCN EngineState files stored Scene objects
105 // directly in the Scenes list. Keep the deserialized instance intact so any
106 // preserved references from other legacy objects still resolve to it.
107 var legacyScene = obj.ToObject<Scene>(serializer)
108 ?? throw new JsonSerializationException(
109 "Legacy EngineState scene entry deserialized to null.");
110
111 return new SceneStateEntry
112 {
113 LegacyScene = legacyScene
114 };

119 SceneStateEntry? value,

452 private static List<SceneStateEntry> CaptureSceneEntries(
453 string? baseDirectory,
454 string engineStatePath,
455 bool separateGscnFiles)
456 {
457 var result = new List<SceneStateEntry>();
458 var usedFileNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
459
460 // Work from a stable snapshot. Scene construction/disposal may occur on
461 // other threads while a state file is being assembled.
462 foreach (var scene in Scene._allScenes.ToList())
463 {
464 if (scene is null)
465 continue;
466
467 if (separateGscnFiles)
468 {
469 var gscnDirectory = GetSceneStateDirectory(engineStatePath);
470 Directory.CreateDirectory(gscnDirectory);
471
472 var baseFileName = SanitizeFileName(scene.ID);
473 var gscnFileName = $"{baseFileName}.gscn";
474 int suffix = 2;
475
476 while (!usedFileNames.Add(gscnFileName))
477 gscnFileName = $"{baseFileName}_{suffix++}.gscn";
478
479 var gscnFullPath = Path.Combine(gscnDirectory, gscnFileName);
480
481 // Use the standalone GSCN serializer so the external file remains a
482 // clean scene definition without EngineState $id/$ref metadata.
483 SceneDefinitionSerializer.Save(gscnFullPath, scene);
484
485 result.Add(new SceneStateEntry
486 {
487 GscnPath = MakeRelativePath(
488 gscnFullPath,
489 baseDirectory)
490 });
491 }
492 else
493 {
494 result.Add(new SceneStateEntry
495 {
496 Definition = SceneDefinitionSerializer.FromScene(scene)
497 });
498 }
499 }
500
501 return result;
502 }

523 private static void ApplySnapshot(
524 EngineStateSnapshot snapshot,
525 bool clearExisting,
526 bool overwriteExisting,
527 EngineStateParts parts,
528 string? baseDirectory)
529 {
530 parts = NormalizeParts(parts);
531
532 // Legacy raw objects may register themselves while the snapshot DTO is
533 // deserialized. Detach those incoming instances before clearing or merging so
534 // they are treated as snapshot data rather than pre-existing live state.
535 DetachLegacySnapshotScenes(snapshot.Scenes);
536 DetachSnapshotSprites(snapshot.Sprites);
537
538 // clear only what we're about to load.
539 if (clearExisting)
540 ClearSelected(parts);
541
542 if (parts.HasFlag(EngineStateParts.AssetsFiles))
543 LoadAssetsFiles(snapshot.AssetsFiles ?? Enumerable.Empty<AssetsFile>(), overwriteExisting);
544
545 if (parts.HasFlag(EngineStateParts.Audio))
546 MergeAudio(snapshot.AssetsFiles, snapshot.SoundResources, overwriteExisting);
547
548 if (parts.HasFlag(EngineStateParts.Tilesheets))
549 MergeTilesheets(snapshot.Tilesheets, overwriteExisting, baseDirectory);
550
551 if (parts.HasFlag(EngineStateParts.Cycles))
552 MergeCycles(snapshot.Cycles, overwriteExisting);
553
554 if (parts.HasFlag(EngineStateParts.Scenes))
555 MergeScenes(snapshot.Scenes, overwriteExisting, baseDirectory);
556
557 if (parts.HasFlag(EngineStateParts.Sprites))
558 MergeSprites(snapshot.Sprites, overwriteExisting);
559 }

585 private static void LoadAssetsFiles(IEnumerable<AssetsFile> resourceFiles, bool overwriteExisting)
586 {
587 // Replace raw deserialized resource files with proper loaded instances
588 if (resourceFiles.Any())
589 {
590 foreach (var raw in resourceFiles)
591 {
592 try
593 {
594 var loaded = AssetsFile.LoadOrCreate(raw.FilePath, raw.Password, raw.UseEncryption);
595
596 if (overwriteExisting)
597 {
598 foreach (var entry in loaded.GetAllEntries().Where(e => e.AssetType == AssetTypes.Svg))
599 SvgResourceManager.Instance.Unload(entry.AssetName);
600 }
601
602 SvgResourceManager.Instance.LoadFromEngineAssetsFile(loaded);
603 }
604 catch (Exception ex)
605 {
606 Engine.Logger.LogError(ex, "Failed to load resource file '{FilePath}'", raw.FilePath);
607 throw;
608 }
609 }
610 }

613 private static void MergeAudio(
614 List<AssetsFile>? assetsFiles,
615 Dictionary<string, AudioResource>? soundSpecs,
616 bool overwriteExisting)
617 {
618 // 1) Load from asset packs
619 if (assetsFiles is not null)
620 {
621 foreach (var af in assetsFiles)
622 {
623 if (overwriteExisting)
624 {
625 foreach (var entry in af.GetAllEntries())
626 {
627 if (entry.AssetType == AssetTypes.Audio)
628 AudioResourceManager.Instance.Unload(entry.AssetName);
629 }
630 }
631
632 AudioResourceManager.Instance.LoadFromEngineAssetsFile(af);
633 }
634 }
635
636 // 2) Apply loose-file specs / overrides
637 if (soundSpecs is null)
638 return;
639
640 foreach (var (key, spec) in soundSpecs)
641 {
642 if (AudioResourceManager.Instance.Contains(key))
643 {
644 if (!overwriteExisting)
645 {
646 var existing = AudioResourceManager.Instance.Get(key);
647 if (existing is not null)
648 {
649 existing.Volume = spec.Volume;
650 existing.Pan = spec.Pan;
651 existing.PlaybackSpeed = spec.PlaybackSpeed;
652 existing.IsLooping = spec.IsLooping;
653 }
654 continue;
655 }
656
657 AudioResourceManager.Instance.Unload(key);
658 }
659
660 // Ensure the audio spec is (re)created/registered in the manager.
661 spec.ReloadIntoManager();
662 }

678 string? baseDirectory,
679 string engineStatePath,
680 bool separateGtsFiles)
681 {
682 var result = new Dictionary<string, TilesheetStateEntry>(StringComparer.Ordinal);
683
684 foreach (var (key, tilesheet) in TilesheetRegistry.Instance.GetAll())

699 tilesheet,
700 makePathsRelative: true);
701
702 var gtsPathForState = MakeRelativePath(
703 gtsFullPath,
704 baseDirectory);
705
706 result[key] = new TilesheetStateEntry
707 {
708 GtsPath = gtsPathForState
709 };
710 }
711 else
712 {
713 result[key] = new TilesheetStateEntry
714 {
715 Definition = TilesheetDefinitionSerializer.FromTilesheet(
716 tilesheet,
717 baseDirectory,
718 makePathsRelative: !string.IsNullOrWhiteSpace(baseDirectory))
719 };
720 }
721 }
722
723 return result;

753 private static string SanitizeFileName(string value)
754 {
755 if (string.IsNullOrWhiteSpace(value))
756 return Guid.NewGuid().ToString("N");
757
758 var invalidChars = Path.GetInvalidFileNameChars();
759 var sanitized = new string(
760 value.Select(ch => invalidChars.Contains(ch) ? '_' : ch).ToArray());
761
762 return string.IsNullOrWhiteSpace(sanitized)
763 ? Guid.NewGuid().ToString("N")
764 : sanitized;

767 private static void MergeTilesheets(
768 Dictionary<string, TilesheetStateEntry>? tilesheets,
769 bool overwriteExisting,
770 string? baseDirectory)
771 {
772 if (tilesheets is null || tilesheets.Count == 0)
773 return;
774
775 var registry = TilesheetRegistry.Instance.GetAll();
776
777 foreach (var (key, entry) in tilesheets)
778 {
779 if (entry is null)
780 continue;
781
782 var existingKey = key;
783
784 if (!overwriteExisting && registry.ContainsKey(existingKey))
785 continue;
786
787 Tilesheet rebuilt;
788
789 if (!string.IsNullOrWhiteSpace(entry.GtsPath))
790 {
791 var gtsPath = ResolvePath(entry.GtsPath, baseDirectory);
792
793 rebuilt = TilesheetFactory.FromDefinitionFile(gtsPath);
794 }
795 else if (entry.Definition is not null)
796 {
797 if (string.IsNullOrWhiteSpace(entry.Definition.Name))
798 entry.Definition.Name = key;
799
800 rebuilt = TilesheetFactory.FromDefinition(
801 entry.Definition,
802 baseDirectory);
803 }
804 else
805 {
806 throw new InvalidDataException(
807 $"Tilesheet state entry '{key}' does not contain a GTS path or inline definition.");
808 }
809
810 TilesheetRegistry.Instance.Register(
811 rebuilt,
812 disposeReplaced: overwriteExisting);
813 }

853 private static void MergeScenes(
854 List<SceneStateEntry>? scenes,
855 bool overwriteExisting,
856 string? baseDirectory)
857 {
858 if (scenes is null || scenes.Count == 0)
859 return;
860
861 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
862 for (int i = 0; i < Scene._allScenes.Count; i++)
863 {
864 var id = Scene._allScenes[i].ID;
865 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
866 existingIndexById.Add(id, i);
867 }
868
869 // Avoid duplicating the same incoming ID twice. A duplicate within one
870 // snapshot is treated as an overwrite so the last incoming entry wins.
871 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
872
873 foreach (var entry in scenes)
874 {
875 if (entry is null)
876 continue;
877
878 Scene incoming;
879 bool materializedFromDefinition = false;
880
881 if (entry.LegacyScene is not null)
882 {
883 incoming = entry.LegacyScene;
884 }
885 else if (!string.IsNullOrWhiteSpace(entry.GscnPath))
886 {
887 var gscnPath = ResolvePath(entry.GscnPath, baseDirectory);
888 incoming = SceneDefinitionSerializer.LoadScene(gscnPath);
889 materializedFromDefinition = true;
890 }
891 else if (entry.Definition is not null)
892 {
893 incoming = SceneDefinitionSerializer.ToScene(entry.Definition);
894 materializedFromDefinition = true;
895 }
896 else
897 {
898 throw new InvalidDataException(
899 "Scene state entry does not contain a GSCN path or inline definition.");
900 }
901
902 // Scene materialization registers runtime scenes immediately. Detach newly
903 // materialized definitions so this method remains the single merge owner.
904 if (materializedFromDefinition)
905 Scene._allScenes.Remove(incoming);
906
907 if (string.IsNullOrWhiteSpace(incoming.ID))
908 incoming.ID = Guid.NewGuid().ToString();
909
910 bool duplicateIncoming = !seenIncoming.Add(incoming.ID);
911
912 if (existingIndexById.TryGetValue(incoming.ID, out int existingIndex))
913 {
914 if (!overwriteExisting && !duplicateIncoming)
915 {
916 if (materializedFromDefinition)
917 incoming.Dispose();
918
919 continue;
920 }
921
922 Scene._allScenes[existingIndex] = incoming;
923 }
924 else
925 {
926 existingIndexById[incoming.ID] = Scene._allScenes.Count;
927 Scene._allScenes.Add(incoming);
928 }
929 }

932 private static void RebindSpriteSceneLayer(Sprite sprite)
933 {
934 var layerId = sprite.SerializedSceneLayerIdForBinding;
935 if (string.IsNullOrWhiteSpace(layerId))
936 {
937 if (!ReferenceEquals(sprite.SceneLayer, SceneLayer.Empty))
938 return;
939
940 throw new InvalidDataException(
941 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' does not contain a SceneLayer identity.");
942 }
943
944 SceneLayer? targetLayer = null;
945 var sceneId = sprite.SerializedSceneIdForBinding;
946
947 if (!string.IsNullOrWhiteSpace(sceneId))
948 {
949 var scene = Scene._allScenes.FirstOrDefault(
950 candidate => string.Equals(candidate.ID, sceneId, StringComparison.Ordinal));
951
952 targetLayer = scene?.GetSceneLayerByID(layerId);
953 }
954 else
955 {
956 // Older data may have a layer ID but no scene ID. Layer IDs are normally
957 // GUIDs; accept a unique match and reject ambiguity.
958 var matches = Scene._allScenes
959 .SelectMany(scene => scene.SceneLayers)
960 .Where(layer => string.Equals(layer.ID, layerId, StringComparison.Ordinal))
961 .Take(2)
962 .ToList();
963
964 if (matches.Count == 1)
965 targetLayer = matches[0];
966 else if (matches.Count > 1)
967 throw new InvalidDataException(
968 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' references ambiguous SceneLayer ID '{layerId}'.");
969 }
970
971 if (targetLayer is null)
972 {
973 throw new InvalidDataException(
974 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' could not resolve SceneLayer '{layerId}'" +
975 (string.IsNullOrWhiteSpace(sceneId) ? "." : $" in Scene '{sceneId}'."));
976 }
977
978 sprite.RebindSceneLayerAfterDeserialization(targetLayer);
979 }

999 for (int i = 0; i < SpriteManager.Instance._spriteList.Count; i++)
1000 {
1001 var id = SpriteManager.Instance._spriteList[i].Nickname;
1002 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
1003 existingIndexById.Add(id, i);
1004 }
1005
1006 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
1007
1008 foreach (var incoming in sprites)
1009 {
1010 if (incoming is null)
1011 continue;
1012
1013 RebindSpriteSceneLayer(incoming);
1014
1015 if (string.IsNullOrWhiteSpace(incoming.Nickname))
1016 incoming.Nickname = Guid.NewGuid().ToString();
1017
1018 if (!seenIncoming.Add(incoming.Nickname))
1019 {
1020 // Same-ID appears again in the incoming list: last one wins.
1021 overwriteExisting = true;
1022 }
1023
1024 if (existingIndexById.TryGetValue(incoming.Nickname, out int existingIndex))
1025 {
1026 if (!overwriteExisting)
1027 continue;
1028
1029 SpriteManager.Instance._spriteList[existingIndex] = incoming;
1030 }
1031 else
1032 {
1033 existingIndexById[incoming.Nickname] = SpriteManager.Instance._spriteList.Count;
1034 SpriteManager.Instance.AddSprite(incoming);
1035 }
1036 }
