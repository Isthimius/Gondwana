Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. 486 private static EngineStateParts NormalizeParts(EngineStateParts parts)
487 {
488 // GANI definitions resolve Frame references through the TilesheetRegistry.
489 if (parts.HasFlag(EngineStateParts.Cycles))
490 parts |= EngineStateParts.Tilesheets;
491
492 // Tilesheets and Audio may depend on AssetsFiles for AssetIdentifier.Data.
493 if (parts.HasFlag(EngineStateParts.Tilesheets) ||
494 parts.HasFlag(EngineStateParts.Audio))
495 {
496 parts |= EngineStateParts.AssetsFiles;
497 }
498
499 return parts;
500 }

595 private static Dictionary<string, AnimationStateEntry> CaptureAnimationEntries(
596 string? baseDirectory,
597 string engineStatePath,
598 bool separateGaniFiles)
599 {
600 var result = new Dictionary<string, AnimationStateEntry>(StringComparer.Ordinal);
601 var usedFileNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
602
603 // Work from a stable registry snapshot in case definitions are changed while
604 // the EngineState snapshot is being assembled.
605 foreach (var (key, cycle) in Cycle._cycles.ToList())
606 {
607 if (cycle is null)
608 continue;
609
610 if (separateGaniFiles)
611 {
612 var ganiDirectory = GetAnimationStateDirectory(engineStatePath);
613 Directory.CreateDirectory(ganiDirectory);
614
615 var baseFileName = SanitizeFileName(key);
616 var ganiFileName = $"{baseFileName}.gani";
617 int suffix = 2;
618
619 while (!usedFileNames.Add(ganiFileName))
620 ganiFileName = $"{baseFileName}_{suffix++}.gani";
621
622 var ganiFullPath = Path.Combine(ganiDirectory, ganiFileName);
623
624 // Use the standalone serializer so .gani files do not acquire
625 // EngineState-specific $id/$ref metadata.
626 AnimationDefinitionSerializer.Save(ganiFullPath, cycle);
627
628 result[key] = new AnimationStateEntry
629 {
630 GaniPath = MakeRelativePath(ganiFullPath, baseDirectory)
631 };
632 }
633 else
634 {
635 result[key] = new AnimationStateEntry
636 {
637 Definition = AnimationDefinitionSerializer.FromCycle(cycle)
638 };
639 }
640 }
641
642 return result;
643 }

645 private static List<SceneStateEntry> CaptureSceneEntries(
646 string? baseDirectory,
647 string engineStatePath,
648 bool separateGscnFiles)
649 {
650 var result = new List<SceneStateEntry>();
651 var usedFileNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
652
653 // Work from a stable snapshot. Scene construction/disposal may occur on
654 // other threads while a state file is being assembled.
655 foreach (var scene in Scene._allScenes.ToList())
656 {
657 if (scene is null)
658 continue;
659
660 if (separateGscnFiles)
661 {
662 var gscnDirectory = GetSceneStateDirectory(engineStatePath);
663 Directory.CreateDirectory(gscnDirectory);
664
665 var baseFileName = SanitizeFileName(scene.ID);
666 var gscnFileName = $"{baseFileName}.gscn";
667 int suffix = 2;
668
669 while (!usedFileNames.Add(gscnFileName))
670 gscnFileName = $"{baseFileName}_{suffix++}.gscn";
671
672 var gscnFullPath = Path.Combine(gscnDirectory, gscnFileName);
673
674 // Use the standalone GSCN serializer so the external file remains a
675 // clean scene definition without EngineState $id/$ref metadata.
676 SceneDefinitionSerializer.Save(gscnFullPath, scene);
677
678 result.Add(new SceneStateEntry
679 {
680 GscnPath = MakeRelativePath(
681 gscnFullPath,
682 baseDirectory)
683 });
684 }
685 else
686 {
687 result.Add(new SceneStateEntry
688 {
689 Definition = SceneDefinitionSerializer.FromScene(scene)
690 });
691 }
692 }
693
694 return result;
695 }

716 private static void ApplySnapshot(
717 EngineStateSnapshot snapshot,
718 bool clearExisting,
719 bool overwriteExisting,
720 EngineStateParts parts,
721 string? baseDirectory)
722 {
723 parts = NormalizeParts(parts);
724
725 // Legacy raw objects may register themselves while the snapshot DTO is
726 // deserialized. Detach those incoming instances before clearing or merging so
727 // they are treated as snapshot data rather than pre-existing live state.
728 DetachLegacySnapshotScenes(snapshot.Scenes);
729 DetachSnapshotSprites(snapshot.Sprites?.LegacySprites);
730
731 // clear only what we're about to load.
732 if (clearExisting)
733 ClearSelected(parts);
734
735 if (parts.HasFlag(EngineStateParts.AssetsFiles))
736 LoadAssetsFiles(snapshot.AssetsFiles ?? Enumerable.Empty<AssetsFile>(), overwriteExisting);
737
738 if (parts.HasFlag(EngineStateParts.Audio))
739 MergeAudio(
740 snapshot.AssetsFiles,
741 snapshot.Audio,
742 snapshot.SoundResources,
743 overwriteExisting,
744 baseDirectory);
745
746 if (parts.HasFlag(EngineStateParts.Tilesheets))
747 MergeTilesheets(snapshot.Tilesheets, overwriteExisting, baseDirectory);
748
749 if (parts.HasFlag(EngineStateParts.Cycles))
750 MergeAnimations(snapshot.Cycles, overwriteExisting, baseDirectory);
751
752 if (parts.HasFlag(EngineStateParts.Scenes))
753 MergeScenes(snapshot.Scenes, overwriteExisting, baseDirectory);
754
755 if (parts.HasFlag(EngineStateParts.Sprites))
756 {
757 var entry = snapshot.Sprites;
758 if (entry is not null)
759 {
760 var incoming = entry.LegacySprites;
761 if (incoming is null)
762 {
763 var definition = !string.IsNullOrWhiteSpace(entry.GsprPath)
764 ? SpriteDefinitionSerializer.Load(ResolvePath(entry.GsprPath, baseDirectory))
765 : entry.Definition ?? throw new InvalidDataException("Sprite state entry requires GsprPath or Definition.");
766 incoming = SpriteDefinitionSerializer.ToSprites(definition, allowDuplicateNicknames: true);
767 DetachSnapshotSprites(incoming);
768 }
769 MergeSprites(incoming, overwriteExisting);
770 }
771 }
772 }

774 private static void ClearSelected(EngineStateParts parts)
775 {
776 if (parts.HasFlag(EngineStateParts.AssetsFiles))
777 {
778 AssetsFile.ClearAll();
779 SvgResourceManager.Instance.Clear();
780 }
781
782 if (parts.HasFlag(EngineStateParts.Tilesheets))
783 TilesheetRegistry.Instance.Clear();
784
785 if (parts.HasFlag(EngineStateParts.Cycles))
786 Cycle.ClearAllAnimationCycles();
787
788 if (parts.HasFlag(EngineStateParts.Scenes))
789 Scene.ClearAllScenes();
790
791 if (parts.HasFlag(EngineStateParts.Sprites))
792 {
793 // EngineState replacement is synchronous; normal game-loop disposal is deferred.
794 foreach (var sprite in SpriteManager.Instance.AllSprites)

796 SpriteManager.Instance._spriteList.Remove(sprite);
797 sprite.DisposeImmediate();
798 }
799 }
800
801 if (parts.HasFlag(EngineStateParts.Audio))
802 AudioResourceManager.Instance.Dispose();
803 }

805 private static void LoadAssetsFiles(IEnumerable<AssetsFile> resourceFiles, bool overwriteExisting)
806 {
807 // Replace raw deserialized resource files with proper loaded instances
808 if (resourceFiles.Any())
809 {
810 foreach (var raw in resourceFiles)
811 {
812 try
813 {
814 var loaded = AssetsFile.LoadOrCreate(raw.FilePath, raw.Password, raw.UseEncryption);
815
816 if (overwriteExisting)
817 {
818 foreach (var entry in loaded.GetAllEntries().Where(e => e.AssetType == AssetTypes.Svg))
819 SvgResourceManager.Instance.Unload(entry.AssetName);
820 }
821
822 SvgResourceManager.Instance.LoadFromEngineAssetsFile(loaded);
823 }
824 catch (Exception ex)
825 {
826 Engine.Logger.LogError(ex, "Failed to load resource file '{FilePath}'", raw.FilePath);
827 throw;
828 }
829 }
830 }

833 private static void MergeAudio(
834 List<AssetsFile>? assetsFiles,
835 AudioStateEntry? audio,
836 Dictionary<string, AudioResource>? soundSpecs,
837 bool overwriteExisting,
838 string? baseDirectory)
839 {
840 if (audio is not null)
841 {
842 AudioDefinition definition;
843
844 if (!string.IsNullOrWhiteSpace(audio.GsndPath))
845 {
846 definition = AudioDefinitionSerializer.Load(
847 ResolvePath(audio.GsndPath, baseDirectory));
848 }
849 else if (audio.Definition is not null)
850 {
851 definition = audio.Definition;
852 }
853 else
854 {
855 throw new InvalidDataException(
856 "Audio state entry does not contain a GSND path or inline definition.");
857 }
858
859 AudioDefinitionSerializer.LoadIntoManager(
860 definition,
861 overwriteExisting,
862 string.IsNullOrWhiteSpace(audio.GsndPath)
863 ? baseDirectory
864 : null);
865 return;
866 }
867
868 // Legacy EngineState compatibility: load old asset packs first, then
869 // apply the serialized AudioResource specs / overrides.
870 if (assetsFiles is not null)
871 {
872 foreach (var af in assetsFiles)
873 {
874 if (overwriteExisting)
875 {
876 foreach (var entry in af.GetAllEntries())
877 {
878 if (entry.AssetType == AssetTypes.Audio)
879 AudioResourceManager.Instance.Unload(entry.AssetName);
880 }
881 }
882
883 AudioResourceManager.Instance.LoadFromEngineAssetsFile(af);
884 }
885 }
886
887 // 2) Apply loose-file specs / overrides
888 if (soundSpecs is null)
889 return;
890
891 foreach (var (key, spec) in soundSpecs)
892 {
893 if (AudioResourceManager.Instance.Contains(key))
894 {
895 if (!overwriteExisting)
896 {
897 var existing = AudioResourceManager.Instance.Get(key);
898 if (existing is not null)
899 {
900 existing.Volume = spec.Volume;
901 existing.Pan = spec.Pan;
902 existing.PlaybackSpeed = spec.PlaybackSpeed;
903 existing.IsLooping = spec.IsLooping;
904 }
905 continue;
906 }
907
908 AudioResourceManager.Instance.Unload(key);
909 }
910
911 // Ensure the audio spec is (re)created/registered in the manager.
912 spec.ReloadIntoManager();
913 }

1096 private static void MergeAnimations(
1097 Dictionary<string, AnimationStateEntry>? animations,
1098 bool overwriteExisting,
1099 string? baseDirectory)
1100 {
1101 if (animations is null || animations.Count == 0)
1102 return;
1103
1104 var definitions = new List<AnimationDefinition>();
1105
1106 foreach (var (key, entry) in animations)
1107 {
1108 if (entry is null)
1109 continue;
1110
1111 if (!overwriteExisting && Cycle._cycles.ContainsKey(key))
1112 continue;
1113
1114 AnimationDefinition definition;
1115
1116 if (!string.IsNullOrWhiteSpace(entry.GaniPath))
1117 {
1118 var ganiPath = ResolvePath(entry.GaniPath, baseDirectory);
1119 definition = AnimationDefinitionSerializer.Load(ganiPath);
1120 }
1121 else if (entry.Definition is not null)
1122 {
1123 definition = entry.Definition;
1124 }
1125 else
1126 {
1127 throw new InvalidDataException(
1128 $"Animation state entry '{key}' does not contain a GANI path or inline definition.");
1129 }
1130
1131 if (string.IsNullOrWhiteSpace(definition.Key))
1132 {
1133 definition.Key = key;
1134 }
1135 else if (!string.Equals(definition.Key, key, StringComparison.Ordinal))
1136 {
1137 throw new InvalidDataException(
1138 $"Animation state entry '{key}' contains definition key '{definition.Key}'.");
1139 }
1140
1141 definitions.Add(definition);
1142 }
1143
1144 // Validate next-cycle identities before mutating the registry. Two-phase
1145 // materialization allows circular transitions such as idle -> blink -> idle.
1146 var availableKeys = new HashSet<string>(
1147 Cycle._cycles.Keys,
1148 StringComparer.Ordinal);
1149
1150 foreach (var definition in definitions)
1151 availableKeys.Add(definition.Key);
1152
1153 foreach (var definition in definitions)
1154 {
1155 if (!string.IsNullOrWhiteSpace(definition.NextCycleKey) &&
1156 !string.Equals(
1157 definition.NextCycleKey,
1158 definition.Key,
1159 StringComparison.Ordinal) &&
1160 !availableKeys.Contains(definition.NextCycleKey))
1161 {
1162 throw new InvalidDataException(
1163 $"GANI animation '{definition.Key}' references next cycle '{definition.NextCycleKey}', but no matching animation is available.");
1164 }
1165 }
1166
1167 var materialized = new List<(Cycle Cycle, AnimationDefinition Definition)>();
1168
1169 foreach (var definition in definitions)
1170 {
1171 var cycle = AnimationDefinitionSerializer.MaterializeCycle(definition);
1172 materialized.Add((cycle, definition));
1173 }
1174
1175 foreach (var (cycle, definition) in materialized)
1176 AnimationDefinitionSerializer.ApplyNextCycle(cycle, definition);
1177 }

1191 private static void MergeScenes(
1192 List<SceneStateEntry>? scenes,
1193 bool overwriteExisting,
1194 string? baseDirectory)
1195 {
1196 if (scenes is null || scenes.Count == 0)
1197 return;
1198
1199 var existingIndexById = new Dictionary<string, int>(StringComparer.Ordinal);
1200 for (int i = 0; i < Scene._allScenes.Count; i++)
1201 {
1202 var id = Scene._allScenes[i].ID;
1203 if (!string.IsNullOrWhiteSpace(id) && !existingIndexById.ContainsKey(id))
1204 existingIndexById.Add(id, i);
1205 }
1206
1207 // Avoid duplicating the same incoming ID twice. A duplicate within one
1208 // snapshot is treated as an overwrite so the last incoming entry wins.
1209 var seenIncoming = new HashSet<string>(StringComparer.Ordinal);
1210
1211 foreach (var entry in scenes)
1212 {
1213 if (entry is null)
1214 continue;
1215
1216 Scene incoming;
1217 bool materializedFromDefinition = false;
1218
1219 if (entry.LegacyScene is not null)
1220 {
1221 incoming = entry.LegacyScene;
1222 }
1223 else if (!string.IsNullOrWhiteSpace(entry.GscnPath))
1224 {
1225 var gscnPath = ResolvePath(entry.GscnPath, baseDirectory);
1226 incoming = SceneDefinitionSerializer.LoadScene(gscnPath);
1227 materializedFromDefinition = true;
1228 }
1229 else if (entry.Definition is not null)
1230 {
1231 incoming = SceneDefinitionSerializer.ToScene(entry.Definition);
1232 materializedFromDefinition = true;
1233 }
1234 else
1235 {
1236 throw new InvalidDataException(
1237 "Scene state entry does not contain a GSCN path or inline definition.");
1238 }
1239
1240 // Scene materialization registers runtime scenes immediately. Detach newly
1241 // materialized definitions so this method remains the single merge owner.
1242 if (materializedFromDefinition)
1243 Scene._allScenes.Remove(incoming);
1244
1245 if (string.IsNullOrWhiteSpace(incoming.ID))
1246 incoming.ID = Guid.NewGuid().ToString();
1247
1248 bool duplicateIncoming = !seenIncoming.Add(incoming.ID);
1249
1250 if (existingIndexById.TryGetValue(incoming.ID, out int existingIndex))
1251 {
1252 if (!overwriteExisting && !duplicateIncoming)
1253 {
1254 if (materializedFromDefinition)
1255 incoming.Dispose();
1256
1257 continue;
1258 }
1259
1260 Scene._allScenes[existingIndex] = incoming;
1261 }
1262 else
1263 {
1264 existingIndexById[incoming.ID] = Scene._allScenes.Count;
1265 Scene._allScenes.Add(incoming);
1266 }
1267 }

1270 private static void RebindSpriteSceneLayer(Sprite sprite)
1271 {
1272 var layerId = sprite.SerializedSceneLayerIdForBinding;
1273 if (string.IsNullOrWhiteSpace(layerId))
1274 {
1275 if (!ReferenceEquals(sprite.SceneLayer, SceneLayer.Empty))
1276 return;
1277
1278 throw new InvalidDataException(
1279 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' does not contain a SceneLayer identity.");
1280 }
1281
1282 SceneLayer? targetLayer = null;
1283 var sceneId = sprite.SerializedSceneIdForBinding;
1284
1285 if (!string.IsNullOrWhiteSpace(sceneId))
1286 {
1287 var scene = Scene._allScenes.FirstOrDefault(
1288 candidate => string.Equals(candidate.ID, sceneId, StringComparison.Ordinal));
1289
1290 targetLayer = scene?.GetSceneLayerByID(layerId);
1291 }
1292 else
1293 {
1294 // Older data may have a layer ID but no scene ID. Layer IDs are normally
1295 // GUIDs; accept a unique match and reject ambiguity.
1296 var matches = Scene._allScenes
1297 .SelectMany(scene => scene.SceneLayers)
1298 .Where(layer => string.Equals(layer.ID, layerId, StringComparison.Ordinal))
1299 .Take(2)
1300 .ToList();
1301
1302 if (matches.Count == 1)
1303 targetLayer = matches[0];
1304 else if (matches.Count > 1)
1305 throw new InvalidDataException(
1306 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' references ambiguous SceneLayer ID '{layerId}'.");
1307 }
1308
1309 if (targetLayer is null)
1310 {
1311 throw new InvalidDataException(
1312 $"Sprite '{sprite.Nickname ?? sprite.Id.ToString()}' could not resolve SceneLayer '{layerId}'" +
1313 (string.IsNullOrWhiteSpace(sceneId) ? "." : $" in Scene '{sceneId}'."));
1314 }
1315
1316 sprite.RebindSceneLayerAfterDeserialization(targetLayer);
1317 }
