Gondwana Game Engine Development (master) - 2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

341 public PopupWidget BindTo(
342 Tile source,
343 WidgetAnchor sourceAnchor = WidgetAnchor.Center,
344 Point? offsetPx = null)
345 {
346 ArgumentNullException.ThrowIfNull(source);
347
348 if (Mode != DirectDrawingMode.SceneLayer)
349 throw new InvalidOperationException("Only a scene-layer popup can bind to a tile.");
350
351 if (!ReferenceEquals(SceneLayer, source.SceneLayer))
352 throw new ArgumentException("The source tile must belong to the popup's SceneLayer.", nameof(source));
353
354 Point offset = offsetPx ?? Point.Empty;
355
356 SourceTile = source;
357 SourceGridLocation = source is SceneLayerTile fixedTile
358 ? fixedTile.GridCoordinatesAbs
359 : null;
360
361 _sourceResolver = () => GetCenteredSourcePosition(
362 source.DrawLocationWorld,
363 GetContentSize(),
364 sourceAnchor,
365 offset);
366
367 SetPosition(_sourceResolver());
368 return this;
369 }

450 public override void Update(long tick)
451 {
452 if (_disposed || tick <= _lastAnimationTick)
453 return;
454
455 float elapsedSec = HighResTimer.GetDuration(_lastAnimationTick, tick);
456 _lastAnimationTick = tick;
457
458 base.Update(tick);
459
460 if (!_isActive)
461 return;
462
463 _elapsedSec += elapsedSec;
464
465 if (_elapsedSec >= LifetimeSec)
466 {
467 CompletePopup();
468 return;
469 }
470
471 float opacity = 1f;
472
473 if (FadeInSec > 0f && _elapsedSec < FadeInSec)
474 opacity = Math.Clamp(_elapsedSec / FadeInSec, 0f, 1f);
475
476 float remainingSec = LifetimeSec - _elapsedSec;
477
478 if (FadeOutSec > 0f && remainingSec < FadeOutSec)
479 opacity = Math.Min(opacity, Math.Clamp(remainingSec / FadeOutSec, 0f, 1f));
480
481 SetOpacity(opacity);
482 }
