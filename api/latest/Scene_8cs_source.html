Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

341 internal void BindRenderSurfaceHost(RenderSurfaceHostBase host)
342 {
343 ArgumentNullException.ThrowIfNull(host);
344
345 // Scene.Empty is a shared null-object placeholder and is never treated as an
346 // actively rendered scene for binding ownership purposes.
347 if (ReferenceEquals(this, Empty))
348 return;
349
350 lock (_renderSurfaceHostSync)
351 {
352 if (_boundRenderSurfaceHost is not null &&
353 !ReferenceEquals(_boundRenderSurfaceHost, host))
354 {
355 throw new InvalidOperationException(
356 $"Scene '{ID}' is already bound to another RenderSurfaceHost.");
357 }
358
359 _boundRenderSurfaceHost = host;
360 }
361
362 // A GPU host redraws the full frame and never consumes RefreshQueue. Remove
363 // anything accumulated before binding; future additions are rejected by the
364 // queue's scene-policy callback.
365 if (!UsesDirtyRegionRendering)
366 {
367 foreach (var layer in _sceneLayers)
368 layer.RefreshQueue.ClearRefreshQueue();
369 }
370 }
371
376 internal void UnbindRenderSurfaceHost(RenderSurfaceHostBase host)
377 {
378 ArgumentNullException.ThrowIfNull(host);
379
380 if (ReferenceEquals(this, Empty))
381 return;
382
383 lock (_renderSurfaceHostSync)
384 {
385 if (ReferenceEquals(_boundRenderSurfaceHost, host))
386 _boundRenderSurfaceHost = null;
387 }
388 }

578 public RectangleF GetWorldBoundsPx()
579 {
580 if (_sceneLayers.Count == 0)
581 return RectangleF.Empty;
582
583 RectangleF result = RectangleF.Empty;
584 bool hasBounds = false;
585
586 foreach (var layer in _sceneLayers)
587 {
588 var lb = layer.GetLayerBoundsPx();
589 if (lb.IsEmpty)
590 continue;
591
592 if (!hasBounds)
593 {
594 result = lb;
595 hasBounds = true;
596 }
597 else
598 {
599 result = RectangleF.Union(result, lb);
600 }
601 }
602
603 return result;
604 }
