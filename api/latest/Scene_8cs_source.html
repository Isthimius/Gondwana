Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

344 internal void BindRenderSurfaceHost(RenderSurfaceHostBase host)
345 {
346 ArgumentNullException.ThrowIfNull(host);
347
348 // Scene.Empty is a shared null-object placeholder and is never treated as an
349 // actively rendered scene for binding ownership purposes.
350 if (ReferenceEquals(this, Empty))
351 return;
352
353 lock (_renderSurfaceHostSync)
354 {
355 if (_boundRenderSurfaceHost is not null &&
356 !ReferenceEquals(_boundRenderSurfaceHost, host))
357 {
358 throw new InvalidOperationException(
359 $"Scene '{ID}' is already bound to another RenderSurfaceHost.");
360 }
361
362 _boundRenderSurfaceHost = host;
363 }
364
365 // A GPU host redraws the full frame and never consumes RefreshQueue. Remove
366 // anything accumulated before binding; future additions are rejected by the
367 // queue's scene-policy callback.
368 if (!UsesDirtyRegionRendering)
369 {
370 foreach (var layer in _sceneLayers)
371 layer.RefreshQueue.ClearRefreshQueue();
372 }
373 }
374
379 internal void UnbindRenderSurfaceHost(RenderSurfaceHostBase host)
380 {
381 ArgumentNullException.ThrowIfNull(host);
382
383 if (ReferenceEquals(this, Empty))
384 return;
385
386 lock (_renderSurfaceHostSync)
387 {
388 if (ReferenceEquals(_boundRenderSurfaceHost, host))
389 _boundRenderSurfaceHost = null;
390 }
391 }

581 public RectangleF GetWorldBoundsPx()
582 {
583 if (_sceneLayers.Count == 0)
584 return RectangleF.Empty;
585
586 RectangleF result = RectangleF.Empty;
587 bool hasBounds = false;
588
589 foreach (var layer in _sceneLayers)
590 {
591 var lb = layer.GetLayerBoundsPx();
592 if (lb.IsEmpty)
593 continue;
594
595 if (!hasBounds)
596 {
597 result = lb;
598 hasBounds = true;
599 }
600 else
601 {
602 result = RectangleF.Union(result, lb);
603 }
604 }
605
606 return result;
