Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

348 internal void BindRenderSurfaceHost(RenderSurfaceHostBase host)
349 {
350 ArgumentNullException.ThrowIfNull(host);
351
352 // Scene.Empty is a shared null-object placeholder and is never treated as an
353 // actively rendered scene for binding ownership purposes.
354 if (ReferenceEquals(this, Empty))
355 return;
356
357 lock (_renderSurfaceHostSync)
358 {
359 if (_boundRenderSurfaceHost is not null &&
360 !ReferenceEquals(_boundRenderSurfaceHost, host))
361 {
362 throw new InvalidOperationException(
363 $"Scene '{ID}' is already bound to another RenderSurfaceHost.");
364 }
365
366 _boundRenderSurfaceHost = host;
367 }
368
369 // A GPU host redraws the full frame and never consumes RefreshQueue. Remove
370 // anything accumulated before binding; future additions are rejected by the
371 // queue's scene-policy callback.
372 if (!UsesDirtyRegionRendering)
373 {
374 foreach (var layer in _sceneLayers)
375 layer.RefreshQueue.ClearRefreshQueue();
376 }
377 }
378
384 internal void UnbindRenderSurfaceHost(RenderSurfaceHostBase host)
385 {
386 ArgumentNullException.ThrowIfNull(host);
387
388 if (ReferenceEquals(this, Empty))
389 return;
390
391 lock (_renderSurfaceHostSync)
392 {
393 if (ReferenceEquals(_boundRenderSurfaceHost, host))
394 _boundRenderSurfaceHost = null;
395 }
396 }

586 public RectangleF GetWorldBoundsPx()
587 {
588 if (_sceneLayers.Count == 0)
589 return RectangleF.Empty;
590
591 RectangleF result = RectangleF.Empty;
592 bool hasBounds = false;
593
594 foreach (var layer in _sceneLayers)
595 {
596 var lb = layer.GetLayerBoundsPx();
597 if (lb.IsEmpty)
598 continue;
599
600 if (!hasBounds)
601 {
602 result = lb;
603 hasBounds = true;
604 }
605 else
606 {
607 result = RectangleF.Union(result, lb);
608 }
609 }
610
611 return result;
612 }
