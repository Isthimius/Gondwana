810 public override void Update(long tick)
811 {
812 if (tick <= _lastTick)
813 return;
814
815 // Compute dt from ticks (seconds)
816 float dt = 0f;
817
818 // no previous tick assume first frame, so skip dt-based updates
819 if (_pulseLastTick > 0)
820 {
821 dt = HighResTimer.GetDuration(_pulseLastTick, tick);
822 _timeSec += dt;
823 }
824
825 _pulseLastTick = tick;
826
827 bool changed = false;
828
829 if (_pulseFillEnabled)
830 {
831 float t = PulseT(_timeSec, _pulseFillPeriodSec, _pulseFillWave);
832 var c = LerpColor(_pulseFillFrom, _pulseFillTo, t);
833 if (_fillPaint.Color != c) { _fillPaint.Color = c; changed = true; }
834 }
835
836 if (_pulseBorderEnabled)
837 {
838 float t = PulseT(_timeSec, _pulseBorderPeriodSec, _pulseBorderWave);
839 var c = LerpColor(_pulseBorderFrom, _pulseBorderTo, t);
840
841 if (_strokePaint.Color != c)
842 {
843 _strokePaint.Color = c; // paint-only override
844 ForceRefresh();
845 }
846 }
847
848 if (changed)
849 {
850 _needsRebuildPaints = false; // we already set paints directly
851 ForceRefresh(); // request redraw
852 }
853
854 base.Update(tick);
855 }

884 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
885 {
886 var canvas = backbuffer.Canvas;
887
888 if (_needsRebuildPaints)
889 RebuildPaints();
890
891 var fillRect = destRectScreen.ToSKRect();
892 var strokeRect = fillRect;
893
894 bool willDrawStroke = !_isFilled || _borderColor.HasValue || _strokePaint.StrokeWidth > 0.01f;
895 float half = _strokePaint.StrokeWidth * 0.5f;
896
897 // 1) APPLY STROKE ALIGNMENT (use HALF the width; path is centered)
898 if (willDrawStroke && _strokeAlign != StrokeAlign.Center)
899 {
900 if (_strokeAlign == StrokeAlign.Inside)
901 strokeRect.Inflate(-half, -half);
902 else if (_strokeAlign == StrokeAlign.Outside)
903 strokeRect.Inflate(half, half);
904 }
905
906 // 1.5) corner radius for the stroke path, adjusted to keep the inner/outer arcs aligned to the fill
907 float strokeCornerRadius;
908 switch (_strokeAlign)
909 {
910 case StrokeAlign.Outside:
911 strokeCornerRadius = MathF.Max(0f, _cornerRadius - half);
912 break;
913 case StrokeAlign.Inside:
914 strokeCornerRadius = _cornerRadius + half;
915 break;
916 case StrokeAlign.Center:
917 default:
918 strokeCornerRadius = _cornerRadius;
919 break;
920 }
921
922 // 2) Dash for stroke only
923 _strokePaint.PathEffect = _dashPattern is { Length: > 0 }
924 ? SKPathEffect.CreateDash(_dashPattern, 0)
925 : null;
926
927 // 3) Draw fill (unmodified rect)
928 if (_isFilled)
929 {
930 if (_fillImage is not null || _fillBitmap is not null)
931 {
932 DrawImageFill(canvas, fillRect);
933 }
934 else if (_cornerRadius > 0)
935 {
936 canvas.DrawRoundRect(fillRect, strokeCornerRadius, strokeCornerRadius, _fillPaint);
937 }
938 else
939 {
940 canvas.DrawRect(fillRect, _fillPaint);
941 }
942 }
943
944 // 4) Draw stroke on its aligned rect (pure/opaque so it stays white)
945 if (willDrawStroke)
946 {
947 var prevBlend = _strokePaint.BlendMode;
948 var prevColor = _strokePaint.Color;
949 var prevAA = _strokePaint.IsAntialias;
950
951 _strokePaint.IsAntialias = false;
952 _strokePaint.BlendMode = SKBlendMode.Src;
953
954 // If pulsing, keep color from Update(),
955 // else use base border color (or previous).
956 var strokeColor = _pulseBorderEnabled ? _strokePaint.Color
957 : (_borderColor ?? prevColor);
958
959 _strokePaint.Color = strokeColor.WithAlpha(255);
960
961 if (_cornerRadius > 0)
962 canvas.DrawRoundRect(strokeRect, _cornerRadius, _cornerRadius, _strokePaint);
963 else
964 canvas.DrawRect(strokeRect, _strokePaint);
965
966 _strokePaint.IsAntialias = prevAA;
967 _strokePaint.BlendMode = prevBlend;
968 _strokePaint.Color = prevColor;
969 _strokePaint.StrokeJoin = SKStrokeJoin.Round;
970 _strokePaint.StrokeCap = SKStrokeCap.Round;
971 }
972 }

1020 private void DrawImageFill(
1021 SKCanvas canvas,
1022 SKRect fillRect)
1023 {
1024 SKImage? image = _fillImage;
1025 SKBitmap? bitmap = _fillBitmap;
1026
1027 int sourceWidth = image?.Width ?? bitmap?.Width ?? 0;
1028 int sourceHeight = image?.Height ?? bitmap?.Height ?? 0;
1029
1030 if (sourceWidth <= 0 || sourceHeight <= 0)
1031 return;
1032
1033 canvas.Save();
1034
1035 try
1036 {
1037 using var clipPath = new SKPath();
1038
1039 if (_cornerRadius > 0f)
1040 clipPath.AddRoundRect(fillRect, _cornerRadius, _cornerRadius);
1041 else
1042 clipPath.AddRect(fillRect);
1043
1044 canvas.ClipPath(
1045 clipPath,
1046 SKClipOperation.Intersect,
1047 antialias: _cornerRadius > 0f);
1048
1049 var source = new SKRect(
1050 0f,
1051 0f,
1052 sourceWidth,
1053 sourceHeight);
1054
1055 if (_imageFillMode == ImageFillMode.Repeat)
1056 {
1057 if (image is not null)
1058 {
1059 DrawRepeatedImage(
1060 canvas,
1061 image,
1062 source,
1063 fillRect);
1064 }
1065 else
1066 {
1067 DrawRepeatedBitmap(
1068 canvas,
1069 bitmap!,
1070 source,
1071 fillRect);
1072 }
1073 }
1074 else
1075 {
1076 SKRect destination = ComputeImageDestination(
1077 fillRect,
1078 source,
1079 _imageFillMode);
1080
1081 if (image is not null)
1082 {
1083 canvas.DrawImage(
1084 image,
1085 source,
1086 destination,
1087 _imagePaint);
1088 }
1089 else
1090 {
1091 canvas.DrawBitmap(
1092 bitmap!,
1093 source,
1094 destination,
1095 _imagePaint);
1096 }
1097 }
1098 }
1099 finally
1100 {
1101 canvas.Restore();
1102 }

1181 private static SKRect ComputeImageDestination(
1182 SKRect bounds,
1183 SKRect source,
1184 ImageFillMode mode)
1185 {
1186 if (mode == ImageFillMode.Stretch)
1187 return bounds;
1188
1189 if (mode == ImageFillMode.Center)
1190 {
1191 float x = bounds.MidX - source.Width * 0.5f;
1192 float y = bounds.MidY - source.Height * 0.5f;
1193
1194 return new SKRect(
1195 x,
1196 y,
1197 x + source.Width,
1198 y + source.Height);
1199 }
1200
1201 float scaleX = bounds.Width / source.Width;
1202 float scaleY = bounds.Height / source.Height;
1203
1204 float scale = mode switch
1205 {
1206 ImageFillMode.Fit => MathF.Min(scaleX, scaleY),
1207 ImageFillMode.Fill => MathF.Max(scaleX, scaleY),
1208 ImageFillMode.PixelPerfect => MathF.Max(
1209 1f,
1210 MathF.Floor(MathF.Min(scaleX, scaleY))),
1211 _ => 1f
1212 };
1213
1214 float width = source.Width * scale;
1215 float height = source.Height * scale;
1216 float left = bounds.MidX - width * 0.5f;
1217 float top = bounds.MidY - height * 0.5f;
1218
1219 return new SKRect(
1220 left,
1221 top,
1222 left + width,
1223 top + height);
1224 }

1292 [MethodImpl(MethodImplOptions.AggressiveInlining)]
1293 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
1294 {
1295 float phase = (timeSec / periodSec) % 1f;
1296 if (wave == PulseWave.Sine)
1297 {
1298 // (sin(2πx)+1)/2 in [0..1]
1299 return 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f));
1300 }
1301 else // Triangle: ramp up 0..1 then down 1..0
1302 {
1303 return phase < 0.5f ? (phase * 2f) : (1f - ((phase - 0.5f) * 2f));
1304 }
1305 }

1307 [MethodImpl(MethodImplOptions.AggressiveInlining)]
1308 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
1309 {
1310 t01 = Math.Clamp(t01, 0f, 1f);
1311
1312 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
1313 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
1314 byte bch = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
1315 byte aA = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
1316
1317 return new SKColor(r, g, bch, aA);
1318 }
