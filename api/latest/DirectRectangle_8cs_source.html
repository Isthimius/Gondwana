879 public override void Update(long tick)
880 {
881 if (tick <= _lastTick)
882 return;
883
884 // no previous tick assume first frame, so skip dt-based updates
885 if (_pulseLastTick > 0)
886 {
887 float dt = HighResTimer.GetDuration(_pulseLastTick, tick);
888 _timeSec += dt;
889 }
890
891 _pulseLastTick = tick;
892
893 bool changed = false;
894
895 if (_pulseFillEnabled)
896 {
897 float t = PulseT(_timeSec, _pulseFillPeriodSec, _pulseFillWave);
898 var c = LerpColor(_pulseFillFrom, _pulseFillTo, t);
899
900 if (_fillPaint.Color != c)
901 {
902 _fillPaint.Color = c;
903 changed = true;
904 }
905 }
906
907 if (_pulseBorderEnabled)
908 {
909 float t = PulseT(_timeSec, _pulseBorderPeriodSec, _pulseBorderWave);
910 var c = LerpColor(_pulseBorderFrom, _pulseBorderTo, t);
911
912 if (_strokePaint.Color != c)
913 {
914 _strokePaint.Color = c; // paint-only override
915 ForceRefresh();
916 }
917 }
918
919 if (changed)
920 {
921 _needsRebuildPaints = false; // we already set paints directly
922 ForceRefresh(); // request redraw
923 }
924
925 base.Update(tick);
926 }

955 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
956 {
957 if (backbuffer is RecordingBackbuffer && _fillImage?.IsTextureBacked == true)
958 throw new NotSupportedException(
959 "Desktop frame recording requires a CPU image, not a context-bound texture.");
960
961 var canvas = backbuffer.Canvas;
962
963 if (_needsRebuildPaints)
964 RebuildPaints();
965
966 var fillRect = destRectScreen.ToSKRect();
967 var strokeRect = fillRect;
968
969 bool willDrawStroke = !_isFilled || _borderColor.HasValue || _strokePaint.StrokeWidth > 0.01f;
970 float half = _strokePaint.StrokeWidth * 0.5f;
971
972 // 1) APPLY STROKE ALIGNMENT (use HALF the width; path is centered)
973 if (willDrawStroke && _strokeAlign != StrokeAlign.Center)
974 {
975 if (_strokeAlign == StrokeAlign.Inside)
976 strokeRect.Inflate(-half, -half);
977 else if (_strokeAlign == StrokeAlign.Outside)
978 strokeRect.Inflate(half, half);
979 }
980
981 // 1.5) corner radius for the stroke path, adjusted to keep the inner/outer arcs aligned to the fill
982 float strokeCornerRadius;
983
984 switch (_strokeAlign)
985 {
986 case StrokeAlign.Outside:
987 strokeCornerRadius = MathF.Max(0f, _cornerRadius - half);
988 break;
989 case StrokeAlign.Inside:
990 strokeCornerRadius = _cornerRadius + half;
991 break;
992 case StrokeAlign.Center:
993 default:
994 strokeCornerRadius = _cornerRadius;
995 break;
996 }
997
998 // 2) Dash for stroke only
999 _strokePaint.PathEffect = _dashPattern is { Length: > 0 }
1000 ? SKPathEffect.CreateDash(_dashPattern, 0)
1001 : null;
1002
1003 // 3) Draw fill (unmodified rect)
1004 if (_isFilled)
1005 {
1006 if (_fillImage is not null || _fillBitmap is not null)
1007 {
1008 DrawImageFill(canvas, fillRect);
1009 }
1010 else if (_cornerRadius > 0)
1011 {
1012 canvas.DrawRoundRect(
1013 fillRect,
1014 strokeCornerRadius,
1015 strokeCornerRadius,
1016 _fillPaint);
1017 }
1018 else
1019 {
1020 canvas.DrawRect(fillRect, _fillPaint);
1021 }
1022 }
1023
1024 // 4) Draw stroke on its aligned rect (pure/opaque so it stays white)
1025 if (willDrawStroke)
1026 {
1027 var prevBlend = _strokePaint.BlendMode;
1028 var prevColor = _strokePaint.Color;
1029 var prevAA = _strokePaint.IsAntialias;
1030
1031 _strokePaint.IsAntialias = false;
1032 _strokePaint.BlendMode = SKBlendMode.Src;
1033
1034 // If pulsing, keep color from Update(),
1035 // else use base border color (or previous).
1036 var strokeColor = _pulseBorderEnabled
1037 ? _strokePaint.Color
1038 : (_borderColor ?? prevColor);
1039
1040 _strokePaint.Color = strokeColor.WithAlpha(255);
1041
1042 if (_cornerRadius > 0)
1043 canvas.DrawRoundRect(
1044 strokeRect,
1045 _cornerRadius,
1046 _cornerRadius,
1047 _strokePaint);
1048 else
1049 canvas.DrawRect(strokeRect, _strokePaint);
1050
1051 _strokePaint.IsAntialias = prevAA;
1052 _strokePaint.BlendMode = prevBlend;
1053 _strokePaint.Color = prevColor;
1054 _strokePaint.StrokeJoin = SKStrokeJoin.Round;
1055 _strokePaint.StrokeCap = SKStrokeCap.Round;
1056 }
1057 }

1105 private void DrawImageFill(
1106 SKCanvas canvas,
1107 SKRect fillRect)
1108 {
1109 SKImage? image = _fillImage;
1110 SKBitmap? bitmap = _fillBitmap;
1111
1112 int sourceWidth = image?.Width ?? bitmap?.Width ?? 0;
1113 int sourceHeight = image?.Height ?? bitmap?.Height ?? 0;
1114
1115 if (sourceWidth <= 0 || sourceHeight <= 0)
1116 return;
1117
1118 canvas.Save();
1119
1120 try
1121 {
1122 using var clipPath = new SKPath();
1123
1124 if (_cornerRadius > 0f)
1125 clipPath.AddRoundRect(fillRect, _cornerRadius, _cornerRadius);
1126 else
1127 clipPath.AddRect(fillRect);
1128
1129 canvas.ClipPath(
1130 clipPath,
1131 SKClipOperation.Intersect,
1132 antialias: _cornerRadius > 0f);
1133
1134 var source = new SKRect(
1135 0f,
1136 0f,
1137 sourceWidth,
1138 sourceHeight);
1139
1140 if (_imageFillMode == ImageFillMode.Repeat)
1141 {
1142 if (image is not null)
1143 {
1144 DrawRepeatedImage(
1145 canvas,
1146 image,
1147 source,
1148 fillRect);
1149 }
1150 else
1151 {
1152 DrawRepeatedBitmap(
1153 canvas,
1154 bitmap!,
1155 source,
1156 fillRect);
1157 }
1158 }
1159 else
1160 {
1161 SKRect destination = ComputeImageDestination(
1162 fillRect,
1163 source,
1164 _imageFillMode);
1165
1166 if (image is not null)
1167 {
1168 canvas.DrawImage(
1169 image,
1170 source,
1171 destination,
1172 _imagePaint);
1173 }
1174 else
1175 {
1176 canvas.DrawBitmap(
1177 bitmap!,
1178 source,
1179 destination,
1180 _imagePaint);
1181 }
1182 }
1183 }
1184 finally
1185 {
1186 canvas.Restore();
1187 }

1266 private static SKRect ComputeImageDestination(
1267 SKRect bounds,
1268 SKRect source,
1269 ImageFillMode mode)
1270 {
1271 if (mode == ImageFillMode.Stretch)
1272 return bounds;
1273
1274 if (mode == ImageFillMode.Center)
1275 {
1276 float x = bounds.MidX - source.Width * 0.5f;
1277 float y = bounds.MidY - source.Height * 0.5f;
1278
1279 return new SKRect(
1280 x,
1281 y,
1282 x + source.Width,
1283 y + source.Height);
1284 }
1285
1286 float scaleX = bounds.Width / source.Width;
1287 float scaleY = bounds.Height / source.Height;
1288
1289 float scale = mode switch
1290 {
1291 ImageFillMode.Fit => MathF.Min(scaleX, scaleY),
1292 ImageFillMode.Fill => MathF.Max(scaleX, scaleY),
1293 ImageFillMode.PixelPerfect => MathF.Max(
1294 1f,
1295 MathF.Floor(MathF.Min(scaleX, scaleY))),
1296 _ => 1f
1297 };
1298
1299 float width = source.Width * scale;
1300 float height = source.Height * scale;
1301 float left = bounds.MidX - width * 0.5f;
1302 float top = bounds.MidY - height * 0.5f;
1303
1304 return new SKRect(
1305 left,
1306 top,
1307 left + width,
1308 top + height);
1309 }

1378 [MethodImpl(MethodImplOptions.AggressiveInlining)]
1379 private static float PulseT(float timeSec, float periodSec, PulseWave wave)
1380 {
1381 float phase = (timeSec / periodSec) % 1f;
1382
1383 if (wave == PulseWave.Sine)
1384 {
1385 // (sin(2πx)+1)/2 in [0..1]
1386 return 0.5f * (1f + MathF.Sin(phase * MathF.PI * 2f));
1387 }
1388 else // Triangle: ramp up 0..1 then down 1..0
1389 {
1390 return phase < 0.5f ? (phase * 2f) : (1f - ((phase - 0.5f) * 2f));
1391 }
1392 }

1394 [MethodImpl(MethodImplOptions.AggressiveInlining)]
1395 private static SKColor LerpColor(SKColor a, SKColor b, float t01)
1396 {
1397 t01 = Math.Clamp(t01, 0f, 1f);
1398
1399 byte r = (byte)(a.Red + (b.Red - a.Red) * t01);
1400 byte g = (byte)(a.Green + (b.Green - a.Green) * t01);
1401 byte bch = (byte)(a.Blue + (b.Blue - a.Blue) * t01);
1402 byte aA = (byte)(a.Alpha + (b.Alpha - a.Alpha) * t01);
1403
1404 return new SKColor(
1405 r,
1406 g,
1407 bch,
1408 aA);
1409 }
