Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

275 private void OnDropDownFocusLost()
276 {
277 if (!_isDropDownOpen)
278 return;
279
280 // Pointer-down moves focus before the header click is dispatched. If focus
281 // moved back to this combo's own header, leave the list open long enough
282 // for that click to toggle it closed normally. Any other focus target means
283 // the user has moved on, so collapse without stealing focus back.
284 WidgetBase? focusedWidget = WidgetInputRouterRegistry.GetFocusedWidget(DropDown);
285 if (ReferenceEquals(focusedWidget, Header) || ReferenceEquals(focusedWidget, DropDown))
286 return;
287
288 CollapseDropDown(restoreHeaderFocus: false);
289 }

377 protected override void OnDraw(BackbufferBase backbuffer, RectangleF destRectScreen)
378 {
379 float width = Math.Min(8f, destRectScreen.Width - 4f);
380 float centerX = destRectScreen.Left + destRectScreen.Width / 2f;
381 float centerY = destRectScreen.Top + destRectScreen.Height / 2f;
382
383 using var paint = new SKPaint

385 Color = _color,
386 IsAntialias = true,
387 Style = SKPaintStyle.Stroke,
388 StrokeWidth = 1.75f,
389 StrokeCap = SKStrokeCap.Round,
390 StrokeJoin = SKStrokeJoin.Round
391 };
392 using var path = new SKPath();
393
394 path.MoveTo(centerX - width / 2f, centerY - 2f);
395 path.LineTo(centerX, centerY + 2f);
396 path.LineTo(centerX + width / 2f, centerY - 2f);

398 backbuffer.Canvas.DrawPath(path, paint);
