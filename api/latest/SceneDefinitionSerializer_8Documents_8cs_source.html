Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

116 private static SceneDefinition PrepareDocumentMember(SceneDefinition definition, string path)
117 {
118 var clone = ParseDefinition(ToJson(definition), null);
119 if (string.IsNullOrWhiteSpace(clone.ID)) clone.ID = Path.GetFileNameWithoutExtension(path);
120 var origin = definition.Source.GscnFilePath ?? definition.Source.AssetsFilePath;
121 var oldDirectory = definition.LoadStamp?.ReferenceBaseDirectory ??
122 (string.IsNullOrWhiteSpace(origin) ? null : Path.GetDirectoryName(Path.GetFullPath(origin)));
123 var newDirectory = Path.GetDirectoryName(path)!;
124 string? RebasePath(string? value) => string.IsNullOrWhiteSpace(value) || (oldDirectory is null && !Path.IsPathRooted(value)) ? value :
125 Path.GetRelativePath(newDirectory, oldDirectory is null ? Path.GetFullPath(value) : Path.GetFullPath(value, oldDirectory));
126 foreach (var source in clone.TilesheetSources)
127 {
128 source.GtsPath = RebasePath(source.GtsPath);
129 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
130 }
131 foreach (var source in clone.AnimationSources)
132 {
133 source.GaniPath = RebasePath(source.GaniPath);
134 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
135 }
136 clone.Source = SceneDefinitionSource.LooseDefinitionFile(path);
137 return clone;
138 }
