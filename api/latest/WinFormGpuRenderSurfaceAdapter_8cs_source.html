Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

172 private void QueueInvalidate()
173 {
174 if (_glControl.IsDisposed || _presentationControl.IsDisposed) return;
175 if (Interlocked.CompareExchange(ref _pendingInvalidate, 1, 0) != 0) return;
176
177 Engine.Instance.UiDispatcher!.Post(() =>
178 {
179 try
180 {
181 if (!_glControl.IsDisposed
182 && !_presentationControl.IsDisposed
183 && _presentationControl.ClientSize.Width > 0
184 && _presentationControl.ClientSize.Height > 0)
185 {
186 _glControl.Invalidate();
187 }
188 }
189 finally
190 {
191 // The dispatcher callback completed. Do not wait for PaintSurface: WinForms is
192 // allowed to coalesce or discard an invalidation, especially at zero size.
193 Interlocked.Exchange(ref _pendingInvalidate, 0);
194 }
195 });
196 }

198 private void OnPaintSurface(object? sender, SKPaintGLSurfaceEventArgs e)
199 {
200 long telemetryGeneration = _host?.Telemetry?.BeginSample() ?? 0;
201 bool collectDiagnostics = FrameDiagnosticsCalculated is not null || telemetryGeneration != 0;
202 long callbackStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
203 long renderAndSnapshotTicks = 0;
204 long blitTicks = 0;
205 long flushTicks = 0;
206
207 // Lazily sync VSync → GLControl.VSync whenever the backbuffer value changes.
208 if (_gpuBackbuffer != null)
209 {
210 bool desiredVSync = _gpuBackbuffer.VSync;
211 if (_appliedVSync != desiredVSync)
212 {
213 _glControl.VSync = desiredVSync;
214 _appliedVSync = desiredVSync;
215 }
216 }
217
218 // Capture/refresh the GRContext so callers can wire the backbuffer to the same one.
219 GrContext = _glControl.GRContext;
220
221 if (GrContext != null)
222 {
223 if (!_grContextReady)
224 {
225 // First time the context is available: clear any resize that happened before the
226 // first paint (the Initialize call will use the current adapter dimensions).
227 _grContextReady = true;
228 Interlocked.Exchange(ref _pendingResize, 0);
229 GrContextFirstAvailable?.Invoke(GrContext);
230 }
231 else if (Interlocked.Exchange(ref _pendingResize, 0) == 1)
232 {
233 // Notify consumers about presentation resource resize on the GL thread.
234 // WinForms controls can report zero size while minimized or before layout completes;
235 // suppress invalid resize requests so downstream GPU initialization is not attempted
236 // with non-positive dimensions.
237 var width = Width;
238 var height = Height;
239 if (width > 0 && height > 0)
240 ResizeRequested?.Invoke(GrContext, width, height);
241 }
242 }
243
244 if (GrContext != null)
245 _gpuBackbuffer?.EnsureInitialized(GrContext);
246
247 var canvas = e.Surface.Canvas;
248
249 // The physical GL drawable is intentionally kept at least 1×1, even when the wrapper has
250 // no presentation area. Avoid doing scene work for that clipped placeholder surface.
251 if (Width <= 0 || Height <= 0)
252 {
253 canvas.Clear(ClearColor);
254 return;
255 }
256
257 // Render + blit entirely on the GL thread.
258 // GlRenderAndSnapshot replays immutable Engine commands on the GPU surface and returns a
259 // lightweight GPU-backed snapshot. Both the snapshot texture and e.Surface share the
260 // same GRContext, so DrawImage is a zero-copy GPU blit.
261 if (_host != null)
262 {
263 long renderStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
264 using var img = _host.GlRenderAndSnapshot();
265 long renderEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
266
267 if (collectDiagnostics)
268 renderAndSnapshotTicks = renderEndTick - renderStartTick;
269
270 if (img != null)
271 {
272 long blitStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
273 DrawImage(canvas, img, ClearColor);
274 long blitEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
275
276 if (collectDiagnostics)
277 blitTicks = blitEndTick - blitStartTick;
278 }
279 else
280 {
281 canvas.Clear(ClearColor);
282 }
283 }
284 else
285 {
286 canvas.Clear(ClearColor);
287 }
288
289 // Optional: flush to ensure work is queued to GPU before we hand new images next frame.
290 long flushStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
291 _glControl.GRContext?.Flush();
292 long flushEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
293
294 if (collectDiagnostics)
295 flushTicks = flushEndTick - flushStartTick;
296
297 // Record the completed frame so the engine's CPS sampler can compute actual rendered FPS.
298 _gpuBackbuffer?.RecordFrame();
299
300 if (collectDiagnostics)
301 {
302 long callbackEndTick = HighResTimer.GetCurrentTick();
303 if (telemetryGeneration != 0)
304 {
305 var source = _host!.Telemetry!;
306 source.Record(telemetryGeneration, "presentation.count", 1);
307 source.Record(telemetryGeneration, "presentation.cpu.ms", HighResTimer.GetDuration(callbackStartTick, callbackEndTick) * 1000d);
308 source.Record(telemetryGeneration, "render.snapshot.cpu.ms", HighResTimer.GetDuration(0, renderAndSnapshotTicks) * 1000d);
309 source.Record(telemetryGeneration, "blit.cpu.ms", HighResTimer.GetDuration(0, blitTicks) * 1000d);
310 source.Record(telemetryGeneration, "flush.cpu.ms", HighResTimer.GetDuration(0, flushTicks) * 1000d);
311 }
312 FrameDiagnosticsCalculated?.Invoke(new WinFormGpuFrameDiagnostics(
313 HighResTimer.GetDuration(callbackStartTick, callbackEndTick) * 1000d,
314 HighResTimer.GetDuration(0, renderAndSnapshotTicks) * 1000d,
315 HighResTimer.GetDuration(0, blitTicks) * 1000d,
316 HighResTimer.GetDuration(0, flushTicks) * 1000d));
317 }
318 }
