Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

264 private static void SetWidgetZOrder(WidgetBase widget, int zOrder)
265 {
266 DirectDrawingBase[] visuals = EnumerateVisuals(widget).ToArray();
267 if (visuals.Length == 0)
268 return;
269
270 int currentMin = visuals.Min(static visual => visual.ZOrder);
271 int delta = zOrder - currentMin;
272
273 foreach (DirectDrawingBase visual in visuals)
274 visual.ZOrder += delta;
275 }

277 private static IEnumerable<DirectDrawingBase> EnumerateVisuals(IDirectCompositeContainer container)
278 {
279 foreach (IDirectCompositeChild child in container.Children)
280 {
281 if (child is DirectDrawingBase drawing)
282 yield return drawing;
283
284 if (child is IDirectCompositeContainer nestedContainer)
285 {
286 foreach (DirectDrawingBase nestedDrawing in EnumerateVisuals(nestedContainer))
287 yield return nestedDrawing;
288 }
289 }
