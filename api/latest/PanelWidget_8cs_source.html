Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

225 private static void SetWidgetZOrder(WidgetBase widget, int zOrder)
226 {
227 DirectDrawingBase[] visuals = EnumerateVisuals(widget).ToArray();
228 if (visuals.Length == 0)
229 return;
230
231 int currentMin = visuals.Min(static visual => visual.ZOrder);
232 int delta = zOrder - currentMin;
233
234 foreach (DirectDrawingBase visual in visuals)
235 visual.ZOrder += delta;
236 }

238 private static IEnumerable<DirectDrawingBase> EnumerateVisuals(IDirectCompositeContainer container)
239 {
240 foreach (IDirectCompositeChild child in container.Children)
241 {
242 if (child is DirectDrawingBase drawing)
243 yield return drawing;
244
245 if (child is IDirectCompositeContainer nestedContainer)
246 {
247 foreach (DirectDrawingBase nestedDrawing in EnumerateVisuals(nestedContainer))
248 yield return nestedDrawing;
249 }
250 }
