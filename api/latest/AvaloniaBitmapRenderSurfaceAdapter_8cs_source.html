Gondwana Game Engine Development (master) - 2.6.0-unreleased
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

71 public override void Present(SKImage bufferImage, SKRectI bufferRect, SKRect destRect)
72 {
73 var sourceRect = SKRectI.Intersect(
74 bufferRect,
75 new SKRectI(0, 0, bufferImage.Width, bufferImage.Height));
76
77 if (sourceRect.IsEmpty)
78 {
79 bufferImage.Dispose();
80 return;
81 }
82
83 bool scheduleBlit;
84 lock (_presentSync)
85 {
86 if (_disposed)
87 {
88 bufferImage.Dispose();
89 return;
90 }
91
92 var old = _currentImage;
93 _currentImage = bufferImage;
94 if (!ReferenceEquals(old, _currentImage) && old is not null)
95 _toDispose.Enqueue(old);
96
97 _pendingSourceRect = Union(_pendingSourceRect, sourceRect);
98 scheduleBlit = !_blitScheduled;
99 _blitScheduled = true;
100 }
101
102 // Several presents may arrive before the UI thread runs. One scheduled blit copies the
103 // union of their dirty regions from the newest (complete) backbuffer snapshot.
104 if (scheduleBlit)
105 Dispatcher.UIThread.Post(BlitAndInvalidate, DispatcherPriority.Render);
106 }
107
108 private void BlitAndInvalidate()
109 {
110 SKImage? img;
111 SKRectI sourceRect;
112
113 lock (_presentSync)
114 {
115 if (_disposed)
116 {
117 _blitScheduled = false;
118 return;
119 }
120
121 img = _currentImage;
122 sourceRect = _pendingSourceRect;
123 _pendingSourceRect = SKRectI.Empty;
124 _blitScheduled = false;
125 }
126
127 if (img == null) return;
128
129 try
130 {
131 var w = img.Width;
132 var h = img.Height;
133 if (w <= 0 || h <= 0) return;
134
135 var imageBounds = new SKRectI(0, 0, w, h);
136 sourceRect = SKRectI.Intersect(sourceRect, imageBounds);
137 if (sourceRect.IsEmpty) return;
138
139 // A new bitmap has no retained pixels, so its first copy must cover the full image.
140 var bitmap = _bitmap;
141 if (bitmap == null || bitmap.PixelSize.Width != w || bitmap.PixelSize.Height != h)
142 {
143 bitmap?.Dispose();
144 bitmap = new WriteableBitmap(
145 new PixelSize(w, h),
146 new Vector(96, 96),
147 PixelFormat.Bgra8888,
148 AlphaFormat.Premul);
149
150 _bitmap = bitmap;
151 sourceRect = imageBounds;
152 }
153
154 // Copy only the accumulated dirty patch into its matching location in the bitmap.
155 bool copied;
156 using (var fb = bitmap.Lock())
157 {
158 var info = new SKImageInfo(
159 sourceRect.Width,
160 sourceRect.Height,
161 SKColorType.Bgra8888,
162 SKAlphaType.Premul);
163
164 var destinationOffset = checked((sourceRect.Top * fb.RowBytes) + (sourceRect.Left * 4));
165 var destination = IntPtr.Add(fb.Address, destinationOffset);
166
167 copied = img.ReadPixels(
168 info,
169 destination,
170 fb.RowBytes,
171 sourceRect.Left,
172 sourceRect.Top);
173 }
174
175 if (!copied) return;
176
177 _control.SetBitmap(bitmap);
178 _control.InvalidateVisual();
179 }
180 finally
181 {
182 DisposeStaleImages();
183 }
184 }

186 private static SKRectI Union(SKRectI left, SKRectI right)
187 {
188 if (left.IsEmpty) return right;
189 if (right.IsEmpty) return left;
190
191 return new SKRectI(
192 Math.Min(left.Left, right.Left),
193 Math.Min(left.Top, right.Top),
194 Math.Max(left.Right, right.Right),
195 Math.Max(left.Bottom, right.Bottom));
196 }
