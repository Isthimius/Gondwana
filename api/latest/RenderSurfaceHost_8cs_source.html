It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. 119 {
120 _renderSurfaceAdapter = renderSurfaceAdapter ?? throw new ArgumentNullException(nameof(renderSurfaceAdapter));

122 // Host layout changes presentation; only the initial valid layout establishes resolution.

124

126 var h = RenderSurfaceAdapter.Height;

130

132 var scale = Engine.Instance.Configuration.RenderScale;

136 RenderSurfaceAdapter.SetBackbufferSize(_logicalWidth, _logicalHeight);
137 Backbuffer.BeginFrame();

142 private static TBackbuffer CreateBackbuffer(int width, int height)
143 {
144 var typeName = typeof(TBackbuffer).FullName;
145 Console.WriteLine($"CreateBackbuffer called for type: {typeName}, dimensions: {width}x{height}");
146
147 // Use explicit type checking instead of reflection to avoid trimming issues in browser/WASM
148 if (typeof(TBackbuffer) == typeof(Backbuffers.BitmapBackbuffer))
149 {
150 Engine.Logger.LogInformation("Using explicit BitmapBackbuffer constructor for type {TypeName}", typeName);
151 return (TBackbuffer)(object)new Backbuffers.BitmapBackbuffer(width, height);
152 }
153
154 if (typeof(TBackbuffer) == typeof(Backbuffers.GpuBackbuffer))
155 {
156 Engine.Logger.LogInformation("Using explicit GpuBackbuffer constructor for type {TypeName}", typeName);
157 return (TBackbuffer)(object)new Backbuffers.GpuBackbuffer(width, height);
158 }
159
160 // Fallback to Activator for any other backbuffer types
161 Engine.Logger.LogInformation("Falling back to Activator.CreateInstance for type {TypeName}", typeName);
162 return (TBackbuffer)Activator.CreateInstance(typeof(TBackbuffer), width, height)!;
163 }

254 public void Bind(Scene newScene, bool limitCameraToWorldBoundPx = true)
255 {
256 ArgumentNullException.ThrowIfNull(newScene, "Cannot bind to null Scene.");
257
258 if (ReferenceEquals(newScene, _scene))
259 return;
260
261 // Claim the new scene before releasing the old one. If another host owns it,
262 // Bind fails without disturbing this host's current scene.
263 newScene.BindRenderSurfaceHost(this);
264
265 var oldScene = _scene;
266
267 try
268 {
269 // Unregister from and release the old scene.
270 if (!ReferenceEquals(oldScene, Scene.Empty))
271 {
272 oldScene.SceneDisposing -= OnSourceDisposing;
273 oldScene.UnbindRenderSurfaceHost(this);
274 }
275
276 _scene = newScene;
277 ViewManager.BindToScene(_scene, limitCameraToWorldBoundPx);
278
279 _scene.SceneDisposing += OnSourceDisposing;
280 _scene.FullRefreshNeeded = true;
281 }
282 catch
283 {
284 // Release the failed new binding and restore the previous scene.
285 newScene.UnbindRenderSurfaceHost(this);
286 _scene = oldScene;
287
288 if (!ReferenceEquals(oldScene, Scene.Empty))
289 {
290 oldScene.BindRenderSurfaceHost(this);
291 oldScene.SceneDisposing += OnSourceDisposing;
292 }
293
294 throw;
295 }
296
297 BindToScene?.Invoke(this, new RenderSurfaceHostBindEventArgs(oldScene, _scene));
298 }

304 internal override void RenderToBackbuffer(long tick)
305 {
306 Backbuffer.BeginFrame();
307 RenderBackbufferBegin?.Invoke();
308
309 // For GL-thread-rendered backbuffers (GpuBackbuffer), the RefreshQueue mechanism is
310 // unreliable: AddWorldRect and ClearRefreshQueue both post to the engine thread, so
311 // enqueued rects are not present when CollectDirtyScreenArea runs, and posted clears
312 // can silently discard dirty rects that were added after the last GL frame.
313 // Instead, always re-render the entire surface each GL paint callback.
314 if (Backbuffer.IsGlThreadRendered)
315 RenderToBackbufferGpuFull(tick);
316 else
317 RenderToBackbufferBitmap(tick);
318
319 RenderBackbufferEnd?.Invoke();
320 }

352 private void RenderToBackbufferGpuFull(long tick)
353 {
354 bool collectDiagnostics = GpuRenderFrameDiagnosticsCalculated is not null;
355 long diagnosticsStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
356 long queryAndSortTicks = 0;
357 long drawTicks = 0;
358 long overlayTicks = 0;
359 int drawableCount = 0;
360 int tileCount = 0;
361 List<GpuLayerRenderDiagnostics>? layerDiagnostics = collectDiagnostics
362 ? new List<GpuLayerRenderDiagnostics>(Scene.VisibleSceneLayers.Count)
363 : null;
364
365 // When there are no views at all, clear the whole surface and bail.
366 // If there ARE views but no scene layers, we fall through so view-mode
367 // DirectDrawings (e.g. a splash screen overlay) are still rendered.
368 if (ViewManager.Views.Count == 0)
369 {
370 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
371 Scene.FullRefreshNeeded = false;
372 return;
373 }
374
375 // A single clear lets translucent, wiped, or translated views reveal the
376 // views beneath them. Each view paints its own background as part of its
377 // presentation group below.
378 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
379
380 foreach (var view in ViewManager.Views)
381 {
382 RenderContext.Push(view, tick);
383
384 try
385 {
386 // 1) View overlays are drawn unconditionally by this full-frame path.
387 // They must not enqueue dirty regions that GL rendering never consumes.
388 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
389
390 var vp = view.GetRenderViewportTargetRectPx();
391
392 // 2) Clip to this view's viewport, excluding areas covered by higher Z-order views.
393 Backbuffer.Canvas.Save();
394 Backbuffer.Canvas.ResetMatrix();
395 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
396
397 foreach (var blocker in ViewManager.GetViewsAbove(view))
398 {
399 if (!blocker.BlocksViewsBelow)
400 continue;
401
402 var overlap = Rectangle.Intersect(vp, blocker.GetRenderViewportTargetRectPx());
403 if (!overlap.IsEmpty)
404 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
405 }
406
407 int viewPresentation = BeginPresentation(
408 view.GetPresentationBoundsPx(),
409 view.EffectOpacity,
410 view.EffectReveal,
411 view.EffectRevealDirection);
412
413 if (viewPresentation > 0)
414 {
415 // The view background belongs inside the group so it fades,
416 // wipes, and slides with the rest of the view.
417 Backbuffer.ClearRect(view.GetPresentationBoundsPx().ToPixelAlignedRect());
418
419 // 4) Render every visible layer for the full viewport extent (layers are drawn
420 // back-to-front by ascending Z-order, which VisibleSceneLayers already provides).
421 var sceneLayers = Scene.VisibleSceneLayers;
422
423 for (int i = 0; i < sceneLayers.Count; i++)
424 {
425 var layer = sceneLayers[i];
426
427 int layerPresentation = BeginPresentation(
428 GetLayerPresentationBounds(view, layer),
429 layer.EffectOpacity,
430 layer.EffectReveal,
431 layer.EffectRevealDirection);
432
433 if (layerPresentation == 0)
434 continue;
435
436 // Compute the world-space rect visible through this viewport for this layer,
437 // expanded by one tile in each direction to cover boundary rounding.
438 var layerWorldRectF = view.ScreenRectToWorldRect(layer, vp);
439 layerWorldRectF.Inflate(layer.TileWidth, layer.TileHeight);
440 var layerWorldRect = layerWorldRectF.ToPixelAlignedRect();
441
442 long queryStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
443 var drawables = layer.GetDrawablesInWorldRect(layerWorldRect);
444 long queryEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
445
446 int layerTileCount = 0;
447 if (collectDiagnostics)
448 {
449 for (int d = 0; d < drawables.Count; d++)
450 {
451 IDrawable drawable = drawables[d] is WrappedDrawable wrapped
452 ? wrapped.Owner
453 : drawables[d];
454
455 if (drawable is Tile)
456 layerTileCount++;
457 }
458 }
459
460 long drawStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
461 Backbuffer.DrawDrawables(view, drawables, vp);
462 long drawEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
463
464 if (collectDiagnostics)
465 {
466 long layerQueryTicks = queryEndTick - queryStartTick;
467 long layerDrawTicks = drawEndTick - drawStartTick;
468 queryAndSortTicks += layerQueryTicks;
469 drawTicks += layerDrawTicks;
470 drawableCount += drawables.Count;
471 tileCount += layerTileCount;
472
473 layerDiagnostics!.Add(new GpuLayerRenderDiagnostics(
474 i,
475 layer.ID,
476 layer.ZOrder,
477 drawables.Count,
478 layerTileCount,
479 layer.TransformedTiles.Count,
480 layer.TileWidth,
481 layer.TileHeight,
482 HighResTimer.GetDuration(0, layerQueryTicks) * 1000d,
483 HighResTimer.GetDuration(0, layerDrawTicks) * 1000d));
484 }
485
486 EndPresentation(layerPresentation);
487 }
488
489 // 5) Render view-based DirectDrawings on top.
490 long overlayStartTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
491 for (int i = 0; i < overlays.Count; i++)
492 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
493 long overlayEndTick = collectDiagnostics ? HighResTimer.GetCurrentTick() : 0;
494
495 if (collectDiagnostics)
496 overlayTicks += overlayEndTick - overlayStartTick;
497
498 EndPresentation(viewPresentation);
499 }
500
501 Backbuffer.Canvas.Restore();
502 }
503 finally
504 {
505 RenderContext.Pop();
506 }
507 }
508
509 Scene.FullRefreshNeeded = false;
510
511 // Notify subscribers that all scene content has been drawn and the canvas is ready for
512 // post-scene effects. For GPU surfaces this runs on the GL thread while GRContext is
513 // current, so subscribers may safely issue Skia GPU draw calls.
514 InvokePostSceneCanvasHooks();
515
516 if (collectDiagnostics)
517 {
518 long diagnosticsEndTick = HighResTimer.GetCurrentTick();
519 GpuRenderFrameDiagnosticsCalculated?.Invoke(new GpuRenderFrameDiagnostics(
520 HighResTimer.GetDuration(diagnosticsStartTick, diagnosticsEndTick) * 1000d,
521 HighResTimer.GetDuration(0, queryAndSortTicks) * 1000d,
522 HighResTimer.GetDuration(0, drawTicks) * 1000d,
523 HighResTimer.GetDuration(0, overlayTicks) * 1000d,
524 drawableCount,
525 tileCount,
526 layerDiagnostics!));
527 }
528 }

565 private void RenderToBackbufferBitmap(long tick)
566 {
567 if (ViewManager.Views.Count == 0)
568 {
569 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
570 Scene.FullRefreshNeeded = false;
571 return;
572 }
573
574 bool fullViewComposition = ViewManager.Views.Any(view => view.HasPresentationEffect)
575 || Scene.VisibleSceneLayers.Any(layer => layer.WrapHorizontally || layer.WrapVertically);
576
577 // 0) If there are no visible SceneLayers, just clear and publish the full frame.
578 if (Scene.CountOfVisibleLayers == 0)
579 {
580 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
581 Scene.FullRefreshNeeded = false;
582 }
583 else
584 {
585 // 1) Handle full scene refresh once (camera moved, zoom changed, etc.): clear and mark all layers as dirty.
586 // This already clears the whole backbuffer and enqueues a full rect per layer.
587 if (Scene.FullRefreshNeeded || (fullViewComposition && Scene.IsDirty))
588 {
589 // this will mark the Scene.IsDirty flag as true
590 EnqueueFullSceneRefresh();
591 }
592 else
593 {
594 // scene is not dirty, this frame is done...
595 if (!Scene.IsDirty)
596 {
597 RenderBackbufferNoOp?.Invoke();
598 return;
599 }
600 }
601 }
602
603 if (fullViewComposition)
604 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
605
606 // 2) Render all views to Backbuffer. Draw layers back -> front (ascending Z).
607 foreach (var view in ViewManager.Views)
608 {
609 RenderContext.Push(view, tick);
610
611 try
612 {
613 // 2.1) Force-refresh all DirectDrawings that overlay this view
614 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
615 foreach (var overlay in overlays)
616 {
617 overlay.ForceRefresh();
618 }
619
620 // 2.2) identify dirty SCREEN areas across all layers for this view
621 var dirtyScreenRects = CollectDirtyScreenArea(view);
622
623 // 2.3) Clip to this view's viewport
624 var vp = view.GetRenderViewportTargetRectPx();
625
626 // clip inclusive to current viewport
627 Backbuffer.Canvas.Save();
628 Backbuffer.Canvas.ResetMatrix();
629 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
630
631 // clip exclusive to higher Z-order views
632 foreach (var blocker in ViewManager.GetViewsAbove(view))
633 {
634 if (!blocker.BlocksViewsBelow)
635 continue;
636
637 var overlap = Rectangle.Intersect(vp, blocker.GetRenderViewportTargetRectPx());
638 if (!overlap.IsEmpty)
639 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
640 }
641
642 int viewPresentation = BeginPresentation(
643 view.GetPresentationBoundsPx(),
644 view.EffectOpacity,
645 view.EffectReveal,
646 view.EffectRevealDirection);
647
648 if (viewPresentation > 0)
649 {
650 if (fullViewComposition)
651 {
652 Backbuffer.ClearRect(view.GetPresentationBoundsPx().ToPixelAlignedRect());
653 }
654 else
655 {
656 // 2.4) Pre-clear dirty areas on backbuffer to Backbuffer.ClearColor
657 PreclearScreenAreas(view, dirtyScreenRects);
658 }
659
660 // 2.5) Render each visible layer's dirty regions for this view
661 // this will draw SceneLayerTiles, Sprites, and SceneLayer-based DirectDrawings
662 var sceneLayers = Scene.VisibleSceneLayers;
663
664 for (int i = 0; i < sceneLayers.Count; i++)
665 {
666 var layer = sceneLayers[i];
667 int layerPresentation = BeginPresentation(
668 GetLayerPresentationBounds(view, layer),
669 layer.EffectOpacity,
670 layer.EffectReveal,
671 layer.EffectRevealDirection);
672
673 if (layerPresentation == 0)
674 continue;
675
676 RenderLayerDirtyRegions(view, layer);
677 EndPresentation(layerPresentation);
678 }
679
680 // 2.6) draw all View-based DirectDrawings for this view
681 for (int i = 0; i < overlays.Count; i++)
682 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
683
684 EndPresentation(viewPresentation);
685 }
686
687 // 2.7) Restore from viewport clip
688 Backbuffer.Canvas.Restore();
689 }
690 finally
691 {
692 RenderContext.Pop();
693 }
694 }
695
696 // 3) Clear layer queues now that we've consumed them (avoids re-drawing same tiles next frame)
697 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
698 Scene.VisibleSceneLayers[i].RefreshQueue.ClearRefreshQueue();
699
700 Scene.FullRefreshNeeded = false;
701
702 // Notify subscribers that all scene content has been drawn and the canvas is ready for
703 // post-scene effects. Fires before RenderBackbufferEnd so subscribers still have a
704 // chance to draw into the canvas before it is presented to the adapter.
705 InvokePostSceneCanvasHooks();
706 }

710 private void EnqueueFullSceneRefresh()
711 {
712 foreach (var view in ViewManager.Views)
713 {
714 var viewport = view.Viewport;
715 var screenRect = viewport.TargetRectPx;
716
717 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
718 {
719 var layer = Scene.VisibleSceneLayers[i];
720
721 // 1) Compute layer-specific visible world rect
722 var layerWorldRect = view.ScreenRectToWorldRect(layer, screenRect);
723
724 // 2) Expand world rect by **one full tile** in all directions.
725 // This compensates for fractional-layer motion (parallax),
726 // camera motion, and tile boundaries.
727 int expandX = layer.TileWidth;
728 int expandY = layer.TileHeight;
729
730 layerWorldRect.Inflate(expandX, expandY);
731
732 // 3) Round to ints and enqueue
733 var worldRectInt = layerWorldRect.ToPixelAlignedRect();
734
735 layer.RefreshQueue.AddWorldRect(worldRectInt);
736 }
737 }
738 }

740 private List<Rectangle> CollectDirtyScreenArea(View view)
741 {
742 var dirty = new List<Rectangle>(64);
743 var viewportRect = view.GetRenderViewportTargetRectPx();
744
745 foreach (var sceneLayer in Scene.VisibleSceneLayers)
746 {
747 var refreshQueue = sceneLayer.RefreshQueue;
748 if (!refreshQueue.IsDirty)
749 continue;
750
751 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
752 {
753 var screenRectF = view.WorldRectToScreenRect(sceneLayer, worldRect);
754 var rect = Rectangle.Intersect(
755 screenRectF.ToPixelAlignedRect(),
756 viewportRect);
757
758 if (rect.IsEmpty)
759 continue;
760
761 dirty.AddDeduped(rect);
762 }
763 }
764
765 return dirty;
766 }

768 private void PreclearScreenAreas(View view, List<Rectangle> screenRects)
769 {
770 if (Backbuffer is null || screenRects is null || screenRects.Count == 0)
771 return;
772
773 foreach (var screenRect in screenRects)
774 {
775 var screenRectViewport = Rectangle.Intersect(screenRect, view.GetRenderViewportTargetRectPx());
776
777 if (screenRectViewport.IsEmpty || screenRectViewport.Width <= 0 || screenRectViewport.Height <= 0)
778 continue;
779
780 // Clear just this patch (overwrite with Backbuffer.ClearColor)
781 Backbuffer.ClearRect(screenRectViewport);
782
783 EnqueueForOverlappingSceneLayers(view, screenRectViewport);
784 }
785 }
786
787 private void EnqueueForOverlappingSceneLayers(View sourceView, Rectangle clearedScreenRect)
788 {
789 var overlap = Rectangle.Intersect(clearedScreenRect, sourceView.GetRenderViewportTargetRectPx());
790
791 if (overlap.IsEmpty)
792 return;
793
794 foreach (var sceneLayer in Scene.VisibleSceneLayers)
795 {
796 var worldRectF = sourceView.ScreenRectToWorldRect(sceneLayer, overlap);
797 var worldRect = worldRectF.ToPixelAlignedRect();
798
799 if (!worldRect.IsEmpty)
800 sceneLayer.RefreshQueue.AddWorldRect(worldRect);
801 }
802 }

804 private void RenderLayerDirtyRegions(View view, SceneLayer layer)
805 {
806 var refreshQueue = layer.RefreshQueue;
807
808 // if this layer has no dirty regions and we are not forcing a full redraw, skip it.
809 if (!refreshQueue.IsDirty)
810 return;
811
812 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
813 {
814 // draw tiles/sprites/direct drawings in this world rect
815 var drawables = layer.GetDrawablesInWorldRect(worldRect);
816
817 // project world → screen for adapter dirty
818 var screenRect = view.WorldRectToScreenRect(layer, worldRect).ToPixelAlignedRect();
819
820 Backbuffer.DrawDrawables(view, drawables, screenRect);
821 }
822 }

824 private int BeginPresentation(
825 RectangleF bounds,
826 float opacity,
827 float reveal,
828 EffectDirection direction)
829 {
830 opacity = Math.Clamp(opacity, 0f, 1f);
831 reveal = Math.Clamp(reveal, 0f, 1f);
832
833 if (opacity <= 0.0001f || reveal <= 0.0001f || bounds.IsEmpty)
834 return 0;
835
836 int saveCount = 1;
837 Backbuffer.Canvas.Save();
838
839 if (reveal < 0.9999f)
840 {
841 RectangleF revealRect = EffectGeometry.GetRevealRect(bounds, direction, reveal);
842 Backbuffer.Canvas.ClipRect(
843 revealRect.ToSKRect(),
844 SKClipOperation.Intersect,
845 antialias: false);
846 }
847
848 if (opacity < 0.9999f)
849 {
850 using var paint = new SKPaint
851 {
852 Color = new SKColor(255, 255, 255, (byte)Math.Round(opacity * 255f))
853 };
854
855 Backbuffer.Canvas.SaveLayer(bounds.ToSKRect(), paint);
856 saveCount++;
857 }
858
859 return saveCount;
860 }

862 private void EndPresentation(int saveCount)
863 {
864 while (saveCount-- > 0)
865 Backbuffer.Canvas.Restore();
866 }

960 internal override void RequestRenderScale(float scale)
961 {
962 if (!_resolutionEstablished || _disposed) return;
963 var adapter = RenderSurfaceAdapter;
964 // Zero-size requests are deferred until a valid presentation size exists.
965 if (adapter.Width <= 0 || adapter.Height <= 0)
966 {
967 _deferredRenderScale = scale;
968 return;
969 }

971 Backbuffer.RequestResize(PresentationTransform.ScaleDimension(adapter.Width, scale),
972 PresentationTransform.ScaleDimension(adapter.Height, scale));
973 }

975 private void OnBackbufferSizeChanged(int width, int height)
976 {
977 float scaleX = (float)width / _logicalWidth;
978 float scaleY = (float)height / _logicalHeight;
