Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

135 private static TBackbuffer CreateBackbuffer(int width, int height)
136 {
137 var typeName = typeof(TBackbuffer).FullName;
138 Console.WriteLine($"CreateBackbuffer called for type: {typeName}, dimensions: {width}x{height}");
139
140 // Use explicit type checking instead of reflection to avoid trimming issues in browser/WASM
141 if (typeof(TBackbuffer) == typeof(Backbuffers.BitmapBackbuffer))
142 {
143 Engine.Logger.LogInformation("Using explicit BitmapBackbuffer constructor for type {TypeName}", typeName);
144 return (TBackbuffer)(object)new Backbuffers.BitmapBackbuffer(width, height);
145 }
146
147 if (typeof(TBackbuffer) == typeof(Backbuffers.GpuBackbuffer))
148 {
149 Engine.Logger.LogInformation("Using explicit GpuBackbuffer constructor for type {TypeName}", typeName);
150 return (TBackbuffer)(object)new Backbuffers.GpuBackbuffer(width, height);
151 }
152
153 // Fallback to Activator for any other backbuffer types
154 Engine.Logger.LogInformation("Falling back to Activator.CreateInstance for type {TypeName}", typeName);
155 return (TBackbuffer)Activator.CreateInstance(typeof(TBackbuffer), width, height)!;
156 }

247 public void Bind(Scene newScene, bool limitCameraToWorldBoundPx = true)
248 {
249 ArgumentNullException.ThrowIfNull(newScene, "Cannot bind to null Scene.");
250
251 if (ReferenceEquals(newScene, _scene))
252 return;
253
254 // Claim the new scene before releasing the old one. If another host owns it,
255 // Bind fails without disturbing this host's current scene.
256 newScene.BindRenderSurfaceHost(this);
257
258 var oldScene = _scene;
259
260 try
261 {
262 // Unregister from and release the old scene.
263 if (!ReferenceEquals(oldScene, Scene.Empty))
264 {
265 oldScene.SceneDisposing -= OnSourceDisposing;
266 oldScene.UnbindRenderSurfaceHost(this);
267 }
268
269 _scene = newScene;
270 ViewManager.BindToScene(_scene, limitCameraToWorldBoundPx);
271
272 _scene.SceneDisposing += OnSourceDisposing;
273 _scene.FullRefreshNeeded = true;
274 }
275 catch
276 {
277 // Release the failed new binding and restore the previous scene.
278 newScene.UnbindRenderSurfaceHost(this);
279 _scene = oldScene;
280
281 if (!ReferenceEquals(oldScene, Scene.Empty))
282 {
283 oldScene.BindRenderSurfaceHost(this);
284 oldScene.SceneDisposing += OnSourceDisposing;
285 }
286
287 throw;
288 }
289
290 BindToScene?.Invoke(this, new RenderSurfaceHostBindEventArgs(oldScene, _scene));
291 }

297 internal override void RenderToBackbuffer(long tick)
298 {
299 Backbuffer.BeginFrame();
300 RenderBackbufferBegin?.Invoke();
301
302 // For GL-thread-rendered backbuffers (GpuBackbuffer), the RefreshQueue mechanism is
303 // unreliable: AddWorldRect and ClearRefreshQueue both post to the engine thread, so
304 // enqueued rects are not present when CollectDirtyScreenArea runs, and posted clears
305 // can silently discard dirty rects that were added after the last GL frame.
306 // Instead, always re-render the entire surface each GL paint callback.
307 if (Backbuffer.IsGlThreadRendered)
308 RenderToBackbufferGpuFull(tick);
309 else
310 RenderToBackbufferBitmap(tick);
311
312 RenderBackbufferEnd?.Invoke();
313 }

345 private void RenderToBackbufferGpuFull(long tick)
346 {
347 // When there are no views at all, clear the whole surface and bail.
348 // If there ARE views but no scene layers, we fall through so view-mode
349 // DirectDrawings (e.g. a splash screen overlay) are still rendered.
350 if (ViewManager.Views.Count == 0)
351 {
352 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
353 Scene.FullRefreshNeeded = false;
354 return;
355 }
356
357 // A single clear lets translucent, wiped, or translated views reveal the
358 // views beneath them. Each view paints its own background as part of its
359 // presentation group below.
360 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
361
362 foreach (var view in ViewManager.Views)
363 {
364 RenderContext.Push(view, tick);
365
366 try
367 {
368 // 1) View overlays are drawn unconditionally by this full-frame path.
369 // They must not enqueue dirty regions that GL rendering never consumes.
370 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
371
372 var vp = view.GetRenderViewportTargetRectPx();
373
374 // 2) Clip to this view's viewport, excluding areas covered by higher Z-order views.
375 Backbuffer.Canvas.Save();
376 Backbuffer.Canvas.ResetMatrix();
377 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
378
379 foreach (var blocker in ViewManager.GetViewsAbove(view))
380 {
381 if (!blocker.BlocksViewsBelow)
382 continue;
383
384 var overlap = Rectangle.Intersect(vp, blocker.GetRenderViewportTargetRectPx());
385 if (!overlap.IsEmpty)
386 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
387 }
388
389 int viewPresentation = BeginPresentation(
390 view.GetPresentationBoundsPx(),
391 view.EffectOpacity,
392 view.EffectReveal,
393 view.EffectRevealDirection);
394
395 if (viewPresentation > 0)
396 {
397 // The view background belongs inside the group so it fades,
398 // wipes, and slides with the rest of the view.
399 Backbuffer.ClearRect(view.GetPresentationBoundsPx().ToPixelAlignedRect());
400
401 // 4) Render every visible layer for the full viewport extent (layers are drawn
402 // back-to-front by ascending Z-order, which VisibleSceneLayers already provides).
403 var sceneLayers = Scene.VisibleSceneLayers;
404
405 for (int i = 0; i < sceneLayers.Count; i++)
406 {
407 var layer = sceneLayers[i];
408
409 int layerPresentation = BeginPresentation(
410 GetLayerPresentationBounds(view, layer),
411 layer.EffectOpacity,
412 layer.EffectReveal,
413 layer.EffectRevealDirection);
414
415 if (layerPresentation == 0)
416 continue;
417
418 // Compute the world-space rect visible through this viewport for this layer,
419 // expanded by one tile in each direction to cover boundary rounding.
420 var layerWorldRectF = view.ScreenRectToWorldRect(layer, vp);
421 layerWorldRectF.Inflate(layer.TileWidth, layer.TileHeight);
422 var layerWorldRect = layerWorldRectF.ToPixelAlignedRect();
423
424 var drawables = layer.GetDrawablesInWorldRect(layerWorldRect);
425 Backbuffer.DrawDrawables(view, drawables, vp);
426
427 EndPresentation(layerPresentation);
428 }
429
430 // 5) Render view-based DirectDrawings on top.
431 for (int i = 0; i < overlays.Count; i++)
432 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
433
434 EndPresentation(viewPresentation);
435 }
436
437 Backbuffer.Canvas.Restore();
438 }
439 finally
440 {
441 RenderContext.Pop();
442 }
443 }
444
445 Scene.FullRefreshNeeded = false;
446
447 // Notify subscribers that all scene content has been drawn and the canvas is ready for
448 // post-scene effects. For GPU surfaces this runs on the GL thread while GRContext is
449 // current, so subscribers may safely issue Skia GPU draw calls.
450 InvokePostSceneCanvasHooks();
451 }

488 private void RenderToBackbufferBitmap(long tick)
489 {
490 if (ViewManager.Views.Count == 0)
491 {
492 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
493 Scene.FullRefreshNeeded = false;
494 return;
495 }
496
497 bool fullViewComposition = ViewManager.Views.Any(view => view.HasPresentationEffect);
498
499 // 0) If there are no visible SceneLayers, just clear and publish the full frame.
500 if (Scene.CountOfVisibleLayers == 0)
501 {
502 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
503 Scene.FullRefreshNeeded = false;
504 }
505 else
506 {
507 // 1) Handle full scene refresh once (camera moved, zoom changed, etc.): clear and mark all layers as dirty.
508 // This already clears the whole backbuffer and enqueues a full rect per layer.
509 if (Scene.FullRefreshNeeded || (fullViewComposition && Scene.IsDirty))
510 {
511 // this will mark the Scene.IsDirty flag as true
512 EnqueueFullSceneRefresh();
513 }
514 else
515 {
516 // scene is not dirty, this frame is done...
517 if (!Scene.IsDirty)
518 {
519 RenderBackbufferNoOp?.Invoke();
520 return;
521 }
522 }
523 }
524
525 if (fullViewComposition)
526 Backbuffer.ClearRect(new Rectangle(0, 0, Backbuffer.Width, Backbuffer.Height));
527
528 // 2) Render all views to Backbuffer. Draw layers back -> front (ascending Z).
529 foreach (var view in ViewManager.Views)
530 {
531 RenderContext.Push(view, tick);
532
533 try
534 {
535 // 2.1) Force-refresh all DirectDrawings that overlay this view
536 var overlays = DirectDrawingManager.Instance.GetDrawingsForView(view);
537 foreach (var overlay in overlays)
538 {
539 overlay.ForceRefresh();
540 }
541
542 // 2.2) identify dirty SCREEN areas across all layers for this view
543 var dirtyScreenRects = CollectDirtyScreenArea(view);
544
545 // 2.3) Clip to this view's viewport
546 var vp = view.GetRenderViewportTargetRectPx();
547
548 // clip inclusive to current viewport
549 Backbuffer.Canvas.Save();
550 Backbuffer.Canvas.ResetMatrix();
551 Backbuffer.Canvas.ClipRect(vp.ToSKRect(), SKClipOperation.Intersect, antialias: false);
552
553 // clip exclusive to higher Z-order views
554 foreach (var blocker in ViewManager.GetViewsAbove(view))
555 {
556 if (!blocker.BlocksViewsBelow)
557 continue;
558
559 var overlap = Rectangle.Intersect(vp, blocker.GetRenderViewportTargetRectPx());
560 if (!overlap.IsEmpty)
561 Backbuffer.Canvas.ClipRect(overlap.ToSKRect(), SKClipOperation.Difference, antialias: false);
562 }
563
564 int viewPresentation = BeginPresentation(
565 view.GetPresentationBoundsPx(),
566 view.EffectOpacity,
567 view.EffectReveal,
568 view.EffectRevealDirection);
569
570 if (viewPresentation > 0)
571 {
572 if (fullViewComposition)
573 {
574 Backbuffer.ClearRect(view.GetPresentationBoundsPx().ToPixelAlignedRect());
575 }
576 else
577 {
578 // 2.4) Pre-clear dirty areas on backbuffer to Backbuffer.ClearColor
579 PreclearScreenAreas(view, dirtyScreenRects);
580 }
581
582 // 2.5) Render each visible layer's dirty regions for this view
583 // this will draw SceneLayerTiles, Sprites, and SceneLayer-based DirectDrawings
584 var sceneLayers = Scene.VisibleSceneLayers;
585
586 for (int i = 0; i < sceneLayers.Count; i++)
587 {
588 var layer = sceneLayers[i];
589 int layerPresentation = BeginPresentation(
590 GetLayerPresentationBounds(view, layer),
591 layer.EffectOpacity,
592 layer.EffectReveal,
593 layer.EffectRevealDirection);
594
595 if (layerPresentation == 0)
596 continue;
597
598 RenderLayerDirtyRegions(view, layer);
599 EndPresentation(layerPresentation);
600 }
601
602 // 2.6) draw all View-based DirectDrawings for this view
603 for (int i = 0; i < overlays.Count; i++)
604 overlays[i].Draw(Backbuffer, overlays[i].GetDrawLocationScreen(view));
605
606 EndPresentation(viewPresentation);
607 }
608
609 // 2.7) Restore from viewport clip
610 Backbuffer.Canvas.Restore();
611 }
612 finally
613 {
614 RenderContext.Pop();
615 }
616 }
617
618 // 3) Clear layer queues now that we've consumed them (avoids re-drawing same tiles next frame)
619 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
620 Scene.VisibleSceneLayers[i].RefreshQueue.ClearRefreshQueue();
621
622 Scene.FullRefreshNeeded = false;
623
624 // Notify subscribers that all scene content has been drawn and the canvas is ready for
625 // post-scene effects. Fires before RenderBackbufferEnd so subscribers still have a
626 // chance to draw into the canvas before it is presented to the adapter.
627 InvokePostSceneCanvasHooks();
628 }

632 private void EnqueueFullSceneRefresh()
633 {
634 foreach (var view in ViewManager.Views)
635 {
636 var viewport = view.Viewport;
637 var screenRect = viewport.TargetRectPx;
638
639 for (int i = 0; i < Scene.CountOfVisibleLayers; i++)
640 {
641 var layer = Scene.VisibleSceneLayers[i];
642
643 // 1) Compute layer-specific visible world rect
644 var layerWorldRect = view.ScreenRectToWorldRect(layer, screenRect);
645
646 // 2) Expand world rect by **one full tile** in all directions.
647 // This compensates for fractional-layer motion (parallax),
648 // camera motion, and tile boundaries.
649 int expandX = layer.TileWidth;
650 int expandY = layer.TileHeight;
651
652 layerWorldRect.Inflate(expandX, expandY);
653
654 // 3) Round to ints and enqueue
655 var worldRectInt = layerWorldRect.ToPixelAlignedRect();
656
657 layer.RefreshQueue.AddWorldRect(worldRectInt);
658 }
659 }
660 }

662 private List<Rectangle> CollectDirtyScreenArea(View view)
663 {
664 var dirty = new List<Rectangle>(64);
665 var viewportRect = view.GetRenderViewportTargetRectPx();
666
667 foreach (var sceneLayer in Scene.VisibleSceneLayers)
668 {
669 var refreshQueue = sceneLayer.RefreshQueue;
670 if (!refreshQueue.IsDirty)
671 continue;
672
673 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
674 {
675 var screenRectF = view.WorldRectToScreenRect(sceneLayer, worldRect);
676 var rect = Rectangle.Intersect(
677 screenRectF.ToPixelAlignedRect(),
678 viewportRect);
679
680 if (rect.IsEmpty)
681 continue;
682
683 dirty.AddDeduped(rect);
684 }
685 }
686
687 return dirty;
688 }

690 private void PreclearScreenAreas(View view, List<Rectangle> screenRects)
691 {
692 if (Backbuffer is null || screenRects is null || screenRects.Count == 0)
693 return;
694
695 foreach (var screenRect in screenRects)
696 {
697 var screenRectViewport = Rectangle.Intersect(screenRect, view.GetRenderViewportTargetRectPx());
698
699 if (screenRectViewport.IsEmpty || screenRectViewport.Width <= 0 || screenRectViewport.Height <= 0)
700 continue;
701
702 // Clear just this patch (overwrite with Backbuffer.ClearColor)
703 Backbuffer.ClearRect(screenRectViewport);
704
705 EnqueueForOverlappingSceneLayers(view, screenRectViewport);
706 }
707 }

709 private void EnqueueForOverlappingSceneLayers(View sourceView, Rectangle clearedScreenRect)
710 {
711 var overlap = Rectangle.Intersect(clearedScreenRect, sourceView.GetRenderViewportTargetRectPx());
712
713 if (overlap.IsEmpty)
714 return;
715
716 foreach (var sceneLayer in Scene.VisibleSceneLayers)
717 {
718 var worldRectF = sourceView.ScreenRectToWorldRect(sceneLayer, overlap);
719 var worldRect = worldRectF.ToPixelAlignedRect();
720
721 if (!worldRect.IsEmpty)
722 sceneLayer.RefreshQueue.AddWorldRect(worldRect);
723 }
724 }

726 private void RenderLayerDirtyRegions(View view, SceneLayer layer)
727 {
728 var refreshQueue = layer.RefreshQueue;
729
730 // if this layer has no dirty regions and we are not forcing a full redraw, skip it.
731 if (!refreshQueue.IsDirty)
732 return;
733
734 foreach (var worldRect in refreshQueue.SnapshotWorldRects())
735 {
736 // draw tiles/sprites/direct drawings in this world rect
737 var drawables = layer.GetDrawablesInWorldRect(worldRect);
738
739 // project world → screen for adapter dirty
740 var screenRect = view.WorldRectToScreenRect(layer, worldRect).ToPixelAlignedRect();
741
742 Backbuffer.DrawDrawables(view, drawables, screenRect);
743 }

746 private int BeginPresentation(
747 RectangleF bounds,
748 float opacity,
749 float reveal,
750 EffectDirection direction)
751 {
752 opacity = Math.Clamp(opacity, 0f, 1f);
753 reveal = Math.Clamp(reveal, 0f, 1f);
754
755 if (opacity <= 0.0001f || reveal <= 0.0001f || bounds.IsEmpty)
756 return 0;
757
758 int saveCount = 1;
759 Backbuffer.Canvas.Save();
760
761 if (reveal < 0.9999f)
762 {
763 RectangleF revealRect = EffectGeometry.GetRevealRect(bounds, direction, reveal);
764 Backbuffer.Canvas.ClipRect(
765 revealRect.ToSKRect(),
766 SKClipOperation.Intersect,
767 antialias: false);
768 }
769
770 if (opacity < 0.9999f)
771 {
772 using var paint = new SKPaint
773 {
774 Color = new SKColor(255, 255, 255, (byte)Math.Round(opacity * 255f))
775 };
776
777 Backbuffer.Canvas.SaveLayer(bounds.ToSKRect(), paint);
778 saveCount++;
779 }
780
781 return saveCount;
782 }

897 private void OnBackbufferSizeChanged(int width, int height)
898 {
899 float scaleX = (float)width / _logicalWidth;
900 float scaleY = (float)height / _logicalHeight;
901 _logicalWidth = width;
902 _logicalHeight = height;
903 RenderSurfaceAdapter.SetBackbufferSize(width, height);
904 foreach (var view in ViewManager.Views)
905 {
906 var old = view.Viewport.TargetRectPx;
907 view.Viewport.TargetRectPx = Rectangle.FromLTRB(
908 (int)Math.Round(old.Left * scaleX), (int)Math.Round(old.Top * scaleY),
909 (int)Math.Round(old.Right * scaleX), (int)Math.Round(old.Bottom * scaleY));
910 }
911 Scene.FullRefreshNeeded = true;
912 Interlocked.Exchange(ref _presentationInvalidated, 1);
