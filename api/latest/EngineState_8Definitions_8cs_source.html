Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

46 internal void Validate(string? loosePath, object? definition, object? legacy = null)
47 {
48 bool hasArchive = AssetsFilePath is not null;
49 bool hasEntry = AssetEntryName is not null;
50 if (hasArchive != hasEntry ||
51 (hasArchive && (string.IsNullOrWhiteSpace(AssetsFilePath) || string.IsNullOrWhiteSpace(AssetEntryName))))
52 throw new InvalidDataException($"Packed definition requires both AssetsFilePath and AssetEntryName: '{AssetsFilePath}' / '{AssetEntryName}'.");
53 int sources = (loosePath is not null ? 1 : 0) + (definition is not null ? 1 : 0) +
54 (hasArchive ? 1 : 0) + (legacy is not null ? 1 : 0);
55 if (sources != 1 || (loosePath is not null && string.IsNullOrWhiteSpace(loosePath)))
56 throw new InvalidDataException("Definition state entry requires exactly one source: inline Definition, loose path, or AssetsFilePath + AssetEntryName.");
57 }

149 private static T LoadPacked<T>(DefinitionStateEntry entry, string? baseDirectory, AssetTypes expectedType, Func<AssetsFile, string, T> load)
150 {
151 var path = ResolvePath(entry.AssetsFilePath!, baseDirectory);
152 try
153 {
154 var archive = FindAssetsFile(path);
155 if (archive is null)
156 {
157 if (!File.Exists(path)) throw new FileNotFoundException("Archive not found.", path);
158 archive = AssetsFile.LoadOrCreate(path);
159 }
160 // EngineState references are exact identities, not the convenience basename
161 // aliases accepted by AssetsFile.Get for interactive runtime lookup.
162 if (!archive.GetAllEntries().Any(candidate => candidate.AssetType == expectedType &&
163 string.Equals(candidate.AssetName, entry.AssetEntryName, StringComparison.Ordinal)))
164 throw new InvalidDataException($"Entry does not exist with expected type '{expectedType}'.");
165 return load(archive, entry.AssetEntryName!);
166 }
167 catch (Exception ex) when (ex is not OutOfMemoryException)
168 {
169 throw new InvalidDataException($"Cannot load packed definition '{entry.AssetEntryName}' from GAF '{path}'.", ex);
170 }
171 }
