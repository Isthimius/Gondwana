Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 internal void Validate(string? loosePath, object? definition, object? legacy = null)
51 {
52 bool hasArchive = AssetsFilePath is not null;
53 bool hasEntry = AssetEntryName is not null;
54
55 if (hasArchive != hasEntry ||
56 (hasArchive && (string.IsNullOrWhiteSpace(AssetsFilePath) ||
57 string.IsNullOrWhiteSpace(AssetEntryName))))
58 throw new InvalidDataException(
59 $"Packed definition requires both AssetsFilePath and AssetEntryName: '{AssetsFilePath}' / '{AssetEntryName}'.");
60
61 int sources = (loosePath is not null ? 1 : 0) + (definition is not null ? 1 : 0) +
62 (hasArchive ? 1 : 0) + (legacy is not null ? 1 : 0);
63
64 if (sources != 1 || (loosePath is not null && string.IsNullOrWhiteSpace(loosePath)))
65 throw new InvalidDataException(
66 "Definition state entry requires exactly one source: inline Definition, loose path, or AssetsFilePath + AssetEntryName.");
67 }

148 private static bool TryPreserve(
149 DefinitionStateEntry entry,
150 DefinitionProvenance? provenance,
151 object definition,
152 DefinitionPersistence persistence,
153 string? baseDirectory,
154 out string? loosePath)
155 {
156 loosePath = null;
157
158 if (persistence != DefinitionPersistence.PreserveSource || provenance is null ||
159 !provenance.Matches(definition))
160 return false;
161
162 entry.DefinitionIdentity = provenance.DefinitionIdentity;
163
164 if (provenance.AssetsFilePath is { } archivePath)
165 {
166 entry.AssetsFilePath = MakeRelativePath(archivePath, baseDirectory);
167 entry.AssetEntryName = provenance.AssetEntryName;
168 return true;
169 }
170
171 if (provenance.FilePath is { } filePath)
172 {
173 loosePath = MakeRelativePath(filePath, baseDirectory);
174 return true;
175 }
176
177 return false;
178 }

205 private static T LoadPacked<T>(
206 DefinitionStateEntry entry,
207 string? baseDirectory,
208 AssetTypes expectedType,
209 Func<AssetsFile, string, T> load)
210 {
211 var path = ResolvePath(entry.AssetsFilePath!, baseDirectory);
212
213 try
214 {
215 var archive = FindAssetsFile(path);
216
217 if (archive is null)
218 {
219 if (!File.Exists(path))
220 throw new FileNotFoundException("Archive not found.", path);
221
222 archive = AssetsFile.LoadOrCreate(path);
223 }
224
225 // EngineState references are exact identities, not the convenience basename
226 // aliases accepted by AssetsFile.Get for interactive runtime lookup.
227 if (!archive.GetAllEntries().Any(candidate => candidate.AssetType == expectedType &&
228 string.Equals(
229 candidate.AssetName,
230 entry.AssetEntryName,
231 StringComparison.Ordinal)))
232 throw new InvalidDataException($"Entry does not exist with expected type '{expectedType}'.");
233
234 return load(archive, entry.AssetEntryName!);
235 }
236 catch (Exception ex) when (ex is not OutOfMemoryException)
237 {
238 throw new InvalidDataException(
239 $"Cannot load packed definition '{entry.AssetEntryName}' from GAF '{path}'.",
240 ex);
241 }
242 }
