Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

423 internal static SKRect ComputeDestRect(
424 RectangleF bounds,
425 int srcW,
426 int srcH,
427 StretchMode mode)
428 {
429 var b = bounds;
430
431 var dst = new SKRect(
432 b.Left,
433 b.Top,
434 b.Right,
435 b.Bottom);
436
437 if (mode == StretchMode.Fill || srcW <= 0 || srcH <= 0)
438 return dst;
439
440 float bw = b.Width, bh = b.Height;
441 float arSrc = (float)srcW / srcH;
442 float arDst = bw / bh;
443
444 if (mode == StretchMode.None)
445 return new SKRect(
446 b.Left,
447 b.Top,
448 b.Left + srcW,
449 b.Top + srcH);
450
451 if (mode == StretchMode.Uniform)
452 {
453 if (arSrc > arDst)
454 {
455 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
456
457 return new SKRect(
458 b.Left,
459 y,
460 b.Left + w,
461 y + h);
462 }
463 else
464 {
465 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
466
467 return new SKRect(
468 x,
469 b.Top,
470 x + w,
471 b.Top + h);
472 }
473 }
474
475 // UniformToFill
476 if (arSrc < arDst)
477 {
478 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
479
480 return new SKRect(
481 b.Left,
482 y,
483 b.Left + w,
484 y + h);
485 }
486 else
487 {
488 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
489
490 return new SKRect(
491 x,
492 b.Top,
493 x + w,
494 b.Top + h);
495 }
496 }
