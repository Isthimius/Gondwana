Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

215 internal static SKRect ComputeDestRect(RectangleF bounds, int srcW, int srcH, StretchMode mode)
216 {
217 var b = bounds;
218 var dst = new SKRect(b.Left, b.Top, b.Right, b.Bottom);
219 if (mode == StretchMode.Fill || srcW <= 0 || srcH <= 0) return dst;
220
221 float bw = b.Width, bh = b.Height;
222 float arSrc = (float)srcW / srcH;
223 float arDst = bw / bh;
224
225 if (mode == StretchMode.None)
226 return new SKRect(b.Left, b.Top, b.Left + srcW, b.Top + srcH);
227
228 if (mode == StretchMode.Uniform)
229 {
230 if (arSrc > arDst)
231 {
232 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
233 return new SKRect(b.Left, y, b.Left + w, y + h);
234 }
235 else
236 {
237 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
238 return new SKRect(x, b.Top, x + w, b.Top + h);
239 }
240 }
241
242 // UniformToFill
243 if (arSrc < arDst)
244 {
245 float w = bw, h = w / arSrc, y = b.Top + (bh - h) / 2f;
246 return new SKRect(b.Left, y, b.Left + w, y + h);
247 }
248 else
249 {
250 float h = bh, w = h * arSrc, x = b.Left + (bw - w) / 2f;
251 return new SKRect(x, b.Top, x + w, b.Top + h);
252 }
253 }
