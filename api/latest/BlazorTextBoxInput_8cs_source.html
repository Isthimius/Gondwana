Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

85 public static char? ResolveCharacter(WidgetKeyboardEventArgs args)
86 {
87 ArgumentNullException.ThrowIfNull(args);
88
89 if ((args.Modifiers & (KeyboardModifierState.Ctrl | KeyboardModifierState.Alt)) != 0)
90 return null;
91
92 bool shift = (args.Modifiers & KeyboardModifierState.Shift) != 0;
93 var key = (BlazorKey)args.Key;
94
95 if (key is >= BlazorKey.KeyA and <= BlazorKey.KeyZ)
96 {
97 char letter = (char)('a' + ((int)key - (int)BlazorKey.KeyA));
98 return shift ? char.ToUpperInvariant(letter) : letter;
99 }
100
101 if (key is >= BlazorKey.Digit1 and <= BlazorKey.Digit9)
102 {
103 int offset = (int)key - (int)BlazorKey.Digit1;
104 if (!shift)
105 return (char)('1' + offset);
106
107 const string shiftedDigits = "!@#$%^&*(";
108 return shiftedDigits[offset];
109 }
110
111 if (key == BlazorKey.Digit0)
112 return shift ? ')' : '0';
113
114 if (key is >= BlazorKey.Numpad0 and <= BlazorKey.Numpad9)
115 return (char)('0' + ((int)key - (int)BlazorKey.Numpad0));
116
117 return key switch
118 {
119 BlazorKey.Space => ' ',
120 BlazorKey.NumpadAdd => '+',
121 BlazorKey.NumpadSubtract => '-',
122 BlazorKey.NumpadMultiply => '*',
123 BlazorKey.NumpadDivide => '/',
124 BlazorKey.NumpadDecimal => '.',
125 BlazorKey.Minus => shift ? '_' : '-',
126 BlazorKey.Equal => shift ? '+' : '=',
127 BlazorKey.BracketLeft => shift ? '{' : '[',
128 BlazorKey.BracketRight => shift ? '}' : ']',
129 BlazorKey.Backslash => shift ? '|' : '\\',
130 BlazorKey.Semicolon => shift ? ':' : ';',
131 BlazorKey.Quote => shift ? '"' : '\'',
132 BlazorKey.Backquote => shift ? '~' : '`',
133 BlazorKey.Comma => shift ? '<' : ',',
134 BlazorKey.Period => shift ? '>' : '.',
135 BlazorKey.Slash => shift ? '?' : '/',
136 _ => null
137 };
138 }
