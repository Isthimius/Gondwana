Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

149 public static void StopAsyncLogging(bool flush = true, TimeSpan? flushTimeout = null)
150 {
151 Channel<LogEvent>? ch;
152 Task? worker;
153 CancellationTokenSource? cts;
154
155 lock (_asyncGate)
156 {
157 ch = _channel;
158 worker = _worker;
159 cts = _cts;
160
161 if (ch == null || worker == null)
162 return;
163
164 // Completing the writer lets the worker drain then exit.
165 try
166 {
167 ch.Writer.TryComplete();
168 }
169 catch
170 {
171 /* ignore */
172 }
173
174 // Clear shared references so other threads can't enqueue on this instance.
175 _channel = null;
176 _worker = null;
177 _cts = null;
178 }
179
180 try
181 {
182 cts?.Cancel();
183 }
184 catch
185 {
186 /* never crash */
187 }
188
189 if (flush)
190 {
191 var timeout = flushTimeout ?? TimeSpan.FromSeconds(2);
192
193 try
194 {
195 worker!.Wait(timeout);
196 }
197 catch
198 {
199 /* never crash */
200 }
201 }
202
203 try
204 {
205 cts?.Dispose();
206 }
207 catch
208 {
209 /* ignore */
210 }

362 // Must be called under _asyncGate.
363 private static void EnsureAsyncStarted_NoLock(bool forceRestart)
364 {
365 if (_mode != EngineLoggingMode.Asynchronous)
366 return;
367
368 if (!forceRestart && _worker != null)
369 return;
370
371 if (forceRestart && _channel != null)
372 {
373 // Best-effort stop without flush; caller controls flush via StopAsyncLogging.
374 try
375 {
376 _channel.Writer.TryComplete();
377 }
378 catch
379 {
380 /* ignore */
381 }
382 }
383
384 int capacityLocal = _capacity;
385
386 var channel = Channel.CreateBounded<LogEvent>(
387 new BoundedChannelOptions(capacityLocal)
388 {
389 SingleReader = true,
390 SingleWriter = false,
391 FullMode = BoundedChannelFullMode.DropWrite // fire-and-forget; drop if full
392 });
393
394 _channel = channel;
395
396 _cts?.Dispose();
397 _cts = new CancellationTokenSource();
398
399 // Pass reader so WorkerLoop never dereferences static _channel (avoids races with Stop).
400 var reader = channel.Reader;
401 var token = _cts.Token;
402 _worker = Task.Run(() => WorkerLoop(reader, token));
403 }

406 {
407 lock (_asyncGate)
408 {
409 if (_mode != EngineLoggingMode.Asynchronous)

424 private static async Task WorkerLoop(ChannelReader<LogEvent> reader, CancellationToken ct)
425 {
426 try
427 {
428 while (await reader.WaitToReadAsync(ct).ConfigureAwait(false))
429 {
430 while (reader.TryRead(out var ev))
431 {
432 try
433 {
434 // Executes formatting + sinks (console) off the engine thread.
435 var logger = GetOrCreateLoggerFactory().CreateLogger(ev.CategoryName);
436
437 logger.Log(
438 ev.LogLevel,
439 ev.EventId,
440 ev.State,
441 ev.Exception,
442 ev.Formatter);
443 }
444 catch (Exception ex)
445 {
446 // Fallback: write to debug output to aid diagnostics without crashing.
447 Debug.WriteLine($"[EngineLogger] Logging failed: {ex.GetType().Name}: {ex.Message}");
448
449 // Raise event for applications that want to monitor logging failures.
450 try
451 {
452 LoggingError?.Invoke(null, new LoggingErrorEventArgs(ex, ev.CategoryName, ev.LogLevel));
453 }
454 catch
455 {
456 // Never let event handlers crash the logging infrastructure.
457 }
458 }
459 }
460 }
461 }
462 catch (System.OperationCanceledException) when (ct.IsCancellationRequested)
463 {
464 // Normal cancellation: no action required.
465 }
466 catch
467 {
468 // Swallow any channel or unexpected exceptions.
469 }
470 }
