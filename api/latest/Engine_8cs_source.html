Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

631 public void Tick()
632 {
633 if (!IsRunning)
634 return;
635
636 if (!_isTimerDriven)
637 {
638 Cycle();
639 return;
640 }
641
642 lock (RenderStateSynchronization.SyncRoot)
643 {
644 EngineDispatcher.Drain();
645
646 long driverTick = HighResTimer.GetCurrentTick();
647 var batch = _timerDrivenSteps.Advance(
648 driverTick,
649 Configuration.TimerDrivenSimulationRate,
650 Configuration.MaxTimerDrivenSimulationSteps);
651
652 bool render = IsTimerDrivenForegroundDue(driverTick);
653 double frameDelta = HighResTimer.GetDuration(_lastForegroundTick, driverTick);
654
655 if (batch.StepCount == 0)
656 {
657 if (render)
658 RenderFrame(driverTick, frameDelta);
659
660 if (Configuration.SamplingTimeForCPS > 0)
661 CalculateCPS(driverTick);
662
663 return;
664 }
665
666 double fixedDelta = batch.StepTicks / (double)HighResTimer.TicksPerSecond;
667
668 for (int step = 0; step < batch.StepCount && IsRunning; step++)
669 {
670 long simulationTick = batch.GetStepTick(step);
671 EngineSimulationClock.SetTimerDrivenTick(simulationTick);
672
673 bool isLastStep = step == batch.StepCount - 1;
674 RunSimulationCycle(
675 simulationTick,
676 fixedDelta,
677 render && isLastStep,
678 driverTick,
679 frameDelta,
680 sampleCps: isLastStep);
681 }
682 }

1017 private bool IsTimerDrivenForegroundDue(long tick)
1018 {
1019 int targetFps = Configuration.TargetFPS;
1020 long elapsedTicks = Math.Max(0, tick - _lastTimerDrivenForegroundDriverTick);
1021 _lastTimerDrivenForegroundDriverTick = tick;
1022
1023 if (targetFps <= 0)
1024 {
1025 _timerDrivenForegroundAccumulatedTicks = 0;
1026 _timerDrivenForegroundTargetFps = targetFps;
1027 return true;
1028 }
1029
1030 if (_timerDrivenForegroundTargetFps != targetFps)
1031 {
1032 _timerDrivenForegroundTargetFps = targetFps;
1033 _timerDrivenForegroundAccumulatedTicks = 0;
1034 }
1035
1036 _timerDrivenForegroundAccumulatedTicks += elapsedTicks;
1037
1038 long frameIntervalTicks = Math.Max(1, HighResTimer.TicksPerSecond / targetFps);
1039 if (_timerDrivenForegroundAccumulatedTicks < frameIntervalTicks)
1040 return false;
1041
1042 // Render at most once per externally driven Tick(), but preserve the fractional
1043 // remainder so small requestAnimationFrame timing jitter cannot permanently lower
1044 // the effective render rate. Whole missed intervals are discarded rather than
1045 // causing an unbounded render catch-up sequence after a long browser stall.
1046 _timerDrivenForegroundAccumulatedTicks %= frameIntervalTicks;
1047 return true;
1048 }

1062 private void DoBackgroundTasks(long tick)
1063 {
1064 // find total real seconds passed since last background loop
1065 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
1066
1067 BeforeBackgroundTasksExecute?.Invoke();
1068
1069 // raise pre-cycle timer events
1070 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
1071
1072 // check for keyboard events
1073 KeyboardEventPoller.Instance?.PollForEvents(tick);
1074
1075 // check for mouse events
1076 MouseEventPoller.Instance?.PollForEvents(tick);
1077
1078 // check for touch events
1079 TouchEventPoller.Instance?.PollForEvents(tick);
1080
1081 // check for gamepad events
1082 GamepadEventPoller.Instance?.PollForEvents(tick);
1083
1084 // cycle Animator frames
1085 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
1086 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
1087
1088 // advance Sprite Movement paths
1089 SpriteManager.Instance.MoveSprites(tick);
1090
1091 // resolve collisions after movement
1092 foreach (var scene in Scenes.Scene.GetAllScenes())
1093 {
1094 foreach (var layer in scene.SceneLayers)
1095 layer.CollisionResolver.Resolve();
1096 }
1097
1098 // update cameras so any movement can mark RefreshNeeded = All.
1099 foreach (var surface in RenderSurfaceHostRegistry.All)
1100 surface.ViewManager.UpdateCameras(deltaSeconds);
1101
1102 AfterBackgroundTasksExecute?.Invoke();
1103
1104 _lastBackgroundTick = tick;
1105 }

1107 private void DoForegroundTasks(long tick)
1108 {
1109 // raise event
1110 BeforeFrameRender?.Invoke();
1111
1112 // Effects are presentation state, so advance them on the foreground/render cadence.
1113 foreach (var surface in RenderSurfaceHostRegistry.All)
1114 surface.Effects.Update(tick);
1115
1116 // update the DirectDrawing instances' states
1117 DirectDrawingManager.Instance.UpdateAll(tick);
1118
1119 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
1120 foreach (var surface in RenderSurfaceHostRegistry.All)
1121 if (!surface.Backbuffer.IsGlThreadRendered)
1122 surface.RenderToBackbuffer(tick);
1123
1124 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
1125 foreach (var surface in RenderSurfaceHostRegistry.All)
1126 if (!surface.Backbuffer.IsGlThreadRendered)
1127 surface.PresentBackbufferToAdapter();
1128
1129 // update state of gamepad(s)
1130 Input.GamepadManager?.Update();
1131
1132 // raise event
1133 // GL thread rendering is done as part of this invocation
1134 AfterFrameRender?.Invoke();
1135
1136 // raise post-cycle timer events
1137 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
1138 }

1140 private void CalculateCPS(long tick)
1141 {
1142 // Has the sampling interval elapsed?
1143 long elapsedTicks = tick - _lastCPSSamplingTick;
1144 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
1145 return;
1146
1147 // SNAPSHOT the counters BEFORE resetting or posting
1148 long grossCycles = _grossCyclesThisMeasure;
1149 long netCycles = _netCyclesThisMeasure;
1150
1151 // Compute using the snapshot
1152 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
1153 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1154 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1155
1156 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
1157 long totalGpuFrames = 0;
1158 int gpuSurfaceCount = 0;
1159 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
1160 {
1161 if (surface.Backbuffer is GpuBackbuffer gpuBb)
1162 {
1163 totalGpuFrames += gpuBb.ConsumeFrameCount();
1164 gpuSurfaceCount++;
1165 }
1166 }
1167 double? gpuFps = gpuSurfaceCount > 0
1168 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
1169 : null;
1170
1171 // Build immutable args NOW (so lambda doesn't read changing fields later)
1172 var args = new CyclesPerSecondCalculatedEventArgs(
1173 grossCycles,
1174 netCycles,
1175 grossCps,
1176 netCps,
1177 elapsedSec,
1178 gpuFps
1179 );
1180
1181 // Post the snapshot
1182 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
1183
1184 _grossCPS = grossCps;
1185 _netFPS = netCps;
1186
1187 // Reset for next window
1188 _lastCPSSamplingTick = tick;
1189 _grossCyclesThisMeasure = 0;
1190 _netCyclesThisMeasure = 0;
1191 }
