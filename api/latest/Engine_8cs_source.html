Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

565 public void Tick()
566 {
567 if (!IsRunning)
568 return;
569
570 if (!_isTimerDriven)
571 {
572 Cycle();
573 return;
574 }
575
576 EngineDispatcher.Drain();
577
578 long driverTick = HighResTimer.GetCurrentTick();
579 var batch = _timerDrivenSteps.Advance(
580 driverTick,
581 Configuration.TimerDrivenSimulationRate,
582 Configuration.MaxTimerDrivenSimulationSteps);
583
584 bool render = IsTimerDrivenForegroundDue(driverTick);
585 double frameDelta = HighResTimer.GetDuration(_lastForegroundTick, driverTick);
586
587 if (batch.StepCount == 0)
588 {
589 if (render)
590 RenderFrame(driverTick, frameDelta);
591
592 if (Configuration.SamplingTimeForCPS > 0)
593 CalculateCPS(driverTick);
594
595 return;
596 }
597
598 double fixedDelta = batch.StepTicks / (double)HighResTimer.TicksPerSecond;
599
600 for (int step = 0; step < batch.StepCount && IsRunning; step++)
601 {
602 long simulationTick = batch.GetStepTick(step);
603 EngineSimulationClock.SetTimerDrivenTick(simulationTick);
604
605 bool isLastStep = step == batch.StepCount - 1;
606 RunSimulationCycle(
607 simulationTick,
608 fixedDelta,
609 render && isLastStep,
610 driverTick,
611 frameDelta,
612 sampleCps: isLastStep);
613 }
614 }

920 private bool IsForegroundDue(long tick) =>
921 Configuration.TargetFPS <= 0
922 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS;

924 private bool IsTimerDrivenForegroundDue(long tick)
925 {
926 int targetFps = Configuration.TargetFPS;
927 long elapsedTicks = Math.Max(0, tick - _lastTimerDrivenForegroundDriverTick);
928 _lastTimerDrivenForegroundDriverTick = tick;
929
930 if (targetFps <= 0)
931 {
932 _timerDrivenForegroundAccumulatedTicks = 0;
933 _timerDrivenForegroundTargetFps = targetFps;
934 return true;
935 }
936
937 if (_timerDrivenForegroundTargetFps != targetFps)
938 {
939 _timerDrivenForegroundTargetFps = targetFps;
940 _timerDrivenForegroundAccumulatedTicks = 0;
941 }
942
943 _timerDrivenForegroundAccumulatedTicks += elapsedTicks;
944
945 long frameIntervalTicks = Math.Max(1, HighResTimer.TicksPerSecond / targetFps);
946 if (_timerDrivenForegroundAccumulatedTicks < frameIntervalTicks)
947 return false;
948
949 // Render at most once per externally driven Tick(), but preserve the fractional
950 // remainder so small requestAnimationFrame timing jitter cannot permanently lower
951 // the effective render rate. Whole missed intervals are discarded rather than
952 // causing an unbounded render catch-up sequence after a long browser stall.
953 _timerDrivenForegroundAccumulatedTicks %= frameIntervalTicks;
954 return true;
955 }

969 private void DoBackgroundTasks(long tick)
970 {
971 // find total real seconds passed since last background loop
972 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
973
974 BeforeBackgroundTasksExecute?.Invoke();
975
976 // raise pre-cycle timer events
977 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
978
979 // check for keyboard events
980 KeyboardEventPoller.Instance?.PollForEvents(tick);
981
982 // check for mouse events
983 MouseEventPoller.Instance?.PollForEvents(tick);
984
985 // check for touch events
986 TouchEventPoller.Instance?.PollForEvents(tick);
987
988 // check for gamepad events
989 GamepadEventPoller.Instance?.PollForEvents(tick);
990
991 // cycle Animator frames
992 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
993 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
994
995 // advance Sprite Movement paths
996 SpriteManager.Instance.MoveSprites(tick);
997
998 // resolve collisions after movement
999 foreach (var scene in Scenes.Scene.GetAllScenes())
1000 {
1001 foreach (var layer in scene.SceneLayers)
1002 layer.CollisionResolver.Resolve();
1003 }
1004
1005 // update cameras so any movement can mark RefreshNeeded = All.
1006 foreach (var surface in RenderSurfaceHostRegistry.All)
1007 surface.ViewManager.UpdateCameras(deltaSeconds);
1008
1009 AfterBackgroundTasksExecute?.Invoke();
1010
1011 _lastBackgroundTick = tick;
1012 }

1014 private void DoForegroundTasks(long tick)
1015 {
1016 // raise event
1017 BeforeFrameRender?.Invoke();
1018
1019 // Effects are presentation state, so advance them on the foreground/render cadence.
1020 foreach (var surface in RenderSurfaceHostRegistry.All)
1021 surface.Effects.Update(tick);
1022
1023 // update the DirectDrawing instances' states
1024 DirectDrawingManager.Instance.UpdateAll(tick);
1025
1026 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
1027 foreach (var surface in RenderSurfaceHostRegistry.All)
1028 if (!surface.Backbuffer.IsGlThreadRendered)
1029 surface.RenderToBackbuffer(tick);
1030
1031 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
1032 foreach (var surface in RenderSurfaceHostRegistry.All)
1033 if (!surface.Backbuffer.IsGlThreadRendered)
1034 surface.PresentBackbufferToAdapter();
1035
1036 // update state of gamepad(s)
1037 Input.GamepadManager?.Update();
1038
1039 // raise event
1040 // GL thread rendering is done as part of this invocation
1041 AfterFrameRender?.Invoke();
1042
1043 // raise post-cycle timer events
1044 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
1045 }

1047 private void CalculateCPS(long tick)
1048 {
1049 // Has the sampling interval elapsed?
1050 long elapsedTicks = tick - _lastCPSSamplingTick;
1051 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
1052 return;
1053
1054 // SNAPSHOT the counters BEFORE resetting or posting
1055 long grossCycles = _grossCyclesThisMeasure;
1056 long netCycles = _netCyclesThisMeasure;
1057
1058 // Compute using the snapshot
1059 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
1060 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1061 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1062
1063 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
1064 long totalGpuFrames = 0;
1065 int gpuSurfaceCount = 0;
1066 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
1067 {
1068 if (surface.Backbuffer is GpuBackbuffer gpuBb)
1069 {
1070 totalGpuFrames += gpuBb.ConsumeFrameCount();
1071 gpuSurfaceCount++;
1072 }
1073 }
1074 double? gpuFps = gpuSurfaceCount > 0
1075 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
1076 : null;
1077
1078 // Build immutable args NOW (so lambda doesn't read changing fields later)
1079 var args = new CyclesPerSecondCalculatedEventArgs(
1080 grossCycles,
1081 netCycles,
1082 grossCps,
1083 netCps,
1084 elapsedSec,
1085 gpuFps
1086 );
1087
1088 // Post the snapshot
1089 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
1090
1091 _grossCPS = grossCps;
1092 _netFPS = netCps;
1093
1094 // Reset for next window
1095 _lastCPSSamplingTick = tick;
1096 _grossCyclesThisMeasure = 0;
1097 _netCyclesThisMeasure = 0;
1098 }

1142 }

1166 }
1167 catch (Exception ex)

1174 }

1177 EngineLogger.StopAsyncLogging(flush: Configuration.FlushAsyncLogsOnShutdown);

1181 }

1183 // unmanaged cleanup...
1184 IsDisposed = true;
1185
1186 if (disposing)

1188 // now signal we're fully torn down
