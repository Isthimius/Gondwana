Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

551 public void Tick()
552 {
553 if (!IsRunning)
554 return;
555
556 if (!_isTimerDriven)
557 {
558 Cycle();
559 return;
560 }
561
562 EngineDispatcher.Drain();
563
564 long driverTick = HighResTimer.GetCurrentTick();
565 var batch = _timerDrivenSteps.Advance(
566 driverTick,
567 Configuration.TimerDrivenSimulationRate,
568 Configuration.MaxTimerDrivenSimulationSteps);
569
570 bool render = IsTimerDrivenForegroundDue(driverTick);
571 double frameDelta = HighResTimer.GetDuration(_lastForegroundTick, driverTick);
572
573 if (batch.StepCount == 0)
574 {
575 if (render)
576 RenderFrame(driverTick, frameDelta);
577
578 if (Configuration.SamplingTimeForCPS > 0)
579 CalculateCPS(driverTick);
580
581 return;
582 }
583
584 double fixedDelta = batch.StepTicks / (double)HighResTimer.TicksPerSecond;
585
586 for (int step = 0; step < batch.StepCount && IsRunning; step++)
587 {
588 long simulationTick = batch.GetStepTick(step);
589 EngineSimulationClock.SetTimerDrivenTick(simulationTick);
590
591 bool isLastStep = step == batch.StepCount - 1;
592 RunSimulationCycle(
593 simulationTick,
594 fixedDelta,
595 render && isLastStep,
596 driverTick,
597 frameDelta,
598 sampleCps: isLastStep);
599 }
600 }

906 private bool IsForegroundDue(long tick) =>
907 Configuration.TargetFPS <= 0
908 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS;
909
910 private bool IsTimerDrivenForegroundDue(long tick)
911 {
912 int targetFps = Configuration.TargetFPS;
913 long elapsedTicks = Math.Max(0, tick - _lastTimerDrivenForegroundDriverTick);
914 _lastTimerDrivenForegroundDriverTick = tick;
915
916 if (targetFps <= 0)
917 {
918 _timerDrivenForegroundAccumulatedTicks = 0;
919 _timerDrivenForegroundTargetFps = targetFps;
920 return true;
921 }
922
923 if (_timerDrivenForegroundTargetFps != targetFps)
924 {
925 _timerDrivenForegroundTargetFps = targetFps;
926 _timerDrivenForegroundAccumulatedTicks = 0;
927 }
928
929 _timerDrivenForegroundAccumulatedTicks += elapsedTicks;
930
931 long frameIntervalTicks = Math.Max(1, HighResTimer.TicksPerSecond / targetFps);
932 if (_timerDrivenForegroundAccumulatedTicks < frameIntervalTicks)
933 return false;
934
935 // Render at most once per externally driven Tick(), but preserve the fractional
936 // remainder so small requestAnimationFrame timing jitter cannot permanently lower
937 // the effective render rate. Whole missed intervals are discarded rather than
938 // causing an unbounded render catch-up sequence after a long browser stall.
939 _timerDrivenForegroundAccumulatedTicks %= frameIntervalTicks;
940 return true;
941 }

955 private void DoBackgroundTasks(long tick)
956 {
957 // find total real seconds passed since last background loop
958 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
959
960 BeforeBackgroundTasksExecute?.Invoke();
961
962 // raise pre-cycle timer events
963 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
964
965 // check for keyboard events
966 KeyboardEventPoller.Instance?.PollForEvents(tick);
967
968 // check for mouse events
969 MouseEventPoller.Instance?.PollForEvents(tick);
970
971 // check for touch events
972 TouchEventPoller.Instance?.PollForEvents(tick);
973
974 // check for gamepad events
975 GamepadEventPoller.Instance?.PollForEvents(tick);
976
977 // cycle Animator frames
978 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
979 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
980
981 // advance Sprite Movement paths
982 SpriteManager.Instance.MoveSprites(tick);
983
984 // resolve collisions after movement
985 foreach (var scene in Scenes.Scene.GetAllScenes())
986 {
987 foreach (var layer in scene.SceneLayers)
988 layer.CollisionResolver.Resolve();
989 }
990
991 // update cameras so any movement can mark RefreshNeeded = All.
992 foreach (var surface in RenderSurfaceHostRegistry.All)
993 surface.ViewManager.UpdateCameras(deltaSeconds);
994
995 AfterBackgroundTasksExecute?.Invoke();
996
997 _lastBackgroundTick = tick;
998 }

1000 private void DoForegroundTasks(long tick)
1001 {
1002 // raise event
1003 BeforeFrameRender?.Invoke();
1004
1005 // Effects are presentation state, so advance them on the foreground/render cadence.
1006 foreach (var surface in RenderSurfaceHostRegistry.All)
1007 surface.Effects.Update(tick);
1008
1009 // update the DirectDrawing instances' states
1010 DirectDrawingManager.Instance.UpdateAll(tick);
1011
1012 // refresh all RenderSurfaceHost backbuffers (skip surfaces rendered on the GL thread)
1013 foreach (var surface in RenderSurfaceHostRegistry.All)
1014 if (!surface.Backbuffer.IsGlThreadRendered)
1015 surface.RenderToBackbuffer(tick);
1016
1017 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
1018 foreach (var surface in RenderSurfaceHostRegistry.All)
1019 if (!surface.Backbuffer.IsGlThreadRendered)
1020 surface.PresentBackbufferToAdapter();
1021
1022 // update state of gamepad(s)
1023 Input.GamepadManager?.Update();
1024
1025 // raise event
1026 // GL thread rendering is done as part of this invocation
1027 AfterFrameRender?.Invoke();
1028
1029 // raise post-cycle timer events
1030 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
1031 }

1033 private void CalculateCPS(long tick)
1034 {
1035 // Has the sampling interval elapsed?
1036 long elapsedTicks = tick - _lastCPSSamplingTick;
1037 if (elapsedTicks < Configuration.SamplingTimeForCPSTicks)
1038 return;
1039
1040 // SNAPSHOT the counters BEFORE resetting or posting
1041 long grossCycles = _grossCyclesThisMeasure;
1042 long netCycles = _netCyclesThisMeasure;
1043
1044 // Compute using the snapshot
1045 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
1046 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1047 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1048
1049 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
1050 long totalGpuFrames = 0;
1051 int gpuSurfaceCount = 0;
1052 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
1053 {
1054 if (surface.Backbuffer is GpuBackbuffer gpuBb)
1055 {
1056 totalGpuFrames += gpuBb.ConsumeFrameCount();
1057 gpuSurfaceCount++;
1058 }
1059 }
1060 double? gpuFps = gpuSurfaceCount > 0
1061 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
1062 : null;
1063
1064 // Build immutable args NOW (so lambda doesn't read changing fields later)
1065 var args = new CyclesPerSecondCalculatedEventArgs(
1066 grossCycles,
1067 netCycles,
1068 grossCps,
1069 netCps,
1070 elapsedSec,
1071 gpuFps
1072 );
1073
1074 // Post the snapshot
1075 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
1076
1077 _grossCPS = grossCps;
1078 _netFPS = netCps;
1079
1080 // Reset for next window
1081 _lastCPSSamplingTick = tick;
1082 _grossCyclesThisMeasure = 0;
1083 _netCyclesThisMeasure = 0;
1084 }

1106 private void Dispose(bool disposing)
1107 {
1108 if (!IsDisposed)
1109 {
1110 if (disposing)
1111 {
1112 IsDisposing = true;
1113
1114 // stop the loop first so handlers don't race the cycle thread
1115 try
1116 {
1117 Stop();
1118 }
1119 catch (Exception ex)
1120 {
1121 Logger.LogError(ex, "Unhandled exception calling Stop()");
1122 }
1123
1124 // wait for the background loop to actually exit
1125 try
1126 {
1127 _cycleTask?.Wait();
1128 }
1129 catch (Exception ex)
1130 {
1131 Logger.LogError(ex, "Error waiting for engine loop to exit.");
1132 }
1133
1134 // raise Disposing on UI thread if possible; otherwise inline
1135 if (UiDispatcher is not null)
1136 UiDispatcher.Post(() => SafeInvoke(Disposing));
1137 else
1138 SafeInvoke(Disposing);
1139
1140 // managed cleanup...
1141 Input.KeyboardEventPoller?.StopMonitoringAllKeys();
1142 MouseEventPoller.Reset();
1143 TouchEventPoller.Reset();
1144
1145 if (Input.GamepadManager is not null)
1146 foreach (var gamepadAdapter in Input.GamepadManager.ConnectedAdapters)
1147 Input.GamepadEventPoller?.StopMonitoringAllButtons(gamepadAdapter.GamepadId);
1148
1149 try
1150 {
1151 _configurationFile?.Dispose();
1152 }
1153 catch (Exception ex)
1154 {
1155 Logger.LogError(ex, "Error saving the engine configuration during disposal.");
1156 }
1157 finally
1158 {
1159 _configurationFile = null;
1160 }
1161
1162 if (EngineLogger.Mode == EngineLoggingMode.Asynchronous)
1163 EngineLogger.StopAsyncLogging(flush: Configuration.FlushAsyncLogsOnShutdown);
1164
1165 Timer.ClearAll();
1166 State.Clear();
1167 }
1168
1169 // unmanaged cleanup...
1170 IsDisposed = true;
1171
1172 if (disposing)
1173 {
1174 // now signal we're fully torn down
1175 if (UiDispatcher is not null)
1176 UiDispatcher.Post(() => SafeInvoke(Disposed));
1177 else
1178 SafeInvoke(Disposed);
1179 }
1180 }
