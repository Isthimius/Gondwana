Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

358 string? configFileName,
359 bool? autoSaveConfig,
360 IKeyboardAdapter? keyboardAdapter,
361 IMouseAdapter? mouseAdapter,
362 ITouchAdapter? touchAdapter,
363 IGamepadManager<IGamepadAdapter>? gamepadManager,
364 IEngineConfigurationStore? configurationStore)
365 {
366 if (_isInitialized || _isInitializing)
367 return;
368
369 // reset in case this instance has been initialized before
370 _initDone.Reset();
371
372 _isInitializing = true;
373
374 try
375 {
376 if (UiDispatcher == null)
377 PreInitialization?.Invoke();
378 else
379 UiDispatcher!.Post(() => PreInitialization?.Invoke());
380
381 try
382 {
383 _configurationStore?.Dispose();
384 }
385 catch (Exception ex)
386 {
387 Logger.LogError(ex, "Error saving the engine configuration during initialization.");
388 }
389
390 _configurationStore = configurationStore ?? EngineConfigurationFile.Load(configFileName, autoSaveConfig);
391
392 if (configurationStore is not null && autoSaveConfig.HasValue)
393 _configurationStore.AutoSave = autoSaveConfig.Value;
394
395 Configuration = _configurationStore.Configuration;
396
397 ConfigureLogging(Configuration);
398
399 if (Configuration.StateFiles?.Any() ?? false)
400 {
401 foreach (var stateFile in Configuration.StateFiles)
402 {
403 EngineState.MergeFromFile(
404 stateFile.File,
405 stateFile.IsCompressed,
406 stateFile.OverwriteExisting,
407 stateFile.EngineStateParts);
408 }
409 }
410
411 if (keyboardAdapter != null)
412 KeyboardEventPoller.Initialize(keyboardAdapter);
413
414 if (mouseAdapter != null)
415 MouseEventPoller.Initialize(mouseAdapter);
416
417 if (touchAdapter != null)
418 Input.TouchAdapter = touchAdapter;
419
420 if (gamepadManager is not null)
421 Input.GamepadManager = gamepadManager;
422
423 if (UiDispatcher == null)
424 PostInitialization?.Invoke();
425 else

427
428 EnginePluginRegistry.InvokeInitialize(this);
429
430 _isInitialized = true;
431
432 if (UiDispatcher == null)
433 InitializationComplete?.Invoke();
434 else

669 public void Tick()
670 {
671 if (!IsRunning)
672 return;
673
674 if (!_isTimerDriven)
675 {
676 Cycle();
677 return;
678 }
679
680 lock (RenderStateSynchronization.SyncRoot)
681 {
682 EngineDispatcher.Drain();
683
684 long driverTick = HighResTimer.GetCurrentTick();
685
686 var batch = _timerDrivenSteps.Advance(
687 driverTick,
688 Configuration.TimerDrivenSimulationRate,
689 Configuration.MaxTimerDrivenSimulationSteps);
690
691 bool render = IsTimerDrivenForegroundDue(driverTick);
692 double frameDelta = HighResTimer.GetDuration(_lastForegroundTick, driverTick);
693
694 if (batch.StepCount == 0)
695 {
696 if (render)
697 RenderFrame(driverTick, frameDelta);
698
699 if (Configuration.LegacyCpsSamplingTime > 0)
700 CalculateCPS(driverTick);
701
702 return;
703 }
704
705 double fixedDelta = batch.StepTicks / (double)HighResTimer.TicksPerSecond;
706
707 for (int step = 0; step < batch.StepCount && IsRunning; step++)
708 {
709 long simulationTick = batch.GetStepTick(step);
710 EngineSimulationClock.SetTimerDrivenTick(simulationTick);
711
712 bool isLastStep = step == batch.StepCount - 1;
713
714 RunSimulationCycle(
715 simulationTick,
716 fixedDelta,
717 render && isLastStep,
718 driverTick,
719 frameDelta,
720 sampleCps: isLastStep);
721 }
722 }

1018 private void Cycle()
1019 {
1020 // Give a GPU paint callback that is already waiting for live render state
1021 // a chance to acquire the shared gate before this unconstrained loop takes
1022 // another simulation cycle.
1023 RenderStateSynchronization.WaitForPendingGpuRenderers();
1024
1025 lock (RenderStateSynchronization.SyncRoot)
1026 {
1027 EngineDispatcher.Drain();
1028
1029 long tick = HighResTimer.GetCurrentTick();
1030 var deltaSeconds = HighResTimer.GetDuration(_lastCycleTick, tick);
1031 _lastCycleTick = tick;
1032
1033 RunSimulationCycle(
1034 tick,
1035 deltaSeconds,
1036 IsForegroundDue(tick),
1037 tick,
1038 deltaSeconds,
1039 sampleCps: true);
1040 }

1043 private void RunSimulationCycle(
1044 long simulationTick,
1045 double simulationDelta,
1046 bool render,
1047 long renderTick,
1048 double frameDelta,
1049 bool sampleCps)
1050 {
1051 // Refresh native gamepad state before any simulation callbacks consume it.
1052 // Use the real driver/render tick rather than the fixed-step simulation clock so
1053 // timer-driven catch-up steps cannot compress or delay hardware polling cadence.
1054 Input.RefreshGamepads(renderTick, Configuration);
1055
1056 long telemetryGeneration = _engineTelemetry.BeginSample();
1057 long telemetryStarted = telemetryGeneration == 0 ? 0 : HighResTimer.GetCurrentTick();
1058 SimulationUpdating?.Invoke(simulationDelta);
1059 EnginePluginRegistry.InvokePreCycle(this, simulationDelta);
1060
1061 DoBackgroundTasks(simulationTick);
1062
1063 if (render)
1064 RenderFrame(renderTick, frameDelta);
1065
1066 _grossCyclesThisMeasure++;
1067
1068 if (sampleCps && Configuration.LegacyCpsSamplingTime > 0)
1069 CalculateCPS(renderTick);
1070
1071 EnginePluginRegistry.InvokePostCycle(this, simulationDelta);
1072
1073 if (telemetryGeneration != 0)
1074 _engineTelemetry.Record(
1075 telemetryGeneration,
1076 "cycle.cpu.ms",
1077 HighResTimer.GetDuration(telemetryStarted, HighResTimer.GetCurrentTick()) * 1000d);
1078 }

1080 private bool IsForegroundDue(long tick) =>
1081 Configuration.TargetFPS <= 0
1082 || (tick - _lastForegroundTick) >= HighResTimer.TicksPerSecond / Configuration.TargetFPS;

1084 private bool IsTimerDrivenForegroundDue(long tick)
1085 {
1086 int targetFps = Configuration.TargetFPS;
1087 long elapsedTicks = Math.Max(0, tick - _lastTimerDrivenForegroundDriverTick);
1088 _lastTimerDrivenForegroundDriverTick = tick;
1089
1090 if (targetFps <= 0)
1091 {
1092 _timerDrivenForegroundAccumulatedTicks = 0;
1093 _timerDrivenForegroundTargetFps = targetFps;
1094 return true;
1095 }
1096
1097 if (_timerDrivenForegroundTargetFps != targetFps)
1098 {
1099 _timerDrivenForegroundTargetFps = targetFps;
1100 _timerDrivenForegroundAccumulatedTicks = 0;
1101 }
1102
1103 _timerDrivenForegroundAccumulatedTicks += elapsedTicks;
1104
1105 long frameIntervalTicks = Math.Max(1, HighResTimer.TicksPerSecond / targetFps);
1106
1107 if (_timerDrivenForegroundAccumulatedTicks < frameIntervalTicks)
1108 return false;
1109
1110 // Render at most once per externally driven Tick(), but preserve the fractional
1111 // remainder so small requestAnimationFrame timing jitter cannot permanently lower
1112 // the effective render rate. Whole missed intervals are discarded rather than
1113 // causing an unbounded render catch-up sequence after a long browser stall.
1114 _timerDrivenForegroundAccumulatedTicks %= frameIntervalTicks;
1115 return true;
1116 }

1138 private void DoBackgroundTasks(long tick)
1139 {
1140 long telemetryGeneration = _engineTelemetry.BeginSample();
1141 long telemetryStarted = telemetryGeneration == 0 ? 0 : HighResTimer.GetCurrentTick();
1142 // find total real seconds passed since last background loop
1143 var deltaSeconds = HighResTimer.GetDuration(_lastBackgroundTick, tick);
1144
1145 BeforeBackgroundTasksExecute?.Invoke();
1146
1147 // raise pre-cycle timer events
1148 Timer.RaiseTimerEvents(TimerType.PreCycle, tick);
1149
1150 // check for keyboard events
1151 KeyboardEventPoller.Instance?.PollForEvents(tick);
1152
1153 // check for mouse events
1154 MouseEventPoller.Instance?.PollForEvents(tick);
1155
1156 // check for touch events
1157 TouchEventPoller.Instance?.PollForEvents(tick);
1158
1159 // check for gamepad events
1160 GamepadEventPoller.Instance?.PollForEvents(tick);
1161
1162 // cycle Animator frames
1163 for (int i = 0; i < Tile.TilesAnimating.Count; i++)
1164 Tile.TilesAnimating[i].TileAnimator.CycleAnimation(tick);
1165
1166 // advance Sprite Movement paths
1167 SpriteManager.Instance.MoveSprites(tick);
1168
1169 // resolve collisions after movement
1170 foreach (var scene in Scenes.Scene.GetAllScenes())
1171 {
1172 foreach (var layer in scene.SceneLayers)
1173 layer.CollisionResolver.Resolve();
1174 }
1175
1176 // update cameras so any movement can mark RefreshNeeded = All.
1177 foreach (var surface in RenderSurfaceHostRegistry.All)
1178 surface.ViewManager.UpdateCameras(deltaSeconds);
1179
1180 AfterBackgroundTasksExecute?.Invoke();
1181
1182 _lastBackgroundTick = tick;
1183
1184 if (telemetryGeneration != 0)
1185 _engineTelemetry.Record(
1186 telemetryGeneration,
1187 "background.cpu.ms",
1188 HighResTimer.GetDuration(telemetryStarted, HighResTimer.GetCurrentTick()) * 1000d);
1189 }

1191 private void DoForegroundTasks(long tick)
1192 {
1193 // raise event
1194 BeforeFrameRender?.Invoke();
1195
1196 // Effects are presentation state, so advance them on the foreground/render cadence.
1197 foreach (var surface in RenderSurfaceHostRegistry.All)
1198 surface.Effects.Update(tick);
1199
1200 // update the DirectDrawing instances' states
1201 DirectDrawingManager.Instance.UpdateAll(tick);
1202
1203 // Resolve desktop GPU visual state at foreground cadence before requesting repaint.
1204 foreach (var surface in RenderSurfaceHostRegistry.All)
1205 surface.ProduceRenderFrameSnapshot(tick);
1206
1207 // Bitmap rendering remains on the engine thread.
1208 foreach (var surface in RenderSurfaceHostRegistry.All)
1209 if (!surface.Backbuffer.IsGlThreadRendered)
1210 surface.RenderToBackbuffer(tick);
1211
1212 // render each Backbuffer to RenderSurfaceHost adapter (skip GL-thread surfaces)
1213 foreach (var surface in RenderSurfaceHostRegistry.All)
1214 if (!surface.Backbuffer.IsGlThreadRendered)
1215 surface.PresentBackbufferToAdapter();
1216
1217 // raise event
1218 // GL thread rendering is done as part of this invocation
1219 AfterFrameRender?.Invoke();
1220
1221 // raise post-cycle timer events
1222 Timer.RaiseTimerEvents(TimerType.PostCycle, tick);
1223 }

1225 private void CalculateCPS(long tick)
1226 {
1227 // Has the sampling interval elapsed?
1228 long elapsedTicks = tick - _lastCPSSamplingTick;
1229
1230 if (elapsedTicks < Configuration.LegacyCpsSamplingTicks)
1231 return;
1232
1233 // SNAPSHOT the counters BEFORE resetting or posting
1234 long grossCycles = _grossCyclesThisMeasure;
1235 long netCycles = _netCyclesThisMeasure;
1236
1237 // Compute using the snapshot
1238 double elapsedSec = elapsedTicks / (double)HighResTimer.TicksPerSecond;
1239 double grossCps = grossCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1240 double netCps = netCycles * HighResTimer.TicksPerSecond / (double)elapsedTicks;
1241
1242 // Collect actual GPU FPS from all registered GPU-rendered backbuffers.
1243 long totalGpuFrames = 0;
1244 int gpuSurfaceCount = 0;
1245
1246 foreach (var surface in RenderSurfaceHostRegistry.All.ToArray())
1247 {
1248 if (surface.Backbuffer is GpuBackbuffer gpuBb)
1249 {
1250 totalGpuFrames += gpuBb.ConsumeFrameCount();
1251 gpuSurfaceCount++;
1252 }
1253 }
1254
1255 double? gpuFps = gpuSurfaceCount > 0
1256 ? totalGpuFrames * HighResTimer.TicksPerSecond / (double)elapsedTicks
1257 : null;
1258
1259 // Build immutable args NOW (so lambda doesn't read changing fields later)
1260 var args = new CyclesPerSecondCalculatedEventArgs(
1261 grossCycles,
1262 netCycles,
1263 grossCps,
1264 netCps,
1265 elapsedSec,
1266 gpuFps
1267 );
1268
1269 // Post the snapshot
1270 UiDispatcher!.Post(() => CPSCalculated?.Invoke(args));
1271
1272 _grossCPS = grossCps;
1273 _netFPS = netCps;
1274
1275 // Reset for next window
1276 _lastCPSSamplingTick = tick;
1277 _grossCyclesThisMeasure = 0;
1278 _netCyclesThisMeasure = 0;
1279 }

1301 private void Dispose(bool disposing)
1302 {
1303 if (!IsDisposed)
1304 {
1305 if (disposing)
1306 {
1307 if (IsDisposing)
1308 return;
1309
1310 IsDisposing = true;
1311
1312 // stop the loop first so handlers don't race the cycle thread
1313 try
1314 {
1315 Stop();
1316 }
1317 catch (Exception ex)
1318 {
1319 Logger.LogError(ex, "Unhandled exception calling Stop()");
1320 }
1321
1322 // wait for the background loop to actually exit unless we're already on it
1323 try
1324 {
1325 var cycleTask = _cycleTask;
1326
1327 if (cycleTask is not null && EngineDispatcher.IsOnEngineThread && !cycleTask.IsCompleted)
1328 {
1329 if (!_deferredDisposeScheduled)
1330 {
1331 _deferredDisposeScheduled = true;
1332
1333 cycleTask.ContinueWith(
1334 _ => CompleteManagedDisposal(),
1335 System.Threading.CancellationToken.None,
1336 System.Threading.Tasks.TaskContinuationOptions.ExecuteSynchronously,
1337 System.Threading.Tasks.TaskScheduler.Default);
1338 }
1339
1340 return;
1341 }
1342
1343 if (cycleTask is not null && !EngineDispatcher.IsOnEngineThread)
1344 cycleTask.Wait();
1345 }
1346 catch (Exception ex)
1347 {
1348 Logger.LogError(ex, "Error waiting for engine loop to exit.");
1349 }
1350
1351 CompleteManagedDisposal();
1352 }
1353 else
1354 {
1355 // unmanaged cleanup...
