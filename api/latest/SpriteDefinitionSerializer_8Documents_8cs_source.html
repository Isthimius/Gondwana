Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

160 private static SpriteDefinition PrepareDocumentMember(SpriteDefinition definition, string path)
161 {
162 var clone = ParseDefinition(ToJson(definition));
163
164 if (string.IsNullOrWhiteSpace(clone.Key))
165 clone.Key = Path.GetFileNameWithoutExtension(path);
166
167 var origin = definition.Source.GsprFilePath ?? definition.Source.AssetsFilePath;
168
169 var oldDirectory = definition.LoadStamp?.ReferenceBaseDirectory ??
170 (string.IsNullOrWhiteSpace(origin) || !Path.IsPathFullyQualified(origin)
171 ? null
172 : Path.GetDirectoryName(origin));
173
174 var newDirectory = Path.GetDirectoryName(path)!;
175
176 string? RebasePath(string? value) =>
177 string.IsNullOrWhiteSpace(value) || (oldDirectory is null && !Path.IsPathRooted(value))
178 ? value
179 : Path.GetRelativePath(
180 newDirectory,
181 oldDirectory is null ? Path.GetFullPath(value) : Path.GetFullPath(value, oldDirectory));
182
183 foreach (var source in clone.TilesheetSources)
184 {
185 source.GtsPath = RebasePath(source.GtsPath);
186 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
187 }
188
189 foreach (var source in clone.SceneSources)
190 {
191 source.GscnPath = RebasePath(source.GscnPath);
192 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
193 }
194
195 foreach (var source in clone.AnimationSources)
196 {
197 source.GaniPath = RebasePath(source.GaniPath);
198 source.AssetsFilePath = RebasePath(source.AssetsFilePath);
199 }
200
201 clone.Source = SpriteDefinitionSource.LooseDefinitionFile(path);
202 return clone;
203 }

273 public static IReadOnlyList<List<Sprite>> ToAllSprites(SpriteDefinitionDocument document)
274 {
275 var errors = ValidateDocument(document);
276
277 if (errors.Count != 0)
278 throw new InvalidDataException(string.Join(Environment.NewLine, errors));
279
280 var created = new List<List<Sprite>>();
281 var previous = SpriteManager.Instance.DefinitionProvenance;
282
283 try
284 {
285 foreach (var definition in document.Definitions)
286 created.Add(ToSprites(definition));
287
288 if (created.Count != 1)
289 SpriteManager.Instance.DefinitionProvenance = null;
290
291 return created;
292 }
293 catch
294 {
295 foreach (var sprite in created.SelectMany(group => group))
296 {
297 SpriteManager.Instance._spriteList.Remove(sprite);
298 sprite.DisposeImmediate();
299 }
300
301 SpriteManager.Instance.DefinitionProvenance = previous;
302 throw;
303 }
