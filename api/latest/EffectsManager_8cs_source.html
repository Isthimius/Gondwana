Gondwana Game Engine Development (master) - 2.6.0-unreleased
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

114 private TEffect RunCore<TEffect>(object target, TEffect effect)
115 where TEffect : DisplayEffect
116 {
117 ObjectDisposedException.ThrowIf(_disposed, this);
118 ArgumentNullException.ThrowIfNull(target);
119 ArgumentNullException.ThrowIfNull(effect);
120
121 if (!OwnsTarget(target))
122 throw new ArgumentException(
123 "The effect target is not owned by this render surface host.",
124 nameof(target));
125
126 if (!effect.SupportsTarget(target))
127 throw new ArgumentException(
128 $"{effect.GetType().Name} does not support {target.GetType().Name} targets.",
129 nameof(effect));
130
131 DisplayEffect? replaced;
132
133 lock (_sync)
134 {
135 if (effect.Status != EffectStatus.Pending)
136 throw new InvalidOperationException("An effect instance can only be run once.");
137
138 replaced = _activeEffects.FirstOrDefault(
139 active => ReferenceEquals(active.Target, target)
140 && active.Channel == effect.Channel);
141
142 if (replaced is not null)
143 _activeEffects.Remove(replaced);
144
145 _activeEffects.Add(effect);
146 }
147
148 // Preserve the current presentation value when replacing an effect so the
149 // new effect can continue from it without a one-frame reset.
150 replaced?.CancelInternal(restoreState: false);
151
152 try
153 {
154 effect.StartInternal(this, target);
155 Invalidate(target);
156 }
157 catch
158 {
159 lock (_sync)
160 _activeEffects.Remove(effect);
161
162 throw;
163 }
164
165 if (effect.Status != EffectStatus.Running)
166 {
167 lock (_sync)
168 _activeEffects.Remove(effect);
169 }
170
171 return effect;
172 }
