Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 public static TelemetrySummary Combine(IEnumerable<TelemetrySummary> summaries)
51 {
52 ArgumentNullException.ThrowIfNull(summaries);
53 long count = 0;
54 double sum = 0;
55 double? min = null, max = null, last = null;
56 double? observed = null, delta = null;
57 TelemetryMetricKind? kind = null;
58 var availability = TelemetryAvailability.Unsupported;
59 foreach (var value in summaries)
60 {
61 if (kind is not null && kind != value.Kind) throw new ArgumentException("Cannot combine different metric kinds.", nameof(summaries));
62 kind = value.Kind;
63 if (value.Availability > availability) availability = value.Availability;
64 if (value.Count == 0) continue;
65 count += value.Count;
66 sum += value.Sum;
67 min = min is null ? value.Minimum : Math.Min(min.Value, value.Minimum!.Value);
68 max = max is null ? value.Maximum : Math.Max(max.Value, value.Maximum!.Value);
69 last = value.Last;
70 observed = value.LastObservedSeconds;
71 if (value.WindowDelta is { } change) delta = (delta ?? 0) + change;
72 }
73 return new(availability, count, sum, min, max, last)
74 {
75 Kind = kind ?? TelemetryMetricKind.Sample,
76 LastObservedSeconds = observed,
77 WindowDelta = delta
78 };
79 }

99 public TimeSpan Interval { get; init; } = TimeSpan.FromMilliseconds(250);
101 public int HistoryCapacity { get; init; } = 120;
103 public int SourceCapacity { get; init; } = 16;
105 public int MetricCapacity { get; init; } = 128;
106
108 internal void Validate()
109 {
110 if (Interval < TimeSpan.FromMilliseconds(10) || Interval > TimeSpan.FromMinutes(1))
111 throw new ArgumentOutOfRangeException(nameof(Interval));
112 if (HistoryCapacity is < 1 or > 1024) throw new ArgumentOutOfRangeException(nameof(HistoryCapacity));
113 if (SourceCapacity is < 1 or > 64) throw new ArgumentOutOfRangeException(nameof(SourceCapacity));
114 if (MetricCapacity is < 8 or > 512) throw new ArgumentOutOfRangeException(nameof(MetricCapacity));
115
116 long retainedMetricSummaries =
117 (long)HistoryCapacity * SourceCapacity * MetricCapacity;
118 if (retainedMetricSummaries > MaxRetainedMetricSummaries)
119 {
120 throw new ArgumentException(
121 $"HistoryCapacity * SourceCapacity * MetricCapacity must not exceed {MaxRetainedMetricSummaries:N0} retained metric summaries.");
122 }
123
124 double publishedMetricSummariesPerSecond =
125 (double)SourceCapacity * MetricCapacity / Interval.TotalSeconds;
126 if (publishedMetricSummariesPerSecond > MaxPublishedMetricSummariesPerSecond)
127 {
128 throw new ArgumentException(
129 $"SourceCapacity * MetricCapacity / Interval.TotalSeconds must not exceed {MaxPublishedMetricSummariesPerSecond:N0} configured metric summaries per second.");
130 }
131 }
