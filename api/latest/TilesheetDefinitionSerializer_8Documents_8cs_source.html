Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

115 private static TilesheetDefinition PrepareDocumentMember(TilesheetDefinition definition, string path)
116 {
117 var clone = ParseDefinition(ToJson(definition), null);
118 var origin = definition.Source.GtsFilePath ?? definition.Source.AssetsFilePath;
119 var oldDirectory = definition.LoadStamp?.ReferenceBaseDirectory ??
120 (string.IsNullOrWhiteSpace(origin) || !Path.IsPathFullyQualified(origin) ? null : Path.GetDirectoryName(origin));
121 var newDirectory = Path.GetDirectoryName(path)!;
122 string? RebasePath(string? value) => string.IsNullOrWhiteSpace(value) || (oldDirectory is null && !Path.IsPathRooted(value)) ? value :
123 Path.GetRelativePath(newDirectory, oldDirectory is null ? Path.GetFullPath(value) : Path.GetFullPath(value, oldDirectory));
124 if (clone.Image is { } image)
125 {
126 if (string.IsNullOrWhiteSpace(image.FilePath) && string.IsNullOrWhiteSpace(image.AssetsFilePath) &&
127 !string.IsNullOrWhiteSpace(image.AssetEntryName))
128 image.AssetsFilePath = definition.LoadStamp?.AssetsFilePath ?? definition.Source.AssetsFilePath;
129 image.FilePath = RebasePath(image.FilePath);
130 image.AssetsFilePath = RebasePath(image.AssetsFilePath);
131 }
132 ApplyDefaultSource(clone, TilesheetDefinitionSource.LooseDefinitionFile(path));
133 return clone;
134 }
