Gondwana Game Engine Development (master) - 2.7.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

497 public void Draw(BackbufferBase backbuffer, RectangleF destRectScreen)
498 {
499 if (!Visible)
500 return;
501
502 var canvas = backbuffer.Canvas;
503
504 // Compute reveal clip rect (screen pixel space) from bounds
505 bool useClip = _revealT < 0.999f;
506
507 if (useClip)
508 {
509 var r = new SKRect(
510 destRectScreen.Left,
511 destRectScreen.Top,
512 destRectScreen.Right,
513 destRectScreen.Bottom);
514
515 SKRect clipRect = _revealDir switch
516 {
517 RevealDirection.LeftToRight => new SKRect(
518 r.Left,
519 r.Top,
520 r.Left + r.Width * _revealT,
521 r.Bottom),
522 RevealDirection.RightToLeft => new SKRect(
523 r.Right - r.Width * _revealT,
524 r.Top,
525 r.Right,
526 r.Bottom),
527 RevealDirection.TopToBottom => new SKRect(
528 r.Left,
529 r.Top,
530 r.Right,
531 r.Top + r.Height * _revealT),
532 RevealDirection.BottomToTop => new SKRect(
533 r.Left,
534 r.Bottom - r.Height * _revealT,
535 r.Right,
536 r.Bottom),
537 _ => r
538 };
539
540 // Early-out if reveal window is empty
541 if (clipRect.Width <= 0f || clipRect.Height <= 0f)
542 return;
543
544 // Outer save owns the clip lifetime
545 canvas.Save();
546
547 // Capture current matrix
548 var m = canvas.TotalMatrix;
549
550 canvas.ResetMatrix();
551 canvas.ClipRect(clipRect, SKClipOperation.Intersect, antialias: false);
552
553 // Put the prior matrix back so Draw() sees the same transform state
554 canvas.SetMatrix(m);
555 }
556
557 if (_opacity >= 0.999f)
558 {
559 OnDraw(backbuffer, destRectScreen);
560 }
561 else
562 {
563 using var layerPaint = new SKPaint
564 {
565 Color = new SKColor(
566 255,
567 255,
568 255,
569 (byte)(_opacity * 255))
570 };
571
572 canvas.SaveLayer(destRectScreen.ToSKRect(), layerPaint);
573 OnDraw(backbuffer, destRectScreen);
574 canvas.Restore(); // end SaveLayer
575 }
576
577 if (useClip)
578 canvas.Restore(); // end outer clip Save
579 }

834 protected internal void ForceRefresh()
835 {
836 switch (Mode)
837 {
838 case DirectDrawingMode.SceneLayer:
839 // bounds is WORLD-space
840 // there will only be one SceneLayer per DirectDrawing in this mode
841 SceneLayer!.RefreshQueue.AddWorldRect(_worldBounds);
842 break;
843
844 case DirectDrawingMode.View:
845 // bounds is SCREEN-space
846 // need to cycle through all SceneLayers for the View to which this DirectDrawing belongs
847 foreach (var sceneLayer in RenderSurfaceHost.Scene.SceneLayers)
848 {
849 sceneLayer.RefreshQueue.AddViewScreenRect(View!, sceneLayer, _screenBounds);
850 }
851
852 break;
853
854 default:
855 break;
856 }
857 }

880 public virtual void Update(long tick)
881 {
882 if (tick <= _lastTick)
883 return;
884
885 // Advance fade tween
886 if (_isFading)
887 {
888 float dt = HighResTimer.GetDuration(_lastTick, tick);
889
890 _fadeElapsedSec += dt;
891 float timeElapsed = Math.Clamp(_fadeElapsedSec / _fadeDurationSec, 0f, 1f);
892 // linear; swap in easing if you like
893 _opacity = _fadeFrom + (_fadeTo - _fadeFrom) * timeElapsed;
894
895 if (HideWhenFullyTransparent)
896 Visible = _opacity > 0f; // hides when hit zero
897
898 ForceRefresh();
899
900 if (timeElapsed >= 1f)
901 {
902 _isFading = false;
903 FadeToCompleted?.Invoke(this, this);
904 }
905 }
906
907 // Advance reveal tween
908 if (_revealAnimating)
909 {
910 float dt = HighResTimer.GetDuration(_lastTick, tick);
911 _revealElapsedSec = Math.Min(_revealElapsedSec + dt, _revealDurationSec);
912
913 float u = _revealElapsedSec / _revealDurationSec;
914 _revealT = (_revealEasing is null ? u : _revealEasing(u));
915 _revealT = _revealStart + (_revealTarget - _revealStart) * _revealT;
916
917 ForceRefresh();
918
919 if (_revealElapsedSec >= _revealDurationSec)
920 _revealAnimating = false;
921 }
922
923 _lastTick = tick;
924 }
