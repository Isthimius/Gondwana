Gondwana Game Engine Development (master) - 2.6.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

219 public TextBoxWidget SetText(string? text)
220 {
221 string normalized = NormalizeText(text);
222
223 if (MaxLength.HasValue && normalized.Length > MaxLength.Value)
224 normalized = normalized[..MaxLength.Value];
225
226 if (_text == normalized)
227 {
228 _caretIndex = Math.Min(_caretIndex, _text.Length);
229 RefreshDisplayedText();
230 return this;
231 }
232
233 _text = normalized;
234 _caretIndex = _text.Length;
235 RefreshDisplayedText();
236 TextChanged?.Invoke(_text);
237 return this;
238 }

243 public TextBoxWidget InsertText(string? text)
244 {
245 if (IsReadOnly || string.IsNullOrEmpty(text))
246 return this;
247
248 string insertion = NormalizeText(text);
249 if (insertion.Length == 0)
250 return this;
251
252 if (MaxLength.HasValue)
253 {
254 int available = MaxLength.Value - _text.Length;
255 if (available <= 0)
256 return this;
257
258 if (insertion.Length > available)
259 insertion = insertion[..available];
260 }
261
262 _text = _text.Insert(_caretIndex, insertion);
263 _caretIndex += insertion.Length;
264 RefreshDisplayedText();
265 TextChanged?.Invoke(_text);
266 return this;
267 }

377 protected override void OnKeyboardInput(WidgetKeyboardEventArgs args)
378 {
379 base.OnKeyboardInput(args);
380
381 if (args.KeyAction is not KeyAction.Pressed and not KeyAction.Repeated)
382 return;
383
384 if (args.Key == SubmitKey)
385 {
386 if (args.KeyAction != KeyAction.Pressed)
387 return;
388
389 args.Handled = true;
390 Submitted?.Invoke(_text);
391 return;
392 }
393
394 if (args.Key == LeftKey)
395 {
396 args.Handled = true;
397 MoveCaret(-1);
398 return;
399 }
400
401 if (args.Key == RightKey)
402 {
403 args.Handled = true;
404 MoveCaret(1);
405 return;
406 }
407
408 if (args.Key == HomeKey)
409 {
410 args.Handled = true;
411 CaretIndex = 0;
412 return;
413 }
414
415 if (args.Key == EndKey)
416 {
417 args.Handled = true;
418 CaretIndex = _text.Length;
419 return;
420 }
421
422 if (args.Key == BackspaceKey)
423 {
424 args.Handled = true;
425 Backspace();
426 return;
427 }
428
429 if (args.Key == DeleteKey)
430 {
431 args.Handled = true;
432 Delete();
433 return;
434 }
435
436 if (IsReadOnly)
437 return;
438
439 char? character = CharacterResolver(args);
440 if (!character.HasValue)
441 return;
442
443 args.Handled = true;
444 InsertText(character.Value.ToString());
445 }

527 private static char? ResolveWindowsVirtualKeyCharacter(WidgetKeyboardEventArgs args)
528 {
529 if ((args.Modifiers & (KeyboardModifierState.Ctrl | KeyboardModifierState.Alt)) != 0)
530 return null;
531
532 bool shift = (args.Modifiers & KeyboardModifierState.Shift) != 0;
533 int key = args.Key;
534
535 if (key is >= 65 and <= 90)
536 {
537 char letter = (char)key;
538 return shift ? letter : char.ToLowerInvariant(letter);
539 }
540
541 if (key is >= 48 and <= 57)
542 {
543 if (!shift)
544 return (char)key;
545
546 const string shiftedDigits = ")!@#$%^&*(";
547 return shiftedDigits[key - 48];
548 }
549
550 if (key is >= 96 and <= 105)
551 return (char)('0' + key - 96);
552
553 return key switch
554 {
555 32 => ' ',
556 106 => '*',
557 107 => '+',
558 109 => '-',
559 110 => '.',
560 111 => '/',
561 186 => shift ? ':' : ';',
562 187 => shift ? '+' : '=',
563 188 => shift ? '<' : ',',
564 189 => shift ? '_' : '-',
565 190 => shift ? '>' : '.',
566 191 => shift ? '?' : '/',
567 192 => shift ? '~' : '`',
568 219 => shift ? '{' : '[',
569 220 => shift ? '|' : '\\',
570 221 => shift ? '}' : ']',
571 222 => shift ? '"' : '\'',
572 _ => null
573 };
574 }
