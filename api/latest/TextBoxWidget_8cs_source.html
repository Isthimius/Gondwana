Gondwana Game Engine Development (master) - 2.6.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

239 public TextBoxWidget SetText(string? text)
240 {
241 string normalized = NormalizeText(text);
242
243 if (MaxLength.HasValue && normalized.Length > MaxLength.Value)
244 normalized = normalized[..MaxLength.Value];
245
246 if (_text == normalized)
247 {
248 _caretIndex = Math.Min(_caretIndex, _text.Length);
249 RefreshDisplayedText();
250 return this;
251 }
252
253 _text = normalized;
254 _caretIndex = _text.Length;
255 RefreshDisplayedText();
256 TextChanged?.Invoke(_text);
257 return this;
258 }

263 public TextBoxWidget InsertText(string? text)
264 {
265 if (IsReadOnly || string.IsNullOrEmpty(text))
266 return this;
267
268 string insertion = NormalizeText(text);
269 if (insertion.Length == 0)
270 return this;
271
272 if (MaxLength.HasValue)
273 {
274 int available = MaxLength.Value - _text.Length;
275 if (available <= 0)
276 return this;
277
278 if (insertion.Length > available)
279 insertion = insertion[..available];
280 }
281
282 _text = _text.Insert(_caretIndex, insertion);
283 _caretIndex += insertion.Length;
284 RefreshDisplayedText();
285 TextChanged?.Invoke(_text);
286 return this;
287 }

397 protected override void OnKeyboardInput(WidgetKeyboardEventArgs args)
398 {
399 base.OnKeyboardInput(args);
400
401 if (args.KeyAction == KeyAction.Released)
402 {
403 if (_lastRepeatedKey == args.Key)
404 {
405 _lastRepeatedKey = null;
406 _lastRepeatTick = 0;
407 }
408
409 return;
410 }
411
412 if (args.KeyAction is not KeyAction.Pressed and not KeyAction.Repeated)
413 return;
414
415 long currentTick = HighResTimer.GetCurrentTick();
416
417 if (args.KeyAction == KeyAction.Pressed)
418 {
419 _lastRepeatedKey = args.Key;
420 _lastRepeatTick = currentTick;
421 }
422 else if (_lastRepeatedKey == args.Key &&
423 _lastRepeatTick != 0 &&
424 HighResTimer.GetDuration(_lastRepeatTick, currentTick) < _repeatedKeyIntervalSeconds)
425 {
426 args.Handled = true;
427 return;
428 }
429 else
430 {
431 _lastRepeatedKey = args.Key;
432 _lastRepeatTick = currentTick;
433 }
434
435 if (args.Key == SubmitKey)
436 {
437 if (args.KeyAction != KeyAction.Pressed)
438 return;
439
440 args.Handled = true;
441 Submitted?.Invoke(_text);
442 return;
443 }
444
445 if (args.Key == LeftKey)
446 {
447 args.Handled = true;
448 MoveCaret(-1);
449 return;
450 }
451
452 if (args.Key == RightKey)
453 {
454 args.Handled = true;
455 MoveCaret(1);
456 return;
457 }
458
459 if (args.Key == HomeKey)
460 {
461 args.Handled = true;
462 CaretIndex = 0;
463 return;
464 }
465
466 if (args.Key == EndKey)
467 {
468 args.Handled = true;
469 CaretIndex = _text.Length;
470 return;
471 }
472
473 if (args.Key == BackspaceKey)
474 {
475 args.Handled = true;
476 Backspace();
477 return;
478 }
479
480 if (args.Key == DeleteKey)
481 {
482 args.Handled = true;
483 Delete();
484 return;
485 }
486
487 if (IsReadOnly)
488 return;
489
490 char? character = CharacterResolver(args);
491 if (!character.HasValue)
492 return;
493
494 args.Handled = true;
495 InsertText(character.Value.ToString());
496 }

578 private static char? ResolveWindowsVirtualKeyCharacter(WidgetKeyboardEventArgs args)
579 {
580 if ((args.Modifiers & (KeyboardModifierState.Ctrl | KeyboardModifierState.Alt)) != 0)
581 return null;
582
583 bool shift = (args.Modifiers & KeyboardModifierState.Shift) != 0;
584 int key = args.Key;
585
586 if (key is >= 65 and <= 90)
587 {
588 char letter = (char)key;
589 return shift ? letter : char.ToLowerInvariant(letter);
590 }
591
592 if (key is >= 48 and <= 57)
593 {
594 if (!shift)
595 return (char)key;
596
597 const string shiftedDigits = ")!@#$%^&*(";
598 return shiftedDigits[key - 48];
599 }
600
601 if (key is >= 96 and <= 105)
602 return (char)('0' + key - 96);
603
604 return key switch
605 {
606 32 => ' ',
607 106 => '*',
608 107 => '+',
609 109 => '-',
610 110 => '.',
611 111 => '/',
612 186 => shift ? ':' : ';',
613 187 => shift ? '+' : '=',
614 188 => shift ? '<' : ',',
615 189 => shift ? '_' : '-',
616 190 => shift ? '>' : '.',
617 191 => shift ? '?' : '/',
618 192 => shift ? '~' : '`',
619 219 => shift ? '{' : '[',
620 220 => shift ? '|' : '\\',
621 221 => shift ? '}' : ']',
622 222 => shift ? '"' : '\'',
623 _ => null
624 };
625 }
