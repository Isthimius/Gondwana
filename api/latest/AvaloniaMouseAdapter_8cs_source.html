Gondwana Game Engine Development (master) - 2.6.2
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop and web platforms using SkiaSharp for graphics and pluggable audio backends for desktop and browser targets.

50 public AvaloniaMouseAdapter(Control control)
51 {
52 _control = control ?? throw new ArgumentNullException(nameof(control));
53
54 // OpenGlControlBase can render in a native child surface (notably WGL).
55 // Route input at the Avalonia TopLevel in the tunnel phase, as keyboard
56 // input already does, so the GL surface cannot swallow pointer events.
57 _control.AttachedToVisualTree += OnAttachedToVisualTree;
58 _control.DetachedFromVisualTree += OnDetachedFromVisualTree;
59
60 if (TopLevel.GetTopLevel(_control) is not null)
61 ConnectToTopLevel();
62 }
63
64 private void OnAttachedToVisualTree(object? sender, VisualTreeAttachmentEventArgs e)
65 => ConnectToTopLevel();
66
67 private void OnDetachedFromVisualTree(object? sender, VisualTreeAttachmentEventArgs e)
68 => DisconnectFromTopLevel();
69
70 private void ConnectToTopLevel()
71 {
72 var topLevel = TopLevel.GetTopLevel(_control);
73 if (topLevel is null || ReferenceEquals(topLevel, _topLevel))
74 return;
75
76 DisconnectFromTopLevel();
77 _topLevel = topLevel;
78 topLevel.AddHandler(InputElement.PointerPressedEvent, OnPointerPressed, RoutingStrategies.Tunnel);
79 topLevel.AddHandler(InputElement.PointerReleasedEvent, OnPointerReleased, RoutingStrategies.Tunnel);
80 topLevel.AddHandler(InputElement.PointerMovedEvent, OnPointerMoved, RoutingStrategies.Tunnel);
81 topLevel.AddHandler(InputElement.PointerWheelChangedEvent, OnPointerWheelChanged, RoutingStrategies.Tunnel);
82 }
83
84 private void DisconnectFromTopLevel()
85 {
86 if (_topLevel is null)
87 return;
88
89 _topLevel.RemoveHandler(InputElement.PointerPressedEvent, OnPointerPressed);
90 _topLevel.RemoveHandler(InputElement.PointerReleasedEvent, OnPointerReleased);
91 _topLevel.RemoveHandler(InputElement.PointerMovedEvent, OnPointerMoved);
92 _topLevel.RemoveHandler(InputElement.PointerWheelChangedEvent, OnPointerWheelChanged);
93 _topLevel = null;
94 }

114 private void UpdatePosition(PointerEventArgs e)
115 {
116 // Pointer events are raised on TopLevel, but positions must remain
117 // local to the render control for DPI/presentation normalization.
118 var pos = e.GetPosition(_control);
119 _currentPosition = new PointF((float)pos.X, (float)pos.Y);
120
121 _modifiers = KeyboardModifierState.None;
122 var km = e.KeyModifiers;
123 if ((km & KeyModifiers.Shift) != 0) _modifiers |= KeyboardModifierState.Shift;
124 if ((km & KeyModifiers.Control) != 0) _modifiers |= KeyboardModifierState.Ctrl;
125 if ((km & KeyModifiers.Alt) != 0) _modifiers |= KeyboardModifierState.Alt;
126 }

128 private static GondwanaMouseButton MapButton(PointerPointProperties props)
129 {
130 if (props.IsLeftButtonPressed) return GondwanaMouseButton.Left;
131 if (props.IsRightButtonPressed) return GondwanaMouseButton.Right;
132 if (props.IsMiddleButtonPressed) return GondwanaMouseButton.Middle;
133 return GondwanaMouseButton.None;

144 private void OnPointerWheelChanged(object? sender, PointerWheelEventArgs e)
145 {
146 // Avalonia Delta.Y is in lines; positive = scroll up (matches WinForms convention).
