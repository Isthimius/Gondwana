Gondwana Game Engine 2.1.0
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

159 public bool SlidePiece(Sprite sprite, float slideTime)
160 {
161 if (FindSpritesAdjToOpenSpace().IndexOf(sprite) == -1)
162 // sprite not eligible to move
163 return false;
164 else
165 {
166 // capture the starting point of the sprite being moved
167 Point startPt = new Point((int)sprite.SceneLayerCoordinates.X, (int)sprite.SceneLayerCoordinates.Y);
168
169 // move the sprite to the open space
170 sprite.Movement.MoveTo(new Vector2(openSpace.X, openSpace.Y), slideTime, null , 0.01f);
171
172 // make the openSpace value equal to the original sprite starting point
173 openSpace = startPt;
174
175 // move was successful
176 return true;
177 }

186 {

193 ShuffleNext();

196 private void ShuffleNext()
197 {
198 Random random = new Random();
199
200 // find all pieces next to open space
201 List<Sprite> sprites = FindSpritesAdjToOpenSpace();
202
203 // pick one of the pieces at random
204 Sprite sprite = sprites[random.Next(0, sprites.Count)];
205
206 // don't move the same sprite 2 times in a row
207 while (sprite == _lastMoved)
208 sprite = sprites[random.Next(0, sprites.Count)];
209
210 // move the piece
211 SlidePiece(sprite, _slideTime);
212 _lastMoved = sprite;
213
214 if (++_moveNumber >= _totalMoves)
215 _isShuffling = false;
216 }

230 {

242 sprite.Movement.ScriptedMovementStarted += delMoveStart;

247 // remove the bottom-right tile; this will be the space for sliding

249 int maxY = numRows - 1;
250 SpriteManager.Instance.GetSpriteByID(maxX.ToString() + "-" + maxY.ToString()).Dispose();
251 openSpace = new Point(maxX, maxY);
252 }

262 private List<Sprite> FindSpritesAdjToOpenSpace()
263 {
264 List<Sprite> adjSprites = new List<Sprite>();
265 List<SceneLayerTile> adjGridPts = new List<SceneLayerTile>();
266
267 var layer = matrixes[0];
268 var centerTile = layer[openSpace];
269
270 adjGridPts.Add(layer.GetAdjacentTile(centerTile, CardinalDirections.N));
271 adjGridPts.Add(layer.GetAdjacentTile(centerTile, CardinalDirections.S));
272 adjGridPts.Add(layer.GetAdjacentTile(centerTile, CardinalDirections.E));
273 adjGridPts.Add(layer.GetAdjacentTile(centerTile, CardinalDirections.W));
274
275 foreach (SceneLayerTile gPt in adjGridPts)
276 {
277 if (gPt != null)
278 adjSprites.AddRange(SpriteManager.Instance.GetSpritesInWorldRectRange(gPt.DrawLocationWorld));
279 }
280
281 return adjSprites;
