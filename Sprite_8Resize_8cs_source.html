Gondwana Game Engine 2.1.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

40 internal void AdvanceResize(float deltaSeconds)
41 {
42 if (!_isResizing)
43 return;
44
45 if (_resizeDurationSeconds <= 0f)
46 {
47 RenderSize = _resizeTarget;
48 ApplyScaledCollisionAdjust(RenderSize);
49 CompleteResizeLeg();
50 return;
51 }
52
53 _resizeElapsedSeconds += deltaSeconds;
54
55 float t = _resizeElapsedSeconds / _resizeDurationSeconds;
56 if (t >= 1f)
57 t = 1f;
58
59 int w = (int)MathF.Round(_resizeStart.Width + ((_resizeTarget.Width - _resizeStart.Width) * t));
60 int h = (int)MathF.Round(_resizeStart.Height + ((_resizeTarget.Height - _resizeStart.Height) * t));
61
62 var next = new Size(Math.Max(1, w), Math.Max(1, h));
63
64 if (next != _renderSize)
65 {
66 RenderSize = next; // invalidation happens in setter
67 ApplyScaledCollisionAdjust(next); // physical pulse: collision scales too
68 }
69
70 if (t >= 1f)
71 {
72 CompleteResizeLeg();
73 }
74 }

76 private void CompleteResizeLeg()
77 {
78 // Snap exactly to the target at the end of the leg
79 RenderSize = _resizeTarget;
80 ApplyScaledCollisionAdjust(RenderSize);
81
82 if (_isPulseMode)
83 {
84 if (_pulseForward)
85 {
86 // Original -> Peak finished; now go Peak -> Original
87 _pulseForward = false;
88 StartResizeLeg(
89 start: _resizePeak,
90 target: _resizeOriginalStart,
91 durationSeconds: _pulseShrinkDurationSeconds,
92 returnToStart: false,
93 returnDurationSeconds: 0f);
94 return;
95 }
96
97 // Peak -> Original finished
98 if (_pulseLoop)
99 {
100 // Start next cycle
101 _pulseForward = true;
102 StartResizeLeg(
103 start: _resizeOriginalStart,
104 target: _resizePeak,
105 durationSeconds: _pulseGrowDurationSeconds,
106 returnToStart: false,
107 returnDurationSeconds: 0f);
108 return;
109 }
110
111 _isPulseMode = false;
112 _isResizing = false;
113
114 ResizeComplete?.Invoke();
115 return;
116 }
117
118 if (_resizeReturnToStart)
119 {
120 StartResizeLeg(
121 start: RenderSize,
122 target: _resizeOriginalStart,
123 durationSeconds: _resizeReturnDurationSeconds,
124 returnToStart: false,
125 returnDurationSeconds: 0f);
126 }
127 else
128 {
129 _isResizing = false;
130 ResizeComplete?.Invoke();
131 }
132 }

156 private void ApplyScaledCollisionAdjust(Size currentSize)
157 {
158 if (_resizeStartSizeForCollision.Width <= 0 || _resizeStartSizeForCollision.Height <= 0)
159 return;
160
161 float sx = (float)currentSize.Width / _resizeStartSizeForCollision.Width;
162 float sy = (float)currentSize.Height / _resizeStartSizeForCollision.Height;
163
164 AdjustCollisionArea = new CollisionDetectionAdjustment(
165 top: (int)MathF.Round(_resizeStartCollisionAdjust.Top * sy),
166 bottom: (int)MathF.Round(_resizeStartCollisionAdjust.Bottom * sy),
167 left: (int)MathF.Round(_resizeStartCollisionAdjust.Left * sx),
168 right: (int)MathF.Round(_resizeStartCollisionAdjust.Right * sx)
169 );
170 }
