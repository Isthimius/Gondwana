Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

149 internal void PollForEvents(long tick)
150 {
151 if (Adapter is null) return;
152 if ((Configuration?.IsPaused ?? true) || !(Configuration?.ReadyForNextEvent(tick) ?? false)) return;
153
154 var currentPos = Adapter.CurrentPosition;
155 var pressed = Adapter.PressedButtons;
156 var scrollDelta = Adapter.ScrollDelta;
157
158 bool anyButtonChange = false;
159 bool isAnyButtonDown = false;
160
161 // update per-button state every cycle
162 foreach (var kvp in _buttonStates.ToList())
163 {
164 var button = kvp.Key;
165 var state = kvp.Value;
166
167 bool isCurrentlyDown = pressed.Contains(button);
168 bool justPressed = isCurrentlyDown && !state.IsDown;
169 bool justReleased = !isCurrentlyDown && state.IsDown;
170
171 state.JustPressed = justPressed;
172 state.JustReleased = justReleased;
173 state.IsDown = isCurrentlyDown;
174
175 anyButtonChange |= justPressed || justReleased;
176 _buttonStates[button] = state;
177
178 if (isCurrentlyDown)
179 isAnyButtonDown = true;
180 }
181
182 // now decide whether to emit an event
183 bool moved = (Configuration?.TrackMouseMovement ?? false) && _lastPosition != currentPos;
184 // Emit a scroll event only when the delta is non-zero and has changed since the last poll.
185 // This works for both adapters that reset delta to 0 after each read and adapters that
186 // hold a persistent delta until it changes, preventing repeated events on every poll.
187 bool scrolled = scrollDelta != 0 && scrollDelta != _lastScrollDelta;
188 _lastScrollDelta = scrollDelta;
189
190 if (anyButtonChange || moved || scrolled || isAnyButtonDown)
191 {
192 MouseEvent?.Invoke(new MouseEventArgs(
193 Configuration!,
194 CurrentKeyboardModifiers,
195 ButtonStates,
196 _lastPosition,
197 currentPos,
198 scrollDelta,
199 tick));
200
201 _lastPosition = currentPos;
202 Configuration!._lastEventTick = tick;
203 }
204 }
