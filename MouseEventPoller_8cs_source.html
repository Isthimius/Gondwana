Gondwana Game Engine v2.2.5
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

138 internal void PollForEvents(long tick)
139 {
140 if (Adapter is null) return;
141 if ((Configuration?.IsPaused ?? true) || !(Configuration?.ReadyForNextEvent(tick) ?? false)) return;
142
143 var currentPos = Adapter.CurrentPosition;
144 var pressed = Adapter.PressedButtons;
145 var scrollDelta = Adapter.ScrollDelta;
146
147 bool anyButtonChange = false;
148 bool isAnyButtonDown = false;
149
150 // update per-button state every cycle
151 foreach (var kvp in _buttonStates.ToList())
152 {
153 var button = kvp.Key;
154 var state = kvp.Value;
155
156 bool isCurrentlyDown = pressed.Contains(button);
157 bool justPressed = isCurrentlyDown && !state.IsDown;
158 bool justReleased = !isCurrentlyDown && state.IsDown;
159
160 state.JustPressed = justPressed;
161 state.JustReleased = justReleased;
162 state.IsDown = isCurrentlyDown;
163
164 anyButtonChange |= justPressed || justReleased;
165 _buttonStates[button] = state;
166
167 if (isCurrentlyDown)
168 isAnyButtonDown = true;
169 }
170
171 // now decide whether to emit an event
172 bool moved = (Configuration?.TrackMouseMovement ?? false) && _lastPosition != currentPos;
173 bool scrolled = _lastScrollDelta != scrollDelta;
174
175 if (anyButtonChange || moved || scrolled || isAnyButtonDown)
176 {
177 MouseEvent?.Invoke(new MouseEventArgs(
178 Configuration!,
179 CurrentKeyboardModifiers,
180 ButtonStates,
181 _lastPosition,
182 currentPos,
183 scrollDelta));
184
185 _lastPosition = currentPos;
186 _lastScrollDelta = scrollDelta;
187 }
188 }
