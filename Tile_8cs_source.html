Gondwana Game Engine v2.5.3
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

422 public int CompareTo(Tile? tile)
423 {
424 if (tile is null)
425 return -1;
426
427 float thisLoc = GetTileLocForCompare(this);
428 float tileLoc = GetTileLocForCompare(tile);
429
430 if (IsPositionFixed && !tile.IsPositionFixed)
431 return -1;
432
433 if (!IsPositionFixed && tile.IsPositionFixed)
434 return 1;
435
436 return (thisLoc, zOrder, SceneLayerCoordinates.X)
437 .CompareTo((tileLoc, tile.zOrder, tile.SceneLayerCoordinates.X));
438 }
439
440 private static float GetTileLocForCompare(Tile tile) =>
441 tile.IsPositionFixed
442 ? tile.DrawLocationWorld.Top + tile.Overhang.Top
443 : tile.DrawLocationWorld.Bottom - tile.Overhang.Bottom - 1;
