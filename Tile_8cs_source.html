Gondwana Game Engine v2.3.1
Gondwana is a cross-platform 2.5D game and rendering engine written in C#/.NET 8. It provides fine-grained control over rendering, timing, and scene composition, with built-in support for parallax, z-ordering, pixel overhang, collision detection, and particle effects. Gondwana targets desktop, mobile, and web platforms using SkiaSharp for graphics and NAudio for sound.

274 public int CompareTo(Tile? tile)
275 {
276 if (tile is null)
277 return -1;
278
279 float thisLoc = GetTileLocForCompare(this);
280 float tileLoc = GetTileLocForCompare(tile);
281
282 // Handle fixed position vs non-fixed first
283 if (IsPositionFixed && !tile.IsPositionFixed)
284 return -1;
285
286 if (!IsPositionFixed && tile.IsPositionFixed)
287 return 1;
288
289 // Use tuple comparison for the rest (Y, Z, X)
290 return (thisLoc, zOrder, SceneLayerCoordinates.X)
291 .CompareTo((tileLoc, tile.zOrder, tile.SceneLayerCoordinates.X));
292 }
293
298 private static float GetTileLocForCompare(Tile tile)
299 {
300 return tile.IsPositionFixed
301 ? tile.DrawLocationWorld.Top + tile.OverhangPixels.Top
302 : tile.DrawLocationWorld.Bottom - tile.OverhangPixels.Bottom - 1;
303 }
